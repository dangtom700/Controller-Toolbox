14 // Discrete PID - backward Euler integral + backward-Euler filtered derivative.
15 //
16 // Derivative filter recurrence (derived from C_D(s) = Kd.N.s/(s+N), backward Euler):
17 // alpha = 1 / (1 + N.Ts)
18 // d[k] = alpha.d[k-1] + Kd.N.alpha.(e[k] - e[k-1])
19 //
20 // Anti-windup back-calculation (Astrom & Wittenmark):
21 // I[k+1] = I[k] + Ki.Ts.e[k] + Kb.(u_sat[k] - u_unsat[k])
22 double DiscretePID::compute(double error)
23 {
24 // Hold last output on bad measurement rather than corrupting integral state
25 if (!std::isfinite(error))
26 return u_prev_;
27
28 // Filtered derivative
29 const double alpha = 1.0 / (1.0 + p_.N * Ts_);
30 const double d_new = alpha * deriv_ + p_.Kd * p_.N * alpha * (error - e_prev_);
31
32 // Integral: backward Euler increment for this step
33 const double ki_update = p_.Ki * Ts_ * error;
34
35 // Unsaturated output - uses I[k] = I[k-1] + ki_update (backward Euler)
36 const double u_unsat = p_.Kp * error + (integral_ + ki_update) + d_new;
37
38 // Output saturation
39 const double u_sat = std::max(p_.uMin, std::min(p_.uMax, u_unsat));
40
41 // Integral state update with anti-windup back-calculation
42 integral_ += ki_update + p_.Kb * (u_sat - u_unsat);
43
44 // State updates
45 deriv_ = d_new;
46 e_prev_ = error;
47 u_prev_ = u_sat;
48
49 notifyObserver(u_sat, error);
50 return u_sat;
51 }

63 // Back-calculate the integral state so that compute(error) approx = u_target.
64 // Derivation: u_target approx = Kp*error + integral + Kd*N*(error-e_prev)/(1+N*Ts)
65 // Solving for integral: integral = u_target - Kp*error - deriv (zeroed here)
66 // On the very first compute() call after this, the output will be close to u_target.
67 void DiscretePID::bumplessInit(double u_target, double error)
68 {
69 deriv_ = 0.0;
70 e_prev_ = error;
71 u_prev_ = u_target;
72 integral_ = u_target - p_.Kp * error; // absorb P term; D starts from rest
73 }

75 // Derivative on measurement: same as compute() but routes the derivative filter
76 // through -y (plant output) instead of e = r - y. This eliminates the derivative
77 // spike that occurs when the setpoint r steps.
78 //
79 // Derivation:
80 // Standard: d[k] = alpha*d[k-1] + Kd*N*alpha*(e[k] - e[k-1])
81 // DoM: d[k] = alpha*d[k-1] - Kd*N*alpha*(y[k] - y[k-1]) <- derivative of -y only
82 // (sign correct: for a step up in r, y increases, d[k] goes negative = less overshoot)
83 //
84 // Equivalent MATLAB: Discrete PID Controller block, DerivativeFilter on 'Measurement'.
85 double DiscretePID::computeDoM(double y, double r)
86 {
87 const double error = r - y;
88
89 if (!std::isfinite(error) || !std::isfinite(y))
90 return u_prev_;
91
92 // Derivative filter on -y (not on error)
93 const double alpha = 1.0 / (1.0 + p_.N * Ts_);
94 const double d_new = alpha * deriv_ - p_.Kd * p_.N * alpha * (y - y_prev_);
95
96 // Integral: backward Euler on error (unchanged)
97 const double ki_update = p_.Ki * Ts_ * error;
98
99 // 2DOF proportional: acts on (b*r - y) rather than (r - y).
100 // b_weight = 1 reduces to standard 1DOF; b_weight = 0 removes the
101 // proportional setpoint kick entirely (Astrom & Hagglund 2006 Section 8.4).
102 const double prop_signal = p_.b_weight * r - y;
103
104 // Unsaturated output
105 const double u_unsat = p_.Kp * prop_signal + (integral_ + ki_update) + d_new;
106
107 // Output saturation
108 const double u_sat = std::max(p_.uMin, std::min(p_.uMax, u_unsat));
109
110 // Integral state update with anti-windup back-calculation
111 integral_ += ki_update + p_.Kb * (u_sat - u_unsat);
112
113 // State updates
114 deriv_ = d_new;
115 e_prev_ = error;
116 y_prev_ = y;
117 u_prev_ = u_sat;
118
119 return u_sat;
120 }
