27double MRACController::compute(double y_plant)
28{
29 // 1. Model tracking error: e_m[k] = y[k] - y_m[k]
30 e_m_ = y_plant - y_m_;
31
32 // 2. Control law: u[k] = theta_r * r + theta_y * y
33 double u = theta_r_ * r_ + theta_y_ * y_plant;
34 u = std::clamp(u, params_.uMin, params_.uMax);
35
36 // 3. Lyapunov adaptation with sigma-modification (discrete gradient descent + damping)
37 // theta[k+1] = theta[k] - Ts * (gamma * e_m * phi + sigma * theta[k])
38 theta_r_ -= Ts_ * (params_.gamma_r * e_m_ * r_ + params_.sigma * theta_r_);
39 theta_y_ -= Ts_ * (params_.gamma_y * e_m_ * y_plant + params_.sigma * theta_y_);
40
41 // 4. Parameter projection: clamp ||theta||2 <= theta_max
42 const double theta_norm = std::sqrt(theta_r_ * theta_r_ + theta_y_ * theta_y_);
43 if (theta_norm > params_.theta_max)
44 {
45 const double scale = params_.theta_max / theta_norm;
46 theta_r_ *= scale;
47 theta_y_ *= scale;
48 }
49
50 // 5. Advance reference model: y_m[k+1] = a_m * y_m[k] + b_m * r[k]
51 y_m_ = params_.a_m * y_m_ + params_.b_m * r_;
52
53 return u;
54}
