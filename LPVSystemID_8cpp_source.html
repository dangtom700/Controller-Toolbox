8LPVModel identifyLPV(const Eigen::MatrixXd& X,
9 const Eigen::MatrixXd& U,
10 const Eigen::MatrixXd& Y,
11 const std::vector<double>& sched,
12 int degree,
13 double Ts)
14{
15 const int n = static_cast<int>(X.rows()); // state dimension
16 const int m = static_cast<int>(U.rows()); // input dimension
17 const int p_out = static_cast<int>(Y.rows()); // output dimension
18 const int N = static_cast<int>(X.cols()); // time steps
19
20 if (N < 2)
21 throw std::invalid_argument("identifyLPV: need at least 2 time steps.");
22 if (static_cast<int>(U.cols()) != N || static_cast<int>(Y.cols()) != N)
23 throw std::invalid_argument("identifyLPV: X, U, Y must have the same number of columns.");
24 if (static_cast<int>(sched.size()) < N)
25 throw std::invalid_argument("identifyLPV: sched.size() must be >= N.");
26 if (degree < 0)
27 throw std::invalid_argument("identifyLPV: degree must be >= 0.");
28
29 const int d = degree;
30 const int d1 = d + 1;
31
32 // -------------------------------------------------------------------------
33 // State equation regression
34 // x[k+1] = A(p[k]) x[k] + B(p[k]) u[k]
35 //
36 // Row regressor phi_k = [x, p*x, ..., p^d*x | u, p*u, ..., p^d*u]
37 // n*(d+1) terms m*(d+1) terms
38 //
39 // Stacked system: Phi [(N-1) x (n+m)*d1] * Theta_AB [(n+m)*d1 x n] = Z_x
40 // Solution: Theta_AB = QR-pinv(Phi) * Z_x
41 //
42 // Theta_AB row layout:
43 // rows 0..n-1 -> A0^T (A_coeffs[0] = rows.T)
44 // rows n..2n-1 -> A1^T
45 // ...
46 // rows d*n..(d+1)*n-1 -> Ad^T
47 // rows (d+1)*n .. (d+1)*n+m-1 -> B0^T
48 // ...
49 // -------------------------------------------------------------------------
50 const int T_x = N - 1;
51 const int cols_AB = (n + m) * d1;
52 Eigen::MatrixXd Phi_AB(T_x, cols_AB);
53 Eigen::MatrixXd Z_x(T_x, n);
54
55 for (int k = 0; k < T_x; ++k) {
56 const double p = sched[k];
57 int col = 0;
58
59 // A-part: p^i * x[k] for i = 0..d
60 double pk = 1.0;
61 for (int i = 0; i <= d; ++i, pk *= p) {
62 Phi_AB.block(k, col, 1, n) = pk * X.col(k).transpose();
63 col += n;
64 }
65
66 // B-part: p^i * u[k] for i = 0..d
67 pk = 1.0;
68 for (int i = 0; i <= d; ++i, pk *= p) {
69 Phi_AB.block(k, col, 1, m) = pk * U.col(k).transpose();
70 col += m;
71 }
72
73 Z_x.row(k) = X.col(k + 1).transpose();
74 }
75
76 // Solve via column-pivot QR (handles near-rank-deficiency when p is narrow)
77 Eigen::MatrixXd Theta_AB = Phi_AB.colPivHouseholderQr().solve(Z_x);
78
79 // -------------------------------------------------------------------------
80 // Output equation regression
81 // y[k] = C(p[k]) x[k] + D(p[k]) u[k]
82 // Same regressor structure over all N steps.
83 // -------------------------------------------------------------------------
84 const int cols_CD = (n + m) * d1;
85 Eigen::MatrixXd Phi_CD(N, cols_CD);
86 Eigen::MatrixXd Z_y(N, p_out);
87
88 for (int k = 0; k < N; ++k) {
89 const double p = sched[k];
90 int col = 0;
91
92 double pk = 1.0;
93 for (int i = 0; i <= d; ++i, pk *= p) {
94 Phi_CD.block(k, col, 1, n) = pk * X.col(k).transpose();
95 col += n;
96 }
97
98 pk = 1.0;
99 for (int i = 0; i <= d; ++i, pk *= p) {
100 Phi_CD.block(k, col, 1, m) = pk * U.col(k).transpose();
101 col += m;
102 }
103
104 Z_y.row(k) = Y.col(k).transpose();
105 }
106
107 Eigen::MatrixXd Theta_CD = Phi_CD.colPivHouseholderQr().solve(Z_y);
108
109 // -------------------------------------------------------------------------
110 // Extract polynomial coefficient matrices from Theta
111 // A_i = Theta_AB[i*n : (i+1)*n, :].transpose() (shape n*n)
112 // B_i = Theta_AB[(d+1)*n + i*m : ..., :].transpose() (shape n*m)
113 // C_i = Theta_CD[i*n : (i+1)*n, :].transpose() (shape p_out*n)
114 // D_i = Theta_CD[(d+1)*n + i*m : ..., :].transpose() (shape p_out*m)
115 // -------------------------------------------------------------------------
116 LPVModel model;
117 model.degree = d;
118 model.n_states = n;
119 model.n_inputs = m;
120 model.n_outputs = p_out;
121 model.Ts = Ts;
122 model.A_coeffs.resize(d1);
123 model.B_coeffs.resize(d1);
124 model.C_coeffs.resize(d1);
125 model.D_coeffs.resize(d1);
126
127 for (int i = 0; i <= d; ++i) {
128 model.A_coeffs[i] = Theta_AB.block(i * n, 0, n, n).transpose();
129 model.B_coeffs[i] = Theta_AB.block((d + 1) * n + i * m, 0, m, n).transpose();
130 model.C_coeffs[i] = Theta_CD.block(i * n, 0, n, p_out).transpose();
131 model.D_coeffs[i] = Theta_CD.block((d + 1) * n + i * m, 0, m, p_out).transpose();
132 }
133
134 return model;
135}

137LPVModel identifyLPVFromIO(const Eigen::MatrixXd& U,
138 const Eigen::MatrixXd& Y,
139 const std::vector<double>& sched,
140 int n_states,
141 int degree,
142 double Ts,
143 int n4sid_i)
144{
145 const int N = static_cast<int>(U.cols());
146
147 // Estimate state sequence using n4sid on the entire dataset.
148 // n4sid gives a linear model valid near the average operating point;
149 // we use it only to obtain state estimates for the LPV regression.
150 SubspaceIDResult id = n4sid(Y, U, n_states, n4sid_i, Ts);
151
152 if (!id.model.has_value())
153 throw std::runtime_error("identifyLPVFromIO: n4sid failed to identify a model.");
154
155 const ctrl::StateSpace& lti = id.model.value();
156
157 // Propagate state sequence forward under the identified LTI model.
158 Eigen::MatrixXd X(n_states, N);
159 X.col(0).setZero();
160 for (int k = 0; k < N - 1; ++k)
161 X.col(k + 1) = lti.A * X.col(k) + lti.B * U.col(k);
162
163 return identifyLPV(X, U, Y, sched, degree, Ts);
164}
