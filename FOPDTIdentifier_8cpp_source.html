14FOPDTIdentifier::FOPDTIdentifier(const std::vector<double> &times,
15 const std::vector<double> &outputs,
16 double stepMagnitude,
17 double stepTime)
18 : step_(stepMagnitude)
19{
20 if (times.size() != outputs.size())
21 throw std::invalid_argument("FOPDTIdentifier: times and outputs must be the same length.");
22 if (times.size() < 4)
23 throw std::invalid_argument("FOPDTIdentifier: need at least 4 data points.");
24 if (std::abs(stepMagnitude) < 1e-14)
25 throw std::invalid_argument("FOPDTIdentifier: stepMagnitude must be non-zero.");
26
27 // Find first index at or after stepTime
28 std::size_t start = 0;
29 while (start < times.size() && times[start] < stepTime - 1e-12)
30 ++start;
31 if (start == times.size())
32 throw std::invalid_argument("FOPDTIdentifier: stepTime is beyond the data range.");
33
34 // Initial output level = value just before step (or at start if no pre-step data)
35 y0_ = (start > 0) ? outputs[start - 1] : outputs[start];
36
37 // Build time-shifted, zero-referenced data
38 t_.reserve(times.size() - start);
39 y_.reserve(times.size() - start);
40 for (std::size_t i = start; i < times.size(); ++i) {
41 t_.push_back(times[i] - stepTime);
42 y_.push_back(outputs[i] - y0_);
43 }
44
45 if (t_.size() < 3)
46 throw std::invalid_argument(
47 "FOPDTIdentifier: fewer than 3 data points remain after stepTime.");
48
49 Ts_data_ = (t_.back() - t_.front()) / static_cast<double>(t_.size() - 1);
50}

76
77PIDParams FOPDTIdentifier::imcTuning(const FOPDTModel &m, double lambdaC,
78 double Ts, bool piOnly)
79{
80 if (lambdaC <= 0.0)
81 throw std::invalid_argument("FOPDTIdentifier::imcTuning: lambdaC must be > 0.");
82
83 const double tau = std::max(m.tau, 1e-9);
84 const double th = std::max(m.theta, 0.0);
85 const double lam = lambdaC;
86
87 // IMC-PID ideal parameters (parallel form)
88 const double Ti = tau + 0.5 * th;
89 const double Td = piOnly ? 0.0 : (tau * th) / (2.0 * tau + th);
90 const double Kp = Ti / (m.K * (lam + 0.5 * th));
91
92 PIDParams pp;
93 pp.Kp = Kp;
94 pp.Ki = (Ti > 1e-12) ? Kp / Ti : 0.0;
95 pp.Kd = Kp * Td;
96 pp.N = 10.0; // derivative filter coefficient (10 = moderate filtering)
97 pp.uMin = -std::numeric_limits<double>::infinity();
98 pp.uMax = std::numeric_limits<double>::infinity();
99 return pp;
100}

103// Private: graphical method (Ziegler-Nichols tangent + 63.2% intercept)
104// =============================================================================
105
106FOPDTModel FOPDTIdentifier::identifyGraphical() const
107{
108 const int N = static_cast<int>(y_.size());
109 const double y_final = y_.back();
110 const double K = y_final / step_;
111
112 // Find steepest tangent using centred finite differences
113 double max_slope = 0.0;
114 int slope_idx = 1;
115 for (int i = 1; i < N - 1; ++i) {
116 const double dt = t_[i + 1] - t_[i - 1];
117 if (dt < 1e-14) continue;
118 const double slope = (y_[i + 1] - y_[i - 1]) / dt;
119 if (slope > max_slope) { max_slope = slope; slope_idx = i; }
120 }
121
122 // Tangent intercept with y=0: t_theta = t[slope_idx] - y[slope_idx] / slope
123 double theta = 0.0;
124 if (max_slope > 1e-14)
125 theta = std::max(0.0, t_[slope_idx] - y_[slope_idx] / max_slope);
126
127 // Time to 63.2% of final value
128 const double y_632 = 0.632 * y_final;
129 double t_632 = t_.back();
130 for (int i = 0; i < N - 1; ++i) {
131 if (y_[i] <= y_632 && y_[i + 1] >= y_632) {
132 const double frac = (y_632 - y_[i]) / (y_[i + 1] - y_[i] + 1e-30);
133 t_632 = t_[i] + frac * (t_[i + 1] - t_[i]);
134 break;
135 }
136 }
137
138 const double tau = std::max(t_632 - theta, Ts_data_);
139
140 // RMSE
141 double sse = 0.0;
142 for (int i = 0; i < N; ++i) {
143 const double dt = t_[i] - theta;
144 const double y_hat = (dt > 0.0) ? K * step_ * (1.0 - std::exp(-dt / tau)) : 0.0;
145 const double e = y_[i] - y_hat;
146 sse += e * e;
147 }
148 return {K, tau, theta, std::sqrt(sse / N)};
149}

151// =============================================================================
152// Private: optimization via golden-section search on theta, then tau
153// =============================================================================
154
155FOPDTModel FOPDTIdentifier::identifyOptimization() const
156{
157 const int N = static_cast<int>(y_.size());
158 const double y_final = y_.back();
159 const double K = y_final / step_;
160 const double T = t_.back();
161
162 // Search bounds
163 const double theta_lo = 0.0;
164 const double theta_hi = 0.5 * T;
165
166 // For a fixed theta, minimise SSE over tau via golden-section
167 auto best_tau_sse = [&](double theta) -> std::pair<double, double> {
168 const double tau_lo = Ts_data_;
169 const double tau_hi = std::max(T - theta, 2.0 * Ts_data_);
170 const double opt_tau = goldenSection(
171 [&](double tau) { return modelSSE(K, tau, theta); },
172 tau_lo, tau_hi);
173 return {opt_tau, modelSSE(K, opt_tau, theta)};
174 };
175
176 // Outer golden-section on theta
177 const double opt_theta = goldenSection(
178 [&](double theta) { return best_tau_sse(theta).second; },
179 theta_lo, theta_hi);
180
181 auto [opt_tau, sse] = best_tau_sse(opt_theta);
182 const double rmse = std::sqrt(sse / N);
183 return {K, opt_tau, opt_theta, rmse};
184}

202// =============================================================================
203// Private: golden-section minimum search on [a, b]
204// =============================================================================
205
206double FOPDTIdentifier::goldenSection(std::function<double(double)> f,
207 double a, double b, double tol)
208{
209 // Golden ratio conjugate: (sqrt(5)-1)/2
210 constexpr double gr = 0.6180339887498949;
211 double c = b - gr * (b - a);
212 double d = a + gr * (b - a);
213 while (std::abs(b - a) > tol) {
214 if (f(c) < f(d)) { b = d; }
215 else { a = c; }
216 c = b - gr * (b - a);
217 d = a + gr * (b - a);
218 }
219 return 0.5 * (a + b);
220}
