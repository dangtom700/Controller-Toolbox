4// ITimer - abstract interface for a monotonic high-resolution clock.
5//
6// Provides nanosecond-resolution timestamps for deadline checking and
7// execution-time measurement inside the control loop. Platform-specific
8// implementations wrap:
9// - Linux RT: clock_gettime(CLOCK_MONOTONIC)
10// - FreeRTOS: xTaskGetTickCount() * portTICK_PERIOD_MS * 1e6
11// - Zephyr: k_cycle_get_32() / sys_clock_hw_cycles_per_sec()
12// - Windows (test): QueryPerformanceCounter()
13//
14// Usage pattern (overrun detection):
15//
16// void control_loop_tick(ITimer& timer, IController& ctrl, double Ts_ns) {
17// uint64_t t0 = timer.nowNs();
18//
19// double u = ctrl.compute(error);
20// actuator.write(u);
21//
22// uint64_t elapsed = timer.nowNs() - t0;
23// if (elapsed > static_cast<uint64_t>(Ts_ns))
24// on_overrun(elapsed, Ts_ns); // log / alarm / fallback

26//
