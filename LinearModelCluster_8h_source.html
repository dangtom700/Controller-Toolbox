67inline ClusterResult clusterByGap(const Eigen::MatrixXd& gapMatrix,
68 double threshold = 0.5)
69{
70 const int N = static_cast<int>(gapMatrix.rows());
71
72 // ---- Union-Find --------------------------------------------------------
73 std::vector<int> parent(N);
74 std::iota(parent.begin(), parent.end(), 0);
75
76 std::function<int(int)> find = [&](int x) -> int {
77 return (parent[x] == x) ? x : (parent[x] = find(parent[x]));
78 };
79 auto unite = [&](int a, int b) { parent[find(a)] = find(b); };
80
81 // Merge all pairs with gap < threshold (single-linkage)
82 for (int i = 0; i < N; ++i)
83 for (int j = i + 1; j < N; ++j)
84 if (gapMatrix(i, j) < threshold)
85 unite(i, j);
86
87 // ---- Assign contiguous cluster labels -----------------------------------
88 std::map<int, int> rootToLabel;
89 std::vector<int> labels(N);
90 int k = 0;
91 for (int i = 0; i < N; ++i) {
92 int root = find(i);
93 if (rootToLabel.find(root) == rootToLabel.end())
94 rootToLabel[root] = k++;
95 labels[i] = rootToLabel[root];
96 }
97
98 // ---- Select representative: minimise average in-cluster gap -------------
99 std::vector<int> reps(k, -1);
100 std::vector<double> bestScore(k, std::numeric_limits<double>::max());
101
102 for (int i = 0; i < N; ++i) {
103 int li = labels[i];
104 double score = 0.0;
105 int cnt = 0;
106 for (int j = 0; j < N; ++j) {
107 if (labels[j] == li) { score += gapMatrix(i, j); ++cnt; }
108 }
109 if (cnt > 1) score /= static_cast<double>(cnt - 1);
110 if (score < bestScore[li]) { bestScore[li] = score; reps[li] = i; }
111 }
112 // Fallback for singleton clusters
113 for (int c = 0; c < k; ++c) {
114 if (reps[c] == -1) {
115 for (int i = 0; i < N; ++i) {
116 if (labels[i] == c) { reps[c] = i; break; }
117 }
118 }
119 }
120
121 // ---- Max intra-cluster gap ----------------------------------------------
122 std::vector<double> maxGap(k, 0.0);
123 for (int i = 0; i < N; ++i)
124 for (int j = i + 1; j < N; ++j)
125 if (labels[i] == labels[j])
126 maxGap[labels[i]] = std::max(maxGap[labels[i]], gapMatrix(i, j));
127
128 ClusterResult res;
129 res.labels = labels;
130 res.representatives = reps;
131 res.maxIntraGap = maxGap;
132 res.numClusters = k;
133 res.threshold = threshold;
134 return res;
135}

164inline double suggestGapThreshold(const Eigen::MatrixXd& gapMatrix,
165 double lo = 0.1,
166 double hi = 0.9,
167 int steps = 20)
168{
169 if (steps < 3) steps = 3;
170 std::vector<double> thresholds(steps);
171 std::vector<int> counts(steps);
172 for (int i = 0; i < steps; ++i) {
173 thresholds[i] = lo + (hi - lo) * i / (steps - 1);
174 counts[i] = clusterByGap(gapMatrix, thresholds[i]).numClusters;
175 }
176 // Find first plateau: |count[i] - count[i-1]| <= 1 for two consecutive steps
177 for (int i = 2; i < steps; ++i) {
178 if (std::abs(counts[i] - counts[i - 1]) <= 1 &&
179 std::abs(counts[i - 1] - counts[i - 2]) <= 1)
180 return thresholds[i - 1];
181 }
182 return thresholds[steps / 2]; // fallback: midpoint
183}
