12// Log-spaced frequency grid from omega_min to pi/Ts.
13std::vector<double> logFreqGrid(double Ts, int N, double omega_min)
14{
15 const double omega_max = 3.14159265358979323846 / Ts;
16 const double log_min = std::log(omega_min);
17 const double log_max = std::log(omega_max);
18 std::vector<double> grid(N);
19 for (int i = 0; i < N; ++i)
20 grid[i] = std::exp(log_min + (log_max - log_min) * i / (N - 1));
21 return grid;
22}

26std::vector<Eigen::MatrixXcd> freqResponseGrid(const StateSpace& sys,
27 const std::vector<double>& omega_grid)
28{
29 const int n = sys.stateSize();
30 const double Ts = sys.Ts;
31 (void)sys.inputSize(); // referenced via B dimensions
32 (void)sys.outputSize(); // referenced via C dimensions
33
34 Eigen::MatrixXcd Ac = sys.A.cast<std::complex<double>>();
35 Eigen::MatrixXcd Bc = sys.B.cast<std::complex<double>>();
36 Eigen::MatrixXcd Cc = sys.C.cast<std::complex<double>>();
37 Eigen::MatrixXcd Dc = sys.D.cast<std::complex<double>>();
38 Eigen::MatrixXcd I = Eigen::MatrixXcd::Identity(n, n);
39
40 std::vector<Eigen::MatrixXcd> result;
41 result.reserve(omega_grid.size());
42
43 for (double omega : omega_grid) {
44 std::complex<double> z = std::exp(std::complex<double>(0.0, omega * Ts));
45 // (z*I - A) is well-conditioned away from poles; fullPivLu handles near-singular.
46 Eigen::MatrixXcd zIA = z * I - Ac;
47 Eigen::MatrixXcd H = Cc * zIA.fullPivLu().solve(Bc) + Dc;
48 result.push_back(std::move(H));
49 }
50 return result;
51}

53double chordalDist(std::complex<double> p1, std::complex<double> p2)
54{
55 double num = std::abs(p1 - p2);
56 double den = std::sqrt((1.0 + std::norm(p1)) * (1.0 + std::norm(p2)));
57 return (den < 1e-15) ? 0.0 : num / den;
58}

60double subspaceDist(const Eigen::MatrixXcd& H1, const Eigen::MatrixXcd& H2)
61{
62 // Form normalised graphs G_i = [H_i; I_{m}] of shape (p+m) x m, then extract
63 // orthonormal column bases U_i via thin SVD. The subspace chordal distance is:
64 // d = sqrt(1 - sigma_min(U1^H * U2)^2)
65 // This reduces to chordalDist() for SISO (p=m=1) -- see header for proof.
66 const int p = static_cast<int>(H1.rows());
67 const int m = static_cast<int>(H1.cols());
68
69 Eigen::MatrixXcd G1(p + m, m), G2(p + m, m);
70 G1.topRows(p) = H1;
71 G1.bottomRows(m) = Eigen::MatrixXcd::Identity(m, m);
72 G2.topRows(p) = H2;
73 G2.bottomRows(m) = Eigen::MatrixXcd::Identity(m, m);
74
75 // Thin SVD: U_i is (p+m) x m with orthonormal columns.
76 Eigen::JacobiSVD<Eigen::MatrixXcd> svd1(G1, Eigen::ComputeThinU);
77 Eigen::JacobiSVD<Eigen::MatrixXcd> svd2(G2, Eigen::ComputeThinU);
78
79 // Smallest singular value of the m x m inner product matrix U1^H * U2.
80 Eigen::JacobiSVD<Eigen::MatrixXcd> svd_cross(
81 svd1.matrixU().adjoint() * svd2.matrixU());
82 const double sigma_min = svd_cross.singularValues().minCoeff();
83 return std::sqrt(std::max(0.0, 1.0 - sigma_min * sigma_min));
84}

86double nuGap(const StateSpace& P1, const StateSpace& P2,
87 int freq_points, double omega_min)
88{
89 if (P1.inputSize() != P2.inputSize() ||
90 P1.outputSize() != P2.outputSize())
91 throw std::invalid_argument(
92 "nuGap: both systems must have the same input/output dimensions.");
93
94 if (std::abs(P1.Ts - P2.Ts) > 1e-10 * P1.Ts)
95 throw std::invalid_argument(
96 "nuGap: both systems must have the same sample time Ts.");
97
98 if (freq_points < 2) freq_points = 2;
99
100 const bool siso = (P1.inputSize() == 1 && P1.outputSize() == 1);
101 auto grid = logFreqGrid(P1.Ts, freq_points, omega_min);
102 auto H1 = freqResponseGrid(P1, grid);
103 auto H2 = freqResponseGrid(P2, grid);
104
105 double gap = 0.0;
106 for (int i = 0; i < freq_points; ++i) {
107 // SISO: use the exact chordal formula (no SVD overhead).
108 // MIMO: subspace chordal distance via normalised-graph SVD (reduces to
109 // chordal for p=m=1 -- see subspaceDist() derivation in header).
110 const double d = siso
111 ? chordalDist(H1[i](0, 0), H2[i](0, 0))
112 : subspaceDist(H1[i], H2[i]);
113 if (d > gap) gap = d;
114 }
115 return gap;
116}

124Eigen::MatrixXd nuGapMatrix(const std::vector<StateSpace>& models, int freq_points)
125{
126 const int N = static_cast<int>(models.size());
127 Eigen::MatrixXd G = Eigen::MatrixXd::Zero(N, N);
128 for (int i = 0; i < N; ++i) {
129 for (int j = i + 1; j < N; ++j) {
130 double d = nuGap(models[i], models[j], freq_points);
131 G(i, j) = d;
132 G(j, i) = d;
133 }
134 }
135 return G;
