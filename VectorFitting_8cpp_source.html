13 // ============================================================================
14 // initPoles: log-spaced real poles inside the unit disk, mapped from
15 // continuous-time: sigma_k = exp(log(omega_min) + k*step) -> p_k = exp(-sigma_k*Ts)
16 // ============================================================================
17 Eigen::VectorXd VectorFitting::initPoles(int n_poles, double omega_max, double Ts)
18 {
19 if (n_poles <= 0)
20 return Eigen::VectorXd();
21 const double omega_min = 1e-3; // avoid DC pole
22 const double sig_lo = omega_min;
23 const double sig_hi = std::min(omega_max * 0.8, M_PI / Ts * 0.9);
24
25 Eigen::VectorXd poles(n_poles);
26 if (n_poles == 1)
27 {
28 poles(0) = std::exp(-0.5 * (sig_lo + sig_hi) * Ts);
29 return poles;
30 }
31 for (int k = 0; k < n_poles; ++k)
32 {
33 const double t = static_cast<double>(k) / (n_poles - 1);
34 // log-spaced sigma
35 const double sigma = std::exp(std::log(sig_lo) * (1.0 - t) + std::log(sig_hi) * t);
36 poles(k) = std::exp(-sigma * Ts);
37 }
38 return poles;
39 }

42 // enforcePoleStability: reflect any pole |p| >= 1 back to 1/p, and ensure p > 0.
43 // ============================================================================
44 void VectorFitting::enforcePoleStability(Eigen::VectorXd &poles)
45 {
46 for (int k = 0; k < poles.size(); ++k)
47 {
48 double p = poles(k);
49 if (std::abs(p) >= 1.0)
50 p = 1.0 / p; // reflect inside unit disk
51 // keep poles real and positive (negative poles cause aliasing issues)
52 if (p <= 0.0)
53 p = 1e-6;
54 poles(k) = p;
55 }
56 }

58 // ============================================================================
59 // buildSKSystem - construct the Sanathanan-Koerner divided LS system.
60 //
61 // Model: H(z) = N(z)/D(z) where D is monic, n_poles-th order in z^{-1}.
62 //
63 // At iteration k: divide by D^{(k)}(z) to linearise:
64 // D^{(k)} * H approx = N -> Sigma_j c_j * phi_j(z) = (D^{(k)}(z) * H(z)) / D^{(k)}(z)
65 //
66 // Basis functions phi_j(z) = z^{-j} / D^{(k)}(z) for j=0..n_poles (numerator)
67 // and psi_j(z) = -H(z) * z^{-j} / D^{(k)}(z) for j=1..n_poles (denominator)
68 //
69 // Because H and phi/psi are all real at real frequencies, we work with real LS.
70 // ============================================================================
71 void VectorFitting::buildSKSystem(
72 const std::vector<std::complex<double>> &z_grid,
73 const std::vector<double> &mag,
74 const Eigen::VectorXd &poles,
75 int n_poles,
76 Eigen::MatrixXd &A_out,
77 Eigen::VectorXd &b_out)
78 {
79 const int N = static_cast<int>(z_grid.size());
80 const int nc = n_poles + 1; // numerator coefficients c_0..c_{n_poles}
81 const int nad = n_poles; // denominator adjustments a_1..a_{n_poles}
82 const int ncols = nc + nad;
83
84 A_out.resize(N, ncols);
85 b_out.resize(N);
86
87 for (int i = 0; i < N; ++i)
88 {
89 const std::complex<double> z = z_grid[i];
90 const double H_i = mag[i];
91
92 // Evaluate D^{(k)}(z) = prod(z - p_k) / z^{n_poles} (monic in z-form)
93 // For the LS problem, we use partial-fractions: 1/D^{(k)} evaluates as
94 // the polynomial Pi(z - p_k) in the denominator.
95 // Simpler: denominator in z^{-1} form: D(z^{-1}) = 1 + Sigma_k a_k * z^{-k}
96 // We compute D^{(k)}(z_i) explicitly from current poles:
97 // D(z) = prod_{k=1..n} (z - p_k) / z^n
98 // On unit circle z=e^{jomega}: this is the product of (z-p_k).
99
100 std::complex<double> Dk(1.0, 0.0);
101 for (int k = 0; k < n_poles; ++k)
102 {
103 // pole p_k is real; cast to complex for multiplication
104 Dk *= (z - std::complex<double>(poles(k), 0.0));
105 }
106 // normalise: Dk is z^{n_poles} * monic-D(z^{-1}) evaluated at z
107 const double Dk_abs2 = std::norm(Dk);
108 if (Dk_abs2 < 1e-30)
109 {
110 A_out.row(i).setZero();
111 b_out(i) = 0.0;
112 continue;
113 }
114
115 // Numerator columns: z^{j} / Dk for j = 0..n_poles (in z basis)
116 // In practice: col j corresponds to z^{j} / prod(z - p_k).
117 // We take the real part of z^j / Dk, weighted by |Dk|^{-1}.
118 std::complex<double> z_pow(1.0, 0.0); // z^0
119 for (int j = 0; j < nc; ++j)
120 {
121 // real part of (z^j / Dk)
122 const std::complex<double> basis = z_pow / Dk;
123 A_out(i, j) = basis.real();
124 z_pow *= z;
125 }
126
127 // Denominator columns: -H(z) * z^{j+1} / Dk for j=0..n_poles-1
128 std::complex<double> z_pow2 = z; // z^1
129 for (int j = 0; j < nad; ++j)
130 {
131 const std::complex<double> basis = -std::complex<double>(H_i, 0.0) * z_pow2 / Dk;
132 A_out(i, nc + j) = basis.real();
133 z_pow2 *= z;
134 }
135
136 b_out(i) = (std::complex<double>(H_i, 0.0) / Dk).real();
137 }
138 }

140 // ============================================================================
141 // buildStateSpace - companion form from real poles and residues.
142 // H(z) = Sigma_k r_k / (z - p_k) + d_gain
143 // = C (zI - A)^{-1} B + d_gain
144 // Diagonal A = diag(p), B = ones, C = r'.
145 // ============================================================================
146 StateSpace VectorFitting::buildStateSpace(
147 const Eigen::VectorXd &poles,
148 const Eigen::VectorXd &residues,
149 double d_gain,
150 double Ts)
151 {
152 const int n = poles.size();
153 Eigen::MatrixXd A = poles.asDiagonal(); // n*n diagonal
154 Eigen::MatrixXd B = Eigen::MatrixXd::Ones(n, 1);
155 Eigen::MatrixXd C = residues.transpose(); // 1*n
156 Eigen::MatrixXd D(1, 1);
157 D(0, 0) = d_gain;
158 return StateSpace(A, B, C, D, Ts);
159 }

162 // evalMagnitude
163 // ============================================================================
164 double VectorFitting::evalMagnitude(const StateSpace &sys, double omega)
165 {
166 const double Ts = sys.Ts;
167 const std::complex<double> z(std::cos(omega * Ts), std::sin(omega * Ts));
168 const int n = sys.A.rows();
169 // const int ny = sys.C.rows();
170 // const int nu = sys.D.cols();
171
172 Eigen::MatrixXcd I = Eigen::MatrixXcd::Identity(n, n);
173 Eigen::MatrixXcd zI = z * I;
174
175 Eigen::FullPivLU<Eigen::MatrixXcd> lu(zI - sys.A.cast<std::complex<double>>());
176 if (!lu.isInvertible())
177 return 0.0;
178
179 const Eigen::MatrixXcd H = sys.C.cast<std::complex<double>>() *
180 lu.solve(sys.B.cast<std::complex<double>>()) +
181 sys.D.cast<std::complex<double>>();
182 return H.norm(); // Frobenius norm = |scalar| for SISO
183 }

185 // ============================================================================
186 // fitMagnitude - main entry point
187 // ============================================================================
188 StateSpace VectorFitting::fitMagnitude(
189 const std::vector<double> &omega_grid,
190 const std::vector<double> &magnitude,
191 int n_poles,
192 double Ts,
193 VectorFittingResult &result,
194 const VectorFittingParams &params)
195 {
196 const int N = static_cast<int>(omega_grid.size());
197 if (N == 0 || magnitude.size() != static_cast<size_t>(N))
198 throw std::invalid_argument("VectorFitting: omega_grid and magnitude must be non-empty and same size.");
199 if (n_poles < 1)
200 throw std::invalid_argument("VectorFitting: n_poles must be >= 1.");
201
202 // Build z-grid on the unit circle
203 std::vector<std::complex<double>> z_grid(N);
204 for (int i = 0; i < N; ++i)
205 z_grid[i] = std::complex<double>(std::cos(omega_grid[i] * Ts),
206 std::sin(omega_grid[i] * Ts));
207
208 // Sanitise magnitudes: replace zeros with floor
209 std::vector<double> mag(N);
210 for (int i = 0; i < N; ++i)
211 mag[i] = std::max(std::abs(magnitude[i]), params.eps_mag);
212
213 // Initialise poles
214 const double omega_max = omega_grid.back();
215 Eigen::VectorXd poles = initPoles(n_poles, omega_max, Ts);
216
217 result.converged = false;
218 result.iterations = 0;
219
220 Eigen::VectorXd residues = Eigen::VectorXd::Ones(n_poles) * 0.1;
221 double d_gain = mag[N / 2]; // rough DC initialisation
222
223 for (int iter = 0; iter < params.max_iter; ++iter)
224 {
225 Eigen::MatrixXd A;
226 Eigen::VectorXd b;
227 buildSKSystem(z_grid, mag, poles, n_poles, A, b);
228
229 // Solve LS: A * x = b
230 Eigen::VectorXd x = A.colPivHouseholderQr().solve(b);
231
232 // Extract numerator coefficients c_0..c_{n_poles} and denominator adjustment a_1..a_{n_poles}
233 const int nc = n_poles + 1;
234 const Eigen::VectorXd c_coeff = x.head(nc); // numerator
235 const Eigen::VectorXd a_coeff = x.tail(n_poles); // denominator a_1..a_n
236
237 // Build denominator polynomial D(z) = z^n + a_1 z^{n-1} + ... + a_n
238 // Companion matrix for root finding
239 Eigen::MatrixXd companion = Eigen::MatrixXd::Zero(n_poles, n_poles);
240 for (int k = 0; k < n_poles - 1; ++k)
241 companion(k + 1, k) = 1.0;
242 for (int k = 0; k < n_poles; ++k)
243 companion(0, k) = -a_coeff(n_poles - 1 - k);
244
245 Eigen::EigenSolver<Eigen::MatrixXd> es(companion, false);
246 Eigen::VectorXd new_poles(n_poles);
247 const auto &evals = es.eigenvalues();
248 for (int k = 0; k < n_poles; ++k)
249 {
250 // Take real part; imaginary part should be ~0 for real-valued signals
251 new_poles(k) = evals(k).real();
252 }
253 enforcePoleStability(new_poles);
254
255 // Convergence check
256 const double max_disp = (new_poles - poles).cwiseAbs().maxCoeff();
257 poles = new_poles;
258 ++result.iterations;
259
260 // After pole update, solve for residues with fixed poles (one LS)
261 // H(z) = Sigma_k r_k / (z - p_k) + d
262 // Build a new LS for residues only
263 Eigen::MatrixXd Ar(N, n_poles + 1);
264 Eigen::VectorXd br(N);
265 for (int i = 0; i < N; ++i)
266 {
267 const std::complex<double> z = z_grid[i];
268 for (int k = 0; k < n_poles; ++k)
269 {
270 const std::complex<double> term = 1.0 / (z - std::complex<double>(poles(k), 0.0));
271 Ar(i, k) = term.real();
272 }
273 Ar(i, n_poles) = 1.0; // constant term d
274 br(i) = mag[i];
275 }
276 Eigen::VectorXd rd = Ar.colPivHouseholderQr().solve(br);
277 residues = rd.head(n_poles);
278 d_gain = rd(n_poles);
279
280 if (max_disp < params.tol)
281 {
282 result.converged = true;
283 break;
284 }
285 }
286
287 // Compute RMS relative error
288 {
289 const StateSpace sys = buildStateSpace(poles, residues, d_gain, Ts);
290 double rms = 0.0;
291 for (int i = 0; i < N; ++i)
292 {
293 const double fit = evalMagnitude(sys, omega_grid[i]);
294 const double rel = (fit - mag[i]) / mag[i];
295 rms += rel * rel;
296 }
297 result.rms_error = std::sqrt(rms / N);
298 }
299
300 return buildStateSpace(poles, residues, d_gain, Ts);
301 }
