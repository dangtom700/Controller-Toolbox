4// Discrete First-Order Sliding Mode Controller (SMC).
5//
6// Sliding surface (SISO, error-based, 1st order):
7// s[k] = c_e . e[k] + c_de . (e[k] - e[k-1])
8// = proportional error + rate-of-error term
9//
10// Control law (boundary-layer saturation to reduce chattering):
11// u[k] = -K . sat(s[k] / phi)
12// sat(x) = x if |x| <= 1 (linear PD inside boundary layer)
13// sat(x) = sign(x) if |x| > 1 (relay switching outside)
14//
15// Inside boundary layer (|s| <= phi): equivalent to a PD controller.
16// Outside (|s| > phi): full switching control - robust to matched disturbances.
17//
18// Setting phi -> 0 recovers ideal SMC with chattering.
19// Setting phi large gives a soft PD approximation.
20//
21// Ref: Utkin "Sliding Modes in Control and Optimization" (1992);
22// Edwards & Spurgeon "Sliding Mode Control: Theory and Applications" (1998);
23// Simulink Variable-Structure Control block.
24namespace ctrl {
25
26struct SMCParams {
27 double c_e = 1.0; // Error weight in sliding surface
28 // Error-rate weight in sliding surface.
29 //
30 // IMPORTANT - sample-time dependence:
31 // In the discrete implementation de/dt is approximated as (e[k] - e[k-1]) / Ts.
32 // c_de is therefore implicitly Ts-dependent: if you tune c_de at Ts=0.01s and then
33 // halve the sample rate to Ts=0.005s, the sliding surface sensitivity doubles without
34 // any parameter change. This is a calibration trap.
35 //
36 // Convention: c_de stores the DISCRETE coefficient (i.e., c_de already includes Ts).
37 // To convert from a continuous-time slope lambda [1/s] (used in control theory):
38 // c_de = lambda * Ts
39 // If you change Ts, recalculate c_de accordingly.
40 //
41 // Larger c_de = faster convergence to sliding surface, more chattering.
42 double c_de = 0.1;
43 double K = 5.0; // Switching gain (larger = more robust, more chattering)
44 double phi = 0.5; // Boundary layer thickness (larger = smoother, slower)
45 double uMin = -1e9; // Output saturation lower limit
46 double uMax = 1e9; // Output saturation upper limit
47};

74// SuperTwistingSMC - Discrete 2nd-order sliding mode (super-twisting algorithm).
75//
76// Eliminates chattering without a boundary layer while retaining finite-time
77// convergence and robustness to matched disturbances.
78//
79// Continuous-time super-twisting law (Levant 1993):
80// u[k] = v[k] - K1 . |s|^{1/2} . sign(s)
81// v.[k] = -K2 . sign(s)
82//
83// Discrete approximation (Euler integration of v):
84// s[k] = c_e . e[k] + c_de . (e[k] - e[k-1])
85// u[k] = v[k] - K1 . |s[k]|^{1/2} . sign(s[k])
86// v[k+1]= v[k] - K2 . Ts . sign(s[k])
87//
88// Gain selection guidance (Moreno & Osorio 2008):
89// K1 > 0, K2 > K1^2 / 4 (sufficient for finite-time convergence in continuous time).
90// In discrete time keep K2 * Ts small ( << K1 ) to limit discretisation error.
91//
92// Ref: Levant "Sliding order and sliding accuracy in SMC" IJRNLC (1993);
93// Moreno & Osorio "A Lyapunov approach to super-twisting" IEEE TAC (2012).
94// ---------------------------------------------------------------------------
95
96struct SuperTwistingParams
97{
98 double c_e = 1.0; // Error weight in sliding surface (same as SMCParams)
99 double c_de = 0.1; // Error-rate weight (discrete; stores lambda*Ts; see SMCParams c_de note)
100 double K1 = 3.0; // Power term gain ( |s|^{1/2} )
101 double K2 = 5.0; // Integral gain ( sign(s) )
102 double uMin = -1e9; // Output saturation lower limit
103 double uMax = 1e9; // Output saturation upper limit
104};
