20 double RelayAutoTuner::step(double y)
21 {
22 step_cnt_++;
23
24 // Relay logic with hysteresis
25 double relay_out;
26 if (relayHigh_ && y > cfg_.hysteresis)
27 {
28 relayHigh_ = false;
29 relay_out = -cfg_.relayAmplitude;
30 collecting_ = true; // ignore transient before first switch
31 }
32 else if (!relayHigh_ && y < -cfg_.hysteresis)
33 {
34 relayHigh_ = true;
35 relay_out = cfg_.relayAmplitude;
36 collecting_ = true;
37 }
38 else
39 {
40 relay_out = relayHigh_ ? cfg_.relayAmplitude : -cfg_.relayAmplitude;
41 }
42
43 // Detect zero crossings after transient (sign change detection)
44 if (collecting_ && (y * y_prev_ < 0.0))
45 crossTimes_.push_back(step_cnt_ * Ts_);
46
47 // Track extrema ONLY after transient; do NOT reset between crossings.
48 // We accumulate global peak/trough over all stable cycles.
49 if (collecting_)
50 {
51 if (y > peak_pos_)
52 peak_pos_ = y;
53 if (y < peak_neg_)
54 peak_neg_ = y;
55 }
56
57 y_prev_ = y;
58
59 // Need 2.cyclesRequired crossings for stable period estimate
60 const int needed = 2 * cfg_.cyclesRequired;
61 if (!done_ && static_cast<int>(crossTimes_.size()) >= needed)
62 {
63 // Average period over last cyclesRequired full cycles
64 const int n = static_cast<int>(crossTimes_.size());
65 Tu_ = 2.0 * (crossTimes_[n - 1] - crossTimes_[n - 1 - cfg_.cyclesRequired]) / cfg_.cyclesRequired;
66
67 // Ku from accumulated peak-to-peak amplitude across all stable cycles
68 const double a_y = (peak_pos_ - peak_neg_) / 2.0;
69 if (a_y > 1e-12)
70 Ku_ = (4.0 * cfg_.relayAmplitude) / (M_PI * a_y);
71
72 done_ = true;
73 }
74
75 return relay_out;
76 }

78 PIDParams RelayAutoTuner::computePIDParams(PIDTuningRule rule, double lambda) const
79 {
80 if (!done_)
81 throw std::runtime_error("RelayAutoTuner: call isDone() first.");
82
83 PIDParams p;
84 switch (rule)
85 {
86 case PIDTuningRule::ZieglerNichols:
87 p.Kp = 0.60 * Ku_;
88 p.Ki = p.Kp * 2.0 / Tu_;
89 p.Kd = p.Kp * Tu_ / 8.0;
90 break;
91
92 case PIDTuningRule::TyreusLuyben:
93 p.Kp = Ku_ / 2.2;
94 p.Ki = p.Kp / (2.2 * Tu_);
95 p.Kd = p.Kp * Tu_ / 6.3;
96 break;
97
98 case PIDTuningRule::AMIGO:
99 // Simplified AMIGO (Astrom 2006) from relay test data alone
100 p.Kp = 0.40 * Ku_;
101 p.Ki = 0.80 * Ku_ / Tu_;
102 p.Kd = 0.10 * p.Kp * Tu_;
103 break;
104
105 case PIDTuningRule::IMC:
106 // IMC from relay test: approximate tau approx = Tu/(2pi), theta approx = Ts
107 {
108 const double tau_est = Tu_ / (2.0 * M_PI);
109 const double lam = (lambda > 0) ? lambda : tau_est;
110 p.Kp = (2.0 * tau_est) / (Ku_ * lam);
111 p.Ki = p.Kp / tau_est;
112 p.Kd = 0.0;
113 }
114 break;
115 }
116
117 p.N = 10.0 / (Tu_ / (2.0 * M_PI)); // derivative filter: cutoff ~ 10* nominal BW
118 p.Kb = std::sqrt(p.Ki * p.Kd + 1e-12);
119 return p;
120 }

126 StepResponseTuner::FOPDTModel
127 StepResponseTuner::identify(const std::vector<double> &time,
128 const std::vector<double> &output,
129 double stepMagnitude)
130 {
131 if (time.size() < 4 || time.size() != output.size())
132 throw std::invalid_argument("StepResponseTuner: need at least 4 matched samples.");
133
134 const double y_inf = output.back();
135 const double K = y_inf / stepMagnitude;
136
137 // Find indices where output crosses 28.3 % and 63.2 % of final value (Smith 1972 tangent)
138 const double y283 = 0.283 * y_inf;
139 const double y632 = 0.632 * y_inf;
140
141 double t283 = 0.0, t632 = 0.0;
142 bool got283 = false, got632 = false;
143
144 for (size_t i = 1; i < output.size(); ++i)
145 {
146 if (!got283 && output[i] >= y283)
147 {
148 t283 = time[i];
149 got283 = true;
150 }
151 if (!got632 && output[i] >= y632)
152 {
153 t632 = time[i];
154 got632 = true;
155 }
156 if (got283 && got632)
157 break;
158 }
159
160 if (!got283 || !got632)
161 {
162 // Integrating (type-1) plants have no finite steady state: the step
163 // response ramps indefinitely and never crosses the y_inf thresholds
164 // computed from output.back(). If the 63.2 % crossing is absent, the
165 // plant is likely integrating or the data record is too short.
166 // For integrating plants use a velocity-form PID or fit a DIPDT model
167 // (integrating plus dead time) rather than FOPDT.
168 throw std::runtime_error(
169 "StepResponseTuner: output did not reach 63.2 % of steady state. "
170 "If the step response is still rising at the end of the data record, "
171 "the plant may be integrating (type-1). FOPDT identification requires "
172 "a finite steady-state value. For integrating plants, use a DIPDT "
173 "model or a velocity-form (incremental) PID structure.");
174 }
175
176 // Process-reaction-curve method (Astrom & Hagglund):
177 // tau = 1.5.(t63.2 - t28.3), theta = t63.2 - tau
178 const double tau = 1.5 * (t632 - t283);
179 const double theta = std::max(0.0, t632 - tau);
180
181 return {K, tau, theta};
182 }

184 PIDParams StepResponseTuner::computePIDParams(const FOPDTModel &m,
185 double /*Ts*/,
186 PIDTuningRule rule,
187 double lambda)
188 {
189 PIDParams p;
190 const double lam = (lambda > 0) ? lambda : 0.5 * m.tau;
191
192 switch (rule)
193 {
194 case PIDTuningRule::ZieglerNichols:
195 p.Kp = 1.2 * m.tau / (m.K * m.theta);
196 p.Ki = p.Kp / (2.0 * m.theta);
197 p.Kd = p.Kp * 0.5 * m.theta;
198 break;
199
200 case PIDTuningRule::TyreusLuyben:
201 p.Kp = 0.45 * m.tau / (m.K * m.theta);
202 p.Ki = p.Kp / (2.63 * m.theta);
203 p.Kd = 0.0;
204 break;
205
206 case PIDTuningRule::IMC:
207 // Rivera, Morari & Skogestad (1986) IMC-PID
208 p.Kp = (2.0 * m.tau + m.theta) / (2.0 * m.K * (lam + m.theta));
209 p.Ki = p.Kp / (m.tau + m.theta / 2.0);
210 p.Kd = p.Kp * m.tau * m.theta / (2.0 * m.tau + m.theta);
211 break;
212
213 case PIDTuningRule::AMIGO:
214 // AMIGO (Astrom & Hagglund 2006) - optimised for robustness
215 {
216 const double ratio = m.theta / m.tau;
217 if (ratio < 0.1 || ratio > 10.0) {
218 std::clog << "[StepResponseTuner] WARNING: AMIGO tuning used outside valid FOPDT ratio range (0.1 < theta/tau < 10.0).\n";
219 }
220 p.Kp = (1.0 / m.K) * (0.2 + 0.45 * m.tau / m.theta);
221 p.Ki = p.Kp * (0.4 * m.theta + 0.8 * m.tau) /
222 (m.theta + 0.1 * m.tau) / (m.tau + 0.8 * m.theta);
223 p.Kd = p.Kp * 0.5 * m.tau * m.theta / (0.3 * m.tau + m.theta);
224 (void)ratio;
225 }
226 break;
227 }
228
229 p.N = 10.0 / m.theta; // derivative filter ~10* dead-time frequency
230 p.Kb = std::sqrt(std::abs(p.Ki * p.Kd) + 1e-12);
231 return p;
232 }

247 LQRParams LQRWeightTuner::polePlacementHint(
248 const StateSpace &plant,
249 const std::vector<std::complex<double>> &desiredPoles,
250 int maxIter)
251 {
252 // Start from Bryson with unit bounds, then iterate Q scaling to steer poles.
253 const int n = plant.stateSize();
254 const int m = plant.inputSize();
255
256 Eigen::VectorXd sx = Eigen::VectorXd::Ones(n);
257 Eigen::VectorXd su = Eigen::VectorXd::Ones(m);
258 LQRParams lp = brysonMethod(sx, su);
259
260 for (int iter = 0; iter < maxIter; ++iter)
261 {
262 DiscreteLQR lqr(plant, lp);
263 Eigen::MatrixXd Acl = plant.A - plant.B * lqr.gainMatrix();
264 Eigen::VectorXcd ev = Acl.eigenvalues();
265
266 // Compute RMS pole-location residual
267 double err = 0.0;
268 for (int i = 0; i < n; ++i)
269 {
270 double best = 1e9;
271 for (auto &dp : desiredPoles)
272 best = std::min(best, std::abs(ev(i) - dp));
273 err += best * best;
274 }
275 err = std::sqrt(err / n);
276 if (err < 1e-4)
277 break;
278
279 // Scale Q by gradient heuristic: increase Q if poles are too slow
280 for (int i = 0; i < n; ++i)
281 {
282 if (std::abs(ev(i)) > std::abs(desiredPoles[i % desiredPoles.size()]))
283 lp.Q(i, i) *= 1.1;
284 else
285 lp.Q(i, i) *= 0.95;
286 }
287 }
288 return lp;
289 }

295 double MPCHorizonTuner::estimateSettlingTime(const StateSpace &plant, int maxSteps)
296 {
297 const int m = plant.inputSize();
298
299 Eigen::VectorXd x = Eigen::VectorXd::Zero(plant.stateSize());
300 Eigen::VectorXd u = Eigen::VectorXd::Ones(m);
301 Eigen::VectorXd y_ss;
302
303 // Run to approximate steady state
304 for (int k = 0; k < maxSteps; ++k)
305 y_ss = plant.C * x + plant.D * u, x = plant.A * x + plant.B * u;
306
307 // Restart and find 99 % settling
308 x.setZero();
309 const double y_final = y_ss(0);
310 const double thr = 0.99 * y_final;
311
312 for (int k = 0; k < maxSteps; ++k)
313 {
314 Eigen::VectorXd y = plant.C * x + plant.D * u;
315 x = plant.A * x + plant.B * u;
316 if (std::abs(y(0)) >= std::abs(thr))
317 return k * plant.Ts;
318 }
319 return maxSteps * plant.Ts;
320 }

353 PIDParams CohenCoonTuner::tuneImpl(const StepResponseTuner::FOPDTModel &m, double /*Ts*/)
354 {
355 if (m.theta < 1e-9)
356 throw std::invalid_argument(
357 "CohenCoonTuner: dead time theta must be > 0."
358 " Cohen-Coon requires finite dead time.");
359
360 const double r = m.theta / m.tau;
361 const double Kp = (m.tau / (m.K * m.theta)) * (4.0 / 3.0 + r / 4.0);
362 const double Ti = m.theta * (32.0 + 6.0 * r) / (13.0 + 8.0 * r);
363 const double Td = 4.0 * m.theta / (11.0 + 2.0 * r);
364
365 PIDParams p;
366 p.Kp = Kp;
367 p.Ki = Kp / Ti;
368 p.Kd = Kp * Td;
369 p.N = 10.0 / m.theta;
370 p.Kb = std::sqrt(std::abs(p.Ki * p.Kd) + 1e-12);
371 return p;
372 }

377
378 LeadLagParams LoopShapingTuner::tuneImpl(const Input &in)
379 {
380 const double phase_rad = in.phase_add_deg * M_PI / 180.0;
381
382 if (phase_rad <= 0.0 || phase_rad >= M_PI / 2.0)
383 {
384 std::clog << "[LoopShapingTuner] Warning: phase_add_deg must be in (0, 90)."
385 " Returning a unity-gain compensator at the requested crossover.\n";
386 const double K = (in.gain_at_wc > 1e-12) ? 1.0 / in.gain_at_wc : 1.0;
387 return {in.omega_c * 0.1, in.omega_c * 10.0, K};
388 }
389
390 const double beta = std::sin(phase_rad);
391 const double alpha = (1.0 + beta) / (1.0 - beta); // p / z
392 const double sqrtA = std::sqrt(alpha);
393
394 LeadLagParams lp;
395 lp.continuousZero = in.omega_c / sqrtA; // z = omega_c / sqrt(alpha)
396 lp.continuousPole = in.omega_c * sqrtA; // p = omega_c * sqrt(alpha)
397 lp.gain = (in.gain_at_wc > 1e-12)
398 ? sqrtA / in.gain_at_wc // K = sqrt(alpha) / |G(jomega_c)|
399 : sqrtA;
400 return lp;
401 }
