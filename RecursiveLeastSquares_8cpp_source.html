17 double RecursiveLeastSquares::update(double y, double u)
18 {
19 // Build regressor phi[k] = [-y[k-1],...,-y[k-na], u[k-1],...,u[k-nb]]
20 // using the current (pre-update) buffers. Negative sign on y terms because
21 // the ARX model is y[k] = -a1*y[k-1] - ... + b1*u[k-1] + ..., so theta = [a1,...,b1,...].
22 Eigen::VectorXd phi(ntheta_);
23 for (int i = 0; i < na_; ++i) phi(i) = -y_buf_(i);
24 for (int i = 0; i < nb_; ++i) phi(na_ + i) = u_buf_(i);
25
26 // A-posteriori prediction error: e = y - phi'*theta_{k-1}
27 const double e = y - phi.dot(theta_);
28
29 // RLS Kalman gain: K = P*phi / (lambda + phi'*P*phi)
30 // Denominator is always > 0 (lambda > 0, phi'*P*phi >= 0, P is PSD).
31 const Eigen::VectorXd Pphi = P_ * phi;
32 const double denom = lambda_ + phi.dot(Pphi);
33 const Eigen::VectorXd K = Pphi / denom;
34
35 // Parameter update: theta = theta + K*e (standard gradient-descent step)
36 theta_ += K * e;
37
38 // Covariance update: P = (P - K*phi'*P) / lambda
39 //
40 // This is equivalent to (I - K*phi')*P / lambda (the standard Kalman form), since
41 // K*phi'*P = (P*phi/denom)*phi'*P, which equals the result of substituting
42 // K = P*phi/denom into (I - K*phi')*P. The factored form avoids forming (I - K*phi')
43 // explicitly (one fewer n*n multiply for large theta_).
44 //
45 // The /lambda term inflates P at each step - equivalent to inflating Q in a Kalman
46 // filter - which keeps P from shrinking to zero and freezing theta_. This gives more
47 // weight to recent data; the effective sliding window length is ~ 1/(1-lambda).
48 // For lambda=1 (no forgetting), P shrinks monotonically and theta_ stops adapting to
49 // time-varying parameters.
50 P_ = (P_ - K * Pphi.transpose()) / lambda_;
51
52 // Symmetrise to suppress numerical drift
53 P_ = 0.5 * (P_ + P_.transpose());
54
55 // Trace capping: when lambda < 1 (forgetting active), scalar exponential
56 // forgetting inflates P isotropically - including unexcited directions -
57 // causing parameter drift. Capping trace(P) at P0_scale * ntheta prevents
58 // runaway covariance growth without introducing directional bias.
59 // Equivalent to "variable-trace" forgetting from Fortescue et al. (1981).
60 if (lambda_ < 1.0)
61 {
62 const double tr = P_.trace();
63 const double tr_max = P0_scale_ * static_cast<double>(ntheta_);
64 if (tr > tr_max)
65 P_ *= tr_max / tr;
66 }
67
68 // Shift buffers: push current y,u to front (index 0 = most recent lag)
69 for (int i = na_ - 1; i > 0; --i) y_buf_(i) = y_buf_(i - 1);
70 if (na_ > 0) y_buf_(0) = y;
71 for (int i = nb_ - 1; i > 0; --i) u_buf_(i) = u_buf_(i - 1);
72 if (nb_ > 0) u_buf_(0) = u;
73
74 ++k_;
75 return e;
76 }

102 TransferFunction RecursiveLeastSquares::toTransferFunction() const
103 {
104 // ARX models require nb <= na so the transfer function B(z)/A(z) is proper.
105 // If nb > na the extra B coefficients cannot be represented in the returned
106 // TransferFunction without adding improper (non-causal) terms.
107 if (nb_ > na_)
108 throw std::logic_error(
109 "RecursiveLeastSquares::toTransferFunction: nb (" + std::to_string(nb_) +
110 ") > na (" + std::to_string(na_) + "). ARX models require nb <= na. "
111 "Increase na or reduce nb.");
112
113 // ARX B-polynomial: B(z^{-1}) = b1*z^{-1} + b2*z^{-2} + ...
114 // There is no z^0 (direct feedthrough) term - input delay of at least one step.
115 // TransferFunction expects num[0] = b0 (the z^0 coefficient). We must prepend 0.
116 // Without this: tf2ss would set D = b1 and corrupt the C vector.
117 Eigen::VectorXd num_e = Eigen::VectorXd::Zero(na_ + 1);
118 num_e.segment(1, nb_) = theta_.segment(na_, nb_); // [0, b1, b2, ..., b_nb, 0, ...]
119
120 const Eigen::VectorXd den_e = denominator();
121
122 // Convert Eigen vectors to std::vector<double> for TransferFunction ctor
123 std::vector<double> num_v(num_e.data(), num_e.data() + num_e.size());
124 std::vector<double> den_v(den_e.data(), den_e.data() + den_e.size());
125 return TransferFunction(num_v, den_v, Ts_);
126 }
