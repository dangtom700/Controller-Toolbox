19HybridModelTrainer::Result
20HybridModelTrainer::trainHybridModel(HybridModel& model,
21 const Eigen::MatrixXd& X_obs,
22 const Eigen::MatrixXd& U_obs,
23 const Eigen::MatrixXd& X_next_obs) const
24{
25 const int N = static_cast<int>(X_obs.cols());
26 const int n_state = static_cast<int>(X_obs.rows());
27 const int n_in = static_cast<int>(U_obs.rows());
28 const int n_feat = n_state + n_in;
29
30 if (N < 1 || U_obs.cols() != N || X_next_obs.cols() != N)
31 throw std::invalid_argument("HybridModelTrainer: inconsistent column counts");
32
33 // Build feature matrix and residual matrix
34 Eigen::MatrixXd Feat(n_feat, N);
35 Eigen::MatrixXd Resid(n_state, N);
36 for (int k = 0; k < N; ++k) {
37 Feat.col(k) << X_obs.col(k), U_obs.col(k);
38 Resid.col(k) = X_next_obs.col(k) - model.predictPhys(X_obs.col(k), U_obs.col(k));
39 }
40
41 HybridModel::DataFunc f_data;
42 std::string method_name;
43
44 switch (p_.method) {
45 case Method::Ridge:
46 f_data = trainRidge(Feat, Resid);
47 method_name = "Ridge";
48 break;
49 case Method::GP:
50 f_data = trainGP(Feat, Resid);
51 method_name = "GP";
52 break;
53 case Method::ESN:
54 f_data = trainESN(Feat, Resid);
55 method_name = "ESN";
56 break;
57 }
58
59 model.setDataModel(std::move(f_data));
60
61 // Compute training RMSE
62 double mse = 0.0;
63 for (int k = 0; k < N; ++k) {
64 const Eigen::VectorXd pred = model.predict(X_obs.col(k), U_obs.col(k));
65 mse += (pred - X_next_obs.col(k)).squaredNorm();
66 }
67 const double rmse = std::sqrt(mse / static_cast<double>(N));
68
69 return Result{true, rmse, method_name, N};
70}

93// Private: Ridge regression
94// =============================================================================
95
96HybridModel::DataFunc
97HybridModelTrainer::trainRidge(const Eigen::MatrixXd& Feat,
98 const Eigen::MatrixXd& Resid) const
99{
100 // Feat: n_feat x N (each column = one feature vector [x;u])
101 // Resid: n_out x N
102 // We want W (n_feat x n_out) such that W' * feat approx = resid
103 // via min ||Feat' W - Resid'||_F + lambda * ||W||_F
104 // Normal equations: (Feat Feat' + lambda*I) W = Feat Resid'
105 Eigen::MatrixXd FtF = Feat * Feat.transpose();
106 FtF.diagonal().array() += p_.ridge_lambda;
107 const Eigen::MatrixXd W = FtF.ldlt().solve(Feat * Resid.transpose()); // n_feat x n_out
108
109 return [W](const Eigen::VectorXd& x, const Eigen::VectorXd& u) -> Eigen::VectorXd {
110 Eigen::VectorXd feat(x.size() + u.size());
111 feat << x, u;
112 return W.transpose() * feat;
113 };
114}

119
120HybridModel::DataFunc
121HybridModelTrainer::trainGP(const Eigen::MatrixXd& Feat,
122 const Eigen::MatrixXd& Resid) const
123{
124 const int N = static_cast<int>(Feat.cols());
125 const int n_out = static_cast<int>(Resid.rows());
126 const int x_dim = static_cast<int>(Feat.rows());
127
128 // Train one GP per state dimension (independent GPs)
129 auto gps = std::make_shared<std::vector<GaussianProcess>>();
130 gps->reserve(n_out);
131
132 for (int d = 0; d < n_out; ++d) {
133 GaussianProcess gp(x_dim, p_.gp);
134 for (int k = 0; k < N; ++k)
135 gp.addPoint(Feat.col(k), Resid(d, k));
136 gp.fit();
137 gps->push_back(std::move(gp));
138 }
139
140 return [gps](const Eigen::VectorXd& x, const Eigen::VectorXd& u) -> Eigen::VectorXd {
141 Eigen::VectorXd feat(x.size() + u.size());
142 feat << x, u;
143 const int n = static_cast<int>(gps->size());
144 Eigen::VectorXd delta(n);
145 for (int d = 0; d < n; ++d)
146 delta(d) = (*gps)[d].predict(feat).mean;
147 return delta;
148 };
149}

155HybridModel::DataFunc
156HybridModelTrainer::trainESN(const Eigen::MatrixXd& Feat,
157 const Eigen::MatrixXd& Resid) const
158{
159 const int N = static_cast<int>(Feat.cols());
160 const int n_in = static_cast<int>(Feat.rows());
161 const int n_out = static_cast<int>(Resid.rows());
162
163 EchoStateNetwork::Params ep = p_.esn;
164 ep.n_in = n_in;
165 ep.n_out = n_out;
166
167 auto esn = std::make_shared<EchoStateNetwork>(ep);
168 for (int k = 0; k < N; ++k) {
169 esn->stepReservoir(Feat.col(k));
170 esn->addTrainingTarget(Resid.col(k));
171 }
172 esn->fitReadout();
173
174 // Zero-state evaluation: reset reservoir before every prediction so
175 // f_data is a stateless mapping (required for MPC multi-step rollout).
176 return [esn, n_in](const Eigen::VectorXd& x, const Eigen::VectorXd& u) -> Eigen::VectorXd {
177 Eigen::VectorXd feat(n_in);
178 feat << x, u;
179 esn->reset();
180 return esn->predict(feat);
181 };
182}
