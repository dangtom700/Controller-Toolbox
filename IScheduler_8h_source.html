5// IScheduler - abstract interface for periodic task management.
6//
7// Registers a fixed-rate callback (the control loop tick) with the platform
8// scheduler and provides a standard way to start, stop, and query the
9// current execution context. Platform-specific implementations wrap:
10// - Linux RT: timer_create(CLOCK_MONOTONIC) + SIGEV_THREAD_ID + SCHED_FIFO
11// - FreeRTOS: xTimerCreate() / vTaskDelayUntil()
12// - Zephyr: k_timer_init() / k_timer_start()
13//
14// The callback signature is void(void) - no arguments. Any state needed
15// by the control loop should be captured in a lambda or stored externally.
16//
17// Example - wiring a PID loop on a 1 ms tick:
18//
19// auto sched = platform::makeScheduler(); // platform-specific factory
20// sched->setPeriodNs(1'000'000); // 1 ms
21// sched->setCallback([&]() {
22// double y = sensor.read();
23// double u = pid.compute(ref - y);
24// actuator.write(u);
25// });
26// sched->start();
27// // ... main thread does non-RT work or sleeps
28// sched->stop();
29//
30// Overrun detection:
31// Pair with ITimer to measure per-tick execution time and call onOverrun()
32// if it exceeds the period. See ITimer.h for the pattern.
33//
34// Thread safety: start()/stop()/setCallback() must be called from the same
35// non-RT setup thread before the loop begins. The callback runs from the RT
36// context; do not call start()/stop() from inside the callback.
37//
38// Ref: POSIX.1-2017 timer_create(); FreeRTOS Software Timer API.
39namespace ctrl {
40
41class IScheduler {
42public:
43 virtual ~IScheduler() = default;

45 // Set the tick period in nanoseconds. Must be called before start().

50 // the RT context; it must not block, allocate memory, or call non-RT-safe APIs.
51 virtual void setCallback(std::function<void()> cb) = 0;

53 // Start the periodic timer. After this call, the callback fires every period_ns.
54 // Calling start() when already running is a no-op.
55 virtual void start() = 0;
56

58 // completes before returning.
59 virtual void stop() = 0;
60

66
67 // Number of timer ticks fired since the last start() call.
68 // Useful for drift measurement and missed-tick detection.
69 virtual uint64_t tickCount() const = 0;
70
71 // Number of ticks where the callback execution time exceeded period_ns.
72 // Reset to zero on each start() call.

74};
