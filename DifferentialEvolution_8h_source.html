90 TunerResult optimize(const CostFn& cost)
91 {
92 const int N = p_.population;
93 const int n = p_.n_dim;
94
95 // --- Initialise population uniformly in [lower, upper] ---
96 std::vector<Eigen::VectorXd> pop(N, Eigen::VectorXd(n));
97 std::vector<double> fitness(N);
98
99 for (int i = 0; i < N; ++i) {
100 for (int j = 0; j < n; ++j) {
101 std::uniform_real_distribution<double> ud(p_.lower[j], p_.upper[j]);
102 pop[i][j] = ud(rng_);
103 }
104 fitness[i] = cost(pop[i]);
105 }
106 int n_evals = N;
107
108 int best_idx = static_cast<int>(
109 std::min_element(fitness.begin(), fitness.end()) - fitness.begin());
110 Eigen::VectorXd best_x = pop[best_idx];
111 double best_cost = fitness[best_idx];
112
113 bool converged = false;
114 int gen = 0;
115
116 std::uniform_int_distribution<int> uid(0, N - 1);
117 std::uniform_real_distribution<double> ud01(0.0, 1.0);
118 std::uniform_int_distribution<int> uid_dim(0, n - 1);
119
120 for (; gen < p_.max_gen; ++gen) {
121 for (int i = 0; i < N; ++i) {
122 // Pick 3 distinct random indices != i
123 int r1, r2, r3;
124 do { r1 = uid(rng_); } while (r1 == i);
125 do { r2 = uid(rng_); } while (r2 == i || r2 == r1);
126 do { r3 = uid(rng_); } while (r3 == i || r3 == r1 || r3 == r2);
127
128 // Mutation: DE/rand/1
129 Eigen::VectorXd v = pop[r1] + p_.F * (pop[r2] - pop[r3]);
130
131 // Boundary handling: reflect out-of-bounds genes
132 for (int j = 0; j < n; ++j) {
133 while (v[j] < p_.lower[j] || v[j] > p_.upper[j]) {
134 if (v[j] < p_.lower[j]) v[j] = 2.0 * p_.lower[j] - v[j];
135 if (v[j] > p_.upper[j]) v[j] = 2.0 * p_.upper[j] - v[j];
136 }
137 }
138
139 // Crossover: binomial (at least one gene from mutant)
140 Eigen::VectorXd trial = pop[i];
141 int j_rand = uid_dim(rng_);
142 for (int j = 0; j < n; ++j) {
143 if (j == j_rand || ud01(rng_) < p_.CR)
144 trial[j] = v[j];
145 }
146
147 // Greedy selection
148 double trial_cost = cost(trial);
149 ++n_evals;
150 if (trial_cost <= fitness[i]) {
151 pop[i] = trial;
152 fitness[i] = trial_cost;
153 if (trial_cost < best_cost) {
154 best_cost = trial_cost;
155 best_x = trial;
156 }
157 }
158 }
159
160 // Convergence: stddev of population fitness
161 double mean = 0.0;
162 for (double f : fitness) mean += f;
163 mean /= N;
164 double var = 0.0;
165 for (double f : fitness) var += (f - mean) * (f - mean);
166 if (std::sqrt(var / N) < p_.tol) { converged = true; break; }
167 }
168
169 TunerResult r;
170 r.params = best_x;
171 r.cost = best_cost;
172 r.nEvals = n_evals;
173 r.nGens = gen + 1;
174 r.converged = converged;
175 return r;
176 }
