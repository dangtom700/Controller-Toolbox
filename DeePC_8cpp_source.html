71// ---------------------------------------------------------------------------
72// Hessian factorisation (called once in constructor)
73//
74// M = rho_y . H_yf'.H_yf
75// + lambda_g . I
76// + lambda_eq. H_up'.H_up
77// + rho . H_uf'.H_uf
78//
79// M is constant because all four Hankel matrices are fixed offline data.
80// Subsequent per-step solves become O(n_col^2) triangular back-substitutions.
81// ---------------------------------------------------------------------------
82
83void DeePC::factoriseHessian()
84{
85 Eigen::MatrixXd M =
86 p_.rho_y * (H_yf_.transpose() * H_yf_)
87 + p_.lambda_g * Eigen::MatrixXd::Identity(n_col_, n_col_)
88 + p_.lambda_eq * (H_up_.transpose() * H_up_)
89 + p_.lambda_eq * (H_yp_.transpose() * H_yp_) // past-output equality
90 + p_.rho_admm * (H_uf_.transpose() * H_uf_);
91
92 M_ldlt_.compute(M);
93 if (M_ldlt_.info() != Eigen::Success)
94 throw std::runtime_error(
95 "DeePC: Hessian factorisation failed - matrix is not positive-definite. "
96 "Check that the offline data is persistently exciting (increase lambda_g if needed).");
97}

103double DeePC::computeIO(double y_meas, double r)
104{
105 // Shift I/O buffer left (drop oldest) and push (u_prev_, y_meas) at the end
106 u_buf_.head(p_.T_ini - 1) = u_buf_.tail(p_.T_ini - 1).eval();
107 y_buf_.head(p_.T_ini - 1) = y_buf_.tail(p_.T_ini - 1).eval();
108 u_buf_(p_.T_ini - 1) = u_prev_;
109 y_buf_(p_.T_ini - 1) = y_meas;
110
111 ++step_cnt_;
112
113 // Return zero during warm-up (buffer not yet representative)
114 if (step_cnt_ <= p_.T_ini) {
115 notifyObserver(0.0, y_meas);
116 return 0.0;
117 }
118
119 // Constant reference trajectory over the horizon
120 const Eigen::VectorXd y_ref = Eigen::VectorXd::Constant(p_.Np, r);
121
122 // Pre-compute constant parts of the g-update RHS that don't change within the loop
123 const Eigen::VectorXd rhs_fixed =
124 p_.rho_y * (H_yf_.transpose() * y_ref)
125 + p_.lambda_eq * (H_up_.transpose() * u_buf_)
126 + p_.lambda_eq * (H_yp_.transpose() * y_buf_); // past-output equality
127
128 // Scalar used in u-update (element-wise for diagonal R = rho_u.I)
129 const double inv_Ru_rho = 1.0 / (p_.rho_u + p_.rho_admm);
130
131 // ADMM iterations
132 // -----------------------------------------------------------------------
133 // g-update: M . g = rhs_fixed + H_uf' . (rho.u - lambda)
134 // u-update: u_free = (rho.H_uf.g + lambda) / (rho_u + rho); u = clip(u_free, uMin, uMax)
135 // lambda-update: lambda += rho . (H_uf.g - u)
136 // -----------------------------------------------------------------------
137 for (int k = 0; k < p_.admm_iter; ++k) {
138 // g-update
139 const Eigen::VectorXd rhs =
140 rhs_fixed + H_uf_.transpose() * (p_.rho_admm * u_opt_ - lam_);
141 g_ = M_ldlt_.solve(rhs);
142
143 // u-update: unconstrained optimum then project
144 const Eigen::VectorXd huf_g = H_uf_ * g_;
145 const Eigen::VectorXd u_free = inv_Ru_rho * (p_.rho_admm * huf_g + lam_);
146 u_opt_ = u_free.cwiseMin(p_.uMax).cwiseMax(p_.uMin);
147
148 // dual update
149 lam_ += p_.rho_admm * (huf_g - u_opt_);
150 }
151
152 primal_res_ = (H_uf_ * g_ - u_opt_).norm();
153
154 const double u_apply = std::clamp(u_opt_(0), p_.uMin, p_.uMax);
155 u_prev_ = u_apply;
156 notifyObserver(u_apply, y_meas);
157 return u_apply;
158}
