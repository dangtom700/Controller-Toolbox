50void DeePC::buildHankel(const Eigen::VectorXd& u_data, const Eigen::VectorXd& y_data) {
51 const int T = p_.T_ini;
52 const int N = p_.N;
53 const int N_data = static_cast<int>(u_data.size());
54 M_ = N_data - T - N + 1;
55
56 H_up_.resize(T, M_);
57 H_yp_.resize(T, M_);
58 H_uf_.resize(N, M_);
59 H_yf_.resize(N, M_);
60
61 for (int j = 0; j < M_; ++j) {
62 for (int i = 0; i < T; ++i) {
63 H_up_(i, j) = u_data(i + j);
64 H_yp_(i, j) = y_data(i + j);
65 }
66 for (int i = 0; i < N; ++i) {
67 H_uf_(i, j) = u_data(T + i + j);
68 H_yf_(i, j) = y_data(T + i + j);
69 }
70 }
71
72 H_up_T_ = H_up_.transpose();
73 H_yp_T_ = H_yp_.transpose();
74 H_uf_T_ = H_uf_.transpose();
75 H_yf_T_ = H_yf_.transpose();
76}

78void DeePC::buildQPMatrices() {
79 const double Q = p_.Q;
80 const double ly = p_.lambda_y;
81 const double lu = p_.lambda_u;
82 const double lg = p_.lambda_g;
83 const double rho = p_.rho;
84
85 // A_g = 2*(Q*H_yf'*H_yf + ly*H_yp'*H_yp + lu*H_up'*H_up + lg*I) + rho*H_uf'*H_uf
86 // Note: R is not in A_g because R penalises u_f (handled in the u_f ADMM step).
87 A_g_.resize(M_, M_);
88 A_g_.noalias() = (2.0 * Q) * (H_yf_T_ * H_yf_);
89 A_g_.noalias() += (2.0 * ly) * (H_yp_T_ * H_yp_);
90 A_g_.noalias() += (2.0 * lu) * (H_up_T_ * H_up_);
91 A_g_.noalias() += rho * (H_uf_T_ * H_uf_);
92 for (int i = 0; i < M_; ++i)
93 A_g_(i, i) += 2.0 * lg;
94
95 // Enforce exact symmetry to avoid LDLT numerical issues
96 A_g_ = 0.5 * (A_g_ + A_g_.transpose()).eval();
97
98 A_g_ldlt_.compute(A_g_);
99
100 // Pre-allocate workspace vectors
101 b_g_.setZero(M_);
102 b_g_ini_.setZero(M_);
103 b_g_ref_.setZero(M_);
104 H_uf_g_.setZero(p_.N);
105 r_stacked_ = Eigen::VectorXd::Constant(p_.N, r_);
106 b_g_ref_.noalias() = (2.0 * Q) * (H_yf_T_ * r_stacked_);
107}

121double DeePC::compute(double y_meas) {
122 if (!std::isfinite(y_meas) || !data_collected_) {
123 notifyObserver(u_last_, y_meas);
124 return u_last_;
125 }
126
127 // Append current measurement and trim buffer to T_ini length
128 y_buf_.push_back(y_meas);
129 if (static_cast<int>(y_buf_.size()) > p_.T_ini)
130 y_buf_.pop_front();

139 u_buf_.pop_front();

145double DeePC::solveADMM() {
146 // Build y_ini and u_ini from deque buffers
147 Eigen::VectorXd y_ini(p_.T_ini), u_ini(p_.T_ini);
148 {
149 int idx = 0;
150 for (double v : y_buf_) y_ini(idx++) = v;
151 }
152 {
153 int idx = 0;
154 for (double v : u_buf_) u_ini(idx++) = v;
155 }
156
157 // Past-trajectory contribution (changes each step, precomputed here)
158 b_g_ini_.noalias() = (2.0 * p_.lambda_y) * (H_yp_T_ * y_ini)
159 + (2.0 * p_.lambda_u) * (H_up_T_ * u_ini);
160
161 // Cold-start u_f and dual (g_ is warm-started from previous step)
162 uf_.setZero();
163 dual_.setZero();
164
165 const double denom_uf = 2.0 * p_.R + p_.rho;
166 const double rho = p_.rho;
167
168 // Boyd et al. (2011) Section 3.3.1 stopping criterion: absolute + relative tolerances.
169 // Scale factors avoid dependence on problem units or matrix norms.
170 const double eps_abs = p_.admm_tol;
171 const double eps_rel = 1e-3;
172 const double sq_N = std::sqrt(static_cast<double>(p_.N));
173 const double sq_M = std::sqrt(static_cast<double>(M_));
174
175 bool converged = false;
176 for (int iter = 0; iter < p_.admm_iters; ++iter) {
177 // ---------------------------------------------------------------
178 // g-update: A_g * g = b_g_ref + b_g_ini + H_uf' * (rho*u_f - nu)
179 // ---------------------------------------------------------------
180 b_g_.noalias() = b_g_ref_ + b_g_ini_ + H_uf_T_ * (rho * uf_ - dual_);
181 const Eigen::VectorXd g_new = A_g_ldlt_.solve(b_g_);
182
183 // ---------------------------------------------------------------
184 // u_f-update: box-project (rho * H_uf * g + nu) / (2R + rho)
185 // ---------------------------------------------------------------
186 H_uf_g_.noalias() = H_uf_ * g_new;
187 const Eigen::VectorXd uf_new = ((rho * H_uf_g_ + dual_) / denom_uf)
188 .cwiseMax(p_.uMin)
189 .cwiseMin(p_.uMax);
190
191 // ---------------------------------------------------------------
192 // Dual update: nu += rho * (H_uf * g - u_f)
193 // ---------------------------------------------------------------
194 const Eigen::VectorXd prim_res = H_uf_g_ - uf_new;
195 const Eigen::VectorXd dual_res = rho * (H_uf_T_ * (uf_new - uf_));
196
197 dual_.noalias() += rho * prim_res;
198 g_ = g_new;
199 uf_ = uf_new;
200
201 // Scaled stopping: eps_prim = eps_abs*sqrt(N) + eps_rel*max(||H_uf g||, ||u_f||)
202 // eps_dual = eps_abs*sqrt(M) + eps_rel*||H_uf^T nu||
203 const double eps_prim = eps_abs * sq_N
204 + eps_rel * std::max(H_uf_g_.norm(), uf_.norm());
205 const double eps_dual = eps_abs * sq_M
206 + eps_rel * (H_uf_T_ * dual_).norm();
207 if (prim_res.norm() < eps_prim && dual_res.norm() < eps_dual) {
208 converged = true;
209 break;
210 }
211 }
212 healthy_ = converged;
213 return uf_(0);
214}
