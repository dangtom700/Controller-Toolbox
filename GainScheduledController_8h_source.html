132 double compute(double error) override {
133 if (schedule_.empty()) {
134 last_output_ = 0.0;
135 notifyObserver(0.0, error);
136 return 0.0;
137 }
138 if (schedule_.size() == 1) {
139 last_output_ = schedule_[0].ctrl->compute(error);
140 active_idx_ = 0;
141 notifyObserver(last_output_, error);
142 return last_output_;
143 }
144
145 int lo = lowerIndex(current_p_);
146 int hi = std::min(lo + 1, static_cast<int>(schedule_.size()) - 1);
147
148 if (mode_ == GainScheduleMode::NearestNeighbor) {
149 int idx = (std::abs(current_p_ - schedule_[lo].p) <=
150 std::abs(current_p_ - schedule_[hi].p)) ? lo : hi;
151 // Bumpless transfer: pre-condition the incoming controller so its
152 // first compute() matches the last output of the outgoing controller.
153 if (idx != active_idx_)
154 schedule_[idx].ctrl->bumplessInit(last_output_, error);
155 active_idx_ = idx;
156 last_output_ = schedule_[idx].ctrl->compute(error);
157 notifyObserver(last_output_, error);
158 return last_output_;
159 }
160
161 // LinearBlend
162 if (lo == hi) {
163 last_output_ = schedule_[lo].ctrl->compute(error);
164 notifyObserver(last_output_, error);
165 return last_output_;
166 }
167 double p0 = schedule_[lo].p, p1 = schedule_[hi].p;
168 double alpha = (p1 > p0) ? (current_p_ - p0) / (p1 - p0) : 0.0;
169 alpha = std::max(0.0, std::min(1.0, alpha));
170 double u0 = schedule_[lo].ctrl->compute(error);
171 double u1 = schedule_[hi].ctrl->compute(error);
172 last_output_ = (1.0 - alpha) * u0 + alpha * u1;
173 notifyObserver(last_output_, error);
174 return last_output_;
175 }

203 int lowerIndex(double p) const noexcept {
204 if (p <= schedule_.front().p) return 0;
205 if (p >= schedule_.back().p)
206 return static_cast<int>(schedule_.size()) - 1;
