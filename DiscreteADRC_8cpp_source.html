31 // ESO backward-Euler update (A-stable for all omega_o * Ts > 0).
32 //
33 // Continuous ESO in matrix form: dz/dt = Ae z + Be_eps * eps + Be_u * u
34 // Ae = [[0,1,0],[0,0,1],[0,0,0]] (nilpotent, Ae^3 = 0)
35 // Be_eps = [beta1, beta2, beta3]'
36 // Be_u = [0, b0, 0]'
37 // eps = y - z1 (observer error, computed from the current z1)
38 //
39 // Backward Euler: (I - Ts.Ae) z_new = z + Ts.(Be_eps.eps + Be_u.u)
40 //
41 // Since Ae is strictly lower-shift, (I - Ts.Ae) is unit upper-triangular and
42 // its inverse is available analytically:
43 // (I - Ts.Ae)^{-1} = [[1, Ts, Ts^2],[0, 1, Ts],[0, 0, 1]]
44 //
45 // eps uses the OLD z1 (semi-implicit); this gives unconditional A-stability at
46 // the cost of one step of phase lag in the observer error - negligible for
47 // omega_o * Ts << 1 and far better than the forward-Euler instability risk.
48 //
49 // PD + disturbance cancellation (unchanged from continuous parameterisation):
50 // u0 = omega_c^2.(r - z1) - 2.omega_c.z2
51 // u = (u0 - z3) / b0
52 double DiscreteADRC::computeTracking(double y, double r)
53 {
54 if (!std::isfinite(y) || !std::isfinite(r))
55 return u_prev_;
56
57 // Observer error (uses current z1 - semi-implicit)
58 const double eps = y - z_(0);
59
60 // RHS of (I - Ts.Ae) z_new = z + Ts.(Be_eps.eps + Be_u.u)
61 const double rhs1 = z_(0) + Ts_ * beta1_ * eps;
62 const double rhs2 = z_(1) + Ts_ * (beta2_ * eps + p_.b0 * u_prev_);
63 const double rhs3 = z_(2) + Ts_ * beta3_ * eps;
64
65 // Apply (I - Ts.Ae)^{-1} via back-substitution (upper triangular, unit diagonal)
66 const double z3n = rhs3;
67 const double z2n = rhs2 + Ts_ * z3n;
68 const double z1n = rhs1 + Ts_ * z2n; // = rhs1 + Ts*rhs2 + Ts^2*rhs3
69 z_ << z1n, z2n, z3n;
70
71 // PD setpoint control + disturbance cancellation
72 const double u0 = p_.omega_c * p_.omega_c * (r - z_(0)) - 2.0 * p_.omega_c * z_(1);
73 double u = (u0 - z_(2)) / p_.b0;
74 u = std::max(p_.uMin, std::min(p_.uMax, u));
75
76 u_prev_ = u;
77 // M3 telemetry: emit ESO state [z1, z2, z3] for observers that need it
78 notifyObserverState("eso", z_);
79 notifyObserver(u, y);
80 return u;
81 }
