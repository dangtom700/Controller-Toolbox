133inline TunerResult AutoTuner::tune(CostFn cost, const Eigen::VectorXd &x0)
134{
135 const int n = p_.n;
136
137 // -----------------------------------------------------------------------
138 // CMA-ES strategy parameters (Hansen 2006 tutorial Table 1)
139 // -----------------------------------------------------------------------
140 const int lambda = 4 + static_cast<int>(std::floor(3.0 * std::log(static_cast<double>(n))));
141 const int mu = lambda / 2;
142
143 // Log-weighted recombination weights (normalised)
144 Eigen::VectorXd w(mu);
145 for (int i = 0; i < mu; ++i)
146 w(i) = std::log(static_cast<double>(mu) + 0.5) - std::log(static_cast<double>(i + 1));
147 w /= w.sum();
148 const double mu_eff = 1.0 / w.squaredNorm(); // variance-effective selection mass
149
150 // Step-size control
151 const double c_sigma = (mu_eff + 2.0) / (static_cast<double>(n) + mu_eff + 5.0);
152 const double d_sigma = 1.0
153 + 2.0 * std::max(0.0, std::sqrt((mu_eff - 1.0) / (static_cast<double>(n) + 1.0)) - 1.0)
154 + c_sigma;
155 const double chi_n = std::sqrt(static_cast<double>(n))
156 * (1.0 - 1.0 / (4.0 * n) + 1.0 / (21.0 * n * n));
157
158 // Covariance adaptation
159 const double c_c = (4.0 + mu_eff / n) / (n + 4.0 + 2.0 * mu_eff / n);
160 const double c_1 = 2.0 / ((n + 1.3) * (n + 1.3) + mu_eff);
161 const double c_mu = std::min(
162 1.0 - c_1,
163 2.0 * (mu_eff - 2.0 + 1.0 / mu_eff) / ((n + 2.0) * (n + 2.0) + mu_eff));
164
165 // Eigendecomposition frequency (at most once per generation for small n)
166 const int eigen_freq = std::max(
167 1, static_cast<int>(std::floor(1.0 / (c_1 + c_mu) / n / 10.0)));
168
169 // -----------------------------------------------------------------------
170 // State variables
171 // -----------------------------------------------------------------------
172 Eigen::VectorXd m = x0; // distribution mean
173 double sigma = p_.sigma0; // step size
174 Eigen::VectorXd p_sigma = Eigen::VectorXd::Zero(n); // CSA path
175 Eigen::VectorXd p_c = Eigen::VectorXd::Zero(n); // covariance path
176 Eigen::MatrixXd C = Eigen::MatrixXd::Identity(n, n); // covariance
177 Eigen::MatrixXd B = Eigen::MatrixXd::Identity(n, n); // eigenvectors of C
178 Eigen::VectorXd D = Eigen::VectorXd::Ones(n); // sqrt eigenvalues of C
179
180 std::normal_distribution<double> normal(0.0, 1.0);
181
182 // -----------------------------------------------------------------------
183 // Tracking
184 // -----------------------------------------------------------------------
185 double best_cost = std::numeric_limits<double>::infinity();
186 Eigen::VectorXd best_params = x0;
187 int n_evals = 0;
188
189 std::vector<Eigen::VectorXd> z_pop(lambda, Eigen::VectorXd(n));
190 std::vector<Eigen::VectorXd> y_pop(lambda, Eigen::VectorXd(n));
191 std::vector<Eigen::VectorXd> x_pop(lambda, Eigen::VectorXd(n));
192 std::vector<double> f_pop(lambda);
193 std::vector<int> idx(lambda);
194
195 for (int gen = 0; gen < p_.maxIter; ++gen)
196 {
197 // -------------------------------------------------------------------
198 // Sample lambda candidate solutions
199 // -------------------------------------------------------------------
200 for (int i = 0; i < lambda; ++i)
201 {
202 for (int j = 0; j < n; ++j) z_pop[i](j) = normal(rng_);
203 // y = B * (D .* z) (covariance-scaled direction)
204 y_pop[i] = B * D.cwiseProduct(z_pop[i]);
205 x_pop[i] = clip(m + sigma * y_pop[i]);
206 }
207
208 // -------------------------------------------------------------------
209 // Evaluate cost and track best
210 // -------------------------------------------------------------------
211 for (int i = 0; i < lambda; ++i)
212 {
213 f_pop[i] = cost(x_pop[i]);
214 ++n_evals;
215 if (f_pop[i] < best_cost)
216 {
217 best_cost = f_pop[i];
218 best_params = x_pop[i];
219 }
220 }
221
222 // -------------------------------------------------------------------
223 // Sort by cost (ascending) to select mu best
224 // -------------------------------------------------------------------
225 std::iota(idx.begin(), idx.end(), 0);
226 std::sort(idx.begin(), idx.end(),
227 [&](int a, int b){ return f_pop[a] < f_pop[b]; });
228
229 // -------------------------------------------------------------------
230 // Update mean
231 // -------------------------------------------------------------------
232 const Eigen::VectorXd m_old = m;
233 m.setZero();
234 for (int i = 0; i < mu; ++i) m += w(i) * x_pop[idx[i]];
235 const Eigen::VectorXd y_w = (m - m_old) / sigma; // mean step direction
236
237 // -------------------------------------------------------------------
238 // Update cumulative step-length path p_sigma
239 // p_sigma <-- (1-cs)*p_sigma + sqrt(cs*(2-cs)*mueff) * C^{-1/2} * yw
240 // where C^{-1/2} = B * D^{-1} * B^T (eigenvector decomposition)
241 // -------------------------------------------------------------------
242 const Eigen::VectorXd Cinv_half_yw =
243 B * (D.cwiseInverse().cwiseProduct(B.transpose() * y_w));
244 p_sigma = (1.0 - c_sigma) * p_sigma
245 + std::sqrt(c_sigma * (2.0 - c_sigma) * mu_eff) * Cinv_half_yw;
246
247 // Step-size update (CSA)
248 sigma *= std::exp((c_sigma / d_sigma) * (p_sigma.norm() / chi_n - 1.0));
249
250 // -------------------------------------------------------------------
251 // Indicator h_sigma (heaviside for rank-1 update)
252 // -------------------------------------------------------------------
253 const double ps_norm_eff = p_sigma.norm()
254 / std::sqrt(1.0 - std::pow(1.0 - c_sigma, 2.0 * (gen + 1)));
255 const double h_threshold = (1.4 + 2.0 / (n + 1.0)) * chi_n;
256 const double h_sigma = (ps_norm_eff < h_threshold) ? 1.0 : 0.0;
257
258 // -------------------------------------------------------------------
259 // Update covariance path p_c
260 // -------------------------------------------------------------------
261 p_c = (1.0 - c_c) * p_c
262 + h_sigma * std::sqrt(c_c * (2.0 - c_c) * mu_eff) * y_w;
263
264 // -------------------------------------------------------------------
265 // Update covariance matrix C (rank-1 + rank-mu)
266 // -------------------------------------------------------------------
267 C = (1.0 - c_1 - c_mu) * C
268 + c_1 * (p_c * p_c.transpose()
269 + (1.0 - h_sigma) * c_c * (2.0 - c_c) * C);
270
271 for (int i = 0; i < mu; ++i)
272 {
273 // Use un-clipped y direction for the update for correct statistics
274 const Eigen::VectorXd yi = (x_pop[idx[i]] - m_old) / sigma;
275 C += c_mu * w(i) * (yi * yi.transpose());
276 }
277
278 // -------------------------------------------------------------------
279 // Eigendecompose C -> B, D (every eigen_freq generations)
280 // -------------------------------------------------------------------
281 if (gen % eigen_freq == 0)
282 {
283 // Enforce symmetry before decomposition
284 C = 0.5 * (C + C.transpose());
285
286 Eigen::SelfAdjointEigenSolver<Eigen::MatrixXd> eig(C);
287 // Floor near-zero eigenvalues for numerical safety
288 D = eig.eigenvalues().cwiseMax(1e-20).cwiseSqrt();
289 B = eig.eigenvectors();
290 }
291
292 // -------------------------------------------------------------------
293 // Convergence check: stop when the step-size-scaled spread is tiny
294 // -------------------------------------------------------------------
295 if ((sigma * D).maxCoeff() < p_.tol)
296 return {best_params, best_cost, n_evals, gen + 1, true};
297 }
298
299 return {best_params, best_cost, n_evals, p_.maxIter, false};
300}
