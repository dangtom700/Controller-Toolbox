8L1AdaptiveController::L1AdaptiveController(const Params& p, double Ts)
9 : p_(p), Ts_(Ts)
10{
11 if (std::abs(p_.a_m) >= 1.0)
12 throw std::invalid_argument("L1AdaptiveController: |a_m| must be < 1");
13 if (p_.b_m == 0.0)
14 throw std::invalid_argument("L1AdaptiveController: b_m must be non-zero");
15 if (p_.Gamma <= 0.0)
16 throw std::invalid_argument("L1AdaptiveController: Gamma must be positive");
17 if (p_.omega_c <= 0.0)
18 throw std::invalid_argument("L1AdaptiveController: omega_c must be positive");
19
20 // Discrete Lyapunov: a_m^2 * P - P + Q = 0 => P = Q / (1 - a_m^2)
21 P_ = p_.Q_lyap / (1.0 - p_.a_m * p_.a_m);
22
23 // Low-pass filter: first-order IIR, C(z) = (1-a_lp)/(z - a_lp)
24 a_lp_ = std::exp(-p_.omega_c * Ts_);
25 b_lp_ = 1.0 - a_lp_;
26}

28double L1AdaptiveController::compute(double y_plant)
29{
30 // Step 1: prediction error
31 pred_err_ = x_hat_ - y_plant;
32
33 // Step 2: adaptation law (gradient descent on Lyapunov function)
34 // sigma^[k+1] = sigma^[k] - Gamma.Ts.(ẽ[k] . b_m . P)
35 sigma_hat_ -= p_.Gamma * Ts_ * (pred_err_ * p_.b_m * P_);
36 sigma_hat_ = std::clamp(sigma_hat_, -p_.sigma_max, p_.sigma_max);
37
38 // Step 3: desired input to the LP filter eta = -sigma^
39 // full signal = k_g*r + eta
40 const double eta = -sigma_hat_;
41 const double signal = p_.k_g * r_ + eta;
42
43 // Step 4: LP filter v[k+1] = a_lp * v[k] + b_lp * signal
44 v_lp_ = a_lp_ * v_lp_ + b_lp_ * signal;
45 double u = std::clamp(v_lp_, p_.uMin, p_.uMax);
46

48 // x^[k+1] = a_m * x^[k] + b_m * (u[k] + sigma^[k])
49 x_hat_ = p_.a_m * x_hat_ + p_.b_m * (u + sigma_hat_);
