12 // ---------------------------------------------------------------------------
13 // ESC step - perturbation + demodulation + gradient integration.
14 //
15 // HPF (backward Euler first-order, removes DC offset from y):
16 // alpha_h = 1 / (1 + omega_h.Ts)
17 // y_h[k] = alpha_h.(y_h[k-1] + y[k] - y[k-1])
18 //
19 // Demodulate by multiplying with reference dither sin(omega_p.k.Ts):
20 // xi[k] = y_h[k] . sin(omega_p.k.Ts)
21 // After LPF: ghat approx = J'(theta).a/2 (gradient of cost w.r.t. operating point)
22 //
23 // LPF (backward Euler first-order):
24 // alpha_l = omega_l.Ts / (1 + omega_l.Ts)
25 // ghat[k] = (1-alpha_l).ghat[k-1] + alpha_l.xi[k]
26 //
27 // Operating-point update (gradient descent / ascent):
28 // theta[k+1] = theta[k] - sign . k_int . Ts . ghat[k]
29 // ---------------------------------------------------------------------------
30 double ExtremumSeeker::compute(double y)
31 {
32 // Advance phase accumulator - stays bounded in [0, 2pi) for arbitrarily long runs,
33 // avoiding the floating-point precision loss of step_ * Ts_ at large step counts
34 // and the 32-bit overflow of a long counter on embedded targets.
35 phase_ = std::fmod(phase_ + 2.0 * M_PI * p_.perturbFreq * Ts_, 2.0 * M_PI);
36
37 // HPF
38 const double wh = 2.0 * M_PI * p_.hpfCutoff;
39 const double alpha_h = 1.0 / (1.0 + wh * Ts_);
40 const double y_h = alpha_h * (hpf_state_ + y - y_prev_);
41 hpf_state_ = y_h;
42 y_prev_ = y;
43
44 // Demodulate using the same phase_ as the dither - coherent demodulation
45 const double demod = y_h * std::sin(phase_);
46
47 // LPF
48 const double wl = 2.0 * M_PI * p_.lpfCutoff;
49 const double alpha_l = wl * Ts_ / (1.0 + wl * Ts_);
50 lpf_state_ = (1.0 - alpha_l) * lpf_state_ + alpha_l * demod;
51
52 // Gradient integration
53 const double sign = p_.seekMinimum ? -1.0 : 1.0;
54 theta_ += sign * p_.integGain * lpf_state_ * Ts_;
55
56 // Return operating point plus dither signal
57 return theta_ + p_.perturbAmp * std::sin(phase_);
58 }
