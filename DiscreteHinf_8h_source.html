7// =============================================================================
8// Discrete-Time H-infinity Controller Synthesis (DGKF two-Riccati method)
9//
10// Synthesises a discrete-time dynamic output-feedback controller K(z) that
11// minimises the closed-loop H-infinity norm:
12//
13// ||F_l(P, K)||_inf < gamma
14//
15// where P is the generalised plant (augmented with weighting functions) and
16// F_l denotes the lower Linear Fractional Transformation.
17//
18// -- Generalised Plant Format ------------------------------------------------
19//
20// P: x[k+1] = A x[k] + B1 w[k] + B2 u[k]
21// z[k] = C1 x[k] + D11 w[k] + D12 u[k] (performance output)
22// y[k] = C2 x[k] + D21 w[k] + D22 u[k] (measurement output)
23//
24// w = exogenous input (disturbances, references)
25// z = performance output (weighted tracking error, control effort)
26// u = control input
27// y = measured output
28//
29// -- Mixed-Sensitivity Design ------------------------------------------------
30//
31// The most common use case: S/KS/T mixed sensitivity. Use MixedSensitivity
32// (below) to build P automatically from a SISO plant G and three weights W1
33// (tracking), W2 (control effort), W3 (complementary sensitivity/robustness).
34//
35// ||[ W1 S ]||
36// ||[ W2 KS ]|| < gamma
37// ||[ W3 T ]|| inf
38//
39// -- Algorithm ---------------------------------------------------------------
40//
41// Discrete DGKF (Doyle, Glover, Khargonekar & Francis 1989 - discrete version
42// by Stoorvogel 1992 / Iglesias & Glover 1991):
43//
44// Assumptions (relaxed form - checked in solve()):
45// (A1) (A, B2) is stabilisable, (A, C2) is detectable
46// (A2) D12 has full column rank, D21 has full row rank
47// (A3) D22 = 0 (standard form; MixedSensitivity guarantees this)
48// D11 may be nonzero; gamma must exceed ||D11||_2 (Iglesias & Glover 1991, Lemma 2.1)
49//
50// Step 1: solve the Control DARE for X_inf:
51// A' X A - X - A' X B (R_tilde)^{-1} B' X A + Q_tilde = 0
52// where B = [B1 B2], R_tilde = [I+D'D D12'], Q_tilde = C1'C1
53//
54// Step 2: solve the Filter DARE for Y_inf:
55// A Y A' - Y - A Y C (R_hat)^{-1} C Y A' + Q_hat = 0
56// where C = [C1; C2], Q_hat = B1 B1'
57//
58// Step 3: compute controller matrices from X_inf, Y_inf.
59// K has order n (same as the plant).
60//
61// The gamma iteration (bisection on gamma) finds the minimum achievable H_inf
62// norm within a specified tolerance.
63//
64// -- Usage -------------------------------------------------------------------
65//
66// // 1. Build augmented plant (use MixedSensitivity helper for S/KS/T design)
67// ctrl::GeneralisedPlant P = ctrl::MixedSensitivity::build(G, W1, W2, W3);
68//
69// // 2. Synthesise controller
70// ctrl::HinfParams hp;
71// hp.gammaInit = 2.0; // initial gamma (bisect down from here)
72// hp.gammaTol = 0.01; // stop when gamma bracket < 0.01
73// auto result = ctrl::DiscreteHinf::solve(P, hp);
74//
75// // 3. Check feasibility
76// if (result.feasible) {
77// ctrl::DiscreteHinf ctrl(result);
78// double u = ctrl.compute(y_measurement);
79// }
80//
81// References:
82// Doyle, Glover, Khargonekar & Francis, IEEE TAC 34(8), 1989.
83// Stoorvogel, "The Discrete-Time H-inf Control Problem," SIAM 1992.
84// Iglesias & Glover, "State-space approach to discrete-time H-inf control,"
85// Int. J. Control 54(5), 1991.
86// Skogestad & Postlethwaite, "Multivariable Feedback Design" Ch. 9, 2005.
87// MATLAB hinfsyn() documentation.
88// =============================================================================

93// ---------------------------------------------------------------------------
94// Generalised Plant
95// x[k+1] = A x + B1 w + B2 u
96// z[k] = C1 x + D11 w + D12 u
97// y[k] = C2 x + D21 w + D22 u
98//
99// Dimensions:
100// n = state size
101// nw = exogenous input size (disturbances + references)
102// nu = control input size
103// nz = performance output size (weighted signals)
104// ny = measurement output size
105// ---------------------------------------------------------------------------
106struct GeneralisedPlant
107{
108 Eigen::MatrixXd A; // n * n
109 Eigen::MatrixXd B1; // n * nw
110 Eigen::MatrixXd B2; // n * nu
111 Eigen::MatrixXd C1; // nz * n
112 Eigen::MatrixXd C2; // ny * n
113 Eigen::MatrixXd D11; // nz * nw
114 Eigen::MatrixXd D12; // nz * nu
115 Eigen::MatrixXd D21; // ny * nw
116 Eigen::MatrixXd D22; // ny * nu
117 double Ts = 0.01;
118
119 int stateSize() const { return static_cast<int>(A.rows()); }
120 int nw() const { return static_cast<int>(B1.cols()); }
121 int nu() const { return static_cast<int>(B2.cols()); }
122 int nz() const { return static_cast<int>(C1.rows()); }
123 int ny() const { return static_cast<int>(C2.rows()); }
124};

244// =============================================================================
245// MixedSensitivity - Helper that builds a generalised plant for S/KS/T design
246//
247// Augments a SISO discrete-time plant G(z) with three weighting transfer
248// functions (as state-space models) to form:
249//
250// Objective: ||[ W1(z).S(z) ]||
251// ||[ W2(z).K(z)S ]|| < gamma
252// ||[ W3(z).T(z) ]|| inf
253//
254// where S = 1/(1+GK), T = GK/(1+GK), KS = K.S.
255//
256// -- Weight Guidelines -------------------------------------------------------
257//
258// W1 (tracking / disturbance rejection):
259// - High gain at low frequencies -> tight reference tracking and disturbance rejection.
260// - Typical: W1 = (s/M + omega_B) / (s + omega_B*eps)
261// => crossover at omega_B, low-freq gain M, DC gain 1/eps.
262// - Discrete: use a 1st or 2nd order filter discretised with Tustin or ZOH.
263//
264// W2 (control effort / actuator limits):
265// - Limits high-frequency control action; prevents actuator saturation.
266// - Typical: constant W2 = 1/u_max, or a high-pass with rolloff at omega_u.
267//
268// W3 (robustness / unmodelled dynamics):
269// - Complements S; ensures T rolls off at high frequencies.
270// - Typical: W3 = (s + omega_B/Mt) / (eps.s + omega_B)
271// => rolls up above omega_B with asymptotic gain 1/eps.
272//
273// -- Usage -------------------------------------------------------------------
274//
275// // Plant: G(s) = 1/(s+1), ZOH at Ts = 0.01
276// auto G = ctrl::c2d(G_continuous, 0.01, ctrl::C2dMethod::ZOH);
277//
278// // W1: integrating weight, crossover at 10 rad/s, eps=0.001
279// auto W1 = ctrl::MixedSensitivity::makeW1(10.0, 1.0, 1e-3, 0.01);
280//
281// // W2: constant gain (limit control to +/- 5)
282// auto W2 = ctrl::MixedSensitivity::makeW2constant(1.0/5.0, 0.01);
283//
284// // W3: high-frequency robustness, rollup above 50 rad/s
285// auto W3 = ctrl::MixedSensitivity::makeW3(50.0, 1.5, 1e-3, 0.01);
286//
287// // Assemble augmented plant
288// auto P = ctrl::MixedSensitivity::build(G, W1, W2, W3);
289//
290// // Synthesise
291// ctrl::HinfParams hp; hp.gammaInit = 5.0;
292// auto result = ctrl::DiscreteHinf::solve(P, hp);
293//
294// =============================================================================
295class MixedSensitivity
296{
297public:
298 // -----------------------------------------------------------------------
299 // Weight factory functions (discrete-time, first-order state-space)
300 // -----------------------------------------------------------------------
301
302 // W1 - sensitivity weight for tracking/disturbance rejection.
303 // Low-freq gain ~ 1/eps (large, enforces tight tracking at DC)
304 // Crossover at omega_B [rad/s]
305 // High-freq gain ~ M (upper bound on |W1| at high frequencies)
306 // Discretised with Tustin at sample time Ts [s].
307 //
308 // Continuous: W1(s) = (s/M + omega_B) / (s + omega_B * eps)
309 static StateSpace makeW1(double omega_B, double M, double eps, double Ts);
310
311 // W2 - scalar constant weight (flat gain at all frequencies).
312 // Useful for simple actuator-level constraints: set gain = 1/u_max.
313 static StateSpace makeW2constant(double gain, double Ts);
314
315 // W2 - first-order high-pass weight for control effort roll-off.
316 // Low-freq gain ~ eps (allows large low-frequency control)
317 // High-freq gain ~ 1 (limits high-frequency control effort)
318 // Rolloff at omega_u [rad/s]
319 // Continuous: W2(s) = (s + omega_u * eps) / (s/1 + omega_u)
320 static StateSpace makeW2highpass(double omega_u, double eps, double Ts);
321
322 // W3 - complementary-sensitivity weight for robustness.
323 // Low-freq gain ~ eps (allows T ~ 1 at low frequencies)
324 // High-freq gain ~ 1/Mt (forces T rolloff above omega_T)
325 // Continuous: W3(s) = (s + omega_T / Mt) / (eps * s + omega_T)
326 static StateSpace makeW3(double omega_T, double Mt, double eps, double Ts);
327
328 // -----------------------------------------------------------------------
329 // Build the generalised plant P from G, W1, W2, W3.
330 // All inputs must be discrete-time (Ts > 0) with the same sample time.
331 // G must be SISO. W1, W2, W3 may be any order (first-order recommended).
332 //
333 // The augmented state is: xa = [xG; xW1; xW2; xW3]
334 //
335 // Generalised plant structure:
336 // Exogenous inputs w = [r; d] (reference and output disturbance) - nw = 2
337 // Performance outputs z = [W1*(r-y); W2*u; W3*y] - nz = 3
338 // Control input u - nu = 1
339 // Measurement output y = G*u + d - ny = 1
340 //
341 // For a pure reference-tracking design, d is left connected but may be
342 // set to zero during simulation.
343 //
344 // Troubleshooting when solve() returns feasible = false:
345 // 1. Increase gammaInit: start at 10 * ||W1||_inf * ||W3||_inf as a rule of thumb.
346 // 2. Check that W1's crossover frequency is below the plant's open-loop bandwidth.
347 // A W1 that demands high gain above the plant's roll-off frequency is infeasible.
348 // 3. For non-minimum-phase plants (RHP zeros or excess delay), the achievable
349 // sensitivity peak S is bounded below by the Poisson integral; even a theoretically
350 // perfect controller cannot drive ||W1*S||_inf below this bound. Reduce W1 gain
351 // or relax the crossover requirement. See Skogestad & Postlethwaite Section 6.3.
352 // 4. If W1 and W3 both have high gain in overlapping frequency bands, the design
353 // objective may be fundamentally infeasible (S + T = 1 sets a hard trade-off).
354 // Reduce |W1(jw)| * |W3(jw)| to satisfy the S-T complementarity constraint.
355 // 5. If gammaInit is too small (below the achievable gamma), bisection cannot start.
356 // Increase gammaInit until solve() returns feasible = true, then tighten.
357 // -----------------------------------------------------------------------
358 static GeneralisedPlant build(const StateSpace &G,
359 const StateSpace &W1,
360 const StateSpace &W2,
361 const StateSpace &W3);
362};
