8// Mamdani / Takagi-Sugeno fuzzy inference engine.
9//
10// -- Architecture -------------------------------------------------------------
11//
12// 1. INPUT VARIABLES - each partitioned into N linguistic terms.
13// Membership functions: triangular, trapezoidal, Gaussian, singleton.
14//
15// 2. RULE BASE - list of if-then rules in the form:
16// IF x1 is A_i AND x2 is B_j ... THEN y is C_k
17// Rules are evaluated using product (t-norm) for AND.
18//
19// 3. INFERENCE METHODS
20// a) Mamdani - output fuzzy sets are clipped/scaled, then aggregated.
21// Defuzzification: Centre of Gravity (CoG) over a discrete universe.
22// b) Takagi-Sugeno (TS) - consequent is a crisp function of inputs (or
23// a singleton value per rule). Defuzzification: weighted average.
24//
25// 4. OUTPUT - single real value (for SISO use) or vector (MIMO via
26// multiple FuzzySystem instances, one per output axis).
27//
28// -- SISO IController wrapper --------------------------------------------------
29// FuzzyController wraps one FuzzySystem in the IController interface so it can
30// be used wherever DiscretePID, DiscreteSMC, etc. are used.
31//
32// -- Terminology ---------------------------------------------------------------
33// MF - membership function
34// crisp - ordinary real-valued number (vs. a fuzzy set)
35// universe - the domain of an input or output variable
36//
37// Ref: Mamdani & Assilian (1975); Takagi & Sugeno (1985);
38// Jang, Sun, Mizutani "Neuro-Fuzzy and Soft Computing" (1997);
39// MATLAB Fuzzy Logic Toolbox documentation.

44// ============================================================================
45// Membership function types
46// ============================================================================
47
48// MF - value-type tagged-union membership function.
49//
50// Replaces std::function<double(double)> to eliminate heap allocation in the
51// RT inference hot path. std::function stores closures on the heap for
52// captures > ~16 bytes (implementation-defined small-buffer threshold), making
53// it unsafe for hard-RT use. MF is a plain struct (sizeof <= 5 doubles) that
54// is trivially copyable and stack-allocatable.
55//
56// Factory functions (mfTriangular, mfTrapezoidal, etc.) return a properly typed
57// MF - the public API is identical to the old std::function interface except
58// that MF is now a concrete value type, not a type-erased callable.
59struct MF {
60 enum class Type {
61 Triangular, // /\ peaks at p[1]=c, zero at p[0]=a and p[2]=b
62 Trapezoidal, // /‾\ flat-top from p[1]=b to p[2]=c, zero outside [p[0]=a, p[3]=d]
63 Gaussian, // exp(-0.5*((x-p[0])/p[1])^2)
64 Singleton, // 1 at x==p[0], 0 elsewhere
65 ShoulderLeft, // 1 for x<=p[0], linear 1->0 from p[0]=a to p[1]=b
66 ShoulderRight // linear 0->1 from p[0]=a to p[1]=b, 1 for x>=p[1]
67 };
68
69 Type type;
70 double p[4] = {}; // parameters; unused slots are zero
71
72 // Evaluate the MF at x. Inlined for zero-overhead in evaluate().
73 double operator()(double x) const;
74};
75

259// ============================================================================
260// FuzzyPID - extends FuzzyPD with a fuzzy integral term.
261//
262// Architecture: three decoupled fuzzy inference blocks:
263// u_P = FuzzyPD(e, de) - proportional + derivative
264// u_I = k_i * integral(e) - crisp accumulator with anti-windup
265// u = clamp(u_P + u_I, uMin, uMax)
266//
267// The integral is identical to the PID back-calculation anti-windup scheme
268// (Ki * Ts * e + Kb * (u_sat - u_unsat)), keeping it numerically safe.
269// ============================================================================
270
271struct FuzzyPIDParams {
272 FuzzyPDParams pd; // FuzzyPD tuning (e_scale, de_scale, u_scale)
273 double Ki = 0.0; // Integral gain (0 = pure FuzzyPD)
274 double Kb = 1.0; // Anti-windup back-calculation gain
275 double uMin = -1e9;
276 double uMax = 1e9;
277};

305// ============================================================================
306// FuzzySupervisor - monitors closed-loop performance and decides whether the
307// underlying controller's linearised plant model needs to be refreshed.
308//
309// Decision logic (Mamdani fuzzy, 2 inputs -> 1 output):
310//
311// Input 1: normalised_error = |e| / e_threshold
312// Terms: Small (0-0.4), Medium (0.3-0.7), Large (0.6-1+)
313//
314// Input 2: error_trend = d|e|/dt / trend_threshold (positive = diverging)
315// Terms: Decreasing (<0), Steady (~0), Increasing (>0)
316//
317// Output: relinearize_signal in [0, 1]
318// 0 = no action, 1 = force relinearisation NOW
319//
320// Representative rules:
321// IF error is Small AND trend is Decreasing THEN signal is None (0.0)
322// IF error is Small AND trend is Steady THEN signal is None (0.0)
323// IF error is Medium AND trend is Increasing THEN signal is Maybe (0.5)
324// IF error is Large AND trend is Steady THEN signal is High (0.8)
325// IF error is Large AND trend is Increasing THEN signal is Full (1.0)
326// (full 9-rule table built internally)
327//
328// Usage pattern in simulation:
329// auto sig = supervisor.update(error_norm, delta_t);
330// if (sig.relinearize) mpc.setPlant(relinearize(plant, current_state));
331//
332// ============================================================================
333
334struct SupervisorParams {
335 double e_threshold = 5.0; // |e| at which error is considered "Large" (same units as error)
336 double trend_threshold = 0.5; // d|e|/dt considered "Increasing fast" (units/s)
337 double signal_threshold = 0.5; // relinearize_signal > this -> trigger action
338 // Hysteresis: once triggered, don't trigger again for cooldown_steps
339 int cooldown_steps = 20;
340};
341
342struct SupervisorDecision {
343 double relinearize_signal; // raw fuzzy output [0, 1]
344 bool relinearize; // threshold decision
345 double error_norm;
346 double trend;
347};
