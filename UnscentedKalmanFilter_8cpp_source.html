9 int n, int p,
10 std::function<Eigen::VectorXd(const Eigen::VectorXd &, const Eigen::VectorXd &)> f,
11 std::function<Eigen::VectorXd(const Eigen::VectorXd &, const Eigen::VectorXd &)> h,
12 const Eigen::MatrixXd &Q_noise,
13 const Eigen::MatrixXd &R_noise,
14 double Ts,
15 const Eigen::MatrixXd &P0,
16 double alpha, double beta, double kappa)
17 : n_states_(n), n_outputs_(p), f_(std::move(f)), h_(std::move(h)),
18 Q_(Q_noise), R_(R_noise), Ts_(Ts)
19 {
20 x_hat_ = Eigen::VectorXd::Zero(n);
21 P_ = P0.rows() == n ? P0 : Eigen::MatrixXd::Identity(n, n);
22
23 // Scaled UT parameters
24 lambda_ = alpha * alpha * (n + kappa) - n;
25
26 // Guard: (n + lambda) must be positive for the Cholesky sqrt in sigmaPoints()
27 // to be well-defined. Equivalent condition: alpha^2*(n+kappa) > 0.
28 if (n + lambda_ <= 0.0)
29 throw std::invalid_argument(
30 "[UnscentedKalmanFilter] n + lambda <= 0: sigma point spread is invalid. "
31 "Increase alpha or set kappa >= 0. "
32 "Typical values: alpha=1e-3, kappa=0 or kappa=3-n.");
33
34 const int L = 2 * n + 1;
35 Wm_.resize(L);
36 Wc_.resize(L);
37 Wm_(0) = lambda_ / (n + lambda_);
38 Wc_(0) = lambda_ / (n + lambda_) + (1.0 - alpha * alpha + beta);
39
40 // Warn when Wc(0) < 0: the zeroth covariance weight is negative, which is
41 // mathematically valid in the sigma-point formulation but can cause P to lose
42 // positive-semidefiniteness numerically. Consider increasing beta or alpha.
43#ifndef NDEBUG
44 if (Wc_(0) < 0.0)
45 std::cerr << "[UnscentedKalmanFilter] WARNING: Wc(0) = " << Wc_(0)
46 << " < 0. Predicted covariance may lose PSD. "
47 << "Consider increasing beta (default 2) or alpha.\n";
48#endif
49
50 const double w_rest = 0.5 / (n + lambda_);
51 for (int i = 1; i < L; ++i)
52 Wm_(i) = Wc_(i) = w_rest;
53 }

55 // Cholesky of (n+lambda)P, columns are the perturbation vectors.
56 Eigen::MatrixXd UnscentedKalmanFilter::sigmaPoints() const
57 {
58 const Eigen::MatrixXd S =
59 ((n_states_ + lambda_) * P_).llt().matrixL(); // lower triangular sqrt
60
61 Eigen::MatrixXd X(n_states_, 2 * n_states_ + 1);
62 X.col(0) = x_hat_;
63 for (int i = 0; i < n_states_; ++i)
64 {
65 X.col(i + 1) = x_hat_ + S.col(i);
66 X.col(i + 1 + n_states_) = x_hat_ - S.col(i);
67 }
68 return X;

71 void UnscentedKalmanFilter::predict(const Eigen::VectorXd &u)
72 {
73 const Eigen::MatrixXd X = sigmaPoints();
74 const int L = 2 * n_states_ + 1;
75
76 // Propagate sigma points through f
77 Eigen::MatrixXd Xp(n_states_, L);
78 for (int i = 0; i < L; ++i)
79 Xp.col(i) = f_(X.col(i), u);
80
81 // Predicted mean
82 x_hat_.setZero();
83 for (int i = 0; i < L; ++i)
84 x_hat_.noalias() += Wm_(i) * Xp.col(i);
85
86 // Predicted covariance - outer-product accumulation; noalias() avoids per-iteration temps.
87 P_ = Q_;
88 for (int i = 0; i < L; ++i)
89 {
90 const Eigen::VectorXd d = Xp.col(i) - x_hat_;
91 P_.noalias() += Wc_(i) * d * d.transpose();
92 }
93 }

95 void UnscentedKalmanFilter::update(const Eigen::VectorXd &y,
96 const Eigen::VectorXd &u)
97 {
98 const Eigen::MatrixXd X = sigmaPoints();
99 const int L = 2 * n_states_ + 1;
100
101 // Propagate sigma points through h
102 Eigen::MatrixXd Y(n_outputs_, L);
103 for (int i = 0; i < L; ++i)
104 Y.col(i) = h_(X.col(i), u);
105
106 // Predicted measurement mean
107 Eigen::VectorXd y_hat = Eigen::VectorXd::Zero(n_outputs_);
108 for (int i = 0; i < L; ++i)
109 y_hat.noalias() += Wm_(i) * Y.col(i);
110
111 Eigen::MatrixXd R_safe = R_;
112 R_safe.diagonal() = R_safe.diagonal().cwiseMax(1e-12);
113
114 // Innovation covariance Syy and cross-covariance Pxy
115 Eigen::MatrixXd Syy = R_safe;
116 Eigen::MatrixXd Pxy = Eigen::MatrixXd::Zero(n_states_, n_outputs_);
117 for (int i = 0; i < L; ++i)
118 {
119 const Eigen::VectorXd dx = X.col(i) - x_hat_;
120 const Eigen::VectorXd dy = Y.col(i) - y_hat;
121 Syy.noalias() += Wc_(i) * dy * dy.transpose();
122 Pxy.noalias() += Wc_(i) * dx * dy.transpose();
123 }
124
125 const auto ldlt = Syy.ldlt();
126 if (ldlt.info() != Eigen::Success)
127 return;
128
129 const Eigen::MatrixXd K = Pxy * ldlt.solve(Eigen::MatrixXd::Identity(n_outputs_, n_outputs_));
130
131 x_hat_.noalias() += K * (y - y_hat);
132
133 // Covariance update: P = P - K*Syy*K'
134 // Unlike the linear KF (which uses the numerically superior Joseph form),
135 // the UKF Joseph form requires an explicit H matrix that does not exist here -
136 // H is replaced by the sigma-point cross-covariance Pxy/Syy. The raw
137 // subtraction is the standard UKF formula (Wan & Van der Merwe 2000, eq. 26).
138 // Round-off in this subtraction can cause P to lose PSD on large measurement
139 // updates; the eigenvalue floor below catches this case.
140 P_.noalias() -= K * Syy * K.transpose();
141
142 // Enforce PSD: symmetrise first, then clamp any negative eigenvalue to zero.
143 // Symmetrisation alone is insufficient - a negative diagonal entry survives it.
144 P_ = 0.5 * (P_ + P_.transpose());
145 Eigen::SelfAdjointEigenSolver<Eigen::MatrixXd> eig(P_);
146 if (eig.eigenvalues().minCoeff() < 0.0)
147 P_ += (-eig.eigenvalues().minCoeff() + 1e-10) *
148 Eigen::MatrixXd::Identity(n_states_, n_states_);
149 }
