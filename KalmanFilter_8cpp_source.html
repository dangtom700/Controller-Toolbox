25 // Update: incorporate measurement y[k], correct estimate, deflate covariance.
26 // Joseph form for P update maintains positive semi-definiteness numerically.
27 void KalmanFilter::update(const Eigen::VectorXd &y,
28 const Eigen::VectorXd &u_current)
29 {
30 const Eigen::MatrixXd &C = plant_.C;
31 const Eigen::MatrixXd &D = plant_.D;
32 const int p = plant_.outputSize();
33 const int n = plant_.stateSize();
34
35 // Enforce minimum noise floor on R to prevent near-singular innovation covariance
36 Eigen::MatrixXd R_safe = R_;
37 R_safe.diagonal() = R_safe.diagonal().cwiseMax(1e-12);
38
39 // Innovation covariance
40 const Eigen::MatrixXd S = C * P_ * C.transpose() + R_safe;
41
42 // Kalman gain - skip update entirely if S is numerically singular
43 const auto ldlt = S.ldlt();
44 if (ldlt.info() != Eigen::Success)
45 return;
46 const Eigen::MatrixXd Kf = P_ * C.transpose() *
47 ldlt.solve(Eigen::MatrixXd::Identity(p, p));
48
49 // State update
50 const Eigen::VectorXd innov = y - C * x_hat_ - D * u_current;
51 x_hat_.noalias() += Kf * innov;
52
53 // Covariance update - Joseph form: P = (I-KC).P.(I-KC)' + K.R.K'
54 const Eigen::MatrixXd IKC = Eigen::MatrixXd::Identity(n, n) - Kf * C;
55 P_ = IKC * P_ * IKC.transpose() + Kf * R_safe * Kf.transpose();
56 }
