21void RecursiveGreyBoxEstimator::initialize(const Eigen::VectorXd& x0,
22 const Eigen::MatrixXd& P0_state)
23{
24 const int n_aug = n_state_ + n_param_;
25
26 // ---- Build augmented noise matrices ------------------------------------
27 Eigen::MatrixXd Q_aug = Eigen::MatrixXd::Zero(n_aug, n_aug);
28 Q_aug.topLeftCorner(n_state_, n_state_) = par_.Q_state;
29 Q_aug.bottomRightCorner(n_param_, n_param_) = par_.Q_param;
30
31 Eigen::MatrixXd P0_p = (par_.P0_param.rows() == n_param_)
32 ? par_.P0_param
33 : (100.0 * par_.Q_param);
34 Eigen::MatrixXd P0_aug = Eigen::MatrixXd::Zero(n_aug, n_aug);
35 P0_aug.topLeftCorner(n_state_, n_state_) = P0_state;
36 P0_aug.bottomRightCorner(n_param_, n_param_) = P0_p;
37
38 // ---- Augmented process function ----------------------------------------
39 // z = [x; p]. Dynamics: integrate f(x,u,p) via RK4; p stays constant.
40 const int n_x = n_state_;
41 const int n_p = n_param_;
42 const double Ts = par_.Ts;
43 const int rk4s = par_.rk4_steps;
44 OdeFn f_copy = f_;
45
46 // Pre-allocate RK4 scratch inside the closure to avoid per-sigma-point heap alloc
47 auto f_aug = [f_copy, n_x, n_p, Ts, rk4s,
48 x_scr = Eigen::VectorXd(n_x),
49 p_scr = Eigen::VectorXd(n_p),
50 k1 = Eigen::VectorXd(n_x),
51 k2 = Eigen::VectorXd(n_x),
52 k3 = Eigen::VectorXd(n_x),
53 k4 = Eigen::VectorXd(n_x),
54 z_next = Eigen::VectorXd(n_x + n_p)](
55 const Eigen::VectorXd& z,
56 const Eigen::VectorXd& u) mutable -> Eigen::VectorXd
57 {
58 x_scr = z.head(n_x);
59 p_scr = z.tail(n_p);
60 const double h = Ts / rk4s;
61 for (int s = 0; s < rk4s; ++s) {
62 k1 = f_copy(x_scr, u, p_scr);
63 k2 = f_copy(x_scr + 0.5*h*k1, u, p_scr);
64 k3 = f_copy(x_scr + 0.5*h*k2, u, p_scr);
65 k4 = f_copy(x_scr + h*k3, u, p_scr);
66 x_scr += (h / 6.0) * (k1 + 2.0*k2 + 2.0*k3 + k4);
67 }
68 z_next.head(n_x) = x_scr;
69 z_next.tail(n_p) = p_scr;
70 return z_next;
71 };
72
73 // ---- Augmented measurement function ------------------------------------
74 MeasFn h_copy = h_;
75 auto h_aug = [h_copy, n_x, n_p](
76 const Eigen::VectorXd& z,
77 const Eigen::VectorXd& /*u*/) -> Eigen::VectorXd
78 {
79 return h_copy(z.head(n_x), z.tail(n_p));
80 };
81
82 // ---- Build UKF ---------------------------------------------------------
83 ukf_ = std::make_unique<UnscentedKalmanFilter>(
84 n_aug, n_meas_,
85 std::move(f_aug), std::move(h_aug),
86 Q_aug, par_.R_meas,
87 Ts, P0_aug,
88 par_.alpha, par_.beta, par_.kappa);
89
90 // Set initial augmented state [x0; p0]
91 Eigen::VectorXd z0(n_aug);
92 z0.head(n_state_) = x0;
93 z0.tail(n_param_) = par_.p0;
94 ukf_->setState(z0);
95
96 initialized_ = true;
97}
