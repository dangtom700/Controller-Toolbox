70struct CUSUMChart {
71 double target = 0.0;
72 double sigma = 1.0;
73 double k = 0.5;
74 double h = 5.0;
75
76 double C_plus = 0.0;
77 double C_minus = 0.0;
78
79 void reset() { C_plus = C_minus = 0.0; }
80
85 bool update(double x) {
86 const double k_abs = k * sigma;
87 const double h_abs = h * sigma;
88 C_plus = std::max(0.0, C_plus + (x - target) - k_abs);
89 C_minus = std::max(0.0, C_minus - (x - target) - k_abs);
90 return (C_plus >= h_abs) || (C_minus >= h_abs);
91 }
92
93 double statistic() const { return std::max(C_plus, C_minus); }
94};

105struct EWMAChart {
106 double target = 0.0;
107 double sigma = 1.0;
108 double lambda = 0.2;
109 double L = 3.0;
110
111 double Z = 0.0;
112
113 void reset() { Z = target; }
114
119 bool update(double x) {
120 Z = lambda * x + (1.0 - lambda) * Z;
121 const double hw = L * sigma * std::sqrt(lambda / (2.0 - lambda));
122 return std::abs(Z - target) >= hw;
123 }
124
125 double statistic() const { return Z; }
126};
