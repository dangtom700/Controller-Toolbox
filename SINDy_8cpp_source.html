12int SINDy::computeNTerms() const
13{
14 const int n = p_.n_state;
15 const int m = p_.n_input;
16 const int d = n + m; // combined dimension for polynomial terms
17
18 switch (p_.library) {
19 case SINDyLibrary::PolyDeg1:
20 // 1 (constant) + d (linear)
21 return 1 + d;
22
23 case SINDyLibrary::PolyDeg2:
24 // 1 + d + d*(d+1)/2 (all monomials up to degree 2)
25 return 1 + d + d * (d + 1) / 2;
26
27 case SINDyLibrary::PolyDeg3:
28 // degree 0 + degree 1 + degree 2 + degree 3 terms
29 // C(d+3,3) via stars-and-bars
30 return (d + 1) * (d + 2) * (d + 3) / 6;
31
32 case SINDyLibrary::PolyTrig:
33 // PolyDeg2 terms + sin(x_i) + cos(x_i) for each state
34 return 1 + d + d * (d + 1) / 2 + 2 * n;
35
36 default:
37 return 1 + d + d * (d + 1) / 2;
38 }
39}

56Eigen::VectorXd SINDy::libraryRow(const Eigen::VectorXd& x,
57 const Eigen::VectorXd& u) const
58{
59 const int n = p_.n_state;
60 const int m = p_.n_input;
61 Eigen::VectorXd row(n_terms_);
62 int idx = 0;
63
64 // Combine [x; u] into a single vector z for polynomial generation
65 Eigen::VectorXd z(n + m);
66 z.head(n) = x;
67 if (m > 0) z.tail(m) = u;
68 const int d = n + m;
69
70 // --- Degree 0: constant 1 ---
71 row(idx++) = 1.0;
72
73 // --- Degree 1: z_i ---
74 for (int i = 0; i < d; ++i)
75 row(idx++) = z(i);
76
77 if (p_.library == SINDyLibrary::PolyDeg1) {
78 return row;
79 }
80
81 // --- Degree 2: z_i * z_j (i <= j) ---
82 for (int i = 0; i < d; ++i)
83 for (int j = i; j < d; ++j)
84 row(idx++) = z(i) * z(j);
85
86 if (p_.library == SINDyLibrary::PolyDeg2) {
87 return row;
88 }
89
90 if (p_.library == SINDyLibrary::PolyTrig) {
91 // Trig: sin(x_i), cos(x_i) for each state component
92 for (int i = 0; i < n; ++i)
93 row(idx++) = std::sin(x(i));
94 for (int i = 0; i < n; ++i)
95 row(idx++) = std::cos(x(i));
96 return row;
97 }
98
99 // --- Degree 3: z_i * z_j * z_k (i <= j <= k) ---
100 for (int i = 0; i < d; ++i)
101 for (int j = i; j < d; ++j)
102 for (int k = j; k < d; ++k)
103 row(idx++) = z(i) * z(j) * z(k);
104
105 return row;
106}

133// ---------------------------------------------------------------------------
134// Sequentially-Thresholded Least Squares (STLS)
135// ---------------------------------------------------------------------------
136
137Eigen::MatrixXd SINDy::stls(const Eigen::MatrixXd& Theta,
138 const Eigen::MatrixXd& Xdot) const
139{
140 const int K = static_cast<int>(Theta.rows()); // snapshots
141 const int L = static_cast<int>(Theta.cols()); // library terms
142 const int S = static_cast<int>(Xdot.cols()); // states
143
144 // Initial least-squares estimate (possibly dense)
145 Eigen::MatrixXd Xi = Theta.jacobiSvd(Eigen::ComputeThinU | Eigen::ComputeThinV)
146 .solve(Xdot);
147
148 if (p_.use_ols)
149 return Xi;
150
151 // STLS: iterate threshold + refit on active terms
152 for (int iter = 0; iter < p_.stls_iter; ++iter) {
153 // Threshold small coefficients
154 for (int s = 0; s < S; ++s)
155 for (int l = 0; l < L; ++l)
156 if (std::abs(Xi(l, s)) < p_.threshold)
157 Xi(l, s) = 0.0;
158
159 // Refit each state using only the active (non-zero) terms
160 for (int s = 0; s < S; ++s) {
161 // Find active indices for column s
162 std::vector<int> active;
163 for (int l = 0; l < L; ++l)
164 if (Xi(l, s) != 0.0)
165 active.push_back(l);
166
167 if (active.empty())
168 continue;
169
170 // Sub-matrix of active columns
171 Eigen::MatrixXd Theta_a(K, static_cast<int>(active.size()));
172 for (int j = 0; j < static_cast<int>(active.size()); ++j)
173 Theta_a.col(j) = Theta.col(active[j]);
174
175 // Least-squares on reduced system
176 Eigen::VectorXd xi_a =
177 Theta_a.jacobiSvd(Eigen::ComputeThinU | Eigen::ComputeThinV)
178 .solve(Xdot.col(s));
179
180 // Write back, zeroing inactive
181 Xi.col(s).setZero();
182 for (int j = 0; j < static_cast<int>(active.size()); ++j)
183 Xi(active[j], s) = xi_a(j);
184 }
185 }
186 return Xi;
187}

193SINDyModel SINDy::fit() const
194{
195 if (theta_rows_.empty())
196 throw std::runtime_error("SINDy::fit(): no snapshots have been added");
197
198 const int K = static_cast<int>(theta_rows_.size());
199 const int L = n_terms_;
200 const int S = p_.n_state;
201
202 // Assemble matrices
203 Eigen::MatrixXd Theta(K, L);
204 Eigen::MatrixXd Xdot(K, S);
205 for (int k = 0; k < K; ++k) {
206 Theta.row(k) = theta_rows_[k].transpose();
207 Xdot .row(k) = xdot_rows_ [k].transpose();
208 }
209
210 // Solve
211 Eigen::MatrixXd Xi = stls(Theta, Xdot); // L * S
212
213 // Package result
214 SINDyModel model;
215 model.n_state_ = p_.n_state;
216 model.n_input_ = p_.n_input;
217 model.lib_type_ = p_.library;
218 model.xi_ = Xi;
219 return model;
220}

233 p_tmp.n_state = n_state_;
234 p_tmp.n_input = n_input_;
235 p_tmp.library = lib_type_;
236 SINDy helper(p_tmp);
