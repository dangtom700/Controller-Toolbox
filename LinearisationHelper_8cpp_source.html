10static Eigen::MatrixXd numJacImpl(
11 const std::function<Eigen::VectorXd(const Eigen::VectorXd &)> &func,
12 const Eigen::VectorXd &x0,
13 double eps_scale)
14{
15 const Eigen::VectorXd f0 = func(x0);
16 const int nx = static_cast<int>(x0.size());
17 const int ny = static_cast<int>(f0.size());
18
19 Eigen::MatrixXd J(ny, nx);
20 Eigen::VectorXd xp = x0, xm = x0;
21
22 for (int i = 0; i < nx; ++i)
23 {
24 const double h = eps_scale * std::max(std::abs(x0(i)), 1.0);
25 xp(i) = x0(i) + h;
26 xm(i) = x0(i) - h;
27 J.col(i) = (func(xp) - func(xm)) / (2.0 * h);
28 xp(i) = x0(i);
29 xm(i) = x0(i);
30 }
31 return J;
32}
