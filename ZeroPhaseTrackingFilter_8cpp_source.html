16// Multiply polynomial (coefficients in descending z powers) by (z - root).
17static std::vector<std::complex<double>> polyMulRoot(
18 const std::vector<std::complex<double>> &p,
19 std::complex<double> root)
20{
21 // p = [p_0, p_1, ..., p_n] representing p_0*z^n + p_1*z^{n-1} + ... + p_n
22 // result = z*p - root*p -> degree n+1
23 std::vector<std::complex<double>> result(p.size() + 1, 0.0);
24 for (std::size_t i = 0; i < p.size(); ++i)
25 {
26 result[i] += p[i]; // z * p[i] shifted up
27 result[i + 1] -= root * p[i]; // -root * p[i]
28 }
29 return result;
30}

32// Build monic polynomial from its roots: [1, c_{n-1}, ..., c_0] in descending z powers.
33// Returns the real part (imaginary cancels for conjugate-pair root sets).
34static Eigen::VectorXd rootsToPoly(const std::vector<std::complex<double>> &roots)
35{
36 std::vector<std::complex<double>> poly = {std::complex<double>(1.0, 0.0)};
37 for (const auto &r : roots)
38 poly = polyMulRoot(poly, r);
39
40 Eigen::VectorXd real_poly(static_cast<int>(poly.size()));
41 for (int i = 0; i < static_cast<int>(poly.size()); ++i)
42 real_poly(i) = poly[static_cast<std::size_t>(i)].real();
43 return real_poly;
44}

46// Evaluate SISO transfer function G(z) = C*(zI-A)^{-1}*B + D at complex z.
47static std::complex<double> evalTF(const StateSpace &sys, std::complex<double> z)
48{
49 const int n = sys.stateSize();
50 const Eigen::MatrixXcd zIA = z * Eigen::MatrixXcd::Identity(n, n)
51 - sys.A.cast<std::complex<double>>();
52 const Eigen::MatrixXcd B_c = sys.B.cast<std::complex<double>>(); // n * m
53 const Eigen::MatrixXcd C_c = sys.C.cast<std::complex<double>>(); // p * n (keep as matrix)
54 const Eigen::MatrixXcd D_c = sys.D.cast<std::complex<double>>(); // p * m
55 // G(z) = C*(zI-A)^{-1}*B + D -> (p*n) * (n*n)^{-1} * (n*m) + (p*m) = (p*m)
56 const Eigen::MatrixXcd G = C_c * zIA.colPivHouseholderQr().solve(B_c) + D_c;
57 return G(0, 0);
58}

64std::vector<std::complex<double>> transmissionZeros(const StateSpace &sys)
65{
66 if (sys.inputSize() != 1 || sys.outputSize() != 1)
67 throw std::invalid_argument("transmissionZeros: only SISO systems are supported.");
68
69 const int n = sys.stateSize();
70 // System matrix M = [[A, B], [C, D]] and E = [[I, 0], [0, 0]]
71 // Transmission zeros = finite eigenvalues of the pencil M*v = lambda*E*v
72 // i.e., solve det([[A-lambdaI, B],[C, D]]) = 0
73
74 const int total = n + 1;
75 Eigen::MatrixXd M(total, total), E = Eigen::MatrixXd::Zero(total, total);
76
77 M.topLeftCorner(n, n) = sys.A;
78 M.topRightCorner(n, 1) = sys.B;
79 M.bottomLeftCorner(1, n) = sys.C;
80 M(n, n) = sys.D(0, 0);
81
82 E.topLeftCorner(n, n) = Eigen::MatrixXd::Identity(n, n);
83 // E(n,n) = 0 (already) - makes the pencil detect zeros at finite eigenvalues
84
85 Eigen::GeneralizedEigenSolver<Eigen::MatrixXd> ges(M, E);
86
87 std::vector<std::complex<double>> zeros;
88 zeros.reserve(n);
89 for (int i = 0; i < total; ++i)
90 {
91 const double beta = std::abs(ges.betas()(i));
92 if (beta > 1e-10)
93 zeros.push_back(ges.alphas()(i) / ges.betas()(i));
94 // beta approx = 0 -> infinite eigenvalue -> not a finite transmission zero
95 }
96 return zeros;
97}

99ZPETCResult designZPETC(const StateSpace &plant)
100{
101 if (plant.inputSize() != 1 || plant.outputSize() != 1)
102 throw std::invalid_argument("designZPETC: only SISO plants are supported.");
103
104 const int n = plant.stateSize();
105
106 // --- Transmission zeros and partition ---
107 const auto all_zeros = transmissionZeros(plant);
108
109 std::vector<std::complex<double>> min_phase_zeros, nmp_zeros;
110 const double nmp_threshold = 1.0 - 1e-8;
111 for (const auto &z : all_zeros)
112 {
113 if (std::abs(z) < nmp_threshold)
114 min_phase_zeros.push_back(z);
115 else
116 nmp_zeros.push_back(z);
117 }
118
119 const bool has_nmp = !nmp_zeros.empty();
120 const int m_plus = static_cast<int>(min_phase_zeros.size()); // degree of B+(z)
121
122 // --- Numerator gain: first non-zero Markov parameter of the plant ---
123 // For G(z) = B(z)/A(z), b_gain = leading coefficient of B(z) in z-polynomial form.
124 // = D (for bi-proper plant) or first non-zero C*A^k*B (for strictly proper).
125 double b_gain = plant.D(0, 0);
126 if (std::abs(b_gain) < 1e-10)
127 {
128 Eigen::VectorXd Ak_B = plant.B;
129 for (int k = 0; k < n; ++k)
130 {
131 b_gain = (plant.C * Ak_B)(0, 0);
132 if (std::abs(b_gain) > 1e-10) break;
133 Ak_B = plant.A * Ak_B;
134 }
135 }
136 if (std::abs(b_gain) < 1e-10)
137 b_gain = 1.0; // fallback for all-pole plant
138
139 // --- DC normalisation: B^-(1) = Pi (1 - z_i) for NMP zeros ---
140 std::complex<double> B_minus_dc(1.0, 0.0);
141 for (const auto &z : nmp_zeros)
142 B_minus_dc *= (1.0 - z);
143 // B-(1) is real and positive for real conjugate-pair zeros
144 const double dc_norm = (std::abs(B_minus_dc) < 1e-12) ? 1.0 : B_minus_dc.real();
145
146 // --- Denominator polynomial: B+(z) * z^{n - m_plus} ---
147 // Start with B+(z) from min-phase zeros
148 Eigen::VectorXd B_plus_coeffs = rootsToPoly(min_phase_zeros);
149 // B_plus_coeffs has length m_plus + 1 (degree m_plus)
150
151 // Pad with n - m_plus trailing zeros to represent multiplication by z^{n-m_plus}
152 const int n_pad = n - m_plus;
153 Eigen::VectorXd den_coeffs(B_plus_coeffs.size() + n_pad);
154 den_coeffs.head(static_cast<int>(B_plus_coeffs.size())) = B_plus_coeffs;
155 den_coeffs.tail(n_pad).setZero();
156 // den_coeffs now represents B+(z) * z^{n-m_plus}, degree = n
157
158 // --- Numerator polynomial: K * A(z) ---
159 // A(z) = characteristic polynomial of plant.A = Pi(z - lambda_i)
160 const auto plant_poles_complex = plant.A.eigenvalues();
161 std::vector<std::complex<double>> poles_vec(plant_poles_complex.data(),
162 plant_poles_complex.data() + n);
163 Eigen::VectorXd A_coeffs = rootsToPoly(poles_vec); // [1, a_1, ..., a_n], degree n
164
165 // K normalises both the plant gain and the NMP DC contribution
166 const double K = 1.0 / (b_gain * dc_norm);
167 Eigen::VectorXd num_coeffs = K * A_coeffs; // degree n
168
169 // Both num and den have degree n -> bi-proper filter (D != 0 via tf2ss)
170
171 // --- Build TransferFunction struct (TF stores coefficients in descending z powers,
172 // denominator must be monic with den[0] = 1) ---
173 // den_coeffs[0] = B_plus_coeffs[0] (leading coefficient of B+(z))
174 // Normalise denominator to be monic:
175 const double den_lead = den_coeffs(0);
176 if (std::abs(den_lead) < 1e-12)
177 {
178 // All zeros are NMP - trivial min-phase inverse = constant gain
179 // Build a simple gain filter: G_ff(z) = K
180 StateSpace gain_filter(
181 Eigen::MatrixXd::Zero(1, 1),
182 Eigen::MatrixXd::Zero(1, 1),
183 Eigen::MatrixXd::Zero(1, 1),
184 Eigen::MatrixXd::Constant(1, 1, K),
185 plant.Ts);
186 return {gain_filter, 0.0, has_nmp, all_zeros, nmp_zeros};
187 }
188
189 Eigen::VectorXd den_norm = den_coeffs / den_lead;
190 Eigen::VectorXd num_norm = num_coeffs / den_lead; // scale num proportionally
191
192 // Convert Eigen vectors to std::vector for TransferFunction constructor
193 const int poly_len = static_cast<int>(num_norm.size());
194 std::vector<double> num_vec(num_norm.data(), num_norm.data() + poly_len);
195 std::vector<double> den_vec(den_norm.data(), den_norm.data() + poly_len);
196
197 TransferFunction ff_tf(num_vec, den_vec, plant.Ts);
198 StateSpace filter = tf2ss(ff_tf);
199
200 // --- Compute DC amplitude error ---
201 // Evaluate |G(e^{jomega}) * G_ff(e^{jomega}) - 1| at 50 frequencies
202 double max_amp_err = 0.0;
203 const int N_freq = 50;
204 for (int i = 0; i < N_freq; ++i)
205 {
206 const double omega = (i + 1.0) * M_PI / (N_freq + 1.0) / plant.Ts;
207 const std::complex<double> z = std::exp(std::complex<double>(0.0, omega * plant.Ts));
208 const std::complex<double> G_val = evalTF(plant, z);
209 const std::complex<double> Gff_val = evalTF(filter, z);
210 max_amp_err = std::max(max_amp_err, std::abs(std::abs(G_val * Gff_val) - 1.0));
211 }
212
213 return {filter, max_amp_err, has_nmp, all_zeros, nmp_zeros};
214}
