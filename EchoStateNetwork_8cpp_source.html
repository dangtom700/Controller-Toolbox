11void EchoStateNetwork::initWeights()
12{
13 std::mt19937 rng(p_.seed);
14 std::uniform_real_distribution<double> u_dist(-1.0, 1.0);
15 std::bernoulli_distribution sparse_mask(1.0 - p_.sparsity);
16
17 // Sparse random reservoir W_res
18 W_res_ = Eigen::MatrixXd::Zero(p_.n_res, p_.n_res);
19 for (int i = 0; i < p_.n_res; ++i)
20 for (int j = 0; j < p_.n_res; ++j)
21 if (sparse_mask(rng))
22 W_res_(i, j) = u_dist(rng);
23
24 // Scale to target spectral radius
25 Eigen::EigenSolver<Eigen::MatrixXd> es(W_res_, false);
26 double rho = 0.0;
27 for (int i = 0; i < p_.n_res; ++i)
28 rho = std::max(rho, std::abs(es.eigenvalues()(i)));
29 if (rho > 1e-12)
30 W_res_ *= p_.spectral_radius / rho;
31
32 // Dense random input weights
33 W_in_ = Eigen::MatrixXd::Zero(p_.n_res, p_.n_in);
34 for (int i = 0; i < p_.n_res; ++i)
35 for (int j = 0; j < p_.n_in; ++j)
36 W_in_(i, j) = p_.input_scaling * u_dist(rng);
37
38 // Readout weights pre-allocated; fitReadout() resizes to n_out x n_res (reservoir only)
39 W_out_ = Eigen::MatrixXd::Zero(p_.n_out, p_.n_res);
40}

65void EchoStateNetwork::stepReservoir(const Eigen::VectorXd& u)
66{
67 // Leaky integration: r[k+1] = (1-alpha)*r[k] + alpha*tanh(W_res*r[k] + W_in*u[k])
68 const Eigen::VectorXd r_new = (W_res_ * r_ + W_in_ * u).array().tanh();
69 r_ = (1.0 - p_.alpha) * r_ + p_.alpha * r_new;
70}

88void EchoStateNetwork::fitReadout()
89{
90 if (states_.empty())
91 throw std::runtime_error("EchoStateNetwork::fitReadout(): no training data");
92
93 const int M = static_cast<int>(states_.size());
94 const int D = p_.n_res; // readout maps from reservoir state r_ only
95
96 Eigen::MatrixXd S(M, D);
97 Eigen::MatrixXd T(M, p_.n_out);
98 for (int k = 0; k < M; ++k) {
99 S.row(k) = states_ [k].transpose();
100 T.row(k) = targets_[k].transpose();
101 }
102
103 // Ridge: W_out' = (S'S + ridge*I)^{-1} S' T
104 Eigen::MatrixXd A = S.transpose() * S
105 + p_.ridge * Eigen::MatrixXd::Identity(D, D);
106 Eigen::MatrixXd rhs = S.transpose() * T;
107
108 Eigen::LDLT<Eigen::MatrixXd> ldlt(A);
109 Eigen::MatrixXd W_out_T = ldlt.solve(rhs); // D * n_out
110
111 // Store as n_out * D (resize W_out_ without the input part for simplified readout)
112 W_out_.resize(p_.n_out, D);
113 W_out_ = W_out_T.transpose();
114
115 fitted_ = true;
116}
