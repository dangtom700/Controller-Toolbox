27 // Full step: KF predict -> KF update -> LQR control.
28 //
29 // Causal ordering note: the Kalman update uses u_prev as its u_current argument.
30 // KalmanFilter::update() uses u_current only in the D*u feedthrough term of the
31 // innovation (y - C*x^ - D*u). Since u[k] has not been computed yet at update
32 // time, u[k-1] is the only available approximation. For D=0 plants (the standard
33 // case for ZOH-discretised models) the D*u term is zero and this has no effect.
34 // The constructor warns when D!=0; avoid D!=0 plants with LQG if accuracy matters.
35 Eigen::VectorXd DiscreteLQG::step(const Eigen::VectorXd &y,
36 const Eigen::VectorXd &u_prev,
37 const Eigen::VectorXd &x_ref)
38 {
39 // 1. Predict: x^[k|k-1] = A*x^[k-1|k-1] + B*u[k-1]
40 kf_->predict(u_prev);
41
42 // 2. Update: x^[k|k] from y[k]. u_prev is used for the D*u term - see note above.
43 kf_->update(y, u_prev);
44
45 // 3. LQR feedback on corrected state estimate
46 const Eigen::VectorXd &xhat = kf_->state();
47 Eigen::VectorXd ref = x_ref.size() == plant_.stateSize() ? x_ref : x_ref_;
48
49 Eigen::VectorXd u = lqr_->compute(xhat, ref);
50
51 // Store for internal convenience
52 u_prev_ = u;
53 return u;
54 }
