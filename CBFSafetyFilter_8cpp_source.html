20double CBFSafetyFilter::compute(double error)
21{
22 // Step 1: nominal controller output
23 u_nom_ = nominal_->compute(error);
24
25 // Step 2: evaluate CBF condition
26 h_val_ = h_(x_);
27 const double dh = dh_dx_(x_);
28 const double f0v = f0_(x_);
29 const double gv = g_(x_);
30
31 // CBF linear constraint (continuous): dh.(f0 + g.u) >= -alpha.(h + slack)
32 // => (dh.g).u >= -dh.f0 - alpha.(h + slack)
33 // Note: Euler discretisation: dh.(x_{k+1}-x_k)/Ts >= -alpha.h(x_k)
34 // => dh.(f0 + g.u) >= -alpha.h (continuous form, acceptable for small Ts)
35
36 const double A_cbf = dh * gv;
37 const double b_cbf = -dh * f0v - p_.alpha * (h_val_ + p_.slack);
38
39 // Step 3: 1D QP with analytical solution
40 // min (u - u_nom)^2 s.t. A_cbf * u >= b_cbf, u in [uMin, uMax]
41 double u_safe = u_nom_;
42 cbf_active_ = false;
43
44 if (std::abs(A_cbf) > 1e-12) {
45 // Minimum u satisfying CBF: u_min_cbf = b_cbf / A_cbf (if A_cbf > 0)
46 // u_max_cbf = b_cbf / A_cbf (if A_cbf < 0)
47 const double u_cbf = b_cbf / A_cbf;
48
49 if (A_cbf > 0.0 && u_nom_ < u_cbf) {
50 u_safe = u_cbf;
51 cbf_active_ = true;
52 } else if (A_cbf < 0.0 && u_nom_ > u_cbf) {
53 u_safe = u_cbf;
54 cbf_active_ = true;
55 }
56 }
57
58 // Hard box constraint
59 u_safe = std::clamp(u_safe, p_.uMin, p_.uMax);
60 notifyObserver(u_safe, x_);
61 return u_safe;
62}
