97// Model-based rollout (used by Python for policy improvement)
98// ---------------------------------------------------------------------------
99
100Eigen::VectorXd DynaController::modelRollout(double e0,
101 const Eigen::VectorXd& u_sequence) const
102{
103 if (!model_fitted_)
104 throw std::logic_error("DynaController::modelRollout: model not fitted - "
105 "call compute() for at least n_collect steps first");
106
107 const int n_steps = static_cast<int>(u_sequence.size());
108 Eigen::VectorXd e_traj(n_steps);
109
110 Eigen::VectorXd e(1); e(0) = e0;
111 Eigen::VectorXd u(1);
112
113 for (int k = 0; k < n_steps; ++k) {
114 u(0) = u_sequence(k);
115 Eigen::VectorXd edot = model_.predict(e, u);
116 e = e + p_.Ts * edot; // Euler forward step
117 e_traj(k) = e(0);
118 }
119 return e_traj;
120}
