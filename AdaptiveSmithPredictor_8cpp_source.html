33
34void AdaptiveSmithPredictor::estimateDelay()
35{
36 const int N = static_cast<int>(u_hist_.size());
37 const int tau_max = std::min(params_.maxDelaySteps, N - 1);
38 if (tau_max < 1) return; // not enough data
39
40 // Demean both buffers before cross-correlation.
41 // Without demeaning, R_uy(tau) is dominated by u_mean*y_mean*(N-tau),
42 // which always peaks at tau=0 regardless of the true delay.
43 double u_mean = 0.0, y_mean = 0.0;
44 for (int i = 0; i < N; ++i) { u_mean += u_hist_[i]; y_mean += y_hist_[i]; }
45 u_mean /= N;
46 y_mean /= N;
47
48 // Normalized cross-covariance: divide by (N-tau) so longer windows at small tau
49 // do not artificially inflate R at tau=0.
50 // rho(tau) = [1/(N-tau)] * sum_{k=tau}^{N-1} (u[k-tau]-u_mean) * (y[k]-y_mean)
51 double best_R = -1e300;
52 int best_tau = current_delay_;
53
54 for (int tau = 0; tau <= tau_max; ++tau)
55 {
56 const int n_terms = N - tau;
57 if (n_terms < 1) continue;
58 double R = 0.0;
59 for (int k = tau; k < N; ++k)
60 R += (u_hist_[k - tau] - u_mean) * (y_hist_[k] - y_mean);
61 R /= static_cast<double>(n_terms);
62
63 if (R > best_R)
64 {
65 best_R = R;
66 best_tau = tau;
67 }
68 }
69
70 if (best_tau != current_delay_)
71 {
72 current_delay_ = best_tau;
73 rebuildSP();
74 }
75}

81double AdaptiveSmithPredictor::compute(double error)
82{
83 // Delegate to the inner Smith Predictor.
84 const double u = sp_->compute(error);
85 last_output_ = u;
86
87 // Determine y for the correlation buffer.
88 // If setPlantOutput() was called this step, use it; otherwise approximate y approx = -error.
89 const double y_buf_val = has_y_ ? last_y_ : -error;
90 has_y_ = false;
91
92 // Append u[k-1] (the control applied at the previous step) and current y[k].
93 u_hist_.push_back(last_u_);
94 y_hist_.push_back(y_buf_val);
95
96 if (static_cast<int>(u_hist_.size()) > params_.bufferLen)
97 {
98 u_hist_.pop_front();
99 y_hist_.pop_front();
100 }
101
102 last_u_ = u; // store for next step
103 ++step_count_;
104
105 // Trigger delay estimation when buffer is sufficiently full.
106 const int min_buf = params_.maxDelaySteps + 10;
107 if (step_count_ % params_.estimateInterval == 0 &&
108 static_cast<int>(u_hist_.size()) >= min_buf)
109 {
110 estimateDelay();
111 }
112
113 notifyObserver(u, error);
114 return u;
115}
