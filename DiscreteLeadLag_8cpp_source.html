14 // Tustin bilinear discretisation of C(s) = K.(s+z_c)/(s+p_c):
15 //
16 // b0 = K.(2/Ts + z_c) / (2/Ts + p_c)
17 // b1 = K.(z_c - 2/Ts) / (2/Ts + p_c)
18 // a1 = (p_c - 2/Ts) / (2/Ts + p_c)
19 //
20 // y[k] = b0.u[k] + b1.u[k-1] - a1.y[k-1]
21 void DiscreteLeadLag::computeCoeffs()
22 {
23 const double two_over_Ts = 2.0 / Ts_;
24 const double denom = two_over_Ts + p_.continuousPole;
25 b0_ = p_.gain * (two_over_Ts + p_.continuousZero) / denom;
26 b1_ = p_.gain * (p_.continuousZero - two_over_Ts) / denom;
27 a1_ = (p_.continuousPole - two_over_Ts) / denom;
28 }

53 // Phase of continuous C(s) = K*(s+z_c)/(s+p_c) evaluated at s = j*omega:
54 // arg[C(j*omega)] = atan(omega/z_c) - atan(omega/p_c) [radians]
55 // Note: this uses the continuous-time formula. The actual discrete-time phase
56 // (from the Tustin coefficients b0, b1, a1) will differ near the Nyquist frequency
57 // due to frequency warping. For frequencies << 1/Ts, the two agree well.
58 double DiscreteLeadLag::phaseAt(double omega) const
59 {
60 return std::atan2(omega, p_.continuousZero) - std::atan2(omega, p_.continuousPole);
61 }
