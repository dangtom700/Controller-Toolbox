60// =============================================================================
61// DARE solver - symplectic pencil generalised eigenvalue method for indefinite R
62//
63// Solves: A'XA - X - A'XB (R + B'XB)^{-1} B'XA + Q = 0
64//
65// The H-inf DARE has an indefinite R (gamma-scaled diagonal blocks), so the
66// standard positive-definite doubling iteration diverges. The Hamiltonian
67// approach forms G = B*Rinv*B', which amplifies ill-conditioning when R is
68// near-singular (as occurs at gamma close to gamma_opt). Instead we use the
69// symplectic pencil (M, N) which defers the R^{-1} product until the
70// eigenvector solve - never forming B*Rinv*B' explicitly:
71//
72// M = [ A, 0 ] N = [ I, B*Rinv*B' ]
73// [-Q, I ] [ 0, A' ]
74//
75// The generalised eigenvalue problem M*v = lambda * N*v has 2n eigenvalues
76// in reciprocal pairs (lambda, 1/lambda). We select the n eigenvalues with
77// |lambda| < 1 (stable subspace) and recover X = V2 * V1^{-1} from the
78// corresponding eigenvectors partitioned as [V1; V2].
79//
80// Ref: Laub (1979) "A Schur method for solving algebraic Riccati equations";
81// Van Dooren (1981) "A generalised eigenvalue approach for solving
82// Riccati equations"; Lancaster & Rodman "Algebraic Riccati Equations".
83// =============================================================================
84DiscreteHinf::DareOut DiscreteHinf::solveHinfDARE(
85 const Eigen::MatrixXd &A,
86 const Eigen::MatrixXd &B,
87 const Eigen::MatrixXd &Q,
88 const Eigen::MatrixXd &R,
89 double /*tol*/, int /*maxIter*/)
90{
91 DareOut out;
92 out.conv = false;
93 out.iters = 1;
94
95 const int n = A.rows();
96
97 // R must be invertible (caller has already verified this).
98 Eigen::FullPivLU<Eigen::MatrixXd> luR(R);
99 if (!luR.isInvertible())
100 {
101 out.X = Eigen::MatrixXd::Zero(n, n);
102 return out;
103 }
104 const Eigen::MatrixXd Rinv = luR.inverse();
105
106 // Symplectic pencil (M, N):
107 // M = [ A, 0 ] N = [ I, B*Rinv*B' ]
108 // [-Q, I ] [ 0, A' ]
109 //
110 // B*Rinv*B' is computed here once; it is n x n and indefinite for H-inf R.
111 const Eigen::MatrixXd BRinvBt = B * Rinv * B.transpose();
112
113 Eigen::MatrixXd M = Eigen::MatrixXd::Zero(2 * n, 2 * n);
114 Eigen::MatrixXd N = Eigen::MatrixXd::Zero(2 * n, 2 * n);
115
116 M.topLeftCorner(n, n) = A;
117 M.bottomLeftCorner(n, n) = -Q;
118 M.bottomRightCorner(n, n) = Eigen::MatrixXd::Identity(n, n);
119
120 N.topLeftCorner(n, n) = Eigen::MatrixXd::Identity(n, n);
121 N.topRightCorner(n, n) = BRinvBt;
122 N.bottomRightCorner(n, n) = A.transpose();
123
124 // Generalised eigenvalue problem: M v = lambda N v.
125 // Eigen's GeneralizedEigenSolver works with real matrices and returns
126 // complex eigenvalue/eigenvector pairs - correct for the indefinite case.
127 Eigen::GeneralizedEigenSolver<Eigen::MatrixXd> ges(M, N);
128 if (ges.info() != Eigen::Success)
129 {
130 out.X = Eigen::MatrixXd::Zero(n, n);
131 return out;
132 }
133
134 const auto &evals = ges.eigenvalues(); // complex (2n,)
135 const auto &evecs = ges.eigenvectors(); // complex (2n, 2n)
136
137 // Select the n eigenvectors whose eigenvalue has |lambda| < 1 (stable subspace).
138 std::vector<int> stable_idx;
139 stable_idx.reserve(n);
140 for (int i = 0; i < 2 * n; ++i)
141 {
142 if (std::abs(evals(i)) < 1.0 - 1e-10)
143 stable_idx.push_back(i);
144 }
145
146 // Need exactly n stable eigenvalues for a unique stabilising solution.
147 if (static_cast<int>(stable_idx.size()) != n)
148 {
149 out.X = Eigen::MatrixXd::Zero(n, n);
150 return out;
151 }
152
153 // Build the stable subspace matrix and partition into [V1; V2] (each n x n).
154 Eigen::MatrixXcd V1(n, n), V2(n, n);
155 for (int j = 0; j < n; ++j)
156 {
157 V1.col(j) = evecs.col(stable_idx[j]).head(n);
158 V2.col(j) = evecs.col(stable_idx[j]).tail(n);
159 }
160
161 // X = real(V2 * V1^{-1})
162 Eigen::FullPivLU<Eigen::MatrixXcd> luV1(V1);
163 if (!luV1.isInvertible())
164 {
165 out.X = Eigen::MatrixXd::Zero(n, n);
166 return out;
167 }
168
169 const Eigen::MatrixXcd X_c = V2 * luV1.inverse();
170 out.X = X_c.real();
171 out.X = 0.5 * (out.X + out.X.transpose()); // enforce symmetry
172
173 // Verify the DARE residual: ||A'XA - X + Q - A'XB * (R+B'XB)^{-1} * B'XA|| / (1+||X||)
174 const Eigen::MatrixXd &X = out.X;
175 const Eigen::MatrixXd Rbar = R + B.transpose() * X * B;
176 Eigen::FullPivLU<Eigen::MatrixXd> luRbar(Rbar);
177 if (!luRbar.isInvertible())
178 {
179 out.X = Eigen::MatrixXd::Zero(n, n);
180 return out;
181 }
182 const Eigen::MatrixXd K = luRbar.inverse() * B.transpose() * X * A;
183 const Eigen::MatrixXd resid = A.transpose() * X * A - X + Q - A.transpose() * X * B * K;
184 const double dare_res = resid.norm() / (1.0 + X.norm());
185 out.conv = (dare_res < 1e-6) && X.allFinite();
186 return out;
187}

189// =============================================================================
190// trySolve - attempt H-inf synthesis at a fixed gamma
191//
192// Returns true iff all DGKF conditions are satisfied:
193// (C1) Control DARE for X_inf has a stabilising solution (X_inf >= 0,
194// spectral radius of closed-loop A - B2 F < 1)
195// (C2) Filter DARE for Y_inf has a stabilising solution (Y_inf >= 0,
196// spectral radius of closed-loop A - L C2 < 1)
197// (C3) rho(X_inf * Y_inf) < gamma^2
198//
199// The controller matrices are assembled only when all three conditions hold.
200//
201// DGKF discrete two-Riccati formulas (Stoorvogel 1992 / Iglesias & Glover 1991):
202//
203// Define:
204// gamma2 = gamma^2
205// B = [B1, B2]
206// D = [D11 D12; D21 D22] (standard form: D11=0, D22=0)
207//
208// Control DARE: Solve for X_inf >= 0 stabilising solution of:
209// A'XA - X + C1'C1 - A'XB * (gamma2*[I 0; 0 -I] + [B1 B2]'*X*[B1 B2])^{-1} * B'*X*A = 0
210//
211// Filter DARE: Solve for Y_inf >= 0 stabilising solution of (transposed):
212// A Y A' - Y + B1 B1' - A Y C' * (gamma2*[I 0; 0 -I] + [C1; C2]*Y*[C1; C2]')^{-1} * C Y A' = 0
213//
214// Controller assembly (Stoorvogel 1992, Lemma 3.1):
215// F_inf = -(R_x)^{-1} * (B2' X_inf A + D12' C1) [state feedback gain]
216// L_inf = -(A Y_inf C2' + B1 D21') * (R_y)^{-1} [observer gain]
217// where R_x = I + D12' D12 + B2' X_inf B2 (positive definite if synthesis feasible)
218// R_y = I + D21 D21' + C2 Y_inf C2' (positive definite if synthesis feasible)
219//
220// Z_inf = (I - gamma^{-2} Y_inf X_inf)^{-1}
221//
222// Ak = A + B2 F_inf + Z_inf L_inf (C2 + D21 D11' ... ) + ...
223// [simplified for D11=0, D22=0 standard form]
224//
225// For the standard form (D11=0, D22=0) the formulas reduce to:
226// Control DARE Q_ctrl = C1'C1, R_ctrl = gamma2*[I 0; 0 -I] + [B1 B2]'*X*[B1 B2]
227// Filter DARE Q_filt = B1*B1', R_filt = gamma2*[I 0; 0 -I] + [C1; C2]*Y*[C1; C2]'
228//
229// F = -(D12'D12 + B2'X B2)^{-1} * B2'X A (standard form, D12'C1 term = 0 when D11=0)
230// L = -A Y C2'*(D21 D21' + C2 Y C2')^{-1}
231// Z = (I - Y X / gamma2)^{-1}
232//
233// Ak = A + B2 F + Z * (L + B2 F... ) [see below for exact assembly]
234// =============================================================================
235bool DiscreteHinf::trySolve(const GeneralisedPlant &P, double gamma,
236 double dareTol, int dareMaxIter,
237 HinfResult &out)
238{
239 const double g2 = gamma * gamma;
240 const int n = P.stateSize();
241 const int nw = P.nw();
242 const int nu = P.nu();
243 const int nz = P.nz();
244 const int ny = P.ny();
245
246 const Eigen::MatrixXd &A = P.A;
247 const Eigen::MatrixXd &B1 = P.B1;
248 const Eigen::MatrixXd &B2 = P.B2;
249 const Eigen::MatrixXd &C1 = P.C1;
250 const Eigen::MatrixXd &C2 = P.C2;
251 const Eigen::MatrixXd &D11 = P.D11;
252 const Eigen::MatrixXd &D12 = P.D12;
253 const Eigen::MatrixXd &D21 = P.D21;
254 // D22 assumed zero (standard form).
255
256 // -----------------------------------------------------------------------
257 // Pre-check: gamma must exceed ||D11||_2 for the standard-form DAREs
258 // to be well-posed (Iglesias & Glover 1991, Lemma 2.1).
259 // -----------------------------------------------------------------------
260 {
261 Eigen::JacobiSVD<Eigen::MatrixXd> svdD11(D11);
262 if (!D11.isZero(1e-14) && svdD11.singularValues()(0) >= gamma)
263 return false;
264 }
265
266 // -----------------------------------------------------------------------
267 // Control DARE - full form with D11, D12 (Iglesias & Glover 1991, Eq. 2)
268 //
269 // Define the block matrix:
270 // Theta_x = [gamma^2 I_nw 0 ] + [D11' ] [D11 D12]
271 // [ 0 -I_nu ] [D12' ]
272 // = [gamma^2*I_nw + D11'D11, D11'D12]
273 // [D12'D11, D12'D12 - I_nu]
274 //
275 // Control DARE: A' X A - X + C1'C1 - A'X [B1 B2] Theta_x(X)^{-1} [B1;B2]'XA
276 // where Theta_x(X) = Theta_x + [B1' X B1 B1' X B2]
277 // [B2' X B1 B2' X B2]
278 //
279 // For the doubling iteration we pass:
280 // Bcat = [B1 B2], Q_x = C1'C1 + [D11'D11 cross-terms], R_x = Theta_x
281 // -----------------------------------------------------------------------
282 const Eigen::MatrixXd Bcat = (Eigen::MatrixXd(n, nw + nu)
283 << B1, B2).finished();
284
285 // R_x = [gamma^2*I + D11'D11, D11'D12]
286 // [D12'D11, D12'D12 - I_nu]
287 Eigen::MatrixXd Rx = Eigen::MatrixXd::Zero(nw + nu, nw + nu);
288 Rx.topLeftCorner(nw, nw) = g2 * Eigen::MatrixXd::Identity(nw, nw) + D11.transpose() * D11;
289 Rx.topRightCorner(nw, nu) = D11.transpose() * D12;
290 Rx.bottomLeftCorner(nu, nw) = D12.transpose() * D11;
291 Rx.bottomRightCorner(nu, nu) = D12.transpose() * D12 - Eigen::MatrixXd::Identity(nu, nu);
292
293 // Q_x = C1'C1 (cross terms D11'C1 enter via the off-diagonal R_x - the
294 // structured doubling algorithm absorbs them when R_x is fully populated.)
295 const Eigen::MatrixXd Qx = C1.transpose() * C1;
296
297 // Check R_x invertibility before solving (required for DARE)
298 Eigen::FullPivLU<Eigen::MatrixXd> luRx0(Rx);
299 if (!luRx0.isInvertible()) return false;
300
301 DareOut dx = solveHinfDARE(A, Bcat, Qx, Rx, dareTol, dareMaxIter);
302 out.dareConvX = dx.conv;
303 out.dareItersX = dx.iters;
304 if (!dx.conv) return false;
305
306 const Eigen::MatrixXd &X = dx.X;
307 // Symmetrise
308 const Eigen::MatrixXd Xs = 0.5 * (X + X.transpose());
309
310 // Check X >= 0 (all eigenvalues >= 0)
311 Eigen::SelfAdjointEigenSolver<Eigen::MatrixXd> esX(Xs);
312 if (esX.eigenvalues().minCoeff() < -1e-6) return false;
313
314 // -----------------------------------------------------------------------
315 // Filter DARE - full form with D11, D21 (dual of control problem)
316 //
317 // R_y = [gamma^2*I_nz + D11*D11', D11*D21']
318 // [D21*D11', D21*D21' - I_ny]
319 // Q_y = B1*B1'
320 // Ccat = [C1; C2]
321 // Solved as DARE(A', Ccat', Q_y, R_y) -> Y_inf
322 // -----------------------------------------------------------------------
323 const Eigen::MatrixXd Ccat = (Eigen::MatrixXd(nz + ny, n)
324 << C1, C2).finished();
325
326 Eigen::MatrixXd Ry = Eigen::MatrixXd::Zero(nz + ny, nz + ny);
327 Ry.topLeftCorner(nz, nz) = g2 * Eigen::MatrixXd::Identity(nz, nz) + D11 * D11.transpose();
328 Ry.topRightCorner(nz, ny) = D11 * D21.transpose();
329 Ry.bottomLeftCorner(ny, nz) = D21 * D11.transpose();
330 Ry.bottomRightCorner(ny, ny) = D21 * D21.transpose() - Eigen::MatrixXd::Identity(ny, ny);
331
332 const Eigen::MatrixXd Qy = B1 * B1.transpose();
333
334 Eigen::FullPivLU<Eigen::MatrixXd> luRy0(Ry);
335 if (!luRy0.isInvertible()) return false;
336
337 DareOut dy = solveHinfDARE(A.transpose(), Ccat.transpose(), Qy, Ry, dareTol, dareMaxIter);
338 out.dareConvY = dy.conv;
339 out.dareItersY = dy.iters;
340
341 if (!dy.conv) return false;
342
343 const Eigen::MatrixXd &Y = dy.X;
344 const Eigen::MatrixXd Ys = 0.5 * (Y + Y.transpose());
345
346 Eigen::SelfAdjointEigenSolver<Eigen::MatrixXd> esY(Ys);
347 if (esY.eigenvalues().minCoeff() < -1e-6) return false;
348
349 // -----------------------------------------------------------------------
350 // Condition (C3): spectral radius of X_inf * Y_inf < gamma^2
351 //
352 // WHY this condition is required (DGKF Theorem 3.1 coupling condition):
353 // The central H-infinity controller is assembled as:
354 // Z_inf = (I - Y_inf * X_inf / gamma^2)^{-1}
355 // which requires (I - Y_inf * X_inf / gamma^2) to be invertible.
356 // This matrix is invertible if and only if none of its eigenvalues are zero,
357 // i.e., none of the eigenvalues of (Y_inf * X_inf / gamma^2) equal 1,
358 // i.e., no eigenvalue of Y_inf * X_inf equals gamma^2,
359 // i.e., rho(X_inf * Y_inf) < gamma^2 (since X_inf*Y_inf and Y_inf*X_inf
360 // share the same nonzero eigenvalues).
361 //
362 // Geometrically: C3 measures the "coupling" between the control and filter
363 // Riccati solutions. When X_inf and Y_inf are both small (plant easy to
364 // control and observe), rho(XY) << gamma^2 easily. As gamma approaches
365 // gamma_opt from above, rho(XY) -> gamma^2 from below; at gamma = gamma_opt,
366 // C3 holds with equality (infimum). For gamma < gamma_opt, C3 fails -
367 // Z_inf becomes singular and the controller assembly is undefined.
368 //
369 // Ref: Doyle, Glover, Khargonekar, Francis (1989) IEEE TAC 34(8), Theorem 3;
370 // Stoorvogel (1992) "The H-infinity Control Problem", Lemma 3.1 condition (iii).
371 // -----------------------------------------------------------------------
372 const Eigen::MatrixXd XY = Xs * Ys;
373 Eigen::EigenSolver<Eigen::MatrixXd> esXY(XY);
374 double rho = 0.0;
375 for (int i = 0; i < esXY.eigenvalues().size(); ++i)
376 rho = std::max(rho, std::abs(esXY.eigenvalues()(i)));
377
378 out.spectralRadius = rho;
379 if (rho >= g2) return false;
380
381 // -----------------------------------------------------------------------
382 // Controller assembly (D22=0; D11 may be nonzero)
383 //
384 // Iglesias & Glover (1991) / Stoorvogel (1992) central controller:
385 //
386 // F = -R_F^{-1} * (B2'*Xs*A + D12'*C1)
387 // L = -(A*Ys*C2' + B1*D21') * R_L^{-1}
388 // Z = (I - Ys*Xs/gamma^2)^{-1}
389 //
390 // Ak = Z * (A + B2*F + L*C2)
391 // Bk = Z * L
392 // Ck = F
393 // Dk = 0_{nu x ny}
394 // -----------------------------------------------------------------------
395
396 // State-feedback gain (D12'*C1 cross-term handles D11 case via DARE structure)
397 const Eigen::MatrixXd R_F = D12.transpose() * D12 + B2.transpose() * Xs * B2;
398 Eigen::FullPivLU<Eigen::MatrixXd> luRF(R_F);
399 if (!luRF.isInvertible()) return false;
400 const Eigen::MatrixXd F = -luRF.inverse() * (B2.transpose() * Xs * A + D12.transpose() * C1);
401
402 // Observer gain (B1*D21' cross-term handles D11 case via dual DARE structure)
403 const Eigen::MatrixXd R_L = D21 * D21.transpose() + C2 * Ys * C2.transpose();
404 Eigen::FullPivLU<Eigen::MatrixXd> luRL(R_L);
405 if (!luRL.isInvertible()) return false;
406 const Eigen::MatrixXd L = -(A * Ys * C2.transpose() + B1 * D21.transpose()) * luRL.inverse();
407
408 // Z = (I - Ys * Xs / gamma^2)^{-1}
409 const Eigen::MatrixXd IminYXg2 = Eigen::MatrixXd::Identity(n, n) - Ys * Xs / g2;
410 Eigen::FullPivLU<Eigen::MatrixXd> luZ(IminYXg2);
411 if (!luZ.isInvertible()) return false;
412 const Eigen::MatrixXd Z = luZ.inverse();
413
414 out.Ak = Z * (A + B2 * F + L * C2);
415 out.Bk = Z * L;
416 out.Ck = F;
417 out.Dk = Eigen::MatrixXd::Zero(nu, ny);
418
419 out.X_inf = Xs;
420 out.Y_inf = Ys;
421
422 // Verify closed-loop stability of the controller (Ak should be stable)
423 Eigen::EigenSolver<Eigen::MatrixXd> esAk(out.Ak);
424 for (int i = 0; i < esAk.eigenvalues().size(); ++i)
425 {
426 if (std::abs(esAk.eigenvalues()(i)) >= 1.0 + 1e-6)
427 return false;
428 }
429
430 return true;
431}

433// =============================================================================
434// solve - bisection on gamma to find minimum achievable H-inf norm
435// =============================================================================
436HinfResult DiscreteHinf::solve(const GeneralisedPlant &P, const HinfParams &params)
437{
438 HinfResult result;
439 result.feasible = false;
440 result.achievedGamma = std::numeric_limits<double>::infinity();
441
442 // Basic dimension checks
443 if (P.stateSize() == 0)
444 throw std::invalid_argument("DiscreteHinf::solve: empty generalised plant.");
445 if (P.nu() == 0 || P.ny() == 0)
446 throw std::invalid_argument("DiscreteHinf::solve: plant must have at least one control and one measurement channel.");
447 if (P.nw() == 0 || P.nz() == 0)
448 throw std::invalid_argument("DiscreteHinf::solve: plant must have exogenous and performance channels.");
449
450 // Verify D12 full column rank and D21 full row rank (DGKF assumption A2)
451 {
452 Eigen::JacobiSVD<Eigen::MatrixXd> svdD12(P.D12);
453 const int rankD12 = (int)(svdD12.singularValues().array() > 1e-10).count();
454 if (rankD12 < P.nu())
455 throw std::invalid_argument("DiscreteHinf::solve: D12 does not have full column rank (DGKF assumption A2 violated).");
456
457 Eigen::JacobiSVD<Eigen::MatrixXd> svdD21(P.D21);
458 const int rankD21 = (int)(svdD21.singularValues().array() > 1e-10).count();
459 if (rankD21 < P.ny())
460 throw std::invalid_argument("DiscreteHinf::solve: D21 does not have full row rank (DGKF assumption A2 violated).");
461 }
462
463 // Warn if D22 is nonzero: the DGKF assembly formulas in trySolve() assume D22=0.
464 // For plants with direct feedthrough, apply a loop-shifting pre-processing step
465 // before calling solve() (see Skogestad & Postlethwaite Section 9.4).
466 if (P.D22.norm() > 1e-12)
467 std::cerr << "[DiscreteHinf] WARNING: generalised plant D22 is nonzero (norm="
468 << P.D22.norm() << "). The standard DGKF assembly assumes D22=0. "
469 << "Apply loop-shifting before calling solve(), or use MixedSensitivity "
470 << "with a plant that has no direct feedthrough (D=0).\n";
471
472 // Gamma bisection: find the smallest gamma in (gammaLo, gammaInit) that is feasible.
473 // gammaLo is set to max(||D11||_2, 1e-4) - the Iglesias-Glover lower bound (Lemma 2.1)
474 // below which no feasible synthesis exists regardless of the DARE solutions.
475 double gammaHi = params.gammaInit;
476 double gammaLo;
477 {
478 Eigen::JacobiSVD<Eigen::MatrixXd> svdD11(P.D11);
479 const double d11_lb = P.D11.isZero(1e-14) ? 0.0 : svdD11.singularValues()(0);
480 gammaLo = std::max(d11_lb + 1e-6, 1e-4);
481 }
482
483 // Check that gammaHi is feasible; if not, double it up to 10 times.
484 HinfResult candidate;
485 bool hiOk = trySolve(P, gammaHi, params.dareTol, params.dareMaxIter, candidate);
486 if (!hiOk)
487 {
488 for (int d = 0; d < 10; ++d)
489 {
490 gammaHi *= 2.0;
491 hiOk = trySolve(P, gammaHi, params.dareTol, params.dareMaxIter, candidate);
492 if (hiOk) break;
493 }
494 if (!hiOk)
495 return result; // infeasible at all tried gammas
496 }
497
498 result = candidate;
499 result.feasible = true;
500 result.achievedGamma = gammaHi;
501 result.Ts = P.Ts;
502
503 // Bisect down
504 for (int iter = 0; iter < params.maxIter; ++iter)
505 {
506 if (gammaHi - gammaLo < params.gammaTol) break;
507
508 const double gammaMid = 0.5 * (gammaLo + gammaHi);
509 HinfResult midResult;
510 if (trySolve(P, gammaMid, params.dareTol, params.dareMaxIter, midResult))
511 {
512 gammaHi = gammaMid;
513 result = midResult;
514 result.feasible = true;
515 result.achievedGamma = gammaMid;
516 result.Ts = P.Ts;
517 }
518 else
519 {
520 gammaLo = gammaMid;
521 }
522 }
523
524 return result;
525}

527// =============================================================================
528// MixedSensitivity - weight factory and generalised plant builder
529//
530// Weight designs follow Skogestad & Postlethwaite Ch. 2-3 (2005).
531// All continuous-time weights are discretised using the Tustin transform.
532// =============================================================================
533
534// Tustin discretisation of a first-order continuous-time system:
535// H_c(s) = (b1 s + b0) / (a1 s + a0)
536// -> H_d(z) = (alpha*b1 + b0*Ts/2) / (alpha*a1 + a0*Ts/2) * (z + beta) / (z + gamma)
537// where alpha = 2/Ts.
538//
539// Returns StateSpace with A = [[p]], B = [[1]], C = [[c_gain]], D = [[d_gain]]
540// encoding H_d(z) = d_gain + c_gain * z^{-1} / (1 - p * z^{-1}) (controller canonical).
541static StateSpace tustinFirstOrder(double b1, double b0, double a1, double a0, double Ts)
542{
543 // Bilinear substitution: s = (2/Ts) * (z-1)/(z+1)
544 // H_c(s) = (b1 s + b0)/(a1 s + a0)
545 // Multiply numerator and denominator by (z+1) and substitute:
546 // num: b1*(2/Ts)*(z-1) + b0*(z+1) = (2b1/Ts + b0)*z + (-2b1/Ts + b0)
547 // den: a1*(2/Ts)*(z-1) + a0*(z+1) = (2a1/Ts + a0)*z + (-2a1/Ts + a0)
548 const double alpha = 2.0 / Ts;
549 const double num_z1 = alpha * b1 + b0;
550 const double num_z0 = -alpha * b1 + b0;
551 const double den_z1 = alpha * a1 + a0;
552 const double den_z0 = -alpha * a1 + a0;
553
554 // Monic: divide by den_z1
555 // H_d(z) = (num_z1/den_z1 * z + num_z0/den_z1) / (z + den_z0/den_z1)
556 // In state-space (scalar, first order):
557 // x[k+1] = (-den_z0/den_z1) x[k] + u[k]
558 // y[k] = (num_z1/den_z1 - D * (-den_z0/den_z1)) x[k] + D * u[k]
559 // with D = num_z1/den_z1.
560 const double A_ss = -den_z0 / den_z1;
561 const double D_ss = num_z1 / den_z1;
562 // y = C*x + D*u => C*A_ss + D*1 = num_z0/den_z1 => C = (num_z0/den_z1 - D*A_ss) / ... wait
563 // For H(z) = D + C*(zI-A)^{-1}*B with B=1:
564 // H(z) = D + C*z^{-1}/(1 - A*z^{-1})
565 // = D*z/(z-A) + C*z^{-1}*z/(z-A) = (D*(z-A) + C) / (z-A) -- only if B=1
566 // Actually for B=1:
567 // H(z) = C*(zI-A)^{-1}*1 + D = C/(z-A) + D = (C + D*(z-A))/(z-A) = (D*z + C - D*A)/(z-A)
568 // Compare with target H_d(z) = (num_z1*z + num_z0)/(den_z1*z + den_z0) [non-monic]
569 // = (num_z1/den_z1 * z + num_z0/den_z1)/(z + den_z0/den_z1)
570 // So: D = num_z1/den_z1, A = -den_z0/den_z1 (pole), C = num_z0/den_z1 - D*(-A) = num_z0/den_z1 - D*A
571 const double C_ss = num_z0 / den_z1 + D_ss * A_ss;
572
573 Eigen::MatrixXd A_m(1, 1); A_m(0, 0) = A_ss;
574 Eigen::MatrixXd B_m(1, 1); B_m(0, 0) = 1.0;
575 Eigen::MatrixXd C_m(1, 1); C_m(0, 0) = C_ss;
576 Eigen::MatrixXd D_m(1, 1); D_m(0, 0) = D_ss;
577
578 return StateSpace(A_m, B_m, C_m, D_m, Ts);
579}

581StateSpace MixedSensitivity::makeW1(double omega_B, double M, double eps, double Ts)
582{
583 // W1(s) = (s/M + omega_B) / (s + omega_B * eps)
584 // Numerator: b1 = 1/M, b0 = omega_B
585 // Denominator: a1 = 1, a0 = omega_B * eps
586 return tustinFirstOrder(1.0 / M, omega_B, 1.0, omega_B * eps, Ts);
587}

600StateSpace MixedSensitivity::makeW2highpass(double omega_u, double eps, double Ts)
601{
602 // W2(s) = (s + omega_u * eps) / (s + omega_u)
603 // Numerator: b1 = 1, b0 = omega_u * eps

608StateSpace MixedSensitivity::makeW3(double omega_T, double Mt, double eps, double Ts)
609{
610 // W3(s) = (s + omega_T / Mt) / (eps * s + omega_T)
611 // Numerator: b1 = 1, b0 = omega_T / Mt
612 // Denominator: a1 = eps, a0 = omega_T

616// =============================================================================
617// MixedSensitivity::build - assemble the generalised plant
618//
619// Augmented state: xa = [xG; xW1; xW2; xW3] (n = nG + nW1 + nW2 + nW3)
620//
621// For a SISO plant G(z) with weighting filters W1, W2, W3:
622//
623// Exogenous inputs w = [r; d] (reference r and output disturbance d)
624// Performance outputs z = [z1; z2; z3] where:
625// z1 = W1 * e = W1 * (r - y) (weighted tracking error)
626// z2 = W2 * u (weighted control effort)
627// z3 = W3 * y (weighted complementary sensitivity)
628// Control input u (scalar)
629// Measurement y = G*u + d (scalar)
630//
631// Plant interconnection (signal flow):
632// e = r - y
633// y = G*u + d (G input: u, G output: yG = G*u)
634// z1 = W1 * e = W1 * (r - yG - d)
635// z2 = W2 * u
636// z3 = W3 * (yG + d)
637// y_meas = yG + d (measured output)
638//
639// Augmented state equations:
640// xG[k+1] = AG xG + BG u
641// xW1[k+1] = AW1 xW1 + BW1 * (r - CG xG - DG u - d)
642// xW2[k+1] = AW2 xW2 + BW2 * u
643// xW3[k+1] = AW3 xW3 + BW3 * (CG xG + DG u + d)
644//
645// Performance outputs:
646// z1 = CW1 xW1 + DW1 * (r - CG xG - DG u - d)
647// z2 = CW2 xW2 + DW2 * u
648// z3 = CW3 xW3 + DW3 * (CG xG + DG u + d)
649//
650// Measurement:
651// y_meas = CG xG + DG u + d
652//
653// Stacking into generalised plant format:
654// xa = [xG; xW1; xW2; xW3]
655// w = [r; d] (nw = 2)
656// z = [z1; z2; z3] (nz = 3 for first-order weights, scalar outputs)
657// u (nu = 1)
658// y_meas (ny = 1)
659// =============================================================================
660GeneralisedPlant MixedSensitivity::build(const StateSpace &G,
661 const StateSpace &W1,
662 const StateSpace &W2,
663 const StateSpace &W3)
664{
665 // Validate SISO plant
666 if (G.inputSize() != 1 || G.outputSize() != 1)
667 throw std::invalid_argument("MixedSensitivity::build: G must be SISO.");
668 if (std::abs(G.Ts - W1.Ts) > 1e-12 || std::abs(G.Ts - W2.Ts) > 1e-12 || std::abs(G.Ts - W3.Ts) > 1e-12)
669 throw std::invalid_argument("MixedSensitivity::build: all systems must have the same sample time.");
670
671 // Extract scalars from state-space matrices
672 const int nG = G.stateSize();
673 const int nW1 = W1.stateSize();
674 const int nW2 = W2.stateSize();
675 const int nW3 = W3.stateSize();
676 const int n = nG + nW1 + nW2 + nW3;
677
678 // Scalar gains from SISO matrices (C and D are 1x1, B is nX x 1)
679 const double cG = G.C(0, 0), dG = G.D(0, 0);
680 const double dW1 = W1.D(0, 0);
681 const double dW2 = W2.D(0, 0);
682 const double dW3 = W3.D(0, 0);
683
684 // -----------------------------------------------------------------------
685 // State transition matrix A_aug (n x n), block diagonal + coupling
686 //
687 // xa[k+1] = A_aug xa + B1_aug w + B2_aug u
688 //
689 // xG[k+1] = AG xG + BG * u
690 // xW1[k+1] = AW1 xW1 - BW1*CG xG + BW1*r - BW1*d - BW1*DG*u
691 // xW2[k+1] = AW2 xW2 + BW2*u
692 // xW3[k+1] = AW3 xW3 + BW3*CG xG + BW3*d + BW3*DG*u
693 // -----------------------------------------------------------------------
694 GeneralisedPlant P;
695 P.Ts = G.Ts;
696
697 // A_aug (n x n)
698 P.A = Eigen::MatrixXd::Zero(n, n);
699 P.A.block(0, 0, nG, nG) = G.A;
700 P.A.block(nG, nG, nW1, nW1) = W1.A;
701 P.A.block(nG+nW1, nG+nW1, nW2, nW2) = W2.A;
702 P.A.block(nG+nW1+nW2, nG+nW1+nW2, nW3, nW3) = W3.A;
703
704 // Coupling: W1 driven by -CG * xG, W3 driven by +CG * xG
705 // xW1[k+1] gets -BW1 * CG * xG
706 P.A.block(nG, 0, nW1, nG) = -W1.B * G.C;
707 // xW3[k+1] gets +BW3 * CG * xG
708 P.A.block(nG+nW1+nW2, 0, nW3, nG) = W3.B * G.C;
709
710 // -----------------------------------------------------------------------
711 // B1_aug (n x nw), nw = 2: [r, d]
712 // -----------------------------------------------------------------------
713 P.B1 = Eigen::MatrixXd::Zero(n, 2);
714 // xG: no exogenous input (G is driven by u only)
715 // xW1: driven by r (+BW1) and d (-BW1)
716 P.B1.block(nG, 0, nW1, 1) = W1.B * Eigen::MatrixXd::Ones(1, 1); // r
717 P.B1.block(nG, 1, nW1, 1) = -W1.B * Eigen::MatrixXd::Ones(1, 1); // d
718 // xW2: no exogenous input
719 // xW3: driven by d (+BW3)
720 P.B1.block(nG+nW1+nW2, 1, nW3, 1) = W3.B * Eigen::MatrixXd::Ones(1, 1); // d
721
722 // -----------------------------------------------------------------------
723 // B2_aug (n x nu), nu = 1: u
724 // -----------------------------------------------------------------------
725 P.B2 = Eigen::MatrixXd::Zero(n, 1);
726 P.B2.block(0, 0, nG, 1) = G.B; // xG driven by BG * u
727 P.B2.block(nG, 0, nW1, 1) = -W1.B * dG; // xW1 driven by -BW1*DG*u
728 P.B2.block(nG+nW1, 0, nW2, 1) = W2.B; // xW2 driven by BW2*u
729 P.B2.block(nG+nW1+nW2, 0, nW3, 1) = W3.B * dG; // xW3 driven by +BW3*DG*u
730
731 // -----------------------------------------------------------------------
732 // Performance outputs z = C1 xa + D11 w + D12 u (nz = 3: [z1, z2, z3])
733 //
734 // z1 = CW1 xW1 + DW1*(r - CG xG - DG u - d)
735 // = -DW1*CG xG + CW1 xW1 + DW1*r - DW1*d - DW1*DG*u
736 //
737 // z2 = CW2 xW2 + DW2 u
738 //
739 // z3 = CW3 xW3 + DW3*(CG xG + DG u + d)
740 // = DW3*CG xG + CW3 xW3 + DW3*DG*u + DW3*d
741 // -----------------------------------------------------------------------
742 const int nz = 3; // scalar outputs from each weight (first-order weights)
743 P.C1 = Eigen::MatrixXd::Zero(nz, n);
744
745 // z1 row
746 P.C1(0, 0) = -dW1 * cG; // from xG (scalar: CG is 1x1 -> cG)
747 if (nW1 > 0) P.C1.block(0, nG, 1, nW1) = W1.C; // from xW1
748
749 // z2 row
750 if (nW2 > 0) P.C1.block(1, nG+nW1, 1, nW2) = W2.C; // from xW2
751
752 // z3 row
753 P.C1(2, 0) = dW3 * cG; // from xG
754 if (nW3 > 0) P.C1.block(2, nG+nW1+nW2, 1, nW3) = W3.C; // from xW3
755
756 // -----------------------------------------------------------------------
757 // D11 (nz x nw): feedthrough from w = [r, d] to z
758 //
759 // z1: DW1*r - DW1*d -> D11(0,0) = DW1, D11(0,1) = -DW1
760 // z2: 0
761 // z3: DW3*d -> D11(2,1) = DW3
762 // -----------------------------------------------------------------------
763 P.D11 = Eigen::MatrixXd::Zero(nz, 2);
764 P.D11(0, 0) = dW1; // z1 from r
765 P.D11(0, 1) = -dW1; // z1 from d
766 P.D11(2, 1) = dW3; // z3 from d
767
768 // -----------------------------------------------------------------------
769 // D12 (nz x nu): feedthrough from u to z
770 //
771 // z1: -DW1*DG*u
772 // z2: DW2*u
773 // z3: DW3*DG*u
774 // -----------------------------------------------------------------------
775 P.D12 = Eigen::MatrixXd::Zero(nz, 1);
776 P.D12(0, 0) = -dW1 * dG;
777 P.D12(1, 0) = dW2;
778 P.D12(2, 0) = dW3 * dG;
779
780 // -----------------------------------------------------------------------
781 // Measurement output y_meas = C2 xa + D21 w + D22 u (ny = 1)
782 //
783 // y_meas = CG xG + d + DG u
784 // = CG xG + 0*r + 1*d + DG*u
785 // -----------------------------------------------------------------------
786 P.C2 = Eigen::MatrixXd::Zero(1, n);
787 P.C2.block(0, 0, 1, nG) = G.C;
788
789 P.D21 = Eigen::MatrixXd::Zero(1, 2);
790 P.D21(0, 0) = 0.0; // no direct r -> y feedthrough
791 P.D21(0, 1) = 1.0; // d -> y
792
793 // Standard DGKF requires D22 = 0. For plants with direct feedthrough (dG != 0),
794 // a loop-shifting pre-processing step must be applied before calling solve()
795 // (see Skogestad & Postlethwaite Section 9.4). Passing such a plant directly
796 // would cause trySolve() to synthesise a controller for the wrong problem, since
797 // D22 is silently ignored in the assembly formulas. We throw here so the caller
798 // learns about the issue at build() time rather than silently getting a wrong K.
799 if (std::abs(dG) > 1e-12)
800 throw std::invalid_argument(
801 "MixedSensitivity::build: plant G has nonzero direct feedthrough (D = " +
802 std::to_string(dG) + "). The standard DGKF assembly assumes D22=0. "
803 "Apply loop-shifting to the plant before building the generalised plant, "
804 "or use a plant model with D = 0.");
805
806 P.D22 = Eigen::MatrixXd::Zero(1, 1); // D22 = 0 guaranteed by the check above
807
808 return P;
809}
