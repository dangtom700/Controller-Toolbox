33 // ---------------------------------------------------------------------------
34 // tf2ss - controllable canonical form
35 //
36 // For H(z^-^1) = (b0 + b1z^-^1 + ... + b_nz^-^n) / (1 + a1z^-^1 + ... + a_nz^-^n):
37 //
38 // A = [[-a1, -a2, ..., -a_n], (companion row)
39 // [ 1, 0, ..., 0 ],
40 // ...
41 // [ 0, 0, ..., 0 ]] (n*n)
42 // B = [[1], [0], ..., [0]] (n*1)
43 // C = [[b1-a1b0, b2-a2b0, ..., b_n-a_nb0]] (1*n)
44 // D = [[b0]] (1*1)
45 //
46 // Equivalent MATLAB: [A,B,C,D] = tf2ss(num,den) in z^-^1 convention.
47 // ---------------------------------------------------------------------------
48 StateSpace tf2ss(const TransferFunction &tf)
49 {
50 int n = tf.order(); // state dimension = denominator degree
51
52 // Pad numerator to exactly n+1 coefficients (prepend zeros if shorter).
53 std::vector<double> num = tf.num;
54 if (static_cast<int>(num.size()) < n + 1) {
55 std::vector<double> padded(n + 1 - static_cast<int>(num.size()), 0.0);
56 padded.insert(padded.end(), num.begin(), num.end());
57 num = std::move(padded);
58 }
59
60 double d0 = num[0]; // feed-through / direct term = b0
61
62 // Build A (n*n companion matrix, first row = -a1...-a_n, sub-diagonal = 1).
63 Eigen::MatrixXd A = Eigen::MatrixXd::Zero(n, n);
64 for (int j = 0; j < n; ++j)
65 A(0, j) = -tf.den[j + 1];
66 for (int i = 1; i < n; ++i)
67 A(i, i - 1) = 1.0;
68
69 // B = [1, 0, ..., 0]'
70 Eigen::MatrixXd B = Eigen::MatrixXd::Zero(n, 1);
71 B(0, 0) = 1.0;
72
73 // C = [b1-a1b0, ..., b_n-a_nb0] (long-division remainder numerator)
74 Eigen::MatrixXd C = Eigen::MatrixXd::Zero(1, n);
75 for (int j = 0; j < n; ++j)
76 C(0, j) = num[j + 1] - d0 * tf.den[j + 1];
77
78 Eigen::MatrixXd D(1, 1);
79 D(0, 0) = d0;
80
81 return StateSpace(A, B, C, D, tf.Ts);
82 }

106 // ---------------------------------------------------------------------------
107 // ss2tf - SISO state-space to transfer function via Markov parameters.
108 //
109 // Denominator: characteristic polynomial of A via root expansion of eigenvalues.
110 // p(z) = Pi (z - lambda_i) -> a = [1, a_1, ..., a_n]
111 //
112 // Numerator via the impulse-response / Markov-parameter identity:
113 // h[0] = D, h[k] = C A^{k-1} B for k >= 1
114 // b[k] = Sigma_{j=0}^{k} a[j] h[k-j]
115 //
116 // This follows from H(z^{-1}) = N(z^{-1}) / D(z^{-1}) and the convolution
117 // relation between the denominator coefficients and the impulse response.
118 // ---------------------------------------------------------------------------
119 TransferFunction ss2tf(const StateSpace &sys)
120 {
121 if (sys.inputSize() != 1 || sys.outputSize() != 1)
122 throw std::invalid_argument("ss2tf: only SISO plants supported (inputSize == outputSize == 1).");
123
124 const int n = sys.stateSize();
125
126 // ---- Denominator: characteristic polynomial via Faddeev-LeVerrier recursion ----
127 // Operates in real arithmetic on A directly; avoids eigenvalue computation
128 // and the catastrophic cancellation of root-product expansion for clustered
129 // or repeated eigenvalues.
130 //
131 // Recurrence (monic, coefficients of z^n + a[1]z^{n-1} + ... + a[n]):
132 // M_0 = I, a[k] = -trace(A * M_{k-1}) / k, M_k = A*M_{k-1} + a[k]*I
133 // At the end, A * M_n = 0 (Cayley-Hamilton check).
134 std::vector<double> a(n + 1, 0.0);
135 {
136 a[0] = 1.0;
137 Eigen::MatrixXd M = Eigen::MatrixXd::Identity(n, n);
138 for (int k = 1; k <= n; ++k)
139 {
140 M = sys.A * M;
141 a[k] = -M.trace() / k;
142 M.diagonal().array() += a[k];
143 }
144 }
145
146 // ---- Markov parameters: h[0]=D, h[k]=C A^{k-1} B for k=1..n ----
147 std::vector<double> h(n + 1);
148 h[0] = sys.D(0, 0);
149 Eigen::MatrixXd Apow = Eigen::MatrixXd::Identity(n, n); // A^0
150 for (int k = 1; k <= n; ++k)
151 {
152 h[k] = (sys.C * Apow * sys.B)(0, 0);
153 Apow = sys.A * Apow; // A^k for next iter
154 }
155
156 // ---- Numerator: b[k] = Sigma_{j=0}^{k} a[j] h[k-j] ----
157 std::vector<double> num(n + 1, 0.0);
158 for (int k = 0; k <= n; ++k)
159 for (int j = 0; j <= k; ++j)
160 num[k] += a[j] * h[k - j];
161
162 // Convert to z^{-1} convention (monic denominator already has a[0]=1)
163 std::vector<double> den(a.begin(), a.end());
164 return TransferFunction(num, den, sys.Ts);
165 }

167 // ---------------------------------------------------------------------------
168 // c2d internal helper: warn when a discrete A matrix has eigenvalues outside
169 // the unit disk despite a stable continuous-time plant. This can happen when
170 // ||Ac|| * Ts is large (stiff system + coarse sample time) or when the matrix
171 // exponential loses precision for ill-conditioned Ac.
172 // ---------------------------------------------------------------------------
173 static void checkDiscreteStability(const Eigen::MatrixXd &Ad, const char *context)
174 {
175 (void)context; // only used inside #ifndef NDEBUG; suppress warning in Release
176 const auto eigs = Ad.eigenvalues();
177 for (int i = 0; i < eigs.size(); ++i)
178 {
179#ifndef NDEBUG
180 if (std::abs(eigs[i]) >= 1.0 + 1e-9)
181 {
182 std::cerr << "[PlantModel::" << context << "] WARNING: A_d eigenvalue "
183 << eigs[i] << " (|lambda| = " << std::abs(eigs[i])
184 << ") lies outside the unit disk.\n"
185 << " This may indicate numerical loss of precision in the matrix "
186 "exponential (ZOH)\n"
187 << " or a bilinear transform of a marginally stable plant (Tustin).\n"
188 << " Check that Ts << 1 / |lambda_max(Ac)| for all fast modes.\n";
189 }
190#endif
191 }

194 // ---------------------------------------------------------------------------
195 // c2d
196 // ---------------------------------------------------------------------------
197 StateSpace c2d(const StateSpace &sys_c, double Ts, C2dMethod method,
198 double prewarp_freq)
199 {
200 if (Ts <= 0.0)
201 throw std::invalid_argument("c2d: Ts must be positive.");
202 if (sys_c.Ts != 0.0)
203 throw std::invalid_argument("c2d: input model must be continuous-time (Ts == 0).");
204
205 const int n = sys_c.stateSize();
206 const int m = sys_c.inputSize();
207
208 if (method == C2dMethod::ZOH)
209 {
210 // ZOH discretisation via the Van Loan (1978) matrix-exponential embedding.
211 //
212 // For a linear system x' = Ac*x + Bc*u with piecewise-constant input u,
213 // the exact discrete-time equivalent is:
214 // x[k+1] = Ad*x[k] + Bd*u[k]
215 // Ad = expm(Ac*Ts), Bd = (integral_0^Ts expm(Ac*t) dt) * Bc
216 //
217 // Both Ad and Bd are extracted in one shot from the (n+m)*(n+m) augmented
218 // matrix exponential (Van Loan 1978, Eq. 2.4):
219 // expm([Ac Bc] * Ts) = [Ad Bd]
220 // ([0 0 ] ) [0 I ]
221 //
222 // This is numerically preferred over Pade-approximant series for stiff Ac
223 // (large spread of eigenvalues): the matrix exponential is computed via
224 // Schur decomposition, which remains stable even when eigenvalues span many
225 // decades (e.g., the Bell-Astrom boiler model where |lambda_fast|/|lambda_slow|
226 // >> 100). A forward-Euler or Pade approximant at such stiffness ratios
227 // produces discrete eigenvalues outside the unit disk.
228 //
229 // Reference: Van Loan, C.F. (1978). "Computing integrals involving the matrix
230 // exponential." IEEE Trans. Automat. Control, 23(3):395-404.
231 Eigen::MatrixXd M = Eigen::MatrixXd::Zero(n + m, n + m);
232 M.topLeftCorner(n, n) = sys_c.A * Ts;
233 M.topRightCorner(n, m) = sys_c.B * Ts;
234 const Eigen::MatrixXd eM = M.exp();
235
236 const Eigen::MatrixXd Ad = eM.topLeftCorner(n, n);
237 const Eigen::MatrixXd Bd = eM.topRightCorner(n, m);
238 StateSpace result(Ad, Bd, sys_c.C, sys_c.D, Ts);
239 checkDiscreteStability(result.A, "c2d (ZOH)");
240 return result;
241 }
242 else // Tustin or TustinPrewarped
243 {
244 // Standard Tustin: s = (2/Ts) * (z-1)/(z+1) -> alpha = Ts/2
245 //
246 // Prewarped Tustin: s = k_p * (z-1)/(z+1)
247 // k_p = prewarp_freq / tan(prewarp_freq * Ts / 2)
248 // Ensures |G_d(e^{j*prewarp_freq*Ts})| = |G_c(j*prewarp_freq)| exactly.
249 // When prewarp_freq = 0 or method == Tustin: k_p = 2/Ts (standard).
250 //
251 // In both cases: alpha = 1/k_p, and the formulas are identical.
252 double alpha;
253 if (method == C2dMethod::TustinPrewarped && prewarp_freq > 0.0)
254 {
255 const double tan_half = std::tan(prewarp_freq * Ts * 0.5);
256 if (tan_half < 1e-300)
257 throw std::invalid_argument("c2d TustinPrewarped: prewarp_freq * Ts / 2 too small.");
258 alpha = tan_half / prewarp_freq; // = 1/k_p
259 }
260 else
261 {
262 alpha = Ts * 0.5; // standard Tustin: alpha = Ts/2
263 }
264
265 const Eigen::MatrixXd Im = Eigen::MatrixXd::Identity(n, n) - alpha * sys_c.A;
266 const Eigen::MatrixXd Ip = Eigen::MatrixXd::Identity(n, n) + alpha * sys_c.A;
267 const Eigen::PartialPivLU<Eigen::MatrixXd> lu(Im);
268 const Eigen::MatrixXd Ad = lu.solve(Ip);
269 const Eigen::MatrixXd Bd = lu.solve(sys_c.B * (2.0 * alpha)); // Bd = (2 alpha)*(I-alpha*A)^{-1}*B
270 StateSpace result(Ad, Bd, sys_c.C, sys_c.D, Ts);
271 checkDiscreteStability(result.A, method == C2dMethod::TustinPrewarped
272 ? "c2d (TustinPrewarped)" : "c2d (Tustin)");
273 return result;
274 }
275 }

277 // ---------------------------------------------------------------------------
278 // minreal - minimal realization via balanced truncation (Ho-Kalman)
279 // ---------------------------------------------------------------------------
280 StateSpace minreal(const StateSpace &sys, double tol)
281 {
282 const int n = sys.stateSize();
283 const int m = sys.inputSize();
284 const int p = sys.outputSize();
285
286 if (n == 0) return sys;
287
288 Eigen::MatrixXd Co(n, n * m);
289 Eigen::MatrixXd Ob(n * p, n);
290
291 Eigen::MatrixXd Apow = Eigen::MatrixXd::Identity(n, n);
292 for (int i = 0; i < n; ++i) {
293 Co.middleCols(i * m, m) = Apow * sys.B;
294 Ob.middleRows(i * p, p) = sys.C * Apow;
295 Apow = sys.A * Apow;
296 }
297
298 Eigen::MatrixXd H = Ob * Co;
299 Eigen::JacobiSVD<Eigen::MatrixXd> svd(H, Eigen::ComputeThinU | Eigen::ComputeThinV);
300
301 int r = 0;
302 for (int i = 0; i < svd.singularValues().size(); ++i) {
303 if (svd.singularValues()(i) > tol) r++;
304 }
305
306 if (r == n) return sys; // Already minimal
307 if (r == 0) {
308 // Completely unobservable/uncontrollable, return 0 state system
309 return StateSpace(Eigen::MatrixXd::Zero(0, 0), Eigen::MatrixXd::Zero(0, m), Eigen::MatrixXd::Zero(p, 0), sys.D, sys.Ts);
310 }
311
312 Eigen::MatrixXd S_inv_sqrt = svd.singularValues().head(r).cwiseInverse().cwiseSqrt().asDiagonal();
313 Eigen::MatrixXd Ur = svd.matrixU().leftCols(r);
314 Eigen::MatrixXd Vr = svd.matrixV().leftCols(r);
315
316 Eigen::MatrixXd H_shift = Ob * sys.A * Co;
317
318 Eigen::MatrixXd A_min = S_inv_sqrt * Ur.transpose() * H_shift * Vr * S_inv_sqrt;
319 Eigen::MatrixXd B_min = S_inv_sqrt * Ur.transpose() * H.leftCols(m);
320 Eigen::MatrixXd C_min = H.topRows(p) * Vr * S_inv_sqrt;
321
322 return StateSpace(A_min, B_min, C_min, sys.D, sys.Ts);
323 }

329 // Central-difference Jacobian of an AlgFunc w.r.t. x2, with x1 and u fixed.
330 // Step: h_i = eps * max(|x2_i|, 1) (same scaled-epsilon strategy as EKF).
331 static Eigen::MatrixXd algJacX2(const DAESystem::AlgFunc &g,
332 const Eigen::VectorXd &x1,
333 const Eigen::VectorXd &x2,
334 double u,
335 double eps = 1e-5)
336 {
337 const Eigen::VectorXd g0 = g(x1, x2, u);
338 const int ny = static_cast<int>(g0.size());
339 const int nx = static_cast<int>(x2.size());
340 Eigen::MatrixXd J(ny, nx);
341 Eigen::VectorXd xp = x2, xm = x2;
342 for (int i = 0; i < nx; ++i)
343 {
344 const double h = eps * std::max(std::abs(x2(i)), 1.0);
345 xp(i) = x2(i) + h;
346 xm(i) = x2(i) - h;
347 J.col(i) = (g(x1, xp, u) - g(x1, xm, u)) / (2.0 * h);
348 xp(i) = x2(i);
349 xm(i) = x2(i);
350 }
351 return J;
352 }

354 // Central-difference Jacobian of an AlgFunc or DiffFunc w.r.t. x1, with x2 and u fixed.
355 static Eigen::MatrixXd algJacX1(const DAESystem::AlgFunc &g,
356 const Eigen::VectorXd &x1,
357 const Eigen::VectorXd &x2,
358 double u,
359 double eps = 1e-5)
360 {
361 const Eigen::VectorXd g0 = g(x1, x2, u);
362 const int ny = static_cast<int>(g0.size());
363 const int nx = static_cast<int>(x1.size());
364 Eigen::MatrixXd J(ny, nx);
365 Eigen::VectorXd xp = x1, xm = x1;
366 for (int i = 0; i < nx; ++i)
367 {
368 const double h = eps * std::max(std::abs(x1(i)), 1.0);
369 xp(i) = x1(i) + h;
370 xm(i) = x1(i) - h;
371 J.col(i) = (g(xp, x2, u) - g(xm, x2, u)) / (2.0 * h);
372 xp(i) = x1(i);
373 xm(i) = x1(i);
374 }
375 return J;
376 }

378 // Central-difference Jacobian w.r.t. scalar u (returns column vector = dy/du).
379 static Eigen::VectorXd algJacU(const DAESystem::AlgFunc &g,
380 const Eigen::VectorXd &x1,
381 const Eigen::VectorXd &x2,
382 double u,
383 double eps = 1e-5)
384 {
385 const double h = eps * std::max(std::abs(u), 1.0);
386 return (g(x1, x2, u + h) - g(x1, x2, u - h)) / (2.0 * h);
387 }

389 // ---------------------------------------------------------------------------
390 // consistentInit - Newton-Raphson solve for g(x1, x2, u) = 0 over x2.
391 // ---------------------------------------------------------------------------
392 Eigen::VectorXd consistentInit(const DAESystem &dae,
393 const Eigen::VectorXd &x1_init,
394 double u0,
395 const Eigen::VectorXd &x2_guess,
396 int max_iter,
397 double tol)
398 {
399 Eigen::VectorXd x2 = x2_guess;
400 for (int iter = 0; iter < max_iter; ++iter)
401 {
402 const Eigen::VectorXd g_val = dae.g(x1_init, x2, u0);
403 if (g_val.norm() < tol) break;
404
405 const Eigen::MatrixXd G2 = algJacX2(dae.g, x1_init, x2, u0);
406 const auto ldlt = G2.ldlt();
407 if (ldlt.info() != Eigen::Success)
408 break; // singular or near-singular G2: return last iterate
409
410 x2 -= ldlt.solve(g_val);
411 }
412 return x2;
413 }

415 // ---------------------------------------------------------------------------
416 // dae2ode - discrete step function via Euler integration + Newton projection.
417 //
418 // The returned function F(x_aug, u) = x_aug_next:
419 // 1. Splits x_aug = [x1; x2].
420 // 2. Newton-projects x2 to satisfy g(x1, x2, u) = 0 (cleans any drift).
421 // 3. Euler-steps x1: x1_next = x1 + Ts * f(x1, x2, u).
422 // 4. Newton-solves x2_next from g(x1_next, x2_next, u) = 0.
423 // 5. Returns [x1_next; x2_next].
424 // ---------------------------------------------------------------------------
425 std::function<Eigen::VectorXd(const Eigen::VectorXd &, double)>
426 dae2ode(const DAESystem &dae, int newton_max_iter, double newton_tol)
427 {
428 if (dae.Ts <= 0.0)
429 throw std::invalid_argument("dae2ode: DAESystem.Ts must be > 0.");
430 if (dae.n_diff <= 0)
431 throw std::invalid_argument("dae2ode: n_diff must be > 0.");
432 if (dae.n_alg < 0)
433 throw std::invalid_argument("dae2ode: n_alg must be >= 0.");
434
435 // Capture by value so the returned lambda is self-contained.
436 return [dae, newton_max_iter, newton_tol](const Eigen::VectorXd &x_aug,
437 double u) -> Eigen::VectorXd
438 {
439 const int n1 = dae.n_diff;
440 const int n2 = dae.n_alg;
441 Eigen::VectorXd x1 = x_aug.head(n1);
442 Eigen::VectorXd x2 = (n2 > 0) ? x_aug.tail(n2)
443 : Eigen::VectorXd();
444
445 // Step 1: project x2 onto constraint manifold (eliminates integration drift)
446 if (n2 > 0)
447 x2 = consistentInit(dae, x1, u, x2, newton_max_iter, newton_tol);
448
449 // Step 2: integrate differential states with forward Euler
450 const Eigen::VectorXd x1_dot = dae.f(x1, x2, u);
451 const Eigen::VectorXd x1_next = x1 + dae.Ts * x1_dot;
452
453 // Step 3: solve for algebraic states at the new differential state
454 Eigen::VectorXd x2_next = x2; // warm start from current x2
455 if (n2 > 0)
456 x2_next = consistentInit(dae, x1_next, u, x2, newton_max_iter, newton_tol);
457
458 // Step 4: assemble augmented state
459 Eigen::VectorXd x_aug_next(n1 + n2);
460 x_aug_next.head(n1) = x1_next;
461 if (n2 > 0) x_aug_next.tail(n2) = x2_next;
462 return x_aug_next;
463 };
464 }

466 // ---------------------------------------------------------------------------
467 // c2d(DAESystem) - linearise, eliminate x2, then discretise via ZOH/Tustin.
468 //
469 // Index-1 algebraic elimination (continuous time):
470 // A11 = df/dx1, A12 = df/dx2, G1 = dg/dx1, G2 = dg/dx2
471 // B1 = df/du (column n1 x 1), B2 = dg/du (column n2 x 1)
472 //
473 // A_red = A11 - A12 * G2^{-1} * G1 (n1 x n1)
474 // B_red = B1 - A12 * G2^{-1} * B2 (n1 x 1)
475 //
476 // Then applies c2d(StateSpace{A_red, B_red, C_red, D_red, 0}, Ts, method).
477 // ---------------------------------------------------------------------------
478 StateSpace c2d(const DAESystem &dae,
479 const Eigen::VectorXd &x1_op,
480 const Eigen::VectorXd &x2_op,
481 double u_op,
482 double Ts,
483 C2dMethod method)
484 {
485 const int n1 = dae.n_diff;
486 const int n2 = dae.n_alg;
487
488 if (n1 <= 0)
489 throw std::invalid_argument("c2d(DAESystem): n_diff must be > 0.");
490
491 // Jacobians of f w.r.t. x1 (A11, n1*n1) and scalar u (B1, n1*1)
492 const Eigen::MatrixXd A11 = algJacX1(dae.f, x1_op, x2_op, u_op);
493 const Eigen::VectorXd B1 = algJacU (dae.f, x1_op, x2_op, u_op);
494
495 Eigen::MatrixXd A_red(A11);
496 Eigen::VectorXd B_red(B1);
497
498 if (n2 > 0)
499 {
500 const Eigen::MatrixXd A12 = algJacX2(dae.f, x1_op, x2_op, u_op); // n1 x n2
501 const Eigen::MatrixXd G1 = algJacX1(dae.g, x1_op, x2_op, u_op); // n2 x n1
502 const Eigen::MatrixXd G2 = algJacX2(dae.g, x1_op, x2_op, u_op); // n2 x n2
503 const Eigen::VectorXd B2 = algJacU (dae.g, x1_op, x2_op, u_op); // n2
504
505 // Check G2 invertibility (Index-1 requirement)
506 Eigen::FullPivLU<Eigen::MatrixXd> lu(G2);
507 if (lu.rcond() < 1e-12)
508 throw std::runtime_error(
509 "c2d(DAESystem): G2 = dg/dx2 is singular at the operating point "
510 "(DAE index > 1 or rank-deficient constraint Jacobian).");
511
512 A_red -= A12 * lu.solve(G1); // A_red = A11 - A12 * G2^{-1} * G1
513 B_red -= A12 * lu.solve(B2); // B_red = B1 - A12 * G2^{-1} * B2
514 }
515
516 // Output Jacobians from h; default C = I (full differential-state output)
517 Eigen::MatrixXd C_red;
518 Eigen::MatrixXd D_red;
519 if (dae.h)
520 {
521 C_red = algJacX1(dae.h, x1_op, x2_op, u_op); // ny x n1
522 // algJacU returns a VectorXd column; MatrixXd assignment widens to ny x 1
523 const Eigen::VectorXd d_col = algJacU(dae.h, x1_op, x2_op, u_op);
524 D_red = d_col; // VectorXd -> MatrixXd (ny x 1) via Eigen implicit conversion
525 }
526 else
527 {
528 C_red = Eigen::MatrixXd::Identity(n1, n1);
529 D_red = Eigen::MatrixXd::Zero(n1, 1);
530 }
