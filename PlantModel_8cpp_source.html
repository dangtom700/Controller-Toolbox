33 // ---------------------------------------------------------------------------
34 // tf2ss - controllable canonical form
35 //
36 // For H(z^-^1) = (b0 + b1z^-^1 + ... + b_nz^-^n) / (1 + a1z^-^1 + ... + a_nz^-^n):
37 //
38 // A = [[-a1, -a2, ..., -a_n], (companion row)
39 // [ 1, 0, ..., 0 ],
40 // ...
41 // [ 0, 0, ..., 0 ]] (n*n)
42 // B = [[1], [0], ..., [0]] (n*1)
43 // C = [[b1-a1b0, b2-a2b0, ..., b_n-a_nb0]] (1*n)
44 // D = [[b0]] (1*1)
45 //
46 // Equivalent MATLAB: [A,B,C,D] = tf2ss(num,den) in z^-^1 convention.
47 // ---------------------------------------------------------------------------
48 StateSpace tf2ss(const TransferFunction &tf)
49 {
50 int n = tf.order(); // state dimension = denominator degree
51
52 // Pad numerator to exactly n+1 coefficients (prepend zeros if shorter).
53 std::vector<double> num = tf.num;
54 if (static_cast<int>(num.size()) < n + 1) {
55 std::vector<double> padded(n + 1 - static_cast<int>(num.size()), 0.0);
56 padded.insert(padded.end(), num.begin(), num.end());
57 num = std::move(padded);
58 }
59
60 double d0 = num[0]; // feed-through / direct term = b0
61
62 // Build A (n*n companion matrix, first row = -a1...-a_n, sub-diagonal = 1).
63 Eigen::MatrixXd A = Eigen::MatrixXd::Zero(n, n);
64 for (int j = 0; j < n; ++j)
65 A(0, j) = -tf.den[j + 1];
66 for (int i = 1; i < n; ++i)
67 A(i, i - 1) = 1.0;
68
69 // B = [1, 0, ..., 0]'
70 Eigen::MatrixXd B = Eigen::MatrixXd::Zero(n, 1);
71 B(0, 0) = 1.0;
72
73 // C = [b1-a1b0, ..., b_n-a_nb0] (long-division remainder numerator)
74 Eigen::MatrixXd C = Eigen::MatrixXd::Zero(1, n);
75 for (int j = 0; j < n; ++j)
76 C(0, j) = num[j + 1] - d0 * tf.den[j + 1];
77
78 Eigen::MatrixXd D(1, 1);
79 D(0, 0) = d0;
80
81 return StateSpace(A, B, C, D, tf.Ts);
82 }

85 // ssStep - advance one discrete step and return current output.
86 // Output is computed before state is updated so that y[k] = C.x[k] + D.u[k].
87 // ---------------------------------------------------------------------------
88 Eigen::VectorXd ssStep(const StateSpace &sys,
89 Eigen::Ref<Eigen::VectorXd> x,
90 const Eigen::VectorXd &u)
91 {
92 Eigen::VectorXd y = sys.C * x + sys.D * u; // y[k]
93 x = sys.A * x + sys.B * u; // x[k+1]

97 // ---------------------------------------------------------------------------
98 // ss2tf - SISO state-space to transfer function via Markov parameters.
99 //
100 // Denominator: characteristic polynomial of A via root expansion of eigenvalues.
101 // p(z) = Pi (z - lambda_i) -> a = [1, a_1, ..., a_n]
102 //
103 // Numerator via the impulse-response / Markov-parameter identity:
104 // h[0] = D, h[k] = C A^{k-1} B for k >= 1
105 // b[k] = Sigma_{j=0}^{k} a[j] h[k-j]
106 //
107 // This follows from H(z^{-1}) = N(z^{-1}) / D(z^{-1}) and the convolution
108 // relation between the denominator coefficients and the impulse response.
109 // ---------------------------------------------------------------------------
110 TransferFunction ss2tf(const StateSpace &sys)
111 {
112 if (sys.inputSize() != 1 || sys.outputSize() != 1)
113 throw std::invalid_argument("ss2tf: only SISO plants supported (inputSize == outputSize == 1).");
114
115 const int n = sys.stateSize();
116
117 // ---- Denominator: characteristic polynomial via Faddeev-LeVerrier recursion ----
118 // Operates in real arithmetic on A directly; avoids eigenvalue computation
119 // and the catastrophic cancellation of root-product expansion for clustered
120 // or repeated eigenvalues.
121 //
122 // Recurrence (monic, coefficients of z^n + a[1]z^{n-1} + ... + a[n]):
123 // M_0 = I, a[k] = -trace(A * M_{k-1}) / k, M_k = A*M_{k-1} + a[k]*I
124 // At the end, A * M_n = 0 (Cayley-Hamilton check).
125 std::vector<double> a(n + 1, 0.0);
126 {
127 a[0] = 1.0;
128 Eigen::MatrixXd M = Eigen::MatrixXd::Identity(n, n);
129 for (int k = 1; k <= n; ++k)
130 {
131 M = sys.A * M;
132 a[k] = -M.trace() / k;
133 M.diagonal().array() += a[k];
134 }
135 }
136
137 // ---- Markov parameters: h[0]=D, h[k]=C A^{k-1} B for k=1..n ----
138 std::vector<double> h(n + 1);
139 h[0] = sys.D(0, 0);
140 Eigen::MatrixXd Apow = Eigen::MatrixXd::Identity(n, n); // A^0
141 for (int k = 1; k <= n; ++k)
142 {
143 h[k] = (sys.C * Apow * sys.B)(0, 0);
144 Apow = sys.A * Apow; // A^k for next iter
145 }
146
147 // ---- Numerator: b[k] = Sigma_{j=0}^{k} a[j] h[k-j] ----
148 std::vector<double> num(n + 1, 0.0);
149 for (int k = 0; k <= n; ++k)
150 for (int j = 0; j <= k; ++j)
151 num[k] += a[j] * h[k - j];
152
153 // Convert to z^{-1} convention (monic denominator already has a[0]=1)
154 std::vector<double> den(a.begin(), a.end());
155 return TransferFunction(num, den, sys.Ts);
156 }

158 // ---------------------------------------------------------------------------
159 // c2d internal helper: warn when a discrete A matrix has eigenvalues outside
160 // the unit disk despite a stable continuous-time plant. This can happen when
161 // ||Ac|| * Ts is large (stiff system + coarse sample time) or when the matrix
162 // exponential loses precision for ill-conditioned Ac.
163 // ---------------------------------------------------------------------------
164 static void checkDiscreteStability(const Eigen::MatrixXd &Ad, const char *context)
165 {
166 const auto eigs = Ad.eigenvalues();
167 for (int i = 0; i < eigs.size(); ++i)
168 {
169 if (std::abs(eigs[i]) >= 1.0 + 1e-9)
170 {
171 std::cerr << "[PlantModel::" << context << "] WARNING: A_d eigenvalue "
172 << eigs[i] << " (|lambda| = " << std::abs(eigs[i])
173 << ") lies outside the unit disk.\n"
174 << " This may indicate numerical loss of precision in the matrix "
175 "exponential (ZOH)\n"
176 << " or a bilinear transform of a marginally stable plant (Tustin).\n"
177 << " Check that Ts << 1 / |lambda_max(Ac)| for all fast modes.\n";
178 }
179 }

182 // ---------------------------------------------------------------------------
183 // c2d
184 // ---------------------------------------------------------------------------
185 StateSpace c2d(const StateSpace &sys_c, double Ts, C2dMethod method,
186 double prewarp_freq)
187 {
188 if (Ts <= 0.0)
189 throw std::invalid_argument("c2d: Ts must be positive.");
190 if (sys_c.Ts != 0.0)
191 throw std::invalid_argument("c2d: input model must be continuous-time (Ts == 0).");
192
193 const int n = sys_c.stateSize();
194 const int m = sys_c.inputSize();
195
196 if (method == C2dMethod::ZOH)
197 {
198 // ZOH discretisation via the Van Loan (1978) matrix-exponential embedding.
199 //
200 // For a linear system x' = Ac*x + Bc*u with piecewise-constant input u,
201 // the exact discrete-time equivalent is:
202 // x[k+1] = Ad*x[k] + Bd*u[k]
203 // Ad = expm(Ac*Ts), Bd = (integral_0^Ts expm(Ac*t) dt) * Bc
204 //
205 // Both Ad and Bd are extracted in one shot from the (n+m)*(n+m) augmented
206 // matrix exponential (Van Loan 1978, Eq. 2.4):
207 // expm([Ac Bc] * Ts) = [Ad Bd]
208 // ([0 0 ] ) [0 I ]
209 //
210 // This is numerically preferred over Pade-approximant series for stiff Ac
211 // (large spread of eigenvalues): the matrix exponential is computed via
212 // Schur decomposition, which remains stable even when eigenvalues span many
213 // decades (e.g., the Bell-Astrom boiler model where |lambda_fast|/|lambda_slow|
214 // >> 100). A forward-Euler or Pade approximant at such stiffness ratios
215 // produces discrete eigenvalues outside the unit disk.
216 //
217 // Reference: Van Loan, C.F. (1978). "Computing integrals involving the matrix
218 // exponential." IEEE Trans. Automat. Control, 23(3):395-404.
219 Eigen::MatrixXd M = Eigen::MatrixXd::Zero(n + m, n + m);
220 M.topLeftCorner(n, n) = sys_c.A * Ts;
221 M.topRightCorner(n, m) = sys_c.B * Ts;
222 const Eigen::MatrixXd eM = M.exp();
223
224 const Eigen::MatrixXd Ad = eM.topLeftCorner(n, n);
225 const Eigen::MatrixXd Bd = eM.topRightCorner(n, m);
226 StateSpace result(Ad, Bd, sys_c.C, sys_c.D, Ts);
227 checkDiscreteStability(result.A, "c2d (ZOH)");
228 return result;
229 }
230 else // Tustin or TustinPrewarped
231 {
232 // Standard Tustin: s = (2/Ts) * (z-1)/(z+1) -> alpha = Ts/2
233 //
234 // Prewarped Tustin: s = k_p * (z-1)/(z+1)
235 // k_p = prewarp_freq / tan(prewarp_freq * Ts / 2)
236 // Ensures |G_d(e^{j*prewarp_freq*Ts})| = |G_c(j*prewarp_freq)| exactly.
237 // When prewarp_freq = 0 or method == Tustin: k_p = 2/Ts (standard).
238 //
239 // In both cases: alpha = 1/k_p, and the formulas are identical.
240 double alpha;
241 if (method == C2dMethod::TustinPrewarped && prewarp_freq > 0.0)
242 {
243 const double tan_half = std::tan(prewarp_freq * Ts * 0.5);
244 if (tan_half < 1e-300)
245 throw std::invalid_argument("c2d TustinPrewarped: prewarp_freq * Ts / 2 too small.");
246 alpha = tan_half / prewarp_freq; // = 1/k_p
247 }
248 else
249 {
250 alpha = Ts * 0.5; // standard Tustin: alpha = Ts/2
251 }
252
253 const Eigen::MatrixXd Im = Eigen::MatrixXd::Identity(n, n) - alpha * sys_c.A;
254 const Eigen::MatrixXd Ip = Eigen::MatrixXd::Identity(n, n) + alpha * sys_c.A;
255 const Eigen::PartialPivLU<Eigen::MatrixXd> lu(Im);
256 const Eigen::MatrixXd Ad = lu.solve(Ip);
257 const Eigen::MatrixXd Bd = lu.solve(sys_c.B * (2.0 * alpha)); // Bd = (2 alpha)*(I-alpha*A)^{-1}*B
258 StateSpace result(Ad, Bd, sys_c.C, sys_c.D, Ts);
259 checkDiscreteStability(result.A, method == C2dMethod::TustinPrewarped
260 ? "c2d (TustinPrewarped)" : "c2d (Tustin)");
261 return result;
262 }
263 }

265 // ---------------------------------------------------------------------------
266 // minreal - minimal realization via balanced truncation (Ho-Kalman)
267 // ---------------------------------------------------------------------------
268 StateSpace minreal(const StateSpace &sys, double tol)
269 {
270 const int n = sys.stateSize();
271 const int m = sys.inputSize();
272 const int p = sys.outputSize();
273
274 if (n == 0) return sys;
275
276 Eigen::MatrixXd Co(n, n * m);
277 Eigen::MatrixXd Ob(n * p, n);
278
279 Eigen::MatrixXd Apow = Eigen::MatrixXd::Identity(n, n);
280 for (int i = 0; i < n; ++i) {
281 Co.middleCols(i * m, m) = Apow * sys.B;
282 Ob.middleRows(i * p, p) = sys.C * Apow;
283 Apow = sys.A * Apow;
284 }
285
286 Eigen::MatrixXd H = Ob * Co;
287 Eigen::JacobiSVD<Eigen::MatrixXd> svd(H, Eigen::ComputeThinU | Eigen::ComputeThinV);
288
289 int r = 0;
290 for (int i = 0; i < svd.singularValues().size(); ++i) {
291 if (svd.singularValues()(i) > tol) r++;
292 }
293
294 if (r == n) return sys; // Already minimal
295 if (r == 0) {
296 // Completely unobservable/uncontrollable, return 0 state system
297 return StateSpace(Eigen::MatrixXd::Zero(0, 0), Eigen::MatrixXd::Zero(0, m), Eigen::MatrixXd::Zero(p, 0), sys.D, sys.Ts);
298 }
299
300 Eigen::MatrixXd S_inv_sqrt = svd.singularValues().head(r).cwiseInverse().cwiseSqrt().asDiagonal();
301 Eigen::MatrixXd Ur = svd.matrixU().leftCols(r);
302 Eigen::MatrixXd Vr = svd.matrixV().leftCols(r);
303
304 Eigen::MatrixXd H_shift = Ob * sys.A * Co;
305
306 Eigen::MatrixXd A_min = S_inv_sqrt * Ur.transpose() * H_shift * Vr * S_inv_sqrt;
307 Eigen::MatrixXd B_min = S_inv_sqrt * Ur.transpose() * H.leftCols(m);
308 Eigen::MatrixXd C_min = H.topRows(p) * Vr * S_inv_sqrt;
309
310 return StateSpace(A_min, B_min, C_min, sys.D, sys.Ts);
311 }
