13TaylorApproximator::TaylorApproximator(const std::vector<double> &xs,
14 const std::vector<double> &ys,
15 int degree)
16{
17 const int N = static_cast<int>(xs.size());
18 if (static_cast<int>(ys.size()) != N)
19 throw std::invalid_argument("TaylorApproximator: xs and ys must have the same length.");
20 if (degree < 0)
21 throw std::invalid_argument("TaylorApproximator: degree must be >= 0.");
22 if (N <= degree)
23 throw std::invalid_argument(
24 "TaylorApproximator: need at least degree+1 data points (have " +
25 std::to_string(N) + ", need " + std::to_string(degree + 1) + ").");
26
27 // Build Vandermonde matrix V (N x (degree+1))
28 // V[i][j] = xs[i]^j
29 const int cols = degree + 1;
30 Eigen::MatrixXd V(N, cols);
31 for (int i = 0; i < N; ++i)
32 {
33 double xpow = 1.0;
34 for (int j = 0; j < cols; ++j)
35 {
36 V(i, j) = xpow;
37 xpow *= xs[i];
38 }
39 }
40
41 Eigen::VectorXd y_vec(N);
42 for (int i = 0; i < N; ++i) y_vec(i) = ys[i];
43
44 // Solve V * a = y in least-squares sense via QR decomposition.
45 // ColPivHouseholderQR handles rank-deficient Vandermonde (e.g., duplicate x-values).
46 const Eigen::VectorXd a = V.colPivHouseholderQr().solve(y_vec);
47
48 coeffs_.resize(cols);
49 for (int j = 0; j < cols; ++j) coeffs_[j] = a(j);
50
51 // RMS residual
52 const Eigen::VectorXd resid = V * a - y_vec;
53 rmse_ = std::sqrt(resid.squaredNorm() / N);
54}

56double TaylorApproximator::evaluate(double x) const
57{
58 // Horner's method: p = a[n] + x*(a[n-1] + x*(...))
59 double p = 0.0;
60 for (int j = static_cast<int>(coeffs_.size()) - 1; j >= 0; --j)
61 p = p * x + coeffs_[j];
62 return p;
63}

66// -----------------------------------------------------------------------------
67// PadeApproximator
68// -----------------------------------------------------------------------------
69
70// Linearised Pade fit (Sanathanan-Koerner approach, one iteration).
71//
72// We seek P(x)/Q(x) approx = f(x) with Q(0)=1 (normalised).
73// Rearranging: P(x) - f(x)*Q(x) = 0
74// p0 + p1x + ... + p_mx^m - f(x)*(1 + q1x + ... + q_nx^n) = 0
75//
76// Writing the unknown vector as theta = [p0 ... p_m, q1 ... q_n]ᵀ, each data point
77// (x_i, y_i) contributes one row to the linear system A.theta = b where:
78// A[i, 0..m] = [1, x_i, x_i^2, ..., x_i^m]
79// A[i, m+1..m+n] = [-y_i.x_i, -y_i.x_i^2, ..., -y_i.x_i^n]
80// b[i] = y_i (the Q(0)=1 constant term absorbs y_i)
81//
82// This is solved in the least-squares sense via QR.
83
84PadeApproximator::PadeApproximator(const std::vector<double> &xs,
85 const std::vector<double> &ys,
86 int num_degree,
87 int den_degree)
88{
89 const int N = static_cast<int>(xs.size());
90 if (static_cast<int>(ys.size()) != N)
91 throw std::invalid_argument("PadeApproximator: xs and ys must have the same length.");
92 if (num_degree < 0 || den_degree < 0)
93 throw std::invalid_argument("PadeApproximator: degrees must be >= 0.");
94 const int unknowns = (num_degree + 1) + den_degree; // p's + q's (q0=1 fixed)
95 if (N <= unknowns)
96 throw std::invalid_argument(
97 "PadeApproximator: underdetermined system - need more data points than unknowns "
98 "(have " + std::to_string(N) + ", need > " + std::to_string(unknowns) + ").");
99
100 const int m = num_degree;
101 const int n = den_degree;
102 const int cols = (m + 1) + n;
103
104 Eigen::MatrixXd A(N, cols);
105 Eigen::VectorXd b(N);
106
107 for (int i = 0; i < N; ++i)
108 {
109 const double xi = xs[i];
110 const double yi = ys[i];
111
112 // Numerator columns: 1, xi, xi^2, ..., xi^m
113 double xpow = 1.0;
114 for (int j = 0; j <= m; ++j)
115 {
116 A(i, j) = xpow;
117 xpow *= xi;
118 }
119
120 // Denominator columns (q1..qn): -yi*xi, -yi*xi^2, ..., -yi*xi^n
121 xpow = xi;
122 for (int j = 1; j <= n; ++j)
123 {
124 A(i, m + j) = -yi * xpow;
125 xpow *= xi;
126 }
127
128 b(i) = yi; // right-hand side: yi * Q(0) = yi * 1
129 }
130
131 const Eigen::VectorXd theta = A.colPivHouseholderQr().solve(b);
132
133 // Unpack numerator coefficients p[0..m]
134 p_.resize(m + 1);
135 for (int j = 0; j <= m; ++j) p_[j] = theta(j);
136
137 // Unpack denominator coefficients q[0..n] with q[0] = 1
138 q_.resize(n + 1);
139 q_[0] = 1.0;
140 for (int j = 1; j <= n; ++j) q_[j] = theta(m + j);
141
142 // Compute RMS residual
143 double sse = 0.0;
144 for (int i = 0; i < N; ++i)
145 {
146 double err = evaluate(xs[i]) - ys[i];
147 sse += err * err;
148 }
149 rmse_ = std::sqrt(sse / N);
150}

152double PadeApproximator::evaluate(double x) const
153{
154 // Evaluate numerator P(x) via Horner
155 double P = 0.0;
156 for (int j = static_cast<int>(p_.size()) - 1; j >= 0; --j)
157 P = P * x + p_[j];
158
159 // Evaluate denominator Q(x) via Horner
160 double Q = 0.0;
161 for (int j = static_cast<int>(q_.size()) - 1; j >= 0; --j)
162 Q = Q * x + q_[j];
163
164 if (std::abs(Q) < 1e-300)
165 throw std::domain_error(
166 "PadeApproximator::evaluate: denominator Q(x) is zero at x = " +
167 std::to_string(x) + " - pole in evaluation domain.");
168
169 return P / Q;
170}

172bool PadeApproximator::hasPoleInDomain(double x_lo, double x_hi, int n_pts) const
173{
174 if (n_pts < 2) n_pts = 2;
175 const double dx = (x_hi - x_lo) / (n_pts - 1);
176
177 // Evaluate Q at the first point
178 auto evalQ = [&](double x) {
179 double Q = 0.0;
180 for (int j = static_cast<int>(q_.size()) - 1; j >= 0; --j)
181 Q = Q * x + q_[j];
182 return Q;
183 };
184
185 double Q_prev = evalQ(x_lo);
186 for (int i = 1; i < n_pts; ++i)
187 {
188 double Q_curr = evalQ(x_lo + i * dx);
189 if (Q_prev * Q_curr < 0.0) return true; // sign change -> real root
190 Q_prev = Q_curr;
191 }
192 return false;
193}

195StateSpace PadeApproximator::toDiscreteFilter(double Ts) const
196{
197 if (numDegree() != 1 || denDegree() != 1)
198 throw std::logic_error(
199 "PadeApproximator::toDiscreteFilter: only [1/1] Pade supported. "
200 "Got [" + std::to_string(numDegree()) + "/" + std::to_string(denDegree()) + "].");
201
202 // Rational function in x-domain: (p0 + p1*x) / (1 + q1*x)
203 // Treat x as z^{-1} and convert to state-space (controllable canonical form).
204 // H(z^{-1}) = (p0 + p1*z^{-1}) / (1 + q1*z^{-1})
205 // Multiply through by z: H(z) = (p0*z + p1) / (z + q1)
206 // State-space: A = -q1, B = 1, C = p1 - p0*q1, D = p0
207 const double p0 = p_[0], p1 = p_[1];
208 const double q1 = q_[1];
209
210 Eigen::MatrixXd A(1, 1), B(1, 1), C(1, 1), D(1, 1);
211 A(0, 0) = -q1;
212 B(0, 0) = 1.0;
213 C(0, 0) = p1 - p0 * q1;
214 D(0, 0) = p0;
215
216 return StateSpace(A, B, C, D, Ts);
217}

224StateSpace padeDelayFilter(double theta_frac, double Ts)
225{
226 if (Ts <= 0.0)
227 throw std::invalid_argument("padeDelayFilter: Ts must be positive.");
228 if (theta_frac < 0.0 || theta_frac >= Ts)
229 throw std::invalid_argument(
230 "padeDelayFilter: theta_frac must be in [0, Ts). Got theta_frac=" +
231 std::to_string(theta_frac) + ", Ts=" + std::to_string(Ts) + ".");
232
233 if (theta_frac < 1e-12)
234 {
235 // No fractional delay - return unity static gain (D=1, no dynamics).
236 Eigen::MatrixXd A(1,1), B(1,1), C(1,1), D(1,1);
237 A(0,0)=0.0; B(0,0)=0.0; C(0,0)=0.0; D(0,0)=1.0;
238 return StateSpace(A, B, C, D, Ts);
239 }
240
241 // Continuous [1/1] Pade approximant of e^{-theta_frac * s}:
242 // H_c(s) = (1 - (theta_frac/2)*s) / (1 + (theta_frac/2)*s)
243 //
244 // Discretise with Tustin (bilinear) transform: s = (2/Ts)*(z-1)/(z+1)
245 // Let r = theta_frac / Ts (fractional ratio, 0 < r < 1)
246 //
247 // H(z) = (1 - r*(z-1)/(z+1)) / (1 + r*(z-1)/(z+1))
248 // = ((z+1) - r*(z-1)) / ((z+1) + r*(z-1))
249 // = ((1-r)*z + (1+r)) ... wait - let me expand cleanly:
250 //
251 // Numerator: (z+1) - r*(z-1) = z(1-r) + (1+r)
252 // Denominator: (z+1) + r*(z-1) = z(1+r) + (1-r)
253 //
254 // Divide by (1+r) to get monic denominator:
255 // b0 = (1-r)/(1+r), b1 = 1
256 // a1 = (1-r)/(1+r) (the z^0 / z^1 ratio in denominator)
257 //
258 // In z^{-1} form (multiply through by z^{-1}):
259 // H(z^{-1}) = ((1-r) + (1+r)*z^{-1}) / ((1+r) + (1-r)*z^{-1})
260 // Normalise denominator: divide through by (1+r):
261 // H(z^{-1}) = (b0_n + b1_n*z^{-1}) / (1 + a1_n*z^{-1})
262 // where b0_n = (1-r)/(1+r), b1_n = 1, a1_n = (1-r)/(1+r)
263 //
264 // State-space (controllable canonical, 1 state):
265 // A = -a1_n, B = 1, C = b1_n - b0_n*a1_n, D = b0_n
266
267 const double r = theta_frac / Ts;
268 const double b0_n = (1.0 - r) / (1.0 + r); // = a1_n by symmetry
269 const double a1_n = b0_n;
270 const double b1_n = 1.0;
271
272 Eigen::MatrixXd A(1,1), B(1,1), C(1,1), D(1,1);
273 A(0,0) = -a1_n;
274 B(0,0) = 1.0;
275 C(0,0) = b1_n - b0_n * a1_n;
276 D(0,0) = b0_n;
277
278 return StateSpace(A, B, C, D, Ts);
279}
