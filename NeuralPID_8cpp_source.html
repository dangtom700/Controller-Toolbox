9{

17 std::normal_distribution<double> nd(0.0, 1.0);
18
19 W1_.resize(p_.n_hidden, 3);
20 b1_.setZero(p_.n_hidden);
21 W2_.resize(3, p_.n_hidden);
22 b2_.setZero(3);
23
24 const double scale1 = std::sqrt(2.0 / 3.0);
25 const double scale2 = std::sqrt(2.0 / p_.n_hidden);

28 for (int j = 0; j < 3; ++j)
29 W1_(i, j) = scale1 * nd(rng);
30 for (int j = 0; j < 3; ++j)
31 W2_(j, i) = scale2 * nd(rng);
32 }
33
34 // Bias initialisation: set b2 so initial gains approx = [Kp0, Ki0, Kd0]
35 // softplus^{-1}(y) = log(exp(y) - 1); for y > 20, softplus(y) approx = y so use y directly
36 // to avoid exp() overflow (e.g. Kp0=2000 would overflow to inf otherwise)
37 auto sp_inv = [](double y) {
38 return (y > 20.0) ? y : std::log(std::expm1(y));
39 };
40 b2_(0) = sp_inv(p_.Kp0);
41 b2_(1) = sp_inv(p_.Ki0 + 1e-6);

51Eigen::Vector3d NeuralPID::forward(const Eigen::Vector3d& z,
52 Eigen::VectorXd& h_out) const
53{
54 // Hidden layer: h = tanh(W1*z + b1)
55 h_out = (W1_ * z + b1_).array().tanh();
56
57 // Output layer: gains = softplus(W2*h + b2)
58 Eigen::Vector3d pre = W2_ * h_out + b2_;
59 return pre.unaryExpr([](double x) { return std::log1p(std::exp(x)); });
60}

66double NeuralPID::compute(double error)
67{
68 if (!std::isfinite(error)) return u_prev_;
69 const double e = error;
70 const double e_dot = (e - e_prev_) / p_.Ts;
71 e_int_ += e * p_.Ts;
72
73 Eigen::Vector3d z(e, e_dot, e_int_);
74
75 // Forward pass
76 Eigen::VectorXd h;
77 Eigen::Vector3d gains = forward(z, h);
78 Kp_ = gains(0);
79 Ki_ = gains(1);
80 Kd_ = gains(2);
81
82 // PID output
83 double u = std::clamp(Kp_ * e + Ki_ * e_int_ + Kd_ * e_dot,
84 p_.uMin, p_.uMax);
85
86 // Online gradient update
87 // Loss J = e^2, dJ/du = 2e * de/du approx = 2e * (-plant_gain) (negative feedback)
88 // dJ/dgains: du/dKp = e, du/dKi = e_int, du/dKd = e_dot
89 const double dJ_du = 2.0 * e * (-p_.plant_gain);
90 Eigen::Vector3d du_dgains(e, e_int_, e_dot);
91 Eigen::Vector3d dJ_dgains = dJ_du * du_dgains;
92
93 // dgains_i/dpre_i = softplus'(pre_i)
94 Eigen::Vector3d pre = W2_ * h + b2_;
95 Eigen::Vector3d sp_d = pre.unaryExpr([](double x) { return 1.0 / (1.0 + std::exp(-x)); });
96
97 // dJ/dpre = dJ_dgains .* sp_d
98 Eigen::Vector3d dJ_dpre = dJ_dgains.array() * sp_d.array();
99
100 // Gradients for W2, b2
101 const Eigen::MatrixXd dW2 = dJ_dpre * h.transpose();
102 const Eigen::VectorXd db2 = dJ_dpre;
103
104 // Backprop through hidden: dJ/dh = W2' * dJ_dpre, dJ/dpre1 = dJ/dh .* (1-h^2)
105 Eigen::VectorXd dJ_dh = W2_.transpose() * dJ_dpre;
106 Eigen::VectorXd dJ_dpre1 = dJ_dh.array() * (1.0 - h.array().square());
107
108 const Eigen::MatrixXd dW1 = dJ_dpre1 * z.transpose();
109 const Eigen::VectorXd db1 = dJ_dpre1;
110
111 // Gradient step
112 W2_ -= p_.lr * dW2;
113 b2_ -= p_.lr * db2;
114 W1_ -= p_.lr * dW1;
115 b1_ -= p_.lr * db1;
116
117 clipWeights();
118
119 e_prev_ = e;
120 u_prev_ = u;
121 notifyObserver(u, e);
122 return u;
123}
