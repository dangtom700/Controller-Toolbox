204inline double BayesianOptimizer::acqUCB(const GaussianProcess& gp,
205 const Eigen::VectorXd& xn) const
206{
207 auto pred = gp.predict(xn);

211inline double BayesianOptimizer::acqEI(const GaussianProcess& gp,
212 const Eigen::VectorXd& xn,
213 double f_best) const
214{
215 auto pred = gp.predict(xn);
216 const double sigma = std::sqrt(std::max(pred.variance, 1e-12));
217 const double z = (f_best - pred.mean - p_.xi) / sigma;
218 static constexpr double kInvSqrt2Pi = 0.3989422804014327;
219 const double phi = kInvSqrt2Pi * std::exp(-0.5 * z * z);
220 const double Phi = 0.5 * (1.0 + std::erf(z * 0.7071067811865476));
221 // EI = sigma * (phi + z*Phi); negate to turn argmax into argmin
222 return -(sigma * (phi + z * Phi));
223}

225inline Eigen::VectorXd
226BayesianOptimizer::maximiseAcquisition(const GaussianProcess& gp,
227 double f_best)
228{
229 std::uniform_real_distribution<double> uniform(0.0, 1.0);
230
231 double best_acq = std::numeric_limits<double>::infinity();
232 Eigen::VectorXd best_xn = Eigen::VectorXd::Constant(p_.n, 0.5);
233
234 Eigen::VectorXd xn(p_.n);
235 for (int i = 0; i < p_.n_acq_restarts; ++i) {
236 for (int j = 0; j < p_.n; ++j)
237 xn(j) = uniform(rng_);
238
239 double acq_val = (p_.acq == BayesOptParams::Acquisition::UCB)
240 ? acqUCB(gp, xn)
241 : acqEI(gp, xn, f_best);
242
243 if (acq_val < best_acq) {

245 best_xn = xn;
246 }
247 }

251inline TunerResult BayesianOptimizer::tune(CostFn cost, const Eigen::VectorXd& x0)
252{
253 const int n = p_.n;
254 GaussianProcess gp(n, p_.gp);
255
256 double best_cost = std::numeric_limits<double>::infinity();
257 Eigen::VectorXd best_params = clamp(x0);
258 int n_evals = 0;
259
260 std::uniform_real_distribution<double> uniform(0.0, 1.0);
261
262 // -----------------------------------------------------------------------
263 // Phase 1: random initialisation (includes x0 as the first point)
264 // -----------------------------------------------------------------------
265 auto evaluate = [&](const Eigen::VectorXd& xn) {
266 Eigen::VectorXd x = clamp(denormalise(xn));
267 double c = cost(x);
268 ++n_evals;
269 gp.addPoint(xn, c);
270 if (c < best_cost) {
271 best_cost = c;
272 best_params = x;
273 }
274 };
275
276 // Evaluate x0 first
277 evaluate(normalise(best_params));
278
279 // Fill remaining n_init slots with random points
280 Eigen::VectorXd xn(n);
281 for (int i = 1; i < p_.n_init; ++i) {
282 for (int j = 0; j < n; ++j) xn(j) = uniform(rng_);
283 evaluate(xn);
284 }
285
286 // -----------------------------------------------------------------------
287 // Phase 2: GP-guided BO iterations
288 // -----------------------------------------------------------------------
289 for (int iter = 0; iter < p_.maxIter; ++iter) {
290 gp.fit();
291
292 // Acquisition maximisation (returns normalised x)
293 Eigen::VectorXd xn_next = maximiseAcquisition(gp, best_cost);
294 evaluate(xn_next);
295 }
296
297 return {best_params, best_cost, n_evals, p_.maxIter, true};
298}
