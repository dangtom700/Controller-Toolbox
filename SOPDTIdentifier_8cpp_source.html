14SOPDTIdentifier::SOPDTIdentifier(const std::vector<double> &times,
15 const std::vector<double> &outputs,
16 double stepMagnitude,
17 double stepTime)
18 : step_(stepMagnitude)
19{
20 if (times.size() != outputs.size())
21 throw std::invalid_argument(
22 "SOPDTIdentifier: times and outputs must have the same length.");
23 if (times.size() < 5)
24 throw std::invalid_argument(
25 "SOPDTIdentifier: need at least 5 data points.");
26 if (std::abs(stepMagnitude) < 1e-14)
27 throw std::invalid_argument(
28 "SOPDTIdentifier: stepMagnitude must be non-zero.");
29
30 std::size_t start = 0;
31 while (start < times.size() && times[start] < stepTime - 1e-12)
32 ++start;
33 if (start == times.size())
34 throw std::invalid_argument(
35 "SOPDTIdentifier: stepTime is beyond the data range.");
36
37 y0_ = (start > 0) ? outputs[start - 1] : outputs[start];
38
39 t_.reserve(times.size() - start);
40 y_.reserve(times.size() - start);
41 for (std::size_t i = start; i < times.size(); ++i) {
42 t_.push_back(times[i] - stepTime);
43 y_.push_back(outputs[i] - y0_);
44 }
45
46 if (t_.size() < 4)
47 throw std::invalid_argument(
48 "SOPDTIdentifier: fewer than 4 data points remain after stepTime.");
49
50 Ts_data_ = (t_.back() - t_.front()) / static_cast<double>(t_.size() - 1);
51}

75
76PIDParams SOPDTIdentifier::imcTuning(const SOPDTModel &m, double lambdaC,
77 double Ts, bool piOnly)
78{
79 (void)Ts; // reserved for future discrete IMC; continuous formula used here
80 if (lambdaC <= 0.0)
81 throw std::invalid_argument(
82 "SOPDTIdentifier::imcTuning: lambdaC must be > 0.");
83
84 const double tau1 = std::max(m.tau1, 1e-9);
85 const double tau2 = std::max(m.tau2, 0.0);
86 const double theta = std::max(m.theta, 0.0);
87 const double tauEq = tau1 + tau2; // equivalent first-order time constant
88
89 // IMC-PID parameters (parallel form, first-order Pade for dead time)
90 const double Ti = tauEq;
91 const double Td = piOnly ? 0.0 : (tau1 * tau2) / tauEq;
92 const double Kp = tauEq / (m.K * (lambdaC + 0.5 * theta));
93
94 PIDParams pp;
95 pp.Kp = Kp;
96 pp.Ki = (Ti > 1e-12) ? Kp / Ti : 0.0;
97 pp.Kd = Kp * Td;
98 pp.N = 10.0;
99 pp.uMin = -std::numeric_limits<double>::infinity();
100 pp.uMax = std::numeric_limits<double>::infinity();
101 return pp;
102}

105// Private: SOPDT step response (zero-referenced, relative to t=theta)
106// =============================================================================
107
108double SOPDTIdentifier::sopdtResponse(double dt, double K_norm,
109 double tau1, double tau2,
110 double /*theta*/) const
111{
112 // dt = t - theta (time past dead-time horizon)
113 if (dt <= 0.0) return 0.0;
114
115 const double tol = 1e-9 * (tau1 + tau2 + 1e-30);
116 double unit_step;
117 if (std::abs(tau1 - tau2) < tol) {
118 // Critically damped: 1 - (1 + dt/tau)*exp(-dt/tau)
119 const double tau = 0.5 * (tau1 + tau2);
120 unit_step = 1.0 - (1.0 + dt / tau) * std::exp(-dt / tau);
121 } else {
122 // Overdamped: 1 - (tau1*exp(-dt/tau1) - tau2*exp(-dt/tau2)) / (tau1-tau2)
123 unit_step = 1.0 - (tau1 * std::exp(-dt / tau1)
124 - tau2 * std::exp(-dt / tau2)) / (tau1 - tau2);
125 }
126 return K_norm * step_ * unit_step;
127}

129// =============================================================================
130// Private: graphical method
131// theta : ZN tangent intercept (same as FOPDT)
132// tau_sum: 63.2%-crossing minus theta
133// tau1/tau2 split: estimated from 28.3%-crossing relative to tau_sum.
134// For FOPDT: t_28.3 / t_63.2 = -ln(1-0.283) / 1.0 = 0.332
135// For SOPDT (equal poles): approximately 0.530
136// Interpolate linearly between these two extremes.
137// =============================================================================
138
139SOPDTModel SOPDTIdentifier::identifyGraphical() const
140{
141 const int N = static_cast<int>(y_.size());
142 const double y_final = y_.back();
143 const double K = y_final / step_;
144
145 // --- Step 1: ZN tangent -> theta ---
146 double max_slope = 0.0;
147 int slope_idx = 1;
148 for (int i = 1; i < N - 1; ++i) {
149 const double dt = t_[i + 1] - t_[i - 1];
150 if (dt < 1e-14) continue;
151 const double slope = (y_[i + 1] - y_[i - 1]) / dt;
152 if (slope > max_slope) { max_slope = slope; slope_idx = i; }
153 }
154
155 double theta = 0.0;
156 if (max_slope > 1e-14)
157 theta = std::max(0.0, t_[slope_idx] - y_[slope_idx] / max_slope);
158
159 // --- Step 2: 63.2% crossing -> tau_sum ---
160 const double y_632 = 0.632 * y_final;
161 double t_632 = t_.back();
162 for (int i = 0; i < N - 1; ++i) {
163 if (y_[i] <= y_632 && y_[i + 1] >= y_632) {
164 const double frac = (y_632 - y_[i]) / (y_[i + 1] - y_[i] + 1e-30);
165 t_632 = t_[i] + frac * (t_[i + 1] - t_[i]);
166 break;
167 }
168 }
169 const double tau_sum = std::max(t_632 - theta, 2.0 * Ts_data_);
170
171 // --- Step 3: 28.3% crossing -> estimate tau1/tau2 ratio ---
172 // 28.3% = 1 - exp(-0.332) for FOPDT -> ratio r = (t28-theta)/tau_sum ~ 0.332
173 // 28.3% for critically-damped SOPDT -> ratio r ~ 0.530
174 // Interpolate: if r is close to 0.332 => tau2 very small (nearly FOPDT),
175 // if r is close to 0.530 => tau1 = tau2 (equal poles, critical).
176 const double y_283 = 0.283 * y_final;
177 double t_283 = t_632 * 0.5; // fallback: midpoint
178 for (int i = 0; i < N - 1; ++i) {
179 if (y_[i] <= y_283 && y_[i + 1] >= y_283) {
180 const double frac = (y_283 - y_[i]) / (y_[i + 1] - y_[i] + 1e-30);
181 t_283 = t_[i] + frac * (t_[i + 1] - t_[i]);
182 break;
183 }
184 }
185
186 const double ratio_28 = std::max(0.0, (t_283 - theta) / (tau_sum + 1e-30));
187 // Map ratio from [0.332, 0.530] to split alpha in [0, 0.5]
188 // alpha=0 -> equal poles (tau1=tau2=tau_sum/2), alpha=0.5 -> FOPDT-like (tau2~0)
189 constexpr double r_fopdt = 0.332;
190 constexpr double r_crit = 0.530;
191 double alpha = 0.5 * (ratio_28 - r_crit) / (r_fopdt - r_crit + 1e-30);
192 alpha = std::max(0.0, std::min(0.45, alpha));
193
194 // tau1 is the dominant (larger) pole
195 const double tau1 = tau_sum * (0.5 + alpha);
196 const double tau2 = std::max(tau_sum * (0.5 - alpha), Ts_data_);
197
198 // --- RMSE ---
199 double sse = 0.0;
200 for (int i = 0; i < N; ++i) {
201 const double y_hat = sopdtResponse(t_[i] - theta, K, tau1, tau2, 0.0);
202 const double e = y_[i] - y_hat;
203 sse += e * e;
204 }
205 return {K, tau1, tau2, theta, std::sqrt(sse / N)};
206}

208// =============================================================================
209// Private: optimization via nested golden-section search on (theta, tau1, tau2)
210// Outer: theta in [0, 0.5*T]
211// Middle: tau1 in [Ts_data, T - theta]
212// Inner: tau2 in [Ts_data, tau1] (enforce tau1 >= tau2)
213// =============================================================================
214
215SOPDTModel SOPDTIdentifier::identifyOptimization() const
216{
217 const int N = static_cast<int>(y_.size());
218 const double y_final = y_.back();
219 const double K = y_final / step_;
220 const double T = t_.back();
221
222 // For fixed (theta, tau1), minimise over tau2 in [Ts_data, tau1]
223 auto best_tau2_sse = [&](double theta, double tau1) -> std::pair<double, double> {
224 const double tau2_lo = Ts_data_;
225 const double tau2_hi = std::max(tau1, tau2_lo + 1e-9);
226 const double opt_tau2 = goldenSection(
227 [&](double tau2) { return modelSSE(K, tau1, tau2, theta); },
228 tau2_lo, tau2_hi);
229 return {opt_tau2, modelSSE(K, tau1, opt_tau2, theta)};
230 };
231
232 // For fixed theta, minimise over tau1 in [Ts_data, T-theta], then over tau2
233 auto best_tau1_sse = [&](double theta) -> std::tuple<double, double, double> {
234 const double tau1_lo = Ts_data_;
235 const double tau1_hi = std::max(T - theta, 2.0 * Ts_data_);
236 const double opt_tau1 = goldenSection(
237 [&](double tau1) { return best_tau2_sse(theta, tau1).second; },
238 tau1_lo, tau1_hi);
239 auto [opt_tau2, sse] = best_tau2_sse(theta, opt_tau1);
240 return {opt_tau1, opt_tau2, sse};
241 };
242
243 // Outer: optimise theta
244 const double opt_theta = goldenSection(
245 [&](double theta) { return std::get<2>(best_tau1_sse(theta)); },
246 0.0, 0.5 * T);
247
248 auto [opt_tau1, opt_tau2, sse] = best_tau1_sse(opt_theta);
249
250 // Ensure tau1 >= tau2 (ordering convention)
251 if (opt_tau1 < opt_tau2) std::swap(opt_tau1, opt_tau2);
252
253 const double rmse = std::sqrt(sse / N);
254 return {K, opt_tau1, opt_tau2, opt_theta, rmse};
255}

273// =============================================================================
274// Private: golden-section search on [a, b]
275// =============================================================================
276
277double SOPDTIdentifier::goldenSection(std::function<double(double)> f,
278 double a, double b, double tol)
279{
280 constexpr double gr = 0.6180339887498949; // (sqrt(5)-1)/2
281 double c = b - gr * (b - a);
282 double d = a + gr * (b - a);
283 while (std::abs(b - a) > tol) {
284 if (f(c) < f(d)) { b = d; }
285 else { a = c; }
286 c = b - gr * (b - a);
287 d = a + gr * (b - a);
288 }
289 return 0.5 * (a + b);
290}
