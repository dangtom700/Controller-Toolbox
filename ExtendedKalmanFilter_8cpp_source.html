23 void ExtendedKalmanFilter::predict(const Eigen::VectorXd &u)
24 {
25 // Linearise f around x^[k|k] to get the state Jacobian F = df/dx.
26 // F is used only for the covariance propagation P = F*P*F' + Q; the
27 // state itself is propagated through the exact nonlinear function f.
28 //
29 // Stability assumption: f_ must be a well-conditioned discrete-time map.
30 // If f_ is internally a forward-Euler step at the control Ts, accuracy and
31 // stability degrade for stiff plants (e.g. hydraulic actuators with high bulk
32 // modulus, electrical systems with fast electrical time constants). For such
33 // plants, implement f_ with RK4 or a fixed-step ODE solver at a sub-step Ts/N.
34 const Eigen::MatrixXd F = Fjac_(x_hat_, u);
35 x_hat_ = f_(x_hat_, u);
36 P_ = F * P_ * F.transpose() + Q_;
37 }

39 void ExtendedKalmanFilter::update(const Eigen::VectorXd &y,
40 const Eigen::VectorXd &u)
41 {
42 // Linearise h around x^[k+1|k] (the post-predict estimate) to get H = dh/dx.
43 // H is evaluated at the predicted state, which is the current x_hat_ after predict().
44 const Eigen::MatrixXd H = Hjac_(x_hat_, u);
45
46 Eigen::MatrixXd R_safe = R_;
47 R_safe.diagonal() = R_safe.diagonal().cwiseMax(1e-12);
48
49 const Eigen::MatrixXd S = H * P_ * H.transpose() + R_safe;
50 const auto ldlt = S.ldlt();
51 if (ldlt.info() != Eigen::Success)
52 return;
53
54 const Eigen::MatrixXd K =
55 P_ * H.transpose() * ldlt.solve(Eigen::MatrixXd::Identity(n_outputs_, n_outputs_));
56
57 x_hat_.noalias() += K * (y - h_(x_hat_, u));
58
59 // Joseph form: P = (I-KH).P.(I-KH)' + K.R.K'
60 const Eigen::MatrixXd IKH = Eigen::MatrixXd::Identity(n_states_, n_states_) - K * H;
61 P_ = IKH * P_ * IKH.transpose() + K * R_safe * K.transpose();
62 }

77 Eigen::MatrixXd ExtendedKalmanFilter::numericalJacobian(
78 const std::function<Eigen::VectorXd(const Eigen::VectorXd &)> &func,
79 const Eigen::VectorXd &x,
80 double eps_scale)
81 {
82 // Central-difference: J(:,i) = (f(x+h_i*ei) - f(x-h_i*ei)) / (2*h_i)
83 // h_i = eps_scale * max(|x_i|, 1) gives O(eps_scale^2) truncation error regardless
84 // of state magnitude - essential for heterogeneous state vectors (positions, angles,
85 // velocities may differ by orders of magnitude).
86 const Eigen::VectorXd f0 = func(x);
87 const int nx = x.size();
88 const int ny = f0.size();
89 Eigen::MatrixXd J(ny, nx);
90 Eigen::VectorXd xp = x, xm = x;
91 for (int i = 0; i < nx; ++i)
92 {
93 const double h = eps_scale * std::max(std::abs(x(i)), 1.0);
94 xp(i) = x(i) + h;
95 xm(i) = x(i) - h;
96 J.col(i) = (func(xp) - func(xm)) / (2.0 * h);
97 xp(i) = x(i);
98 xm(i) = x(i);
99 }
100 return J;
101 }
