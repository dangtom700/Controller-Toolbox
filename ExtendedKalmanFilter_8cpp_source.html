23 void ExtendedKalmanFilter::predict(const Eigen::VectorXd &u)
24 {
25 // Linearise f around x^[k|k] to get the state Jacobian F = df/dx.
26 // F is used only for the covariance propagation P = F*P*F' + Q; the
27 // state itself is propagated through the exact nonlinear function f.
28 //
29 // Stability assumption: f_ must be a well-conditioned discrete-time map.
30 // If f_ is internally a forward-Euler step at the control Ts, accuracy and
31 // stability degrade for stiff plants (e.g. hydraulic actuators with high bulk
32 // modulus, electrical systems with fast electrical time constants). For such
33 // plants, implement f_ with RK4 or a fixed-step ODE solver at a sub-step Ts/N.
34 const Eigen::MatrixXd F = Fjac_(x_hat_, u);
35 x_hat_ = f_(x_hat_, u);
36 P_ = F * P_ * F.transpose() + Q_;
37 }

39 void ExtendedKalmanFilter::update(const Eigen::VectorXd &y,
40 const Eigen::VectorXd &u)
41 {
42 // Linearise h around x^[k+1|k] (the post-predict estimate) to get H = dh/dx.
43 // H is evaluated at the predicted state, which is the current x_hat_ after predict().
44 const Eigen::MatrixXd H = Hjac_(x_hat_, u);
45
46 Eigen::MatrixXd R_safe = R_;
47 R_safe.diagonal() = R_safe.diagonal().cwiseMax(1e-12);
48
49 const Eigen::MatrixXd S = H * P_ * H.transpose() + R_safe;
50 const auto ldlt = S.ldlt();
51 if (ldlt.info() != Eigen::Success)
52 return;
53
54 const Eigen::MatrixXd K =
55 P_ * H.transpose() * ldlt.solve(Eigen::MatrixXd::Identity(n_outputs_, n_outputs_));
56
57 x_hat_.noalias() += K * (y - h_(x_hat_, u));
58
59 // Joseph form: P = (I-KH).P.(I-KH)' + K.R.K'
60 const Eigen::MatrixXd IKH = Eigen::MatrixXd::Identity(n_states_, n_states_) - K * H;
61 P_ = IKH * P_ * IKH.transpose() + K * R_safe * K.transpose();
62
63 // P3: project algebraic states onto constraint manifold (if constraint attached)
64 if (alg_set_)
65 projectAlgebraicStates(u);
66 }

71 void ExtendedKalmanFilter::setAlgebraicConstraint(
72 AlgConstraintFn g_alg, int n_diff, int n_alg, double newton_tol)
73 {
74 if (n_diff + n_alg != n_states_)
75 throw std::invalid_argument(

77 if (n_diff <= 0 || n_alg <= 0)

80 alg_g_ = std::move(g_alg);
81 n_diff_alg_ = n_diff;
82 n_alg_ = n_alg;
83 alg_tol_ = newton_tol;
84 alg_set_ = true;
85 }

87 // ---------------------------------------------------------------------------
88 // projectAlgebraicStates - Newton-project x2 block, then project covariance.
89 //
90 // Newton iterations: x2 <- x2 - G2^{-1} * g(x1, x2, u)
91 // Covariance projection: P <- J_proj * P * J_proj'
92 // where J_proj = [[I_{n1}, 0]; [-G2^{-1}*G1, 0_{n2}]]
93 // ---------------------------------------------------------------------------
94 void ExtendedKalmanFilter::projectAlgebraicStates(const Eigen::VectorXd &u)
95 {
96 const int n1 = n_diff_alg_;
97 const int n2 = n_alg_;
98 const double u_s = u.size() > 0 ? u(0) : 0.0; // SISO scalar input
99
100 Eigen::VectorXd x1 = x_hat_.head(n1);
101 Eigen::VectorXd x2 = x_hat_.tail(n2);
102
103 // Newton-Raphson: solve g(x1, x2, u) = 0 for x2
104 for (int iter = 0; iter < 20; ++iter)
105 {
106 const Eigen::VectorXd g_val = alg_g_(x1, x2, u_s);
107 if (g_val.norm() < alg_tol_) break;
108
109 // Central-difference dg/dx2 (n2 x n2)
110 const Eigen::MatrixXd G2 = numericalJacobian(
111 [&](const Eigen::VectorXd &x2v) { return alg_g_(x1, x2v, u_s); }, x2);
112
113 const auto ldlt = G2.ldlt();
114 if (ldlt.info() != Eigen::Success) break;
115 x2 -= ldlt.solve(g_val);
116 }
117 x_hat_.tail(n2) = x2;
118
119 // Covariance projection: J_proj = [[I, 0]; [-G2^{-1}*G1, 0]]
120 // Numerically compute G1 = dg/dx1 and G2 = dg/dx2 at the projected point
121 const Eigen::MatrixXd G1 = numericalJacobian(
122 [&](const Eigen::VectorXd &x1v) { return alg_g_(x1v, x2, u_s); }, x1); // n2 x n1
123 const Eigen::MatrixXd G2 = numericalJacobian(
124 [&](const Eigen::VectorXd &x2v) { return alg_g_(x1, x2v, u_s); }, x2); // n2 x n2
125
126 Eigen::MatrixXd G2inv_G1 = G2.ldlt().solve(G1); // n2 x n1 (best-effort; not checked)
127
128 Eigen::MatrixXd J_proj = Eigen::MatrixXd::Zero(n1 + n2, n1 + n2);
129 J_proj.topLeftCorner(n1, n1) = Eigen::MatrixXd::Identity(n1, n1);
130 J_proj.bottomLeftCorner(n2, n1) = -G2inv_G1;
131 // bottomRightCorner remains zero: x2 variance explained by x1 variance via constraint
132
133 P_ = J_proj * P_ * J_proj.transpose();
134 }

149 Eigen::MatrixXd ExtendedKalmanFilter::numericalJacobian(
150 const std::function<Eigen::VectorXd(const Eigen::VectorXd &)> &func,
151 const Eigen::VectorXd &x,
152 double eps_scale)
153 {
154 // Central-difference: J(:,i) = (f(x+h_i*ei) - f(x-h_i*ei)) / (2*h_i)
155 // h_i = eps_scale * max(|x_i|, 1) gives O(eps_scale^2) truncation error regardless
156 // of state magnitude - essential for heterogeneous state vectors (positions, angles,
157 // velocities may differ by orders of magnitude).
158 const Eigen::VectorXd f0 = func(x);
159 const int nx = x.size();
160 const int ny = f0.size();
161 Eigen::MatrixXd J(ny, nx);
162 Eigen::VectorXd xp = x, xm = x;
163 for (int i = 0; i < nx; ++i)
164 {
165 const double h = eps_scale * std::max(std::abs(x(i)), 1.0);
166 xp(i) = x(i) + h;
167 xm(i) = x(i) - h;
168 J.col(i) = (func(xp) - func(xm)) / (2.0 * h);
169 xp(i) = x(i);
170 xm(i) = x(i);
171 }
172 return J;
173 }
