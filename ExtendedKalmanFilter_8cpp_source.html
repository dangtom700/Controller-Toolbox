23 void ExtendedKalmanFilter::predict(const Eigen::VectorXd &u)
24 {
25 // Linearise f around x^[k|k] to get the state Jacobian F = df/dx.
26 // F is used only for the covariance propagation P = F*P*F' + Q; the
27 // state itself is propagated through the exact nonlinear function f.
28 const Eigen::MatrixXd F = Fjac_(x_hat_, u);
29 x_hat_ = f_(x_hat_, u);
30 P_ = F * P_ * F.transpose() + Q_;
31 }

33 void ExtendedKalmanFilter::update(const Eigen::VectorXd &y,
34 const Eigen::VectorXd &u)
35 {
36 // Linearise h around x^[k+1|k] (the post-predict estimate) to get H = dh/dx.
37 // H is evaluated at the predicted state, which is the current x_hat_ after predict().
38 const Eigen::MatrixXd H = Hjac_(x_hat_, u);
39
40 Eigen::MatrixXd R_safe = R_;
41 R_safe.diagonal() = R_safe.diagonal().cwiseMax(1e-12);
42
43 const Eigen::MatrixXd S = H * P_ * H.transpose() + R_safe;
44 const auto ldlt = S.ldlt();
45 if (ldlt.info() != Eigen::Success)
46 return;
47
48 const Eigen::MatrixXd K =
49 P_ * H.transpose() * ldlt.solve(Eigen::MatrixXd::Identity(p_, p_));
50
51 x_hat_ += K * (y - h_(x_hat_, u));
52
53 // Joseph form: P = (I-KH).P.(I-KH)' + K.R.K'
54 const Eigen::MatrixXd IKH = Eigen::MatrixXd::Identity(n_, n_) - K * H;
55 P_ = IKH * P_ * IKH.transpose() + K * R_safe * K.transpose();
56 }

71 Eigen::MatrixXd ExtendedKalmanFilter::numericalJacobian(
72 const std::function<Eigen::VectorXd(const Eigen::VectorXd &)> &func,
73 const Eigen::VectorXd &x,
74 double eps)
75 {
76 // Central-difference approximation: J(:,i) = (f(x+eps*ei) - f(x-eps*ei)) / (2*eps)
77 // Error is O(eps^2) vs O(eps) for forward-difference. Default eps=1e-5 gives
78 // ~1e-10 truncation error for smooth functions. Scale eps with |x(i)| for
79 // states with very large or very small magnitudes (not done here for simplicity).
80 const Eigen::VectorXd f0 = func(x);
81 const int nx = x.size();
82 const int ny = f0.size();
83 Eigen::MatrixXd J(ny, nx);
84 Eigen::VectorXd xp = x, xm = x;
85 for (int i = 0; i < nx; ++i)
86 {
87 xp(i) = x(i) + eps;
88 xm(i) = x(i) - eps;
89 J.col(i) = (func(xp) - func(xm)) / (2.0 * eps);
90 xp(i) = x(i);
91 xm(i) = x(i);
92 }
93 return J;
94 }
