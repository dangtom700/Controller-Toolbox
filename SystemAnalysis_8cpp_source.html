39 Eigen::MatrixXd SystemAnalysis::solveDiscreteLyapunov(const Eigen::MatrixXd &A,
40 const Eigen::MatrixXd &Q)
41 {
42 // Vectorisation approach: apply vec() to both sides of A*P*A' - P + Q = 0.
43 // Using the Kronecker identity vec(A*P*A') = (A\otimesA)*vec(P):
44 // (A\otimesA - I) * vec(P) = -vec(Q) => (I - A\otimesA) * vec(P) = vec(Q)
45 // Solve this n^2 x n^2 linear system for vec(P), then reshape.
46 // Cost: O(n^6) - suitable for small systems only (n <= ~10).
47 // For n > 10, the Bartels-Stewart O(n^3) algorithm is strongly preferred.
48 // See header comment for references.
49 int n = A.rows();
50 if (n != A.cols() || n != Q.rows() || n != Q.cols())
51 {
52 throw std::invalid_argument("solveDiscreteLyapunov: Matrix dimensions mismatch.");
53 }
54
55 // Kronecker product: A \otimes A
56 Eigen::MatrixXd kronAA(n * n, n * n);
57 for (int i = 0; i < n; ++i)
58 {
59 for (int j = 0; j < n; ++j)
60 {
61 kronAA.block(i * n, j * n, n, n) = A(i, j) * A;
62 }
63 }
64
65 Eigen::MatrixXd I = Eigen::MatrixXd::Identity(n * n, n * n);
66 Eigen::MatrixXd M = I - kronAA;
67
68 // vec(Q)
69 Eigen::VectorXd vecQ(n * n);
70 for (int j = 0; j < n; ++j)
71 {
72 for (int i = 0; i < n; ++i)
73 {
74 vecQ(j * n + i) = Q(i, j);
75 }
76 }
77
78 // Solve M * vec(P) = vec(Q)
79 Eigen::VectorXd vecP = M.colPivHouseholderQr().solve(vecQ);
80
81 // Reshape back to matrix P
82 Eigen::MatrixXd P(n, n);
83 for (int j = 0; j < n; ++j)
84 {
85 for (int i = 0; i < n; ++i)
86 {
87 P(i, j) = vecP(j * n + i);
88 }
89 }
90 return P;
91 }

94 const std::vector<double> &freqs)

105

120 StabilityMargins SystemAnalysis::calculateMargins(const StateSpace &sys)
121 {
122 // Precompute complex matrices once for all frequency evaluations
123 const int n = sys.A.rows();
124 const Eigen::MatrixXcd I = Eigen::MatrixXcd::Identity(n, n);
125 const Eigen::MatrixXcd Ac = sys.A.cast<std::complex<double>>();
126 const Eigen::MatrixXcd Bc = sys.B.cast<std::complex<double>>();
127 const Eigen::MatrixXcd Cc = sys.C.cast<std::complex<double>>();
128 const Eigen::MatrixXcd Dc = sys.D.cast<std::complex<double>>();
129
130 // Evaluate G(e^{jwTs}) at a single frequency - shared by coarse scan and bisection
131 auto evalAt = [&](double w) -> std::complex<double>
132 {
133 std::complex<double> z = std::polar(1.0, w * sys.Ts);
134 Eigen::MatrixXcd G = Cc * (z * I - Ac).inverse() * Bc + Dc;
135 return G(0, 0);
136 };
137
138 // Step 1: Coarse logarithmic grid to locate crossover brackets (~200 evals).
139 // Phase is unwrapped continuously across consecutive grid points so that
140 // sign-change detection for the -180^\circ crossing works correctly for higher-order
141 // plants and non-minimum-phase systems.
142 const double w_min = 1e-3;
143 const double w_max = kPi / sys.Ts;
144 const int coarse = 200;
145
146 std::vector<double> fw(coarse), mag(coarse), pha(coarse);
147 pha[0] = std::arg(evalAt(w_min * std::pow(w_max / w_min, 0.0))) * 180.0 / kPi;
148 fw[0] = w_min;
149 mag[0] = std::abs(evalAt(fw[0]));
150 for (int i = 1; i < coarse; ++i)
151 {
152 fw[i] = w_min * std::pow(w_max / w_min, static_cast<double>(i) / (coarse - 1));
153 auto g = evalAt(fw[i]);
154 mag[i] = std::abs(g);
155 // Unwrap: add the smallest possible +/-360^\circ shift to keep the phase continuous
156 double raw = std::arg(g) * 180.0 / kPi;
157 double diff = raw - pha[i - 1];
158 while (diff > 180.0) diff -= 360.0;
159 while (diff < -180.0) diff += 360.0;
160 pha[i] = pha[i - 1] + diff;
161 }
162
163 // Step 2: Bisection refinement over each bracket - converges in ~50 evals per crossing.
164 // Bisection tracks the unwrapped phase by carrying the last known unwrapped value at
165 // the left endpoint (p_lo) and shifting the midpoint evaluation by +/-360^\circ to match.
166 const int bisect_iters = 50;
167
168 StabilityMargins margins;
169 margins.gainMarginDb = std::numeric_limits<double>::infinity();
170 margins.phaseMarginDeg = std::numeric_limits<double>::infinity();
171 margins.wCrossoverGain = 0.0;
172 margins.wCrossoverPhase = 0.0;
173
174 // Helper: unwrap a raw arg() result relative to a reference unwrapped value.
175 auto unwrapRelative = [](double raw_deg, double ref_deg) -> double {
176 double diff = raw_deg - ref_deg;
177 while (diff > 180.0) diff -= 360.0;
178 while (diff < -180.0) diff += 360.0;
179 return ref_deg + diff;
180 };
181
182 // Gain margin: locate every phase = -180^\circ crossing; keep worst-case (smallest GM)
183 for (int i = 0; i + 1 < coarse; ++i)
184 {
185 if ((pha[i] - (-180.0)) * (pha[i + 1] - (-180.0)) >= 0.0) continue;
186
187 double lo = fw[i], hi = fw[i + 1];
188 double p_lo = pha[i];
189 for (int k = 0; k < bisect_iters; ++k)
190 {
191 double mid = std::sqrt(lo * hi); // geometric midpoint preserves log scale
192 double p_mid = unwrapRelative(std::arg(evalAt(mid)) * 180.0 / kPi, p_lo);
193 if ((p_lo - (-180.0)) * (p_mid - (-180.0)) < 0.0)
194 hi = mid;
195 else { lo = mid; p_lo = p_mid; }
196 }
197 const double w_cross = std::sqrt(lo * hi);
198 const double mag_cross = std::abs(evalAt(w_cross));
199 const double gm_cand = (mag_cross >= 1e-300)
200 ? -20.0 * std::log10(mag_cross)
201 : std::numeric_limits<double>::infinity();
202 if (gm_cand < margins.gainMarginDb)
203 {
204 margins.gainMarginDb = gm_cand;
205 margins.wCrossoverGain = w_cross;
206 }
207 }
208
209 // Phase margin: locate every |G| = 1 crossing; keep worst-case (smallest PM)
210 for (int i = 0; i + 1 < coarse; ++i)
211 {
212 if ((mag[i] - 1.0) * (mag[i + 1] - 1.0) >= 0.0) continue;
213
214 double lo = fw[i], hi = fw[i + 1];
215 double m_lo = mag[i];
216 for (int k = 0; k < bisect_iters; ++k)
217 {
218 double mid = std::sqrt(lo * hi);
219 double m_mid = std::abs(evalAt(mid));
220 if ((m_lo - 1.0) * (m_mid - 1.0) < 0.0)
221 hi = mid;
222 else { lo = mid; m_lo = m_mid; }
223 }
224 // Find the unwrapped phase at the crossing by locating the bracket in the
225 // coarse grid and unwrapping relative to the left grid neighbour.
226 const double w_cross = std::sqrt(lo * hi);
227 const double phase_raw = std::arg(evalAt(w_cross)) * 180.0 / kPi;
228 const double phase_cross = unwrapRelative(phase_raw, pha[i]);
229 const double pm_cand = 180.0 + phase_cross;
230 if (pm_cand < margins.phaseMarginDeg)
231 {
232 margins.phaseMarginDeg = pm_cand;
233 margins.wCrossoverPhase = w_cross;
234 }
235 }
236
237 return margins;
238 }

240 double SystemAnalysis::calculateHInfinityNorm(const StateSpace &sys)
241 {
242 // Step 1: coarse logarithmic grid to locate the approximate peak frequency.
243 const double w_min = 1e-3;
244 const double w_max = kPi / sys.Ts;
245 const int num_points = 500;
246
247 const int n = sys.A.rows();
248 const Eigen::MatrixXcd I = Eigen::MatrixXcd::Identity(n, n);
249 const Eigen::MatrixXcd Ac = sys.A.cast<std::complex<double>>();
250 const Eigen::MatrixXcd Bc = sys.B.cast<std::complex<double>>();
251 const Eigen::MatrixXcd Cc = sys.C.cast<std::complex<double>>();
252 const Eigen::MatrixXcd Dc = sys.D.cast<std::complex<double>>();
253
254 // Evaluate max singular value at frequency w (log-domain: w in natural frequency)
255 auto evalSV = [&](double w) -> double
256 {
257 const std::complex<double> z = std::polar(1.0, w * sys.Ts);
258 const Eigen::MatrixXcd G = Cc * (z * I - Ac).inverse() * Bc + Dc;
259 Eigen::JacobiSVD<Eigen::MatrixXcd> svd(G);
260 return svd.singularValues()(0);
261 };
262
263 double peak_mag = 0.0;
264 int peak_i = 0;
265 std::vector<double> grid_w(num_points);
266
267 for (int i = 0; i < num_points; ++i)
268 {
269 grid_w[i] = w_min * std::pow(w_max / w_min, static_cast<double>(i) / (num_points - 1));
270 const double sv = evalSV(grid_w[i]);
271 if (sv > peak_mag) { peak_mag = sv; peak_i = i; }
272 }
273
274 // Step 2: golden-section search in the log-frequency bracket around the grid peak.
275 // This refines the grid lower bound to the true (or near-true) Hinf norm in O(50) evals.
276 const double phi = (std::sqrt(5.0) - 1.0) * 0.5; // golden ratio reciprocal
277 double a = (peak_i > 0) ? grid_w[peak_i - 1] : w_min;
278 double b = (peak_i < num_points - 1) ? grid_w[peak_i + 1] : w_max;
279
280 // Use log-domain for better numerical behaviour across wide frequency ranges.
281 double la = std::log(a), lb_w = std::log(b);
282 double lc = lb_w - phi * (lb_w - la);
283 double ld = la + phi * (lb_w - la);
284 double fc = evalSV(std::exp(lc));
285 double fd = evalSV(std::exp(ld));
286
287 const int gs_iters = 60; // typically reduces interval by phi^60 ~ 1e-13
288 for (int k = 0; k < gs_iters; ++k)
289 {
290 if (fc < fd)
291 {
292 la = lc;
293 lc = ld; fc = fd;
294 ld = la + phi * (lb_w - la);
295 fd = evalSV(std::exp(ld));
296 }
297 else
298 {
299 lb_w = ld;
300 ld = lc; fd = fc;
301 lc = lb_w - phi * (lb_w - la);
302 fc = evalSV(std::exp(lc));
303 }
304 }
305 const double refined = std::max(fc, fd);
306 return std::max(peak_mag, refined); // take the best of grid and refined peak
307 }
