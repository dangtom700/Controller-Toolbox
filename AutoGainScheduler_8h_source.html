90inline Eigen::VectorXd findEquilibrium(const StateFunc& f,
91 const Eigen::VectorXd& u_eq,
92 const Eigen::VectorXd& x0,
93 int max_iter = 50,
94 double tol = 1e-8)
95{
96 Eigen::VectorXd x = x0;
97 for (int iter = 0; iter < max_iter; ++iter) {
98 Eigen::VectorXd fx = f(x, u_eq);
99 if (fx.norm() < tol) return x;
100 Eigen::MatrixXd J = jacobianX(f, x, u_eq);
101 // Solve J*dx = -f(x) via QR; fallback to pseudoinverse for degenerate J.
102 Eigen::VectorXd dx = J.colPivHouseholderQr().solve(-fx);
103 x += dx;
104 }
105 // Accept if residual is within a loose tolerance
106 if (f(x, u_eq).norm() < tol * 1e3)
107 return x;
108 throw std::runtime_error(
109 "findEquilibrium: Newton-Raphson failed to converge. "
110 "Check that f(x_eq, u_eq)=0 has a solution near x0.");
111}

131buildAutoGainScheduler(
132 const StateFunc& f,
133 double p_min, double p_max, int grid_density,
134 const std::function<Eigen::VectorXd(double p)>& u_eq_fn,
135 const std::function<Eigen::VectorXd(double p)>& x0_fn,
136 const std::function<std::shared_ptr<IController>(const StateSpace&, double p)>& design_fn,
137 double Ts,
138 double gap_threshold = 0.5,
139 int freq_points = 200)
140{
141 if (grid_density < 2)
142 throw std::invalid_argument("buildAutoGainScheduler: grid_density must be >= 2.");
143
144 // ---- Phase 1-2: grid sweep + equilibrium + linearisation ----------------
145 std::vector<double> p_grid(grid_density);
146 std::vector<StateSpace> models;
147 models.reserve(grid_density);
148
149 for (int i = 0; i < grid_density; ++i) {
150 double p = p_min + (p_max - p_min) * i / (grid_density - 1);
151 p_grid[i] = p;
152
153 Eigen::VectorXd u_eq = u_eq_fn(p);
154 Eigen::VectorXd x0 = x0_fn(p);
155 Eigen::VectorXd x_eq = findEquilibrium(f, u_eq, x0);
156
157 models.push_back(lineariseAtPoint(f, x_eq, u_eq, Ts));
158 }
159
160 // ---- Phase 3-4: nu-gap matrix + clustering --------------------------------
161 Eigen::MatrixXd G = nuGapMatrix(models, freq_points);
162 ClusterResult clusters = clusterByGap(G, gap_threshold);
163
164 // ---- Phase 5-6: design + assemble ----------------------------------------
165 auto scheduler = std::make_shared<GainScheduledController>(
166 Ts, GainScheduleMode::LinearBlend);
167
168 for (int c = 0; c < clusters.numClusters; ++c) {
169 int rep_idx = clusters.representatives[c];
170 double p_rep = p_grid[rep_idx];
171 auto ctrl = design_fn(models[rep_idx], p_rep);
172 scheduler->addSchedulePoint(p_rep, ctrl);
173 }
174
175 return scheduler;
176}
