35// ---------------------------------------------------------------------------
36// Norm-optimal pre-computation
37// L_opt = (G'*rho_e*G + rho_u*I)^{-1} * G' * rho_e
38// Q_e = I - L_opt * G
39// ---------------------------------------------------------------------------
40
41void ILC::buildNormOptimal(const Eigen::MatrixXd& G)
42{
43 const int N = p_.N;
44 const Eigen::MatrixXd GtG = G.transpose() * G;
45 const Eigen::MatrixXd M = p_.rho_e * GtG
46 + p_.rho_u * Eigen::MatrixXd::Identity(N, N);
47
48 Eigen::LDLT<Eigen::MatrixXd> ldlt(M);
49 if (ldlt.info() != Eigen::Success)
50 throw std::runtime_error("ILC (norm-optimal): failed to factor (G'G + rho_u.I)");
51
52 // L_opt = M^{-1} * G' * rho_e -> solve M * L_opt' = G * rho_e
53 const Eigen::MatrixXd rhs = p_.rho_e * G.transpose();
54 // ldlt.solve(rhs) solves M * X = rhs for X = L_opt' -> transpose
55 L_opt_ = ldlt.solve(rhs).transpose(); // N x N
56
57 Q_e_ = Eigen::MatrixXd::Identity(N, N) - L_opt_ * G;
58}

77void ILC::updateFeedforward()
78{
79 rms_last_ = std::sqrt(e_curr_.squaredNorm() / p_.N);
80
81 Eigen::VectorXd u_new(p_.N);
82
83 if (p_.mode == Mode::NormOptimal && has_G_) {
84 // Norm-optimal: u_{j+1} = Q_e * u_j + L_opt * e_j
85 u_new = Q_e_ * u_ff_ + L_opt_ * e_curr_;
86
87 } else if (p_.mode == Mode::DType) {
88 // D-type: u_{j+1}[k] = Q * u_j[k] + Lp * e_j[k] + Ld * (e_j[k] - e_j[k-1]) / Ts
89 for (int k = 0; k < p_.N; ++k) {
90 double de = (k > 0) ? (e_curr_(k) - e_curr_(k - 1)) / p_.Ts
91 : (e_curr_(k) - e_prev_(p_.N - 1)) / p_.Ts;
92 u_new(k) = p_.Q_filter * u_ff_(k)
93 + p_.Lp * e_curr_(k)
94 + p_.Ld * de;
95 }
96
97 } else {
98 // P-type (default): u_{j+1}[k] = Q * u_j[k] + Lp * e_j[k]
99 for (int k = 0; k < p_.N; ++k)
100 u_new(k) = p_.Q_filter * u_ff_(k) + p_.Lp * e_curr_(k);
101 }
102
103 // Saturate feedforward
104 u_new = u_new.cwiseMin(p_.uMax).cwiseMax(p_.uMin);
105
106 // Shift: previous error becomes e_prev for D-type
107 e_prev_ = e_curr_;
108 e_curr_.setZero();
109
110 u_ff_ = u_new;
111 ++trial_;
112}
