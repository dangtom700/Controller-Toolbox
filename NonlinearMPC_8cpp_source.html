12static Eigen::MatrixXd jacX_discrete(
13 const NonlinearMPC::DiscreteDynamics &f,
14 const Eigen::VectorXd &x,
15 const Eigen::VectorXd &u,
16 double eps = 1e-5)
17{
18 // Wrap the discrete dynamics into the StateFunc signature (same as LinearisationHelper)
19 // and apply central-difference Jacobian w.r.t. x.
20 const int n = static_cast<int>(x.size());
21 const int out_dim = static_cast<int>(f(x, u).size());
22 Eigen::MatrixXd J(out_dim, n);
23
24 for (int i = 0; i < n; ++i)
25 {
26 const double h = eps * std::max(std::abs(x(i)), 1.0);
27 Eigen::VectorXd xp = x, xm = x;
28 xp(i) += h;
29 xm(i) -= h;
30 J.col(i) = (f(xp, u) - f(xm, u)) / (2.0 * h);
31 }
32 return J;
33}

35static Eigen::MatrixXd jacU_discrete(
36 const NonlinearMPC::DiscreteDynamics &f,
37 const Eigen::VectorXd &x,
38 const Eigen::VectorXd &u,
39 double eps = 1e-5)
40{
41 const int m = static_cast<int>(u.size());
42 const int out_dim = static_cast<int>(f(x, u).size());
43 Eigen::MatrixXd J(out_dim, m);
44
45 for (int i = 0; i < m; ++i)
46 {
47 const double h = eps * std::max(std::abs(u(i)), 1.0);
48 Eigen::VectorXd up = u, um = u;
49 up(i) += h;
50 um(i) -= h;
51 J.col(i) = (f(x, up) - f(x, um)) / (2.0 * h);
52 }
53 return J;
54}

106void NonlinearMPC::buildAndSolve()
107{
108 const int n = p_.n_states;
109 const int m = p_.n_inputs;
110 const int p = C_.rows();
111 const int Np = p_.Np;
112 const int Nu = p_.Nu;
113
114 // Step 1 - simulate nominal trajectory from x_current_ under U_warm_
115 std::vector<Eigen::VectorXd> x_traj(Np + 1);
116 x_traj[0] = x_current_;
117 for (int j = 0; j < Np; ++j)
118 {
119 const int k = std::min(j, Nu - 1);
120 x_traj[j + 1] = f_(x_traj[j], U_warm_.col(k));
121 }
122
123 // Step 2 - linearise at each step: A_j = df/dx, B_j = df/du
124 std::vector<Eigen::MatrixXd> A_list(Np), B_list(Np);
125 for (int j = 0; j < Np; ++j)
126 {
127 const int k = std::min(j, Nu - 1);
128 A_list[j] = jacX_discrete(f_, x_traj[j], U_warm_.col(k));
129 B_list[j] = jacU_discrete(f_, x_traj[j], U_warm_.col(k));
130 }
131
132 // Step 3 - build xi: nominal output trajectory minus reference (stacked Np times)
133 for (int j = 0; j < Np; ++j)
134 xi_.segment(j * p, p) = C_ * x_traj[j + 1] - y_ref_;
135
136 // Step 4 - build Theta (p*Np x m*Nu): time-varying step-response matrix.
137 //
138 // Row block j (output Deltay_{j+1}), col block k (from Deltau_k):
139 // Theta[j, k] = C * A_j * A_{j-1} * ... * A_{k+1} * B_k for k <= j
140 //
141 // For k >= Nu the input is held at Deltau_{Nu-1}, so contributions are
142 // accumulated into column block Nu-1.
143 //
144 // Incremental product: for fixed j iterate k from j down to 0.
145 // Phi starts at I, then after each k: Phi = Phi * A_k.
146 // This gives Phi[k-1] = A_j * ... * A_k correctly.
147 Theta_.setZero();
148 for (int j = 0; j < Np; ++j)
149 {
150 Eigen::MatrixXd Phi = Eigen::MatrixXd::Identity(n, n);
151 for (int k = j; k >= 0; --k)
152 {
153 const Eigen::MatrixXd contrib = C_ * Phi * B_list[k];
154 const int col_k = std::min(k, Nu - 1);
155 Theta_.block(j * p, col_k * m, p, m) += contrib;
156 // Update Phi for k-1: Phi_new = Phi_old * A_k
157 Phi = Phi * A_list[k];
158 }
159 }
160
161 // Step 5 - build QP in terms of DeltaU (deviation from warm start)
162 // H = rho_y * Theta^T Theta + rho_u * I
163 // g = rho_y * Theta^T * xi
164 const int Nu_total = m * Nu;
165 H_qp_.noalias() = p_.rho_y * (Theta_.transpose() * Theta_);
166 H_qp_ += p_.rho_u * Eigen::MatrixXd::Identity(Nu_total, Nu_total);
167 g_qp_.noalias() = p_.rho_y * (Theta_.transpose() * xi_);
168
169 // Box constraints on DeltaU: uMin - u_bar_k <= DeltaU_k <= uMax - u_bar_k
170 for (int k = 0; k < Nu; ++k)
171 {
172 lb_qp_.segment(k * m, m).setConstant(p_.uMin);
173 ub_qp_.segment(k * m, m).setConstant(p_.uMax);
174 lb_qp_.segment(k * m, m) -= U_warm_.col(k);
175 ub_qp_.segment(k * m, m) -= U_warm_.col(k);
176 }
177
178 // Lipschitz constant and LDLT (must be recomputed each RTI step)
179 L_qp_ = H_qp_.selfadjointView<Eigen::Upper>().eigenvalues().maxCoeff();
180 if (L_qp_ < 1e-12) L_qp_ = 1.0; // guard against degenerate H
181 ldlt_ = H_qp_.ldlt();
182
183 // Step 6 - solve QP via FISTA for DeltaU
184 auto res = solveGradientProjectionQP(H_qp_, g_qp_, lb_qp_, ub_qp_,
185 ldlt_, L_qp_,
186 p_.qpMaxIter, p_.qpTol,
187 dU_sol_, tmp1_, tmp2_);
188 last_qp_converged_ = res.converged;
189 last_qp_iters_ = res.iters;
190
191 // Step 7 - reconstruct U* = U_warm + DeltaU; extract first control
192 Eigen::MatrixXd U_opt(m, Nu);
193 for (int k = 0; k < Nu; ++k)
194 U_opt.col(k) = U_warm_.col(k) + dU_sol_.segment(k * m, m);
195
196 u_opt_ = U_opt.col(0);
197 last_output_ = u_opt_(0);
198
199 // Shift warm start: [u*_1, u*_2, ..., u*_{Nu-1}, u*_{Nu-1}]
200 for (int k = 0; k < Nu - 1; ++k)
201 U_warm_.col(k) = U_opt.col(k + 1);
202 U_warm_.col(Nu - 1) = U_opt.col(Nu - 1);
203}
