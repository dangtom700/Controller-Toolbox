94 TunerResult optimize(const CostFn& cost)
95 {
96 const int N = p_.population;
97 const int n = p_.n_dim;
98 const int n_elite = std::max(1, static_cast<int>(std::round(p_.elite_frac * N)));
99
100 // --- Initialise population uniformly in [lower, upper] ---
101 std::vector<Eigen::VectorXd> pop(N, Eigen::VectorXd(n));
102 std::vector<double> fitness(N);
103 for (int i = 0; i < N; ++i) {
104 for (int j = 0; j < n; ++j) {
105 std::uniform_real_distribution<double> ud(p_.lower[j], p_.upper[j]);
106 pop[i][j] = ud(rng_);
107 }
108 fitness[i] = cost(pop[i]);
109 }
110 int n_evals = N;
111
112 Eigen::VectorXd best_x = pop[0];
113 double best_cost = fitness[0];
114 for (int i = 1; i < N; ++i) {
115 if (fitness[i] < best_cost) { best_cost = fitness[i]; best_x = pop[i]; }
116 }
117
118 bool converged = false;
119 int gen = 0;
120 for (; gen < p_.max_gen; ++gen) {
121 // --- Sort by fitness (ascending = better) ---
122 std::vector<int> idx(N);
123 std::iota(idx.begin(), idx.end(), 0);
124 std::sort(idx.begin(), idx.end(),
125 [&](int a, int b){ return fitness[a] < fitness[b]; });
126
127 std::vector<Eigen::VectorXd> new_pop(N, Eigen::VectorXd(n));
128 std::vector<double> new_fit(N);
129
130 // --- Elitism: copy best n_elite unchanged ---
131 for (int e = 0; e < n_elite; ++e) {
132 new_pop[e] = pop[idx[e]];
133 new_fit[e] = fitness[idx[e]];
134 }
135
136 // --- Fill remainder via selection + crossover + mutation ---
137 for (int child = n_elite; child < N; ++child) {
138 // Tournament selection for two parents
139 Eigen::VectorXd p1 = tournamentSelect(pop, fitness);
140 Eigen::VectorXd p2 = tournamentSelect(pop, fitness);
141
142 Eigen::VectorXd offspring(n);
143 std::uniform_real_distribution<double> ud01(0.0, 1.0);
144 if (ud01(rng_) < p_.crossover) {
145 // BLX-alpha crossover
146 for (int j = 0; j < n; ++j) {
147 double lo = std::min(p1[j], p2[j]);
148 double hi = std::max(p1[j], p2[j]);
149 double range = hi - lo;
150 std::uniform_real_distribution<double> blx(
151 lo - p_.alpha * range, hi + p_.alpha * range);
152 offspring[j] = blx(rng_);
153 }
154 } else {
155 offspring = (ud01(rng_) < 0.5) ? p1 : p2;
156 }
157
158 // Gaussian mutation
159 for (int j = 0; j < n; ++j) {
160 if (ud01(rng_) < p_.mutation) {
161 double sigma = 0.1 * (p_.upper[j] - p_.lower[j]);
162 std::normal_distribution<double> nd(0.0, sigma);
163 offspring[j] += nd(rng_);
164 }
165 // Clip to bounds
166 offspring[j] = std::max(p_.lower[j], std::min(p_.upper[j], offspring[j]));
167 }
168
169 new_pop[child] = offspring;
170 new_fit[child] = cost(offspring);
171 ++n_evals;
172 if (new_fit[child] < best_cost) {
173 best_cost = new_fit[child];
174 best_x = new_pop[child];
175 }
176 }
177
178 pop = std::move(new_pop);
179 fitness = std::move(new_fit);
180
181 // --- Convergence check: stddev of fitness ---
182 double mean = 0.0;
183 for (double f : fitness) mean += f;
184 mean /= N;
185 double var = 0.0;
186 for (double f : fitness) var += (f - mean) * (f - mean);
187 if (std::sqrt(var / N) < p_.tol) { converged = true; break; }
188 }
189
190 TunerResult r;
191 r.params = best_x;
192 r.cost = best_cost;
193 r.nEvals = n_evals;
194 r.nGens = gen + 1;
195 r.converged = converged;
196 return r;
197 }

203 // Tournament selection (size 3) — returns copy of winner individual
204 Eigen::VectorXd tournamentSelect(const std::vector<Eigen::VectorXd>& pop,
205 const std::vector<double>& fitness)
206 {
207 const int N = static_cast<int>(pop.size());
208 std::uniform_int_distribution<int> uid(0, N - 1);
209 int best = uid(rng_);
210 for (int t = 1; t < 3; ++t) {
211 int cand = uid(rng_);
212 if (fitness[cand] < fitness[best]) best = cand;
213 }
214 return pop[best];
215 }
