6// SafeSensor - ISensor decorator that enforces the isValid() contract.
7//
8// Problem: ISensor::isValid() documents that "controllers should freeze when
9// this returns false," but no controller calls isValid() - it only receives a
10// double from the caller. A sensor glitch (encoder loss, CAN timeout, ADC
11// saturation) can silently feed NaN, 0.0, or a rail value into the estimator,
12// corrupting Kalman covariance and MPC state estimates before any guard fires.
13//
14// Solution: wrap any ISensor in SafeSensor. read() returns the last valid
15// measurement whenever the inner sensor reports isValid() == false, instead of
16// forwarding the bad sample. isValid() on SafeSensor still reflects the inner
17// state so the caller can react (log, alarm, switch to safe mode).
18//
19// Usage:
20// // Hardware sensor with fault detection:
21// ctrl::SafeSensor safe(my_encoder);
22//
23// // In the control loop:
24// double y = safe.read(); // last-valid hold on fault; never NaN
25// if (!safe.isValid())
26// alarm.trigger(AlarmCode::SensorFault);
27// double u = pid.compute(ref - y);
28//
29// For simulation (SimSensor), isValid() is always true so SafeSensor is a
30// transparent pass-through - no overhead.
31//
32// Staleness limit:
33// If max_stale_steps > 0, SafeSensor calls onMaxStale() (if set) after that
34// many consecutive invalid samples, signalling that the hold value is too
35// old to be trusted. The default is unlimited (hold forever).
36namespace ctrl {
37
38class SafeSensor : public ISensor {
39public:
40 // inner: the sensor being wrapped (must outlive this object)
41 // max_stale_steps: how many consecutive invalid samples before onMaxStale() fires.
42 // 0 = unlimited (hold forever, never call onMaxStale).
43 explicit SafeSensor(ISensor& inner, int max_stale_steps = 0)

45 max_stale_(max_stale_steps)

48 // Returns the inner sensor's reading when isValid() == true,
49 // or the last valid reading when isValid() == false (hold-last).
50 // Never returns NaN or Inf even when the inner sensor does.
51 double read() override
52 {

55 // Accept the reading only if it is finite; reject NaN/Inf silently.
56 if (std::isfinite(v)) {

63 } else {
64 ++stale_count_;

68 }

72 // or listen for the onMaxStale callback for that.
