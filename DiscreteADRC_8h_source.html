5// Active Disturbance Rejection Controller - 2nd-order Linear ADRC (LADRC).
6//
7// Models the controlled plant as a double integrator with unknown total disturbance:
8// y = f(y, ydot, d, t) + b0.u (b0 is an approximate input gain)
9//
10// A 3rd-order Extended State Observer (ESO) estimates:
11// z1 approx = y (output)
12// z2 approx = ydot (rate)
13// z3 approx = f(...) (total disturbance - lumped unknown dynamics + external)
14//
15// ESO (bandwidth-parameterised, backward Euler, bandwidth omega0):
16// Discrete update solves (I - Ts.Ae) z_new = z + Ts.(Be_eps.eps + Be_u.u)
17// where Ae = [[0,1,0],[0,0,1],[0,0,0]] (nilpotent). The analytical inverse of
18// (I - Ts.Ae) is [[1,Ts,Ts^2],[0,1,Ts],[0,0,1]], yielding:
19// z3[k+1] = z3[k] + Ts.beta3.epsilon[k]
20// z2[k+1] = z2[k] + Ts.(beta2.epsilon[k] + b0.u[k]) + Ts.z3[k+1]
21// z1[k+1] = z1[k] + Ts.beta1.epsilon[k] + Ts.z2[k+1]
22// epsilon[k] = y[k] - z1[k] (observer error, using old z1 - semi-implicit)
23//
24// A-stable: stable for all omega0.Ts > 0 (unlike the prior forward-Euler scheme).
25// beta1 = 3omega0, beta2 = 3omega0^2, beta3 = omega0^3 (characteristic poly (s+omega0)^3)
26//
27// PD control + disturbance cancellation (bandwidth omega_c):
28// u0[k] = omega_c^2.(r[k] - z1[k]) - 2omega_c.z2[k]
29// u[k] = (u0[k] - z3[k]) / b0
30//
31// IController::compute(error) follows the standard contract (error = r - y).
32// For the full interface use computeTracking(y, r) directly.
33//
34// Ref: Han "From PID to Active Disturbance Rejection Control" (2009);
35// Gao "Scaling and bandwidth-parameterization based controller tuning" (2003).

39 struct ADRCParams
40 {
41 double omega_o = 20.0; // ESO bandwidth [rad/s] - typically 3-10* omega_c
42 double omega_c = 5.0; // Controller BW [rad/s]
43 // Approximate plant input gain b0. The control law u = (u0 - z3) / b0 cancels
44 // the estimated total disturbance z3; if b0 is badly wrong, cancellation degrades
45 // and the ESO can diverge.
46 //
47 // Robustness: ADRC tolerates b0 uncertainty within roughly a factor of 3 for
48 // omega_o >= 5*omega_c (Gao 2003, Section IV). Beyond that, disturbance
49 // cancellation degrades and the ESO can destabilise.
50 //
51 // Practical starting estimates:
52 // - Motors: b0 = Km / J (torque constant / inertia)
53 // - Integrating plants: b0 = DC_gain / Ts^2
54 // - Unknown plants: b0 = 1/tau where tau is the open-loop time constant
55 //
56 // Verify by running open-loop: the step response slope at t=0+ is approx b0*u.
57 double b0 = 1.0;
58 double uMin = -1e9;
59 double uMax = 1e9;
60 };
