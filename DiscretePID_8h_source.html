4// Discrete-time PID controller with derivative filter and back-calculation anti-windup.
5// Discretisation: backward Euler for integral; backward Euler for derivative filter.
6// ISA standard form: u[k] = Kp.e[k] + I[k] + D[k]
7//
8// Derivative filter (backward Euler, ref MATLAB 'Filter coefficient' N):
9// d[k] = (1/(1+N.Ts)).d[k-1] + (Kd.N/(1+N.Ts)).(e[k] - e[k-1])
10//
11// Anti-windup - back-calculation:
12// I[k] = I[k-1] + Ki.Ts.e[k] + Kb.(u_sat[k] - u_unsat[k])
13// u[k] = Kp.e[k] + I[k] + D[k] (I[k] includes current e[k])
14//
15// Ref: Astrom & Wittenmark "Computer Controlled Systems" Ch 3;
16// MATLAB pid(), pidtune(), Simulink Discrete PID Controller block.

20 // Tuning parameters. All gains follow MATLAB's parallel PID form.
21 struct PIDParams
22 {
23 double Kp = 1.0; // Proportional gain
24 double Ki = 0.0; // Integral gain (Ki = Kp / Ti)
25 double Kd = 0.0; // Derivative gain (Kd = Kp . Td)
26 // Derivative filter coefficient N [rad/s] - higher = less filtering (sharper derivative).
27 // The derivative pole is placed at z = 1/(1 + N*Ts) in the backward-Euler discretisation,
28 // corresponding to a continuous-time pole at s = -N. This attenuates derivative action
29 // above N rad/s. Setting N too large causes the derivative to amplify high-frequency noise:
30 // - Nyquist limit: N < pi / Ts (e.g., N < 314 rad/s for Ts=0.01s)
31 // - Practical limit: N < pi / (2 * Ts * omega_c) where omega_c is the closed-loop BW
32 // (ensures the derivative pole stays below Nyquist with a 2x margin)
33 // The default N=100 is appropriate for many processes; reduce if derivative action is noisy.
34 double N = 100.0;
35 double uMin = -1e9; // Lower output saturation limit
36 double uMax = 1e9; // Upper output saturation limit
37 // Anti-windup back-calculation gain (0 = disabled, 1 = default).
38 // Recommended tuning range (Astrom & Wittenmark §3.5): 1/Ti <= Kb <= 1/Td
39 // where Ti = Kp/Ki and Td = Kd/Kp (i.e., Kb in [Ki/Kp, Kp/Kd]).
40 // - Kb = Ki/Kp (slow reset): gentle recovery; good for dead-time-dominated plants.
41 // - Kb = Kp/Kd (fast reset): fastest recovery; may cause actuator spike on re-entry.
42 // - Kb = sqrt(Ki*Kp/Kd) is a geometric-mean rule of thumb for balanced reset.
43 // - Kb = 0: disables anti-windup entirely; use only when saturation cannot occur.
44 // - Kb = 1.0 (default): reasonable for many processes; may be slow for high-Ki loops.
45 double Kb = 1.0;
46 // Two-degree-of-freedom proportional setpoint weight b in [0, 1].
47 // Applied only by computeDoM(y, r): proportional term acts on (b*r - y) while
48 // the integral term acts on (r - y). This decouples setpoint-tracking bandwidth
49 // from disturbance-rejection bandwidth without changing any gains.
50 //
51 // Tuning guide (Skogestad 2003 / Astrom & Hagglund 2006 §8.4):
52 // b = 1.0 (default): standard 1DOF — maximum setpoint gain.
53 // b = 0.5: reduces overshoot ~10-15 % vs b=1 for ZN-tuned loops.
54 // b = 0.0: setpoint feed-forward entirely through integrator;
55 // no proportional kick on step — slowest but smoothest.
56 //
57 // Has no effect on compute(error) because that overload receives only the
58 // pre-computed error (r - y) and cannot separate r from y.
59 double b_weight = 1.0;
60 };
