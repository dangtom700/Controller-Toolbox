13int KoopmanEDMD::computeNLifted() const
14{
15 const int n = p_.n_state;
16 const int m = p_.n_input;
17 const int d = n + m;
18
19 switch (p_.dict) {
20 case Dict::PolyDeg1:
21 return 1 + n + m;
22
23 case Dict::PolyDeg2:
24 // 1 + n + m + all degree-2 terms of [x;u]: d*(d+1)/2
25 return 1 + n + m + d * (d + 1) / 2;
26
27 case Dict::RBF:
28 // n (original state) + n_centres (RBF outputs) + m (inputs passed through)
29 return n + p_.rbf_n_cent + m;
30
31 default:
32 return 1 + n + m;
33 }
34}

51Eigen::VectorXd KoopmanEDMD::liftPoly(const Eigen::VectorXd& x,
52 const Eigen::VectorXd& u) const
53{
54 const int n = p_.n_state, m = p_.n_input;
55 const int d = n + m;
56 Eigen::VectorXd z(n + m);
57 z.head(n) = x;
58 if (m > 0) z.tail(m) = u;
59
60 Eigen::VectorXd psi(n_lifted_);
61 int idx = 0;
62
63 psi(idx++) = 1.0;
64 psi.segment(idx, n) = x; idx += n;
65 if (m > 0) { psi.segment(idx, m) = u; idx += m; }
66
67 if (p_.dict == Dict::PolyDeg2) {
68 for (int i = 0; i < d; ++i)
69 for (int j = i; j < d; ++j)
70 psi(idx++) = z(i) * z(j);
71 }
72 return psi;
73}

75Eigen::VectorXd KoopmanEDMD::liftRBF(const Eigen::VectorXd& x,
76 const Eigen::VectorXd& u) const
77{
78 const int n = p_.n_state, m = p_.n_input;
79 Eigen::VectorXd psi(n_lifted_);
80 int idx = 0;
81
82 psi.segment(idx, n) = x; idx += n;
83
84 const double inv2sig2 = 0.5 / (p_.rbf_width * p_.rbf_width);
85 const int nc = std::min(p_.rbf_n_cent, static_cast<int>(rbf_centres_.cols()));
86 for (int c = 0; c < nc; ++c) {
87 double d2 = (x - rbf_centres_.col(c)).squaredNorm();
88 psi(idx++) = std::exp(-d2 * inv2sig2);
89 }
90 // Fill remaining centres slots with 0 if not yet set
91 while (idx < n + p_.rbf_n_cent)
92 psi(idx++) = 0.0;
93
94 if (m > 0) { psi.segment(idx, m) = u; idx += m; }
95 return psi;
96}

144// ---------------------------------------------------------------------------
145// Fit: solve min ||Ψ_{k+1} - K.Ψ_k||_F (Tikhonov regularised)
146//
147// K = Ψ_{k+1} . Ψ_k' . (Ψ_k . Ψ_k' + alpha.I)^{-1}
148// ---------------------------------------------------------------------------
149
150StateSpace KoopmanEDMD::fit()
151{
152 const int K = snapshotCount();
153 if (K == 0)
154 throw std::runtime_error("KoopmanEDMD::fit(): no snapshots added");
155
156 const int nz = n_lifted_;
157
158 // Assemble Ψ_k (nz x K) and Ψ_{k+1} (nz x K)
159 Eigen::MatrixXd Psi_k (nz, K), Psi_k1(nz, K);
160 for (int j = 0; j < K; ++j) {
161 Psi_k .col(j) = psi_k_ [j];
162 Psi_k1.col(j) = psi_k1_[j];
163 }
164
165 // G = Ψ_k . Ψ_k' + alpha.I (nz x nz)
166 Eigen::MatrixXd G = Psi_k * Psi_k.transpose()
167 + p_.tikhonov * Eigen::MatrixXd::Identity(nz, nz);
168
169 // K_op = Ψ_{k+1} . Ψ_k' . G^{-1}
170 Eigen::LDLT<Eigen::MatrixXd> ldlt(G);
171 if (ldlt.info() != Eigen::Success)
172 throw std::runtime_error("KoopmanEDMD: EDMD factorisation failed");
173
174 // Solve G' . K_op' = (Ψ_{k+1} . Ψ_k')' -> K_op = solution'
175 const Eigen::MatrixXd A_lift = ldlt.solve((Psi_k * Psi_k1.transpose())).transpose();
176 // A_lift : nz * nz (lifted state transition)
177
178 // Split A_lift into [A_zz | B_zu] parts:
179 // The last p_.n_input columns of ψ carry the input (for poly dict)
180 const int n_x = nz - p_.n_input; // lifted state part
181 Eigen::MatrixXd A = A_lift.topLeftCorner(n_x, n_x);
182 Eigen::MatrixXd B = A_lift.topRightCorner(n_x, p_.n_input);
183 Eigen::MatrixXd C = Eigen::MatrixXd::Identity(n_x, n_x); // full lifted output
184 Eigen::MatrixXd D = Eigen::MatrixXd::Zero(n_x, p_.n_input);
185
186 return StateSpace(A, B, C, D, 1.0); // Ts=1 (data-driven, discrete)
187}

189StateSpace KoopmanEDMD::fitProjected()
190{
191 // Fit full lifted model
192 StateSpace ss_lifted = fit();
193
194 // Projection matrix: maps lifted state -> original state
195 // P = X_{k+1} . Ψ_k' . (Ψ_k . Ψ_k' + alpha.I)^{-1} (first n_state rows of K_op)
196 const int K = snapshotCount();
197 const int nz = n_lifted_ - p_.n_input; // lifted state size (excl. input columns)
198
199 Eigen::MatrixXd Psi_k(nz, K), X_k1(p_.n_state, K);
200 for (int j = 0; j < K; ++j) {
201 Psi_k .col(j) = psi_k_[j].head(nz);
202 X_k1 .col(j) = x_k1_[j];
203 }
204
205 Eigen::MatrixXd G = Psi_k * Psi_k.transpose()
206 + p_.tikhonov * Eigen::MatrixXd::Identity(nz, nz);
207 Eigen::LDLT<Eigen::MatrixXd> ldlt(G);
208
209 // Output matrix C_proj: n_state * nz
210 Eigen::MatrixXd C_proj = ldlt.solve(Psi_k * X_k1.transpose()).transpose();
211
212 // Build projected state-space: n_state states, n_input inputs
213 const Eigen::MatrixXd& A_l = ss_lifted.A;
214 const Eigen::MatrixXd& B_l = ss_lifted.B;
215 Eigen::MatrixXd A_proj = C_proj * A_l; // n_state * nz (approximate)
216 // For full projected model we need A: n_state * n_state
217 // Use A_proj * pseudo-inverse(C_proj) as approximation
218 Eigen::MatrixXd A_out = A_proj * C_proj.completeOrthogonalDecomposition()
219 .pseudoInverse();
220 Eigen::MatrixXd B_out = C_proj * B_l;
221 Eigen::MatrixXd C_out = Eigen::MatrixXd::Identity(p_.n_state, p_.n_state);
222 Eigen::MatrixXd D_out = Eigen::MatrixXd::Zero(p_.n_state, p_.n_input);
223
224 return StateSpace(A_out, B_out, C_out, D_out, 1.0);
225}
