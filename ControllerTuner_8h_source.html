12// ============================================================
13// ControllerTuner.h - Offline and online tuning utilities
14//
15// Every tuner exposes two interfaces:
16//
17// tuneImpl(...) - unchecked, for internal use or when
18// the controller type is already known.
19//
20// tuneFor<C>(...) - template wrapper; triggers a
21// static_assert (hard compiler error)
22// when C is incompatible, with a
23// diagnostic naming the correct tuner.
24//
25// Existing tuners (RelayAutoTuner, StepResponseTuner,
26// LQRWeightTuner, MPCHorizonTuner) gain an additional
27// *For<C>() overload - their original methods are unchanged.
28//
29// New tuners added here:
30// ZieglerNicholsTuner - standalone ZN from (Ku, Tu)
31// CohenCoonTuner - FOPDT-based PID (Cohen & Coon 1953)
32// LoopShapingTuner - crossover freq + phase margin -> LeadLagParams
33// KalmanWeightTuner - noise levels -> Qf / Rf for DiscreteLQG
34//
35// Strategies from the literature that require external solvers
36// or adaptive runtime infrastructure are NOT implemented here:
37// Hinf / mu-synthesis -> requires SLICOT, Python-control, or MATLAB
38// QFT -> requires CAD tool (QFT Toolbox, QDESIGN)
39// GA / PSO / DE -> use an optimisation framework (Optuna, DEAP)
40// Bayesian opt. -> use BoTorch / GPyOpt / scikit-optimize
41// MRAC / STR / IFT / VRFT -> adaptive control; separate library
42//
43// Ref: Astrom & Hagglund (1988, 2006); Bryson & Ho (1975);
44// Cohen & Coon (1953); Rivera, Morari & Skogestad (1986);
45// Camacho & Bordons MPC Ch 3; Franklin, Powell & Emami-Naeini.
46// ============================================================

61 // ============================================================
62 // Relay Auto-Tuner (Astrom-Hagglund relay feedback test)
63 // ============================================================
64 // Applies a relay +/-d to the closed plant, records the resulting limit cycle,
65 // then derives ultimate gain Ku and ultimate period Tu.
66 // Ku = (4.d) / (pi.a_y) where a_y = peak-to-peak output amplitude / 2
67 // Tu = average oscillation period from zero-crossing times
68 //
69 // Usage pattern (simulation loop):
70 // while (!tuner.isDone()) {
71 // u = tuner.step(plant_output);
72 // apply u, advance plant;
73 // }
74 // auto params = tuner.computePIDParams(PIDTuningRule::TyreusLuyben);
75 // ============================================================
76 struct RelayTunerConfig
77 {
78 double relayAmplitude = 1.0; // Relay output amplitude +/-d
79 double hysteresis = 0.0; // Dead-band to prevent noise-driven chatter
80 int cyclesRequired = 3; // Stable oscillation cycles needed before exit
81 };

357 // Cohen-Coon Tuner (FOPDT process-reaction-curve method)
358 // ============================================================
359 // More accurate than ZN for processes where dead time theta is a
360 // significant fraction of the lag tau (theta/tau in [0.1, 1.0]).
361 //

397 // ============================================================
398 // Loop-Shaping Tuner (frequency-domain -> LeadLagParams)
399 // ============================================================
400 // Designs a lead compensator C(s) = K.(s+z)/(s+p) so that:
401 // - The gain crossover occurs at omega_c (|L(j.omega_c)| = 1)
402 // - The compensator contributes phase_add_deg of phase lead at omega_c
403 //
404 // Derivation (lead, phase_add_deg > 0):
405 // beta = sin(phi), alpha = (1+beta)/(1-beta)
406 // z = omega_c/\sqrtalpha, p = omega_c.\sqrtalpha, K = \sqrtalpha / |G(j.omega_c)|
407 //
408 // where |G(j.omega_c)| = gain_at_wc (open-loop plant magnitude before
409 // adding the compensator).
410 //
411 // Ref: Franklin, Powell & Emami-Naeini "Feedback Control of Dynamic
412 // Systems" Section 9.3.
413 // ============================================================
414 class LoopShapingTuner
415 {
416 public:
417 struct Input
418 {
419 double omega_c; // desired gain crossover frequency [rad/s]
420 double phase_add_deg; // phase lead to add at omega_c [deg], must be in (0, 90)
421 double gain_at_wc; // open-loop |G(j.omega_c)| without the compensator
422 };
423
424 static LeadLagParams tuneImpl(const Input &in);
425
426 template <typename C>
427 static LeadLagParams tuneFor(const Input &in)
428 {
429 static_assert(ControllerTraits<C>::supports_freq_tuning,
430 "\n[LoopShapingTuner::tuneFor<C>]"

432 " LoopShapingTuner returns LeadLagParams;"
433 " only valid for DiscreteLeadLag.\n"
434 " --> DiscretePID :"
435 " ZieglerNicholsTuner or"
436 " StepResponseTuner::computePIDParamsFor<DiscretePID>()\n"
437 " --> DiscreteLQR :"
438 " LQRWeightTuner::brysonMethodFor<DiscreteLQR>(xmax, umax)\n"
439 " --> DiscreteLQG :"
440 " LQRWeightTuner::brysonMethodFor<DiscreteLQG>()"

450 // ============================================================
451 // Kalman Weight Tuner (noise covariance selection for DiscreteLQG)
452 // ============================================================
453 // Provides heuristic starting points for Qf (process noise covariance)
454 // and Rf (measurement noise covariance). Both scale as sigma^2, where sigma is

461 // The separation principle guarantees that the LQR and Kalman gains
462 // can be tuned independently; use LQRWeightTuner for the LQR part.
463 //
464 // Ref: Bryson & Ho "Applied Optimal Control" (1975) Section 14.3.
465 // ============================================================
466 class KalmanWeightTuner
467 {
468 public:

470 static KalmanNoiseParams fromNoise(const Eigen::VectorXd &maxProcessNoise,
471 const Eigen::VectorXd &maxMeasNoise);
472
