76Eigen::VectorXd MovingHorizonEstimator::estimate(const Eigen::VectorXd &y,
77 const Eigen::VectorXd &u)
78{
79 if (y.size() != p_)
80 throw std::invalid_argument("MHE::estimate: y must have size p.");
81 if (u.size() != m_)
82 throw std::invalid_argument("MHE::estimate: u must have size m.");
83
84 const int N = params_.N;
85
86 // Shift history: oldest entry falls off, new y and u go at the back.
87 // y_hist_ has N+1 slots: indices [0..N], where 0 is the oldest.
88 // u_hist_ has N slots: u_hist_[i] drives the transition x[i] -> x[i+1].
89 for (int i = 0; i < N; ++i) y_hist_[i] = y_hist_[i + 1];
90 y_hist_[N] = y;
91 for (int i = 0; i < N - 1; ++i) u_hist_[i] = u_hist_[i + 1];
92 if (N >= 1) u_hist_[N - 1] = u;
93
94 ++step_count_;
95
96 // Effective horizon: ramp up from 1 to N during the first N steps
97 const int eff_N = std::min(step_count_, N);
98 const int eff_Np1 = eff_N + 1;
99 const int dz = n_ * (eff_N + 1); // decision variable size
100
101 // --- Build QP linear cost g = H_mhe*prior_z - Psi'*C_bar'*Rinv*y_hist ---
102 // Decision variable z = [x_bar_; 0; ...; 0] (prior on the process-noise window)
103
104 // Stack measurements from window tail (most recent eff_N steps + 1 measurement)
105 Eigen::VectorXd Y_hist(p_ * eff_Np1);
106 for (int i = 0; i < eff_Np1; ++i) {
107 // y_hist_[N - eff_N + i] maps to window position i
108 const int idx = N - eff_N + i;
109 Y_hist.segment(i * p_, p_) = y_hist_[idx];
110 }
111
112 // Prior z_prior = [x_bar_; 0; ...; 0]
113 Eigen::VectorXd z_prior = Eigen::VectorXd::Zero(dz);
114 z_prior.head(n_) = x_bar_;
115
116 // Stack inputs for the effective window
117 Eigen::VectorXd U_hist_eff((eff_N) * m_);
118 for (int i = 0; i < eff_N; ++i) {
119 const int idx = N - eff_N + i;
120 U_hist_eff.segment(i * m_, m_) = u_hist_[idx];
121 }
122
123 // For the effective horizon, build Psi, Gamma_u, C_bar sub-matrices
124 // Psi_eff: (n*eff_Np1) x (n*eff_Np1)
125 Eigen::MatrixXd Psi_eff = Eigen::MatrixXd::Zero(n_ * eff_Np1, n_ * eff_Np1);
126 Eigen::MatrixXd Gamma_u_eff = Eigen::MatrixXd::Zero(n_ * eff_Np1, m_ * eff_N);
127 Eigen::MatrixXd C_bar_eff = Eigen::MatrixXd::Zero(p_ * eff_Np1, n_ * eff_Np1);
128
129 // x_0 row
130 Psi_eff.block(0, 0, n_, n_) = Eigen::MatrixXd::Identity(n_, n_);
131 C_bar_eff.block(0, 0, p_, n_) = plant_.C;
132
133 // Precompute powers of A
134 std::vector<Eigen::MatrixXd> A_pow(eff_Np1);
135 A_pow[0] = Eigen::MatrixXd::Identity(n_, n_);
136 for (int i = 1; i < eff_Np1; ++i) A_pow[i] = plant_.A * A_pow[i - 1];
137
138 for (int i = 1; i < eff_Np1; ++i) {
139 // x_i: Psi block [i] col 0 = A^i (x_0 contribution)
140 Psi_eff.block(i * n_, 0, n_, n_) = A_pow[i];
141 // w_j contributes A^{i-1-j} to x_i for j = 0..i-1
142 for (int j = 0; j < i; ++j) {
143 // w_j is decision variable index j+1 (after x_0)
144 Psi_eff.block(i * n_, (j + 1) * n_, n_, n_) = A_pow[i - 1 - j];
145 }
146 // u_j contributes A^{i-1-j}*B*u_j (known, goes into Gamma_u)
147 for (int j = 0; j < i; ++j) {
148 Gamma_u_eff.block(i * n_, j * m_, n_, m_) = A_pow[i - 1 - j] * plant_.B;
149 }
150 // C_bar block
151 C_bar_eff.block(i * p_, i * n_, p_, n_) = plant_.C;
152 }
153
154 // Build Hessian for effective horizon
155 // H_eff = Hprior + Psi'*C_bar'*Rinv*C_bar*Psi
156 // Hprior = block_diag(P0inv, Qinv_block) (only for effective size)
157 Eigen::MatrixXd Qinv = Q_proc_.ldlt().solve(Eigen::MatrixXd::Identity(n_, n_));
158 Eigen::MatrixXd Rinv = R_meas_.ldlt().solve(Eigen::MatrixXd::Identity(p_, p_));
159
160 Eigen::MatrixXd H_prior = Eigen::MatrixXd::Zero(dz, dz);
161 H_prior.block(0, 0, n_, n_) = P0inv_;
162 for (int i = 1; i < eff_Np1; ++i)
163 H_prior.block(i * n_, i * n_, n_, n_) = Qinv;
164
165 // R_bar = block_diag(Rinv, ..., Rinv), size p*eff_Np1
166 Eigen::MatrixXd R_bar = Eigen::MatrixXd::Zero(p_ * eff_Np1, p_ * eff_Np1);
167 for (int i = 0; i < eff_Np1; ++i)
168 R_bar.block(i * p_, i * p_, p_, p_) = Rinv;
169
170 Eigen::MatrixXd CTPsi = C_bar_eff.transpose() * R_bar * C_bar_eff;
171 Eigen::MatrixXd H_eff = H_prior + Psi_eff.transpose() * CTPsi * Psi_eff;
172
173 // Linear cost: f = -H_prior*z_prior - Psi'*C_bar'*Rinv*(Y - C_bar*Gamma_u*U)
174 // (Y - C_bar*Gamma_u*U) is the residual from the known input contributions
175 Eigen::VectorXd Y_pred_u = C_bar_eff * Gamma_u_eff * U_hist_eff;
176 Eigen::VectorXd Y_res = Y_hist - Y_pred_u;
177
178 Eigen::VectorXd f_eff = -H_prior * z_prior
179 - Psi_eff.transpose() * C_bar_eff.transpose() * R_bar * Y_res;
180
181 // Box constraints on z: x_0 state bounds + per-element w noise bounds.
182 // Decision vector layout: z = [x_0 (n); w_0 (n); w_1 (n); ...; w_{eff_N-1} (n)]
183 Eigen::VectorXd lb_eff = Eigen::VectorXd::Constant(dz, -1e30);
184 Eigen::VectorXd ub_eff = Eigen::VectorXd::Constant(dz, 1e30);
185
186 // x_0 state bounds (MHEParams::xMin / xMax), if provided.
187 if (params_.xMin.size() == n_) lb_eff.head(n_) = params_.xMin;
188 if (params_.xMax.size() == n_) ub_eff.head(n_) = params_.xMax;
189
190 // Process-noise bounds for w_1..w_{eff_N}.
191 for (int i = 1; i < eff_Np1; ++i) {
192 lb_eff.segment(i * n_, n_).fill(params_.wMin);
193 ub_eff.segment(i * n_, n_).fill(params_.wMax);
194 }
195
196 // Solve QP
197 Eigen::LDLT<Eigen::MatrixXd> ldlt_eff(H_eff);
198 if (ldlt_eff.info() != Eigen::Success) {
199 // Fallback: return prior estimate
200 x_est_ = x_bar_;
201 last_converged_ = false;
202 last_qp_iters_ = 0;
203 return x_est_;
204 }
205
206 const double L_eff = H_eff.eigenvalues().real().maxCoeff();
207 if (L_eff <= 0.0) {
208 x_est_ = x_bar_;
209 last_converged_ = false;
210 return x_est_;
211 }
212
213 Eigen::VectorXd z_sol_eff(dz), t1(dz), t2(dz);
214 z_sol_eff.setZero();
215 auto result = solveGradientProjectionQP(
216 H_eff, f_eff, lb_eff, ub_eff,
217 ldlt_eff, L_eff,
218 params_.qpMaxIter, params_.qpTol,
219 z_sol_eff, t1, t2);
220
221 last_converged_ = result.converged;
222 last_qp_iters_ = result.iters;
223
224 // Extract x_0 from solution, propagate to get current x estimate
225 const Eigen::VectorXd x0_opt = z_sol_eff.head(n_);
226
227 // Propagate x_0_opt through the horizon using stored inputs and estimated w
228 Eigen::VectorXd x_cur = x0_opt;
229 for (int i = 0; i < eff_N; ++i) {
230 const int idx = N - eff_N + i;
231 const Eigen::VectorXd w_i = z_sol_eff.segment((i + 1) * n_, n_);
232 x_cur = plant_.A * x_cur + plant_.B * u_hist_[idx] + w_i;
233 }
234
235 x_est_ = x_cur;
236
237 // Update arrival state: x_bar_ for the next step is x_0 shifted by one
238 // (the oldest state in the window advances by one step)
239 if (step_count_ >= N) {
240 // x_bar_ = A*x0_opt + B*u_hist_[N-eff_N] + w_0
241 const Eigen::VectorXd w0 = z_sol_eff.segment(n_, n_);
242 const int u0_idx = N - eff_N;
243 x_bar_ = plant_.A * x0_opt + plant_.B * u_hist_[u0_idx] + w0;
244 } else {
245 x_bar_ = x0_opt;
246 }
247
248 return x_est_;
249}

293// buildCondensedMatrices
294// Precomputes the full-horizon matrices for the QP.
295// estimate() uses sub-matrices for the effective (ramped-up) horizon.
296// =============================================================================
297
298void MovingHorizonEstimator::buildCondensedMatrices()
299{
300 const int N = params_.N;
301 const int Np1 = N + 1;
302 const int dz = n_ * Np1;
303
304 // Psi_: (n*Np1) x (n*Np1)
305 // Row block i: x_i = sum_j Psi_[i,j]*z_j + Gamma_u terms
306 Psi_ = Eigen::MatrixXd::Zero(n_ * Np1, n_ * Np1);
307 Gamma_u_= Eigen::MatrixXd::Zero(n_ * Np1, m_ * N);
308 C_bar_ = Eigen::MatrixXd::Zero(p_ * Np1, n_ * Np1);
309
310 // x_0 row
311 Psi_.block(0, 0, n_, n_) = Eigen::MatrixXd::Identity(n_, n_);
312 C_bar_.block(0, 0, p_, n_) = plant_.C;
313
314 std::vector<Eigen::MatrixXd> A_pow(Np1);
315 A_pow[0] = Eigen::MatrixXd::Identity(n_, n_);
316 for (int i = 1; i < Np1; ++i) A_pow[i] = plant_.A * A_pow[i - 1];
317
318 for (int i = 1; i < Np1; ++i) {
319 Psi_.block(i * n_, 0, n_, n_) = A_pow[i];
320 for (int j = 0; j < i; ++j) {
321 Psi_.block(i * n_, (j + 1) * n_, n_, n_) = A_pow[i - 1 - j];
322 Gamma_u_.block(i * n_, j * m_, n_, m_) = A_pow[i - 1 - j] * plant_.B;
323 }
324 C_bar_.block(i * p_, i * n_, p_, n_) = plant_.C;
325 }
326
327 // Inverses
328 Eigen::MatrixXd Qinv = Q_proc_.ldlt().solve(Eigen::MatrixXd::Identity(n_, n_));
329 Eigen::MatrixXd Rinv = R_meas_.ldlt().solve(Eigen::MatrixXd::Identity(p_, p_));
330
331 // H_prior = block_diag(P0inv, Qinv, ..., Qinv) size dz x dz
332 Eigen::MatrixXd H_prior = Eigen::MatrixXd::Zero(dz, dz);
333 H_prior.block(0, 0, n_, n_) = P0inv_;
334 for (int i = 1; i < Np1; ++i)
335 H_prior.block(i * n_, i * n_, n_, n_) = Qinv;
336
337 // R_bar = block_diag(Rinv, ...) size p*Np1
338 Eigen::MatrixXd R_bar = Eigen::MatrixXd::Zero(p_ * Np1, p_ * Np1);
339 for (int i = 0; i < Np1; ++i)
340 R_bar.block(i * p_, i * p_, p_, p_) = Rinv;
341
342 H_mhe_ = H_prior + Psi_.transpose() * C_bar_.transpose() * R_bar * C_bar_ * Psi_;
343
344 ldlt_.compute(H_mhe_);
345 L_ = H_mhe_.eigenvalues().real().maxCoeff();
346 if (L_ <= 0.0) L_ = 1.0;
347
348 // Pre-allocate work vectors
349 z_sol_.resize(dz); z_sol_.setZero();
350 g_qp_ .resize(dz); g_qp_ .setZero();
351 lb_z_ .resize(dz); lb_z_ .fill(-1e30);
352 ub_z_ .resize(dz); ub_z_ .fill( 1e30);
353 tmp1_ .resize(dz); tmp1_ .setZero();
354 tmp2_ .resize(dz); tmp2_ .setZero();
355}
