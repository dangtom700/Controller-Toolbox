82Eigen::VectorXd MovingHorizonEstimator::estimate(const Eigen::VectorXd &y,
83 const Eigen::VectorXd &u)
84{
85 if (y.size() != p_)
86 throw std::invalid_argument("MHE::estimate: y must have size p.");
87 if (u.size() != m_)
88 throw std::invalid_argument("MHE::estimate: u must have size m.");
89
90 const int N = params_.N;
91
92 // Shift history: oldest entry falls off, new y and u go at the back.
93 // y_hist_ has N+1 slots: indices [0..N], where 0 is the oldest.
94 // u_hist_ has N slots: u_hist_[i] drives the transition x[i] -> x[i+1].
95 for (int i = 0; i < N; ++i) y_hist_[i] = y_hist_[i + 1];
96 y_hist_[N] = y;
97 for (int i = 0; i < N - 1; ++i) u_hist_[i] = u_hist_[i + 1];
98 if (N >= 1) u_hist_[N - 1] = u;
99
100 ++step_count_;
101
102 // Effective horizon: ramp up from 1 to N during the first N steps
103 const int eff_N = std::min(step_count_, N);
104 const int eff_Np1 = eff_N + 1;
105 const int dz = n_ * (eff_N + 1); // decision variable size
106
107 // --- Build QP linear cost g = H_mhe*prior_z - Psi'*C_bar'*Rinv*y_hist ---
108 // Decision variable z = [x_bar_; 0; ...; 0] (prior on the process-noise window)
109
110 // Stack measurements from window tail (most recent eff_N steps + 1 measurement)
111 Eigen::VectorXd Y_hist(p_ * eff_Np1);
112 for (int i = 0; i < eff_Np1; ++i) {
113 // y_hist_[N - eff_N + i] maps to window position i
114 const int idx = N - eff_N + i;
115 Y_hist.segment(i * p_, p_) = y_hist_[idx];
116 }
117
118 // Prior z_prior = [x_bar_; 0; ...; 0]
119 Eigen::VectorXd z_prior = Eigen::VectorXd::Zero(dz);
120 z_prior.head(n_) = x_bar_;
121
122 // Stack inputs for the effective window
123 Eigen::VectorXd U_hist_eff((eff_N) * m_);
124 for (int i = 0; i < eff_N; ++i) {
125 const int idx = N - eff_N + i;
126 U_hist_eff.segment(i * m_, m_) = u_hist_[idx];
127 }
128
129 // For the effective horizon, build Psi, Gamma_u, C_bar sub-matrices
130 // Psi_eff: (n*eff_Np1) x (n*eff_Np1)
131 Eigen::MatrixXd Psi_eff = Eigen::MatrixXd::Zero(n_ * eff_Np1, n_ * eff_Np1);
132 Eigen::MatrixXd Gamma_u_eff = Eigen::MatrixXd::Zero(n_ * eff_Np1, m_ * eff_N);
133 Eigen::MatrixXd C_bar_eff = Eigen::MatrixXd::Zero(p_ * eff_Np1, n_ * eff_Np1);
134
135 // x_0 row
136 Psi_eff.block(0, 0, n_, n_) = Eigen::MatrixXd::Identity(n_, n_);
137 C_bar_eff.block(0, 0, p_, n_) = plant_.C;
138
139 // Precompute powers of A
140 std::vector<Eigen::MatrixXd> A_pow(eff_Np1);
141 A_pow[0] = Eigen::MatrixXd::Identity(n_, n_);
142 for (int i = 1; i < eff_Np1; ++i) A_pow[i] = plant_.A * A_pow[i - 1];
143
144 for (int i = 1; i < eff_Np1; ++i) {
145 // x_i: Psi block [i] col 0 = A^i (x_0 contribution)
146 Psi_eff.block(i * n_, 0, n_, n_) = A_pow[i];
147 // w_j contributes A^{i-1-j} to x_i for j = 0..i-1
148 for (int j = 0; j < i; ++j) {
149 // w_j is decision variable index j+1 (after x_0)
150 Psi_eff.block(i * n_, (j + 1) * n_, n_, n_) = A_pow[i - 1 - j];
151 }
152 // u_j contributes A^{i-1-j}*B*u_j (known, goes into Gamma_u)
153 for (int j = 0; j < i; ++j) {
154 Gamma_u_eff.block(i * n_, j * m_, n_, m_) = A_pow[i - 1 - j] * plant_.B;
155 }
156 // C_bar block
157 C_bar_eff.block(i * p_, i * n_, p_, n_) = plant_.C;
158 }
159
160 // Build Hessian for effective horizon
161 // H_eff = Hprior + Psi'*C_bar'*Rinv*C_bar*Psi
162 // Hprior = block_diag(P0inv, Qinv_block) (only for effective size)
163 Eigen::MatrixXd Qinv = Q_proc_.ldlt().solve(Eigen::MatrixXd::Identity(n_, n_));
164 Eigen::MatrixXd Rinv = R_meas_.ldlt().solve(Eigen::MatrixXd::Identity(p_, p_));
165
166 Eigen::MatrixXd H_prior = Eigen::MatrixXd::Zero(dz, dz);
167 H_prior.block(0, 0, n_, n_) = P0inv_;
168 for (int i = 1; i < eff_Np1; ++i)
169 H_prior.block(i * n_, i * n_, n_, n_) = Qinv;
170
171 // R_bar = block_diag(Rinv, ..., Rinv), size p*eff_Np1
172 Eigen::MatrixXd R_bar = Eigen::MatrixXd::Zero(p_ * eff_Np1, p_ * eff_Np1);
173 for (int i = 0; i < eff_Np1; ++i)
174 R_bar.block(i * p_, i * p_, p_, p_) = Rinv;
175
176 Eigen::MatrixXd CTPsi = C_bar_eff.transpose() * R_bar * C_bar_eff;
177 Eigen::MatrixXd H_eff = H_prior + Psi_eff.transpose() * CTPsi * Psi_eff;
178
179 // Linear cost: f = -H_prior*z_prior - Psi'*C_bar'*Rinv*(Y - C_bar*Gamma_u*U)
180 // (Y - C_bar*Gamma_u*U) is the residual from the known input contributions
181 Eigen::VectorXd Y_pred_u = C_bar_eff * Gamma_u_eff * U_hist_eff;
182 Eigen::VectorXd Y_res = Y_hist - Y_pred_u;
183
184 Eigen::VectorXd f_eff = -H_prior * z_prior
185 - Psi_eff.transpose() * C_bar_eff.transpose() * R_bar * Y_res;
186
187 // Box constraints on z: x_0 state bounds + per-element w noise bounds.
188 // Decision vector layout: z = [x_0 (n); w_0 (n); w_1 (n); ...; w_{eff_N-1} (n)]
189 Eigen::VectorXd lb_eff = Eigen::VectorXd::Constant(dz, -1e30);
190 Eigen::VectorXd ub_eff = Eigen::VectorXd::Constant(dz, 1e30);
191
192 // x_0 state bounds (MHEParams::xMin / xMax), if provided.
193 if (params_.xMin.size() == n_) lb_eff.head(n_) = params_.xMin;
194 if (params_.xMax.size() == n_) ub_eff.head(n_) = params_.xMax;
195
196 // Process-noise bounds for w_1..w_{eff_N}.
197 for (int i = 1; i < eff_Np1; ++i) {
198 lb_eff.segment(i * n_, n_).fill(params_.wMin);
199 ub_eff.segment(i * n_, n_).fill(params_.wMax);
200 }
201
202 // Solve QP
203 Eigen::LDLT<Eigen::MatrixXd> ldlt_eff(H_eff);
204 if (ldlt_eff.info() != Eigen::Success) {
205 // Fallback: return prior estimate
206 x_est_ = x_bar_;
207 last_converged_ = false;
208 last_qp_iters_ = 0;
209 return x_est_;
210 }
211
212 const double L_eff = H_eff.eigenvalues().real().maxCoeff();
213 if (L_eff <= 0.0) {
214 x_est_ = x_bar_;
215 last_converged_ = false;
216 return x_est_;
217 }
218
219 Eigen::VectorXd z_sol_eff(dz), t1(dz), t2(dz);
220 z_sol_eff.setZero();
221 auto result = solveGradientProjectionQP(
222 H_eff, f_eff, lb_eff, ub_eff,
223 ldlt_eff, L_eff,
224 params_.qpMaxIter, params_.qpTol,
225 z_sol_eff, t1, t2);
226
227 last_converged_ = result.converged;
228 last_qp_iters_ = result.iters;
229
230 // Extract x_0 from solution; apply polytopic projection (E4) if constraints are set.
231 Eigen::VectorXd x0_opt = projectX0Polytope(z_sol_eff.head(n_));
232
233 // Propagate x_0_opt through the horizon using stored inputs and estimated w
234 Eigen::VectorXd x_cur = x0_opt;
235 for (int i = 0; i < eff_N; ++i) {
236 const int idx = N - eff_N + i;
237 const Eigen::VectorXd w_i = z_sol_eff.segment((i + 1) * n_, n_);
238 x_cur = plant_.A * x_cur + plant_.B * u_hist_[idx] + w_i;
239 }
240
241 x_est_ = x_cur;
242
243 // D1: feed one-step-ahead residual into CUSUM mismatch detector (if enabled).
244 if (mismatch_det_) {
245 const Eigen::VectorXd resid = y_hist_[N] - plant_.C * x_est_;
246 mismatch_det_->update(resid);
247 }
248
249 // Update arrival state: x_bar_ for the next step is x_0 shifted by one
250 // (the oldest state in the window advances by one step)
251 if (step_count_ >= N) {
252 // x_bar_ = A*x0_opt + B*u_hist_[N-eff_N] + w_0
253 const Eigen::VectorXd w0 = z_sol_eff.segment(n_, n_);
254 const int u0_idx = N - eff_N;
255 x_bar_ = plant_.A * x0_opt + plant_.B * u_hist_[u0_idx] + w0;
256 } else {
257 x_bar_ = x0_opt;
258 }
259
260 return x_est_;
261}

304// =============================================================================
305// buildCondensedMatrices
306// Precomputes the full-horizon matrices for the QP.
307// estimate() uses sub-matrices for the effective (ramped-up) horizon.
308// =============================================================================
309
310void MovingHorizonEstimator::buildCondensedMatrices()
311{
312 const int N = params_.N;
313 const int Np1 = N + 1;
314 const int dz = n_ * Np1;
315
316 // Psi_: (n*Np1) x (n*Np1)
317 // Row block i: x_i = sum_j Psi_[i,j]*z_j + Gamma_u terms
318 Psi_ = Eigen::MatrixXd::Zero(n_ * Np1, n_ * Np1);
319 Gamma_u_= Eigen::MatrixXd::Zero(n_ * Np1, m_ * N);
320 C_bar_ = Eigen::MatrixXd::Zero(p_ * Np1, n_ * Np1);
321
322 // x_0 row
323 Psi_.block(0, 0, n_, n_) = Eigen::MatrixXd::Identity(n_, n_);
324 C_bar_.block(0, 0, p_, n_) = plant_.C;
325
326 std::vector<Eigen::MatrixXd> A_pow(Np1);
327 A_pow[0] = Eigen::MatrixXd::Identity(n_, n_);
328 for (int i = 1; i < Np1; ++i) A_pow[i] = plant_.A * A_pow[i - 1];
329
330 for (int i = 1; i < Np1; ++i) {
331 Psi_.block(i * n_, 0, n_, n_) = A_pow[i];
332 for (int j = 0; j < i; ++j) {
333 Psi_.block(i * n_, (j + 1) * n_, n_, n_) = A_pow[i - 1 - j];
334 Gamma_u_.block(i * n_, j * m_, n_, m_) = A_pow[i - 1 - j] * plant_.B;
335 }
336 C_bar_.block(i * p_, i * n_, p_, n_) = plant_.C;
337 }
338
339 // Inverses
340 Eigen::MatrixXd Qinv = Q_proc_.ldlt().solve(Eigen::MatrixXd::Identity(n_, n_));
341 Eigen::MatrixXd Rinv = R_meas_.ldlt().solve(Eigen::MatrixXd::Identity(p_, p_));
342
343 // H_prior = block_diag(P0inv, Qinv, ..., Qinv) size dz x dz
344 Eigen::MatrixXd H_prior = Eigen::MatrixXd::Zero(dz, dz);
345 H_prior.block(0, 0, n_, n_) = P0inv_;
346 for (int i = 1; i < Np1; ++i)
347 H_prior.block(i * n_, i * n_, n_, n_) = Qinv;
348
349 // R_bar = block_diag(Rinv, ...) size p*Np1
350 Eigen::MatrixXd R_bar = Eigen::MatrixXd::Zero(p_ * Np1, p_ * Np1);
351 for (int i = 0; i < Np1; ++i)
352 R_bar.block(i * p_, i * p_, p_, p_) = Rinv;
353
354 H_mhe_ = H_prior + Psi_.transpose() * C_bar_.transpose() * R_bar * C_bar_ * Psi_;
355
356 ldlt_.compute(H_mhe_);
357 L_ = H_mhe_.eigenvalues().real().maxCoeff();
358 if (L_ <= 0.0) L_ = 1.0;
359
360 // Pre-allocate work vectors
361 z_sol_.resize(dz); z_sol_.setZero();
362 g_qp_ .resize(dz); g_qp_ .setZero();
363 lb_z_ .resize(dz); lb_z_ .fill(-1e30);
364 ub_z_ .resize(dz); ub_z_ .fill( 1e30);
365 tmp1_ .resize(dz); tmp1_ .setZero();
366 tmp2_ .resize(dz); tmp2_ .setZero();
367}

369// =============================================================================
370// projectX0Polytope (E4 - polytopic inequality constraints on arrival state)
371// =============================================================================
372
373Eigen::VectorXd MovingHorizonEstimator::projectX0Polytope(
374 const Eigen::VectorXd &x0) const
375{
376 const auto &C = params_.C_ineq;
377 const auto &d = params_.d_ineq;
378
379 // No polytopic constraint: just apply box, which mirrors what FISTA already did.
380 if (C.rows() == 0 || d.size() == 0) {
381 Eigen::VectorXd x = x0;
382 if (params_.xMin.size() == n_) x = x.cwiseMax(params_.xMin);
383 if (params_.xMax.size() == n_) x = x.cwiseMin(params_.xMax);
384 return x;
385 }
386
387 Eigen::VectorXd x = x0;
388 const int m_c = static_cast<int>(C.rows());
389
390 // Pre-compute squared row norms (constant across iterations).
391 Eigen::VectorXd row_sq(m_c);
392 for (int i = 0; i < m_c; ++i)
393 row_sq(i) = C.row(i).squaredNorm();
394
395 // Hildreth's cyclic half-space projections:
396 // if c_i' * x > d_i, project x onto the half-space {x : c_i' * x = d_i}.
397 for (int iter = 0; iter < params_.ineq_proj_iters; ++iter) {
398 bool all_satisfied = true;
399 for (int i = 0; i < m_c; ++i) {
400 if (row_sq(i) < 1e-14) continue;
401 const double viol = C.row(i).dot(x) - d(i);
402 if (viol > 0.0) {
403 x -= (viol / row_sq(i)) * C.row(i).transpose();
404 all_satisfied = false;
405 }
406 }
407 if (all_satisfied) break;
408 }
409
410 // Re-apply box constraints so both are simultaneously satisfied.
411 if (params_.xMin.size() == n_) x = x.cwiseMax(params_.xMin);
412 if (params_.xMax.size() == n_) x = x.cwiseMin(params_.xMax);
413
414 return x;
415}
