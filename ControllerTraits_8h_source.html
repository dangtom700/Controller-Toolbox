4// ============================================================
5// ControllerTraits.h
6//
7// Compile-time metadata that maps each controller type to the
8// tuning strategies it supports.
9//
10// Two enforcement levels:
11//
12// Hard error (static_assert) - tuner<C>() in ControllerTuner.h
13// fires when ControllerTraits<C>::supports_<X> == false.
14// The message names the compatible controllers and
15// suggests the correct tuner to use instead.
16//
17// Soft warning ([[deprecated]] struct) - instantiated inside
18// an `if constexpr` branch in the tuner template method.
19// Warns without blocking compilation when a strategy is
20// technically valid but leaves part of the controller
21// un-tuned (e.g. pole placement on LQG ignores Kalman Qf/Rf).
22//
23// To register a new controller:
24// 1. Forward-declare it below.
25// 2. Add a ctrl::ControllerTraits<YourType> specialisation.
26// ============================================================

107 // -- Primary template - undefined for unknown types ------------
108 // Any use of ControllerTraits<T> for an unregistered T triggers
109 // a clear compile error that names the missing specialisation.
110 template <typename C>
111 struct ControllerTraits
112 {
113 static_assert(sizeof(C) == 0,
114 "\n[ControllerTraits] No traits registered for this controller type.\n"
115 " Add a ctrl::ControllerTraits<YourType> specialisation\n"
116 " in lib/ControllerTraits.h before using it with any tuner.\n");
117 };
