97// ---------------------------------------------------------------------------
98// Log-sum-exp normalisation for numerical stability:
99// log w_i_unnorm = log w_i + log p(y|x_i)
100// Subtract max before exponentiating to avoid underflow.
101// ---------------------------------------------------------------------------
102void ParticleFilter::update(const Eigen::VectorXd &y,
103 const Eigen::VectorXd &u)
104{
105 Eigen::VectorXd log_w = w_.array().log(); // log of current weights
106
107 for (int i = 0; i < p_.n_particles; ++i) {
108 const Eigen::VectorXd innov = y - h_(particles_[i], u);
109 // log p(y|x_i) = -0.5 * innov' * R_inv * innov (Gaussian, dropping constant)
110 const double log_lik = -0.5 * innov.dot(R_inv_ * innov);
111 log_w(i) += log_lik;
112 }
113
114 // Normalise (log-sum-exp trick)
115 const double log_max = log_w.maxCoeff();
116 w_ = (log_w.array() - log_max).exp();
117 const double sum_w = w_.sum();
118 if (sum_w > 0.0)
119 w_ /= sum_w;
120 else
121 w_.setConstant(1.0 / p_.n_particles); // degenerate fallback
122
123 // Resample if effective sample size falls below threshold
124 if (resample_thresh_ > 0.0 && effectiveSampleSize() < resample_thresh_)
125 resample();
126}

136// ---------------------------------------------------------------------------
137// Systematic resampling (Kitagawa 1996) - O(N), lower variance than multinomial.
138// Generates N equally-spaced points on [0,1) offset by a single uniform draw.
139// ---------------------------------------------------------------------------
140void ParticleFilter::resample()
141{
142 const int N = p_.n_particles;
143
144 // Build CDF into pre-allocated member
145 cdf_(0) = w_(0);
146 for (int i = 1; i < N; ++i)
147 cdf_(i) = cdf_(i - 1) + w_(i);
148
149 // Single uniform draw U ~ Uniform[0, 1/N)
150 const double inv_N = 1.0 / static_cast<double>(N);
151 double u0 = uniform_(rng_) * inv_N;
152
153 int j = 0;
154 for (int i = 0; i < N; ++i) {
155 const double ui = u0 + static_cast<double>(i) * inv_N;
156 while (j < N - 1 && cdf_(j) < ui)
157 ++j;
158 resample_buf_[i] = particles_[j];
159 }
160
161 particles_.swap(resample_buf_);
162 w_.setConstant(inv_N);
163 ++resample_count_;
164}
