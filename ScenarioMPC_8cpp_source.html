9ScenarioMPC::ScenarioMPC(const StateSpace &sys, const ScenarioMPCParams &p)
10 : sys_(sys), p_(p),
11 n_(static_cast<int>(sys.A.rows())),
12 m_(static_cast<int>(sys.B.cols())),
13 pp_(static_cast<int>(sys.C.rows())),
14 rng_(p.seed)
15{
16 // --- dimension checks ----------------------------------------------------
17 if (p_.Nu > p_.Np)
18 throw std::invalid_argument("ScenarioMPC: Nu must be <= Np.");
19 if (p_.Q.rows() != pp_ || p_.Q.cols() != pp_)
20 throw std::invalid_argument("ScenarioMPC: Q must be pp x pp.");
21 if (p_.R.rows() != m_ || p_.R.cols() != m_)
22 throw std::invalid_argument("ScenarioMPC: R must be m x m.");
23 if (p_.Sigma_w.rows() != n_ || p_.Sigma_w.cols() != n_)
24 throw std::invalid_argument("ScenarioMPC: Sigma_w must be n x n.");
25 if (p_.uMin.size() != m_ || p_.uMax.size() != m_)
26 throw std::invalid_argument("ScenarioMPC: uMin/uMax must have m elements.");
27 if (p_.N_samples < 1)
28 throw std::invalid_argument("ScenarioMPC: N_samples must be >= 1.");
29
30 // --- Cholesky of Sigma_w for sampling w = L * z, z ~ N(0, I) -----------
31 // For Sigma_w = 0 (deterministic limit) L = 0.
32 Eigen::LLT<Eigen::MatrixXd> llt(p_.Sigma_w);
33 if (llt.info() == Eigen::Success) {
34 Sigma_w_chol_ = llt.matrixL();
35 } else {
36 // Fall back: treat as zero noise (pure deterministic MPC)
37 Sigma_w_chol_ = Eigen::MatrixXd::Zero(n_, n_);
38 }
39
40 // --- build condensed matrices and QP ------------------------------------
41 buildCondensedMatrices();
42
43 // --- initialise online state --------------------------------------------
44 x_nom_ = Eigen::VectorXd::Zero(n_);
45 y_ref_ = Eigen::VectorXd::Zero(pp_);
46 V_warm_ = Eigen::VectorXd::Zero(p_.Nu * m_);
47 avg_W_ = Eigen::VectorXd::Zero(p_.Np * pp_);
48 g_ = Eigen::VectorXd::Zero(p_.Nu * m_);
49 tmp1_ = Eigen::VectorXd::Zero(p_.Nu * m_);
50 tmp2_ = Eigen::VectorXd::Zero(p_.Nu * m_);
51 y_fista_ = Eigen::VectorXd::Zero(p_.Nu * m_);
52 x_noise_ = Eigen::VectorXd::Zero(n_);
53 z_noise_ = Eigen::VectorXd::Zero(n_);

58void ScenarioMPC::buildCondensedMatrices()
59{
60 const int Np = p_.Np;
61 const int Nu = p_.Nu;
62
63 F_.resize(Np * pp_, n_);
64 Phi_.resize(Np * pp_, Nu * m_);
65 Phi_.setZero();
66
67 // Precompute C * A^k * B for k = 0..Np-1
68 Eigen::MatrixXd Ak(n_, n_);
69 Ak.setIdentity();
70 std::vector<Eigen::MatrixXd> CAkB(Np);
71 for (int k = 0; k < Np; ++k) {
72 CAkB[k] = sys_.C * Ak * sys_.B;
73 Ak = sys_.A * Ak;
74 }
75
76 // F: row block i = C * A^(i+1) * x
77 Eigen::MatrixXd Aipow = sys_.A;
78 for (int i = 0; i < Np; ++i) {
79 F_.block(i * pp_, 0, pp_, n_) = sys_.C * Aipow;
80 Aipow = sys_.A * Aipow;
81 }
82
83 // Phi: standard hold-last condensed input matrix
84 for (int j = 0; j < Nu; ++j) {
85 const bool is_last = (j == Nu - 1);
86 for (int i = j; i < Np; ++i)
87 Phi_.block(i * pp_, j * m_, pp_, m_) += CAkB[i - j];
88
89 if (is_last) {
90 for (int t = Nu; t < Np; ++t)
91 for (int i = t; i < Np; ++i)
92 Phi_.block(i * pp_, j * m_, pp_, m_) += CAkB[i - t];
93 }
94 }
95
96 // Build Q_blk and R_blk for cost
97 Eigen::MatrixXd Qy = Eigen::MatrixXd::Zero(Np * pp_, Np * pp_);
98 for (int k = 0; k < Np; ++k)
99 Qy.block(k * pp_, k * pp_, pp_, pp_) = p_.Q;
100
101 Eigen::MatrixXd Ru = Eigen::MatrixXd::Zero(Nu * m_, Nu * m_);
102 for (int k = 0; k < Nu; ++k)
103 Ru.block(k * m_, k * m_, m_, m_) = p_.R;
104
105 PhiTQy_ = Phi_.transpose() * Qy; // precompute for online use
106
107 H_ = 2.0 * (PhiTQy_ * Phi_ + Ru);
108 H_ = 0.5 * (H_ + H_.transpose()); // symmetrise
109
110 ldlt_.compute(H_);
111
112 const Eigen::SelfAdjointEigenSolver<Eigen::MatrixXd> eig(H_);
113 const double L = eig.eigenvalues().maxCoeff();
114 L_inv_ = (L > 1e-12) ? (1.0 / L) : 1.0;
115
116 lb_.resize(Nu * m_);
117 ub_.resize(Nu * m_);
118 for (int k = 0; k < Nu; ++k) {
119 lb_.segment(k * m_, m_) = p_.uMin;
120 ub_.segment(k * m_, m_) = p_.uMax;
121 }
122}

146Eigen::VectorXd ScenarioMPC::computeControl()
147{
148 const int Np = p_.Np;
149 const int Nu = p_.Nu;
150
151 // --- sample N_s noise trajectories and compute avg_W -------------------
152 // avg_W[j] = C * (mean over s of: sum_{l=0}^{j-1} A^(j-1-l) * w^(s)[l])
153 avg_W_.setZero();
154
155 for (int s = 0; s < p_.N_samples; ++s) {
156 x_noise_.setZero();
157 for (int j = 0; j < Np; ++j) {
158 // Accumulate C * x_noise for this scenario's step j
159 avg_W_.segment(j * pp_, pp_) += sys_.C * x_noise_;
160
161 // Sample w_j ~ N(0, Sigma_w) = Sigma_w_chol * z, z ~ N(0,I) (reuse z_noise_)
162 for (int i = 0; i < n_; ++i)
163 z_noise_(i) = normal_(rng_);
164 x_noise_ = sys_.A * x_noise_ + Sigma_w_chol_ * z_noise_;
165 }
166 }
167 avg_W_ /= static_cast<double>(p_.N_samples);
168
169 // --- stacked reference (reuse pre-allocated R_stacked_) -----------------
170 for (int k = 0; k < Np; ++k)
171 R_stacked_.segment(k * pp_, pp_) = y_ref_;
172
173 // --- linear cost: g = 2 * Phi'*Q_blk * (F*x_nom + avg_W - R) -----------
174 const Eigen::VectorXd err = F_ * x_nom_ + avg_W_ - R_stacked_;
175
176 // g = 2 * Phi' * Q_blk * err (PhiTQy_ = Phi' * Q_blk precomputed at construction)
177 g_ = 2.0 * (PhiTQy_ * err);
178
179 // --- solve FISTA QP -----------------------------------------------------
180 QPSolveResult res = solveGradientProjectionQP(
181 H_, g_, lb_, ub_, ldlt_, 1.0 / L_inv_, p_.qpMaxIter, p_.qpTol,
182 V_warm_, tmp1_, tmp2_, y_fista_);
183 qp_converged_ = res.converged;
184
185 // --- extract first nominal move -----------------------------------------
186 Eigen::VectorXd u = V_warm_.head(m_);
187 for (int i = 0; i < m_; ++i)
188 u(i) = std::clamp(u(i), p_.uMin(i), p_.uMax(i));
189
190 // --- advance nominal model ----------------------------------------------
191 x_nom_ = sys_.A * x_nom_ + sys_.B * u;
192
193 // --- shift warm start ---------------------------------------------------
194 if (Nu > 1) {
195 V_warm_.head((Nu - 1) * m_) = V_warm_.segment(m_, (Nu - 1) * m_);
196 V_warm_.tail(m_) = u;
197 }
198
199 notifyObserver(u(0), y_ref_(0));
200 return u;
201}
