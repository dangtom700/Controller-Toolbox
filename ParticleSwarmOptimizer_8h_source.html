89 TunerResult optimize(const CostFn& cost)
90 {
91 const int N = p_.n_particles;
92 const int n = p_.n_dim;
93
94 // V_max per dimension
95 Eigen::VectorXd v_max = p_.v_frac * (p_.upper - p_.lower);
96
97 // --- Initialise positions and velocities ---
98 std::vector<Eigen::VectorXd> pos(N, Eigen::VectorXd(n));
99 std::vector<Eigen::VectorXd> vel(N, Eigen::VectorXd(n));
100 std::vector<Eigen::VectorXd> pbest(N, Eigen::VectorXd(n));
101 std::vector<double> pbest_cost(N, std::numeric_limits<double>::max());
102
103 std::uniform_real_distribution<double> ud01(0.0, 1.0);
104
105 for (int i = 0; i < N; ++i) {
106 for (int j = 0; j < n; ++j) {
107 std::uniform_real_distribution<double> udx(p_.lower[j], p_.upper[j]);
108 pos[i][j] = udx(rng_);
109 std::uniform_real_distribution<double> udv(-v_max[j], v_max[j]);
110 vel[i][j] = udv(rng_);
111 }
112 pbest_cost[i] = cost(pos[i]);
113 pbest[i] = pos[i];
114 }
115 int n_evals = N;
116
117 // Global best
118 Eigen::VectorXd gbest(n);
119 double gbest_cost = std::numeric_limits<double>::max();
120 for (int i = 0; i < N; ++i) {
121 if (pbest_cost[i] < gbest_cost) {
122 gbest_cost = pbest_cost[i];
123 gbest = pbest[i];
124 }
125 }
126
127 bool converged = false;
128 int iter = 0;
129 for (; iter < p_.max_iter; ++iter) {
130 for (int i = 0; i < N; ++i) {
131 for (int j = 0; j < n; ++j) {
132 double r1 = ud01(rng_);
133 double r2 = ud01(rng_);
134 vel[i][j] = p_.w * vel[i][j]
135 + p_.c1 * r1 * (pbest[i][j] - pos[i][j])
136 + p_.c2 * r2 * (gbest[j] - pos[i][j]);
137 // Velocity clamping
138 vel[i][j] = std::max(-v_max[j], std::min(v_max[j], vel[i][j]));
139
140 pos[i][j] += vel[i][j];
141 // Position clamping + zero velocity on boundary hit
142 if (pos[i][j] < p_.lower[j]) { pos[i][j] = p_.lower[j]; vel[i][j] = 0.0; }
143 if (pos[i][j] > p_.upper[j]) { pos[i][j] = p_.upper[j]; vel[i][j] = 0.0; }
144 }
145
146 double c = cost(pos[i]);
147 ++n_evals;
148 if (c < pbest_cost[i]) {
149 pbest_cost[i] = c;
150 pbest[i] = pos[i];
151 if (c < gbest_cost) { gbest_cost = c; gbest = pos[i]; }
152 }
153 }
154
155 // Convergence: stddev of personal best costs
156 double mean = 0.0;
157 for (double c : pbest_cost) mean += c;
158 mean /= N;
159 double var = 0.0;
160 for (double c : pbest_cost) var += (c - mean) * (c - mean);
161 if (std::sqrt(var / N) < p_.tol) { converged = true; break; }
162 }
163
164 TunerResult r;
165 r.params = gbest;
166 r.cost = gbest_cost;
167 r.nEvals = n_evals;
168 r.nGens = iter + 1;
169 r.converged = converged;
170 return r;
171 }
