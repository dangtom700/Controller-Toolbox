10 // ---------------------------------------------------------------------------
11 // PBH (Popov-Belevitch-Hautus) stabilizability test for discrete-time (A, B).
12 // A system is stabilisable iff for every eigenvalue lambda of A with |lambda| >= 1,
13 // rank([lambdaI - A, B]) = n. Returns false if any unstable mode is uncontrollable.
14 // ---------------------------------------------------------------------------
15 static bool isPBHStabilizable(const Eigen::MatrixXd &A, const Eigen::MatrixXd &B)
16 {
17 const int n = A.rows();
18 Eigen::EigenSolver<Eigen::MatrixXd> es(A, /*computeEigenvectors=*/false);
19 const Eigen::VectorXcd &eigs = es.eigenvalues();
20
21 for (int i = 0; i < eigs.size(); ++i)
22 {
23 if (std::abs(eigs[i]) < 1.0) continue; // stable mode - skip
24
25 // Build [lambdaI - A | B] and check its rank
26 Eigen::MatrixXcd PBH(n, n + B.cols());
27 PBH.leftCols(n) = eigs[i] * Eigen::MatrixXcd::Identity(n, n)
28 - A.cast<std::complex<double>>();
29 PBH.rightCols(B.cols()) = B.cast<std::complex<double>>();
30
31 Eigen::JacobiSVD<Eigen::MatrixXcd> svd(PBH);
32 const double threshold = 1e-10 * svd.singularValues()(0);
33 const int rnk = (svd.singularValues().array().abs() > threshold).count();
34 if (rnk < n) return false; // unstable uncontrollable mode found
35 }
36 return true;
37 }

39 // ---------------------------------------------------------------------------
40 // Doubling algorithm for DARE (Smith / Pappas-Laub-Sandell iteration).
41 //
42 // Maintains a triplet (X_k, L_k, G_k) such that X_k = T^{2^k}(Q) where T
43 // is the Riccati operator. Each iteration doubles the effective horizon:
44 //
45 // W = I + G_k X_k
46 // Z1 = W^{-1} L_k (W^{-1} applied on the right of L_k)
47 // Z2 = W^{-1} G_k
48 // X_{k+1} = X_k + L_k' X_k Z1 (2^{k+1}-step cost)
49 // G_{k+1} = G_k + L_k Z2 L_k' (2^{k+1}-step controllability Gramian)
50 // L_{k+1} = L_k Z1 (2^{k+1}-step closed-loop A)
51 //
52 // Converges quadratically (doubles correct digits each iteration); typically
53 // 30-50 iterations suffice for double precision, vs. thousands for value iteration.
54 //
55 // Ref: Pappas, Laub & Sandell "On a Numerical Solution of the Discrete-Time
56 // Algebraic Riccati Equation" IEEE TAC (1980);
57 // Varga "On solving discrete-time periodic Riccati equations" (2006).
58 // ---------------------------------------------------------------------------
59 DareResult DiscreteLQR::solveDARE(const Eigen::MatrixXd &A,
60 const Eigen::MatrixXd &B,
61 const Eigen::MatrixXd &Q,
62 const Eigen::MatrixXd &R)
63 {
64 const int n = A.rows();
65 const int maxIter = 100; // quadratic convergence: far fewer needed than value-iter
66 const double tol = 1e-12;
67
68 const Eigen::MatrixXd In = Eigen::MatrixXd::Identity(n, n);
69
70 // G0 = B R^{-1} B' (initial "controllability Gramian" factor)
71 const Eigen::MatrixXd G0 = B * R.ldlt().solve(B.transpose());
72
73 Eigen::MatrixXd X = Q; // DARE solution estimate (X_0 = Q = T^1(0))
74 Eigen::MatrixXd Lk = A; // reduced closed-loop A (L_0 = A)
75 Eigen::MatrixXd Gk = G0; // reduced Gramian (G_0 = BR^{-1}B')
76
77 for (int iter = 0; iter < maxIter; ++iter)
78 {
79 const Eigen::MatrixXd W = In + Gk * X; // I + G_k X_k
80 const Eigen::PartialPivLU<Eigen::MatrixXd> Wlu(W);
81
82 // W Z1 = Lk --> Z1 = W^{-1} Lk
83 const Eigen::MatrixXd Z1 = Wlu.solve(Lk);
84 // W Z2 = Gk --> Z2 = W^{-1} Gk
85 const Eigen::MatrixXd Z2 = Wlu.solve(Gk);
86
87 const Eigen::MatrixXd X_new = X + Lk.transpose() * X * Z1;
88 const Eigen::MatrixXd G_new = Gk + Lk * Z2 * Lk.transpose();
89 const Eigen::MatrixXd L_new = Lk * Z1;
90
91 const double err = (X_new - X).norm() / (1.0 + X.norm());
92 // Enforce symmetry at each iteration: the Riccati solution is symmetric by
93 // construction, but each matrix multiply introduces O(eps_mach) asymmetry.
94 // Without this, X can drift asymmetric over 30-50 iterations, which causes
95 // LDLT on S = R + B'*P*B to report NumericalIssue on near-rank-deficient systems.
96 X = 0.5 * (X_new + X_new.transpose());
97 Gk = G_new;
98 Lk = L_new;
99
100 if (err < tol)
101 return {X, true, iter + 1};
102 }
103
104 return {X, false, maxIter};
105 }
106
107 // ---------------------------------------------------------------------------
108 // PBH detectability test for (A, Q).
109 // Detectability requires (A, Q^{1/2}) to satisfy: for every eigenvalue lambda
110 // of A with |lambda| >= 1, rank([lambda*I - A; Q^{1/2}]) = n.
111 // Equivalently, (A', (Q^{1/2})') must be stabilizable (dual test).
112 // We extract Q^{1/2} via eigendecomposition (Q is symmetric PSD).
113 // ---------------------------------------------------------------------------
114 static bool isPBHDetectable(const Eigen::MatrixXd &A, const Eigen::MatrixXd &Q)
115 {
116 Eigen::SelfAdjointEigenSolver<Eigen::MatrixXd> eig(Q);
117 const Eigen::VectorXd &sv = eig.eigenvalues();
118 const Eigen::MatrixXd &V = eig.eigenvectors();
119
120 // Collect rows of Q^{1/2} for non-trivial eigenvalues
121 const double eps = 1e-10 * sv.maxCoeff();
122 std::vector<int> nz;
123 for (int i = 0; i < sv.size(); ++i)
124 if (sv(i) > eps) nz.push_back(i);
125
126 if (nz.empty()) return false; // Q = 0: no information about any state
127
128 const int n = A.rows();
129 Eigen::MatrixXd CQ(static_cast<int>(nz.size()), n);
130 for (int i = 0; i < static_cast<int>(nz.size()); ++i)
131 CQ.row(i) = std::sqrt(sv(nz[i])) * V.col(nz[i]).transpose();
132
133 // Detectability of (A, CQ) <=> stabilizability of (A', CQ')
134 return isPBHStabilizable(A.transpose(), CQ.transpose());
135 }

137 DiscreteLQR::DiscreteLQR(const StateSpace &plant, const LQRParams &params)
138 : Ts_(plant.Ts), n_(plant.stateSize()), m_(plant.inputSize()),
139 dare_converged_(false), dare_iterations_(0)
140 {
141 if (!isPBHStabilizable(plant.A, plant.B))
142 {
143 std::cerr << "[DiscreteLQR] WARNING: (A,B) failed the PBH stabilizability test. "
144 << "DARE may not converge - check that all unstable modes are controllable.\n";
145 }
146
147 if (!isPBHDetectable(plant.A, params.Q))
148 {
149 std::cerr << "[DiscreteLQR] WARNING: (A,Q) failed the PBH detectability test. "
150 << "DARE may not converge - check that all unstable modes are penalised "
151 << "by Q (no zero-row in Q corresponding to an unstable state).\n";
152 }
153
154 DareResult res = solveDARE(plant.A, plant.B, params.Q, params.R);
155 dare_converged_ = res.converged;
156 dare_iterations_ = res.iterations;
157
158 if (!res.converged)
159 {
160 std::cerr << "[DiscreteLQR] WARNING: DARE did not converge in "
161 << res.iterations << " iterations. "
162 << "Using last iterate - verify (A,B) is stabilisable "
163 << "and (A,sqrt(Q)) is detectable.\n";
164 }
165
166 P_ = res.P;
167 const Eigen::MatrixXd S = params.R + plant.B.transpose() * P_ * plant.B;
168 K_ = S.ldlt().solve(plant.B.transpose() * P_ * plant.A);
169 }

172 Eigen::VectorXd DiscreteLQR::compute(const Eigen::VectorXd &x,
173 const Eigen::VectorXd &x_ref,
174 const Eigen::VectorXd &u_ff) const
175 {
176 if (x_ref.size() != 0 && x_ref.size() != n_)
177 throw std::invalid_argument(
178 "DiscreteLQR::compute: x_ref has wrong size (" +
179 std::to_string(x_ref.size()) + "), expected 0 or " + std::to_string(n_));
180 if (u_ff.size() != 0 && u_ff.size() != m_)
181 throw std::invalid_argument(
182 "DiscreteLQR::compute: u_ff has wrong size (" +
183 std::to_string(u_ff.size()) + "), expected 0 or " + std::to_string(m_));
184
185 Eigen::VectorXd xe = x;
186 if (x_ref.size() == n_)
187 xe -= x_ref;
188
189 Eigen::VectorXd u = -K_ * xe;
190 if (u_ff.size() == m_)
191 u += u_ff;
192 return u;
193 }
