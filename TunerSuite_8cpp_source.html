34 // Return the tier for a given (tuner, controller) pair.
35 // Tuner names match the cheatsheet sections.
36 static Tier tier(const char *tuner, CtrlKind k)
37 {
38 // ---------- Relay / ZN (cheatsheet Section 1) ----------
39 if (!strcmp(tuner, "RelayZN"))
40 {
41 switch (k)
42 {
43 case CtrlKind::PID:
44 return Tier::IDEAL;
45 case CtrlKind::Smith:
46 return Tier::SOFT; // tune the inner PID
47 case CtrlKind::LeadLag:
48 return Tier::SOFT; // Ku/Tu can give a gain hint
49 default:
50 return Tier::FALLBACK;
51 }
52 }
53 // ---------- IMC-PID (cheatsheet Section 2) ----------
54 if (!strcmp(tuner, "IMCPID"))
55 {
56 switch (k)
57 {
58 case CtrlKind::PID:
59 return Tier::IDEAL;
60 case CtrlKind::Smith:
61 return Tier::SOFT;
62 case CtrlKind::LeadLag:
63 return Tier::SOFT; // Kp -> gain K hint
64 default:
65 return Tier::FALLBACK;
66 }
67 }
68 // ---------- Cohen-Coon (cheatsheet Section 2 variant) ----------
69 if (!strcmp(tuner, "CohenCoon"))
70 {
71 switch (k)
72 {
73 case CtrlKind::PID:
74 return Tier::IDEAL;
75 case CtrlKind::Smith:
76 return Tier::SOFT;
77 default:
78 return Tier::FALLBACK;
79 }
80 }
81 // ---------- Bryson's Rule (cheatsheet Section 3) ----------
82 if (!strcmp(tuner, "Bryson"))
83 {
84 switch (k)
85 {
86 case CtrlKind::LQR:
87 return Tier::IDEAL;
88 case CtrlKind::LQG:
89 return Tier::IDEAL;
90 case CtrlKind::MPC:
91 return Tier::SOFT; // rho_y/rho_u analogy
92 case CtrlKind::SMC:
93 return Tier::SOFT; // Q ratio -> c_e/c_de hint
94 default:
95 return Tier::FALLBACK;
96 }
97 }
98 // ---------- Kalman Noise (cheatsheet Section 4 observer part) ----------
99 if (!strcmp(tuner, "KalmanNoise"))
100 {
101 switch (k)
102 {
103 case CtrlKind::LQG:
104 return Tier::IDEAL;
105 case CtrlKind::Generic:
106 return Tier::SOFT; // standalone KalmanFilter
107 default:
108 return Tier::FALLBACK;
109 }
110 }
111 // ---------- MPC Horizon (cheatsheet Section 5) ----------
112 if (!strcmp(tuner, "MPCHorizon"))
113 {
114 switch (k)
115 {
116 case CtrlKind::MPC:
117 return Tier::IDEAL;
118 case CtrlKind::LQR:
119 return Tier::SOFT; // settling-time info is still useful
120 default:
121 return Tier::FALLBACK;
122 }
123 }
124 // ---------- Loop Shaping (cheatsheet Section 6) ----------
125 if (!strcmp(tuner, "LoopShaping"))
126 {
127 switch (k)
128 {
129 case CtrlKind::LeadLag:
130 return Tier::IDEAL;
131 case CtrlKind::PID:
132 return Tier::SOFT; // omega_c -> bandwidth / N hint
133 default:
134 return Tier::FALLBACK;
135 }
136 }
137 // ---------- Optimisation-based (cheatsheet Section 7) ----------
138 // Always IDEAL: the method is controller-agnostic.
139 if (!strcmp(tuner, "Optimise"))
140 return Tier::IDEAL;
141
142 return Tier::FALLBACK;
143 }

508 // ============================================================
509 // 8. Optimisation-based Nelder-Mead (cheatsheet Section 7)
510 // ============================================================
511
512 // ---- Nelder-Mead simplex minimiser ----------------------------------------
513 // Standard algorithm: Nelder & Mead (1965) as described in
514 // Lagarias et al. "Convergence Properties of the Nelder-Mead Simplex
515 // Method in Low Dimensions" (SIAM J. Optim. 1998).
516 //
517 // Parameters: alpha=1 (reflect), gamma=2 (expand), rho=0.5 (contract), sigma=0.5 (shrink).
518 // Bounds are enforced by clamping candidates before each evaluation.
519 // -------------------------------------------------------------------------
520 std::vector<double> TunerSuite::nelderMead(
521 std::function<double(const std::vector<double> &)> f,
522 std::vector<double> x0,
523 const std::vector<std::pair<double, double>> &bounds,
524 int maxEvals,
525 double tol,
526 int *evalCount_out)
527 {
528 const int n = static_cast<int>(x0.size());
529 if (n == 0)
530 return x0;
531
532 // Clamp helper
533 auto clamp = [&](std::vector<double> v)
534 {
535 for (int i = 0; i < n; ++i)
536 v[i] = std::max(bounds[i].first, std::min(bounds[i].second, v[i]));
537 return v;
538 };
539
540 // Build initial simplex: x0 + n perturbed vertices.
541 // Perturbation = 5 % of range (or 0.05 if degenerate).
542 using Vertex = std::pair<std::vector<double>, double>;
543 std::vector<Vertex> simplex;
544 simplex.reserve(n + 1);
545
546 simplex.push_back({clamp(x0), f(clamp(x0))});
547 for (int i = 0; i < n; ++i)
548 {
549 auto xi = x0;
550 double range = bounds[i].second - bounds[i].first;
551 xi[i] += (range > 1e-12) ? 0.05 * range : 0.05;
552 auto cxi = clamp(xi);
553 simplex.push_back({cxi, f(cxi)});
554 }
555
556 int evals = n + 1;
557
558 const double alpha = 1.0;
559 const double gamma = 2.0;
560 const double rho = 0.5;
561 const double sigma = 0.5;
562
563 auto centroid = [&](int exclude)
564 {
565 std::vector<double> c(n, 0.0);
566 for (int i = 0; i <= n; ++i)
567 if (i != exclude)
568 for (int j = 0; j < n; ++j)
569 c[j] += simplex[i].first[j];
570 for (auto &v : c)
571 v /= n;
572 return c;
573 };
574
575 auto eval = [&](std::vector<double> v) -> Vertex
576 {
577 auto cv = clamp(v);
578 return {cv, f(cv)};
579 };
580
581 while (evals < maxEvals)
582 {
583 // Sort: best first (smallest cost)
584 std::sort(simplex.begin(), simplex.end(),
585 [](const Vertex &a, const Vertex &b)
586 { return a.second < b.second; });
587
588 // Convergence check: simplex diameter
589 double diam = 0.0;
590 for (int i = 1; i <= n; ++i)
591 for (int j = 0; j < n; ++j)
592 diam = std::max(diam, std::abs(simplex[i].first[j] - simplex[0].first[j]));
593 if (diam < tol)
594 break;
595
596 auto xo = centroid(n); // centroid of best n vertices
597 auto [xr, fr] = eval([&] { // reflect
598 std::vector<double> v(n);
599 for (int j = 0; j < n; ++j)
600 v[j] = xo[j] + alpha * (xo[j] - simplex[n].first[j]);
601 return v;
602 }());
603 ++evals;
604
605 if (fr < simplex[0].second)
606 {
607 // Try expansion
608 auto [xe, fe] = eval([&]
609 {
610 std::vector<double> v(n);
611 for (int j=0;j<n;++j) v[j]=xo[j]+gamma*(xr[j]-xo[j]);
612 return v; }());
613 ++evals;
614 simplex[n] = (fe < fr) ? Vertex{xe, fe} : Vertex{xr, fr};
615 }
616 else if (fr < simplex[n - 1].second)
617 {
618 simplex[n] = {xr, fr};
619 }
620 else
621 {
622 // Contraction
623 bool outer = (fr < simplex[n].second);
624 auto [xc, fc] = eval([&]
625 {
626 std::vector<double> v(n);
627 const auto& xw = outer ? xr : simplex[n].first;
628 for (int j=0;j<n;++j) v[j]=xo[j]+rho*(xw[j]-xo[j]);
629 return v; }());
630 ++evals;
631
632 if (fc < (outer ? fr : simplex[n].second))
633 {
634 simplex[n] = {xc, fc};
635 }
636 else
637 {
638 // Shrink
639 auto &best = simplex[0].first;
640 for (int i = 1; i <= n && evals < maxEvals; ++i)
641 {
642 std::vector<double> v(n);
643 for (int j = 0; j < n; ++j)
644 v[j] = best[j] + sigma * (simplex[i].first[j] - best[j]);
645 simplex[i] = eval(v);
646 ++evals;
647 }
648 }
649 }
650 }
651
652 std::sort(simplex.begin(), simplex.end(),
653 [](const Vertex &a, const Vertex &b)
654 { return a.second < b.second; });
655
656 if (evalCount_out)
657 *evalCount_out = evals;
658 return simplex[0].first;
659 }

661 // ---- Public optimise() wrapper --------------------------------------------
662 OptimTuneResult TunerSuite::optimise(
663 CtrlKind target,
664 const std::vector<std::pair<double, double>> &paramBounds,
665 std::function<double(const std::vector<double> &)> costFn,
666 std::vector<double> x0,
667 int maxEvals,
668 double tol)
669 {
670 OptimTuneResult res;
671
672 if (paramBounds.empty())
673 {
674 softWarn(res, "TunerSuite::optimise: paramBounds is empty - nothing to optimise.");
675 res.success = false;
676 return res;
677 }
678
679 if (!costFn)
680 {
681 softWarn(res, "TunerSuite::optimise: costFn is null - provide a simulation cost function.");
682 res.success = false;
683 return res;
684 }
685
686 const int n = static_cast<int>(paramBounds.size());
687
688 // Default initial guess: midpoint of each bound
689 if (x0.empty())
690 {
691 x0.resize(n);
692 for (int i = 0; i < n; ++i)
693 x0[i] = 0.5 * (paramBounds[i].first + paramBounds[i].second);
694 }
695 else if (static_cast<int>(x0.size()) != n)
696 {
697 softWarn(res, "TunerSuite::optimise: x0 size mismatch with paramBounds - using midpoints.");
698 x0.resize(n);
699 for (int i = 0; i < n; ++i)
700 x0[i] = 0.5 * (paramBounds[i].first + paramBounds[i].second);
701 }
702
703 int evals = 0;
704 auto best = nelderMead(costFn, x0, paramBounds, maxEvals, tol, &evals);
705
706 res.bestParams = best;
707 res.bestCost = costFn(best);
708 res.evalCount = evals + 1;
709 res.success = true;
710
711 // All controller types are valid for optimisation - no warning needed.
712 // Just record which target was requested (informational).
713 (void)target;
714
715 return res;
716 }

718 // ============================================================
719 // makeISECost / makeITAECost
720 // ============================================================
721 std::function<double(const std::vector<double> &)>
722 TunerSuite::makeISECost(
723 const StateSpace &plant,
724 double ref,
725 int N,
726 std::function<std::unique_ptr<IController>(const std::vector<double> &)> factory)
727 {
728 return [plant, ref, N, factory](const std::vector<double> &params) -> double
729 {
730 std::unique_ptr<IController> ctrl;
731 try
732 {
733 ctrl = factory(params);
734 }
735 catch (...)
736 {
737 return 1e30; // construction failed - reject this candidate
738 }
739 if (!ctrl)
740 return 1e30;
741
742 StateSpace p2 = plant; // local copy so state is independent
743 Eigen::VectorXd x = Eigen::VectorXd::Zero(p2.stateSize());
744 double y = 0.0;
745 double ise = 0.0;
746 const double Ts = p2.Ts;
747
748 for (int k = 0; k < N; ++k)
749 {
750 double e = ref - y;
751 double u;
752 try
753 {
754 u = ctrl->compute(e);
755 }
756 catch (...)
757 {
758 return 1e30;
759 }
760 if (!std::isfinite(u))
761 return 1e30;
762
763 Eigen::VectorXd uv(1);
764 uv << u;
765 Eigen::VectorXd yv = ssStep(p2, x, uv);
766 y = yv(0);
767 if (!std::isfinite(y))
768 return 1e30;
769
770 ise += e * e * Ts;
771 }
772 return ise;
773 };
774 }

780 int N,
781 std::function<std::unique_ptr<IController>(const std::vector<double> &)> factory)
782 {
783 return [plant, ref, N, factory](const std::vector<double> &params) -> double
784 {
785 std::unique_ptr<IController> ctrl;
786 try
787 {
788 ctrl = factory(params);
789 }
790 catch (...)
791 {
792 return 1e30;
793 }
794 if (!ctrl)
795 return 1e30;
796
797 StateSpace p2 = plant;
798 Eigen::VectorXd x = Eigen::VectorXd::Zero(p2.stateSize());
799 double y = 0.0;
800 double itae = 0.0;
801 const double Ts = p2.Ts;
802
803 for (int k = 0; k < N; ++k)
804 {
805 double t = k * Ts;
806 double e = ref - y;
807 double u;
808 try
809 {
810 u = ctrl->compute(e);
811 }
812 catch (...)
813 {
814 return 1e30;
815 }
816 if (!std::isfinite(u))
817 return 1e30;
818
819 Eigen::VectorXd uv(1);
820 uv << u;
821 Eigen::VectorXd yv = ssStep(p2, x, uv);
822 y = yv(0);
823 if (!std::isfinite(y))
824 return 1e30;
825
826 itae += t * std::abs(e) * Ts;
827 }
828 return itae;
829 };
