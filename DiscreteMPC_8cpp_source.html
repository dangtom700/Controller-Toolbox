25 // ---------------------------------------------------------------------------
26 // buildPredictionMatrices - plant-dependent only (A, B, C).
27 // Call when the state-space model changes (setPlant).
28 //
29 // F(i.p : (i+1).p, :) = C . A^{i+1} i = 0...Np-1
30 // Phi(i.p, j.m) = C . A^{i-j} . B j <= i, 0 otherwise
31 // Gu(i,:) = Sigma_{j=0}^{i} C.A^j.B
32 // ---------------------------------------------------------------------------
33 void DiscreteMPC::buildPredictionMatrices()
34 {
35 const int n = plant_.stateSize();
36 const int m = plant_.inputSize();
37 const int p = plant_.outputSize();
38 const int Np = p_.Np;
39 const int Nc = p_.Nc;
40
41 // Powers of A: Apow[k] = A^k
42 std::vector<Eigen::MatrixXd> Apow(Np + 1);
43 Apow[0] = Eigen::MatrixXd::Identity(n, n);
44 for (int k = 1; k <= Np; ++k)
45 Apow[k] = plant_.A * Apow[k - 1];
46
47 // F: (Np.p) * n
48 F_.resize(Np * p, n);
49 for (int i = 0; i < Np; ++i)
50 F_.block(i * p, 0, p, n) = plant_.C * Apow[i + 1];
51
52 // Phi: (Np.p) * (Nc.m)
53 Phi_.resize(Np * p, Nc * m);
54 Phi_.setZero();
55 for (int i = 0; i < Np; ++i)
56 for (int j = 0; j <= std::min(i, Nc - 1); ++j)
57 Phi_.block(i * p, j * m, p, m) = plant_.C * Apow[i - j] * plant_.B;
58
59 // Gu: (Np.p) * m - cumulative step response for u_prev offset
60 Gu_.resize(Np * p, m);
61 Gu_.block(0, 0, p, m) = plant_.C * plant_.B;
62 for (int i = 1; i < Np; ++i)
63 Gu_.block(i * p, 0, p, m) = Gu_.block((i - 1) * p, 0, p, m) + plant_.C * Apow[i] * plant_.B;
64 }
65
66 // ---------------------------------------------------------------------------
67 // buildCostMatrix - weight-dependent only (rho_y, rho_u).
68 // Requires Phi_ to be current. Call after buildPredictionMatrices or
69 // whenever rho_y / rho_u change.
70 // ---------------------------------------------------------------------------
71 void DiscreteMPC::buildCostMatrix()
72 {
73 const int m = plant_.inputSize();
74 const int p = plant_.outputSize();
75 const int Np = p_.Np;
76 const int Nc = p_.Nc;
77
78 // Weight matrices
79 Qy_ = p_.rho_y * Eigen::MatrixXd::Identity(Np * p, Np * p);
80 Ru_ = p_.rho_u * Eigen::MatrixXd::Identity(Nc * m, Nc * m);
81
82 // Precompute Hessian (positive definite for rho_u > 0)
83 H_ = Phi_.transpose() * Qy_ * Phi_ + Ru_;
84
85 // Lipschitz constant = max eigenvalue of H (gradient-projection step 1/L)
86 L_ = H_.selfadjointView<Eigen::Upper>().eigenvalues().maxCoeff();
87
88 // Pre-factor H_ once; reused every computeRef() call without re-factorisation.
89 ldlt_ = H_.ldlt();
90
91 // Pre-allocate work vectors so computeRef() is allocation-free per step
92 R_stack_.resize(Np * p);
93 pred_err_.resize(Np * p);
94 grad_.resize(Nc * m);
95 DeltaU_.resize(Nc * m);
96 grad_k_.resize(Nc * m);
97 DU_new_.resize(Nc * m);
98 lb_.resize(Nc * m);
99 ub_.resize(Nc * m);
100 cumMin_.resize(m);
101 cumMax_.resize(m);
102 }

119 // Full interface: optimise and return u[k].
120 //
121 // Solves the box-constrained QP:
122 // min_{DeltaU} 0.5 DeltaU'HDeltaU + g'DeltaU
123 // s.t. lb_ <= DeltaU <= ub_
124 //
125 // via gradient projection with constant step alpha = 1/L (L = max eigenvalue of H).
126 // Warm-started from the clamped unconstrained optimum; typically converges in
127 // a handful of iterations for MPC horizon sizes.
128 //
129 // Bounds per horizon step j (rolling worst-case tightening):
130 // u[j] = u_prev + Sigma_{i=0}^{j} Deltau_i must satisfy uMin <= u[j] <= uMax.
131 //
132 // At step j, Deltau_j is bounded by the worst-case contribution from prior steps:
133 // lb_(j,k) = max(duMin, uMin_k - u_prev_k - cumMax_k)
134 // ub_(j,k) = min(duMax, uMax_k - u_prev_k - cumMin_k)
135 // where cumMin/cumMax track the tightest possible cumulative Deltau range.
136 // This guarantees u feasibility across the full horizon without a full-QP reformulation.
137 Eigen::VectorXd DiscreteMPC::computeRef(const Eigen::VectorXd &x,
138 const Eigen::VectorXd &r_ref)
139 {
140 const int p = plant_.outputSize();
141 const int m = plant_.inputSize();
142 const int Np = p_.Np;
143 const int Nc = p_.Nc;
144
145 // Stack reference for all prediction steps
146 for (int i = 0; i < Np; ++i)
147 R_stack_.segment(i * p, p) = r_ref;
148
149 // Gradient at DeltaU = 0: g = Phi'.Qy.(F.x + Gu.u_prev - R)
150 pred_err_.noalias() = F_ * x + Gu_ * u_prev_ - R_stack_;
151 grad_.noalias() = Phi_.transpose() * (Qy_ * pred_err_);
152
153 // Build rolling worst-case tightened box bounds lb_/ub_ on DeltaU \in R^{Nc*m}.
154 // cumMin_(k) / cumMax_(k) track the range of cumulative Deltau for input channel k
155 // up to (but not including) the current step, assuming each prior Deltau_i was at
156 // its own tightened lb/ub. This is conservative (over-tightens slightly) but
157 // guarantees that u = u_prev + Sigma Deltau stays within [uMin, uMax] for every step.
158 cumMin_.setZero();
159 cumMax_.setZero();
160 for (int j = 0; j < Nc; ++j)
161 {
162 for (int k = 0; k < m; ++k)
163 {
164 const int idx = j * m + k;
165 const double lo = std::max(p_.duMin, p_.uMin - u_prev_(k) - cumMax_(k));
166 const double hi = std::min(p_.duMax, p_.uMax - u_prev_(k) - cumMin_(k));
167 lb_(idx) = lo;
168 ub_(idx) = (hi >= lo) ? hi : lo; // collapse infeasible interval to lo
169 cumMin_(k) += lo;
170 cumMax_(k) += ub_(idx);
171 }
172 }
173
174 // Warm-start: clamped unconstrained optimum (ldlt_ pre-factored in buildCondensedMatrices)
175 if (ldlt_.info() != Eigen::Success)
176 return u_prev_; // degenerate Hessian - hold previous input
177
178 // Gradient projection - all work vectors pre-allocated; zero per-step allocation.
179 const auto qp = solveGradientProjectionQP(
180 H_, grad_, lb_, ub_, ldlt_, L_, p_.qpMaxIter, p_.qpTol,
181 DeltaU_, grad_k_, DU_new_);
182 last_qp_converged_ = qp.converged;
183 last_qp_iters_ = qp.iters;
184
185 // Non-convergence is queryable via lastQPConverged() / isHealthy().
186 // The clog print is debug-only: it produces ~50 k lines per boiler run at
187 // default qpMaxIter=200 and corrupts stdout. Use lastQPConverged() in
188 // application code to detect and handle non-convergence.
189#ifndef NDEBUG
190 if (!last_qp_converged_)
191 std::clog << "[DiscreteMPC] WARNING: QP solver reached max iterations ("
192 << p_.qpMaxIter << "). Use lastQPConverged() to check per step.\n";
193#endif
194
195 // Apply first control increment
196 const Eigen::VectorXd du = DeltaU_.head(m);
197 const Eigen::VectorXd u = (u_prev_ + du).cwiseMax(p_.uMin).cwiseMin(p_.uMax);
198
199 // Advance open-loop state estimate (used by compute() wrapper next step).
200 // This is a pure-prediction step with no measurement correction. For plants
201 // with model mismatch, persistent disturbances, or integrating modes, x_hat_
202 // will drift away from the real plant state over time. Call setState() with an
203 // external observer estimate (e.g., KalmanFilter::state()) before each step to
204 // avoid this. The compute(error) wrapper is only reliable for D=0, well-matched
205 // models without significant unmeasured disturbances.
206 x_hat_ = plant_.A * x + plant_.B * u;
207 u_prev_ = u;
208
209 return u;
210 }
