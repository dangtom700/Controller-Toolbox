9TubeMPC::TubeMPC(const StateSpace &sys, const TubeMPCParams &p)
10 : sys_(sys), p_(p),
11 n_(static_cast<int>(sys.A.rows())),
12 m_(static_cast<int>(sys.B.cols())),
13 pp_(static_cast<int>(sys.C.rows()))
14{
15 // --- dimension checks ---
16 if (p_.Nu > p_.Np)
17 throw std::invalid_argument("TubeMPC: Nu must be <= Np.");
18 if (p_.Q.rows() != pp_ || p_.Q.cols() != pp_)
19 throw std::invalid_argument("TubeMPC: Q must be pp x pp.");
20 if (p_.R.rows() != m_ || p_.R.cols() != m_)
21 throw std::invalid_argument("TubeMPC: R must be m x m.");
22 if (p_.K.rows() != m_ || p_.K.cols() != n_)
23 throw std::invalid_argument("TubeMPC: K must be m x n.");
24 if (p_.wMax.size() != n_)
25 throw std::invalid_argument("TubeMPC: wMax must have n elements.");
26 if (p_.uMin.size() != m_ || p_.uMax.size() != m_)
27 throw std::invalid_argument("TubeMPC: uMin/uMax must have m elements.");
28 if ((p_.wMax.array() < 0.0).any())
29 throw std::invalid_argument("TubeMPC: wMax elements must be >= 0.");
30
31 // --- spectral radius check: rho(A_cl) < 1 ---
32 const Eigen::MatrixXd A_cl = sys_.A + sys_.B * p_.K;
33 const double rho = A_cl.eigenvalues().cwiseAbs().maxCoeff();
34 if (rho >= 1.0)
35 throw std::invalid_argument(
36 "TubeMPC: K does not stabilise A+B*K (spectral radius = " +
37 std::to_string(rho) + " >= 1). Choose a stabilising gain K.");
38
39 // --- compute mRPI and build condensed matrices ---
40 computeMRPI();
41 buildCondensedMatrices();
42
43 // --- initialise online state ---
44 x_actual_ = Eigen::VectorXd::Zero(n_);
45 x_nom_ = Eigen::VectorXd::Zero(n_);
46 y_ref_ = Eigen::VectorXd::Zero(pp_);
47 V_warm_ = Eigen::VectorXd::Zero(p_.Nu * m_);
48 g_ = Eigen::VectorXd::Zero(p_.Nu * m_);
49 tmp1_ = Eigen::VectorXd::Zero(p_.Nu * m_);
50 tmp2_ = Eigen::VectorXd::Zero(p_.Nu * m_);
51}

53// ---------------------------------------------------------------------------
54// Minimal Robust Positively Invariant set via iterated box-summation.
55//
56// For the closed-loop system e[k+1] = A_cl * e[k] + w[k],
57// the mRPI set is the limit of the Minkowski sum:
58// Z_inf = ⊕_{k=0}^{inf} A_cl^k * W
59//
60// For box W = {w : |w_i| <= wMax_i} the element-wise bound is:
61// z_max_new = |A_cl| * z_max + wMax
62// (where |A_cl| is the matrix of absolute values of A_cl elements).
63// This converges because rho(A_cl) < 1 (verified in constructor).
64// ---------------------------------------------------------------------------
65void TubeMPC::computeMRPI()
66{
67 const Eigen::MatrixXd A_cl = sys_.A + sys_.B * p_.K;
68 const Eigen::MatrixXd A_cl_abs = A_cl.cwiseAbs();
69
70 z_max_ = Eigen::VectorXd::Zero(n_);
71 for (int iter = 0; iter < p_.mRPI_maxIter; ++iter) {
72 const Eigen::VectorXd z_new = A_cl_abs * z_max_ + p_.wMax;
73 if ((z_new - z_max_).cwiseAbs().maxCoeff() < p_.mRPI_tol)
74 {
75 z_max_ = z_new;
76 break;
77 }
78 z_max_ = z_new;
79 }
80
81 // Tighten input constraints: v \in [uMin + |K|*z_max, uMax - |K|*z_max]
82 const Eigen::MatrixXd K_abs = p_.K.cwiseAbs();
83 const Eigen::VectorXd ku = K_abs * z_max_;
84 v_min_ = p_.uMin + ku;
85 v_max_ = p_.uMax - ku;
86
87 // Guard against infeasible tightening (would mean wMax or K is too large)
88 for (int i = 0; i < m_; ++i) {
89 if (v_min_(i) > v_max_(i)) {
90 // Fall back to zero-width interval at midpoint (safest fallback)
91 const double mid = 0.5 * (p_.uMin(i) + p_.uMax(i));
92 v_min_(i) = mid;
93 v_max_(i) = mid;
94 }
95 }
96}

98// ---------------------------------------------------------------------------
99// Condensed prediction matrices.
100//
101// Y_nom = F * x_nom_0 + Phi * V
102//
103// F : (Np*pp) x n -- free-response (no input)
104// Phi : (Np*pp) x (Nu*m)-- Markov response with hold-last at j = Nu-1
105//
106// Phi[i, j] = C * A^(i-1-j) * B for j < Nu-1 and j <= i-1
107// Phi[i, Nu-1] = C * sum_{t=Nu-1}^{i-1} A^(i-1-t) * B (accumulated hold-last)
108//
109// Hessian: H = 2*(Phi' * Qy * Phi + Ru) where Qy = block-diag(Q,..Q), Ru = block-diag(R,..R)
110// ---------------------------------------------------------------------------
111void TubeMPC::buildCondensedMatrices()
112{
113 const int Np = p_.Np;
114 const int Nu = p_.Nu;
115
116 F_.resize(Np * pp_, n_);
117 Phi_.resize(Np * pp_, Nu * m_);
118 Phi_.setZero();
119
120 // Precompute A^k * B for k=0..Np-1
121 Eigen::MatrixXd Ak(n_, n_);
122 Ak.setIdentity();
123 std::vector<Eigen::MatrixXd> CB_list(Np); // CB_list[k] = C * A^k * B
124
125 for (int k = 0; k < Np; ++k) {
126 CB_list[k] = sys_.C * Ak * sys_.B; // pp x m
127 Ak = sys_.A * Ak; // A^(k+1)
128 }
129
130 // F: row block i (0-based) = C * A^(i+1) * x0 (i=0..Np-1)
131 Eigen::MatrixXd Aipow(n_, n_);
132 Aipow = sys_.A;
133 for (int i = 0; i < Np; ++i) {
134 F_.block(i * pp_, 0, pp_, n_) = sys_.C * Aipow;
135 Aipow = sys_.A * Aipow; // A^(i+2) for next iteration
136 }
137
138 // Phi: column block j (0-based) covers input at time j (or hold-last at j=Nu-1)
139 // Phi[i, j] for i >= j: CB_list[i-j-1] (since C*A^(i-j-1)*B)
140 for (int j = 0; j < Nu; ++j) {
141 const bool is_last = (j == Nu - 1);
142 for (int i = j; i < Np; ++i) {
143 // At step i, input at step j contributes C*A^(i-j-1)*B
144 // For hold-last at j=Nu-1, accumulate from j onward
145 // CB_list[k] = C * A^k * B
146 // Phi[row i, col j] = C * A^(i-j) * B for j <= i (0-based block indices)
147 Phi_.block(i * pp_, j * m_, pp_, m_) += CB_list[i - j];
148
149 if (is_last && i < Np - 1) {
150 // For hold-last: input v[Nu-1] is repeated at steps Nu, Nu+1, ..., Np-1.
151 // The accumulated contribution is already handled via the loop below.
152 // Nothing extra here - the hold-last accumulation is done in the inner loop.
153 }
154 }
155
156 // For hold-last at j = Nu-1: add contributions for t = Nu, ..., Np-1
157 // where v[t] = v[Nu-1] (held), and these affect rows i >= t.
158 if (is_last) {
159 for (int t = Nu; t < Np; ++t) { // extra held steps
160 for (int i = t; i < Np; ++i) {
161 Phi_.block(i * pp_, j * m_, pp_, m_) += CB_list[i - t];
162 }
163 }
164 }
165 }
166
167 // Build cost matrices
168 // Qy = block-diag(Q, Q, ..., Q) size (Np*pp) x (Np*pp)
169 // Ru = block-diag(R, R, ..., R) size (Nu*m) x (Nu*m)
170 // H = 2*(Phi'*Qy*Phi + Ru)
171 Eigen::MatrixXd Qy = Eigen::MatrixXd::Zero(Np * pp_, Np * pp_);
172 for (int k = 0; k < Np; ++k)
173 Qy.block(k * pp_, k * pp_, pp_, pp_) = p_.Q;
174
175 Eigen::MatrixXd Ru = Eigen::MatrixXd::Zero(Nu * m_, Nu * m_);
176 for (int k = 0; k < Nu; ++k)
177 Ru.block(k * m_, k * m_, m_, m_) = p_.R;
178
179 const Eigen::MatrixXd PhiTQy = Phi_.transpose() * Qy;
180 H_ = 2.0 * (PhiTQy * Phi_ + Ru);
181 H_ = 0.5 * (H_ + H_.transpose()); // symmetrise
182
183 ldlt_.compute(H_);
184
185 // Lipschitz constant = lambda_max(H)
186 const Eigen::SelfAdjointEigenSolver<Eigen::MatrixXd> eig(H_);
187 const double L = eig.eigenvalues().maxCoeff();
188 L_inv_ = (L > 1e-12) ? (1.0 / L) : 1.0;
189
190 // Stacked box bounds for FISTA
191 lb_.resize(Nu * m_);
192 ub_.resize(Nu * m_);
193 for (int k = 0; k < Nu; ++k) {
194 lb_.segment(k * m_, m_) = v_min_;
195 ub_.segment(k * m_, m_) = v_max_;
196 }
197}

224// ---------------------------------------------------------------------------
225Eigen::VectorXd TubeMPC::computeControl()
226{
227 const int Nu = p_.Nu;
228
229 // --- build stacked reference ---
230 Eigen::VectorXd R_stacked(p_.Np * pp_);
231 for (int k = 0; k < p_.Np; ++k)
232 R_stacked.segment(k * pp_, pp_) = y_ref_;
233
234 // --- linear cost: g = 2*Phi'*Qy*(F*x_nom - R_stacked) ---
235 // (Qy factored into Phi'*Q block products for efficiency)
236 const Eigen::VectorXd Fx = F_ * x_nom_;
237 const Eigen::VectorXd err = Fx - R_stacked; // (Np*pp) x 1
238
239 // g = 2 * Phi' * Qy * err
240 // Qy = block-diag(Q) - apply block-wise to avoid explicit Qy allocation
241 Eigen::VectorXd Qy_err(p_.Np * pp_);
242 for (int k = 0; k < p_.Np; ++k)
243 Qy_err.segment(k * pp_, pp_) = p_.Q * err.segment(k * pp_, pp_);
244 g_ = 2.0 * (Phi_.transpose() * Qy_err);
245
246 // --- solve QP ---
247 QPSolveResult res = solveGradientProjectionQP(
248 H_, g_, lb_, ub_, ldlt_, 1.0 / L_inv_, p_.qpMaxIter, p_.qpTol,
249 V_warm_, tmp1_, tmp2_);
250 qp_converged_ = res.converged;
251
252 // --- extract nominal first move ---
253 const Eigen::VectorXd v_nom = V_warm_.head(m_);
254
255 // --- composite tube control: u = K*(x - x_nom) + v_nom ---
256 Eigen::VectorXd u = p_.K * (x_actual_ - x_nom_) + v_nom;
257 // Hard clamp to physical limits (tube feedback may slightly exceed tightened limits)
258 for (int i = 0; i < m_; ++i)
259 u(i) = std::clamp(u(i), p_.uMin(i), p_.uMax(i));
260
261 // --- advance nominal model ---
262 x_nom_ = sys_.A * x_nom_ + sys_.B * v_nom;
263
264 state_set_ = false;
265
266 // Shift warm start (move-blocking)
267 if (Nu > 1) {
268 V_warm_.head((Nu - 1) * m_) = V_warm_.segment(m_, (Nu - 1) * m_);
269 V_warm_.tail(m_) = v_nom; // hold last
270 }
271
272 const double u_scalar = u(0);
273 notifyObserver(u_scalar, y_ref_(0));
274 return u;
275}
