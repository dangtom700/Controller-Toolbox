14 // Sliding surface: s[k] = c_e.e[k] + c_de.(e[k] - e[k-1])
15 //
16 // Boundary layer saturation avoids chattering:
17 // |s| <= phi -> continuous PD law (sat = s/phi)
18 // |s| > phi -> full relay switching (sat = sign(s))
19 double DiscreteSMC::compute(double error)
20 {
21 if (!std::isfinite(error))
22 return u_prev_;
23
24 const double s = p_.c_e * error + p_.c_de * (error - e_prev_);
25
26 double sat_val;
27 if (p_.phi > 1e-12)
28 sat_val = std::max(-1.0, std::min(1.0, s / p_.phi));
29 else
30 sat_val = (s > 0.0) ? 1.0 : (s < 0.0 ? -1.0 : 0.0);
31
32 double u = -p_.K * sat_val;
33 u = std::max(p_.uMin, std::min(p_.uMax, u));
34
35 e_prev_ = error;
36 s_prev_ = s;
37 u_prev_ = u;
38 // M3 telemetry: sliding surface value
39 notifyObserverState("surface", Eigen::VectorXd::Constant(1, s));
40 notifyObserver(u, error);
41 return u;
42 }

55 : p_(params), Ts_(sampleTime)

60 // Discrete super-twisting update:
61 // s[k] = c_e.e[k] + c_de.(e[k] - e[k-1])
62 // u[k] = v_ - K1.|s|^{1/2}.sign(s) (power term - chattering-free)
63 // v_new = v_ - K2.Ts.sign(s) (integral term - Euler)
64 //
65 // The integrator v_ provides the equivalent of a sliding-mode integral that
66 // removes steady-state error without a boundary layer.
67 double SuperTwistingSMC::compute(double error)
68 {
69 if (!std::isfinite(error))
70 return std::max(p_.uMin, std::min(p_.uMax, v_));
71
72 const double s = p_.c_e * error + p_.c_de * (error - e_prev_);
73
74 const double sign_s = (s > 0.0) ? 1.0 : (s < 0.0 ? -1.0 : 0.0);
75 const double sqrt_s = std::sqrt(std::abs(s));
76
77 // Control output
78 double u = v_ - p_.K1 * sqrt_s * sign_s;
79 u = std::max(p_.uMin, std::min(p_.uMax, u));
80
81 // Integrator update (Euler - keep K2*Ts small for accuracy)
82 v_ -= p_.K2 * Ts_ * sign_s;
83
84 e_prev_ = error;
85 s_prev_ = s;
86 return u;
87 }
