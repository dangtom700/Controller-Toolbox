9TruncationResult balancedTruncate(const StateSpace &sys, int r)
10{
11 const int n = sys.stateSize();
12
13 if (r < 1 || r >= n)
14 throw std::invalid_argument(
15 "balancedTruncate: r must satisfy 1 <= r < n (n=" + std::to_string(n) + ").");
16
17 // --- Step 0: Check stability ---
18 if (!SystemAnalysis::isDiscreteStable(sys))
19 throw std::invalid_argument(
20 "balancedTruncate: system must be strictly stable (all |eigenvalues(A)| < 1). "
21 "Use a stabilising pre-compensator if needed.");
22
23 const Eigen::MatrixXd &A = sys.A;
24 const Eigen::MatrixXd &B = sys.B;
25 const Eigen::MatrixXd &C = sys.C;
26 const Eigen::MatrixXd &D = sys.D;
27
28 // --- Step 1: Gramians ---
29 // Controllability: A*Pc*A' - Pc + B*B' = 0
30 const Eigen::MatrixXd Pc = SystemAnalysis::solveDiscreteLyapunov(A, B * B.transpose());
31
32 // Observability: A'*Po*A - Po + C'*C = 0
33 const Eigen::MatrixXd Po = SystemAnalysis::solveDiscreteLyapunov(A.transpose(),
34 C.transpose() * C);
35
36 // --- Step 2: Cholesky of Pc -> Pc = L_c * L_c' ---
37 Eigen::LLT<Eigen::MatrixXd> llt(Pc);
38 if (llt.info() != Eigen::Success)
39 throw std::invalid_argument(
40 "balancedTruncate: controllability gramian Pc is not positive definite. "
41 "The system may have uncontrollable modes. Add a small regularisation: "
42 "Pc += eps*I before calling.");
43
44 const Eigen::MatrixXd L_c = llt.matrixL(); // lower triangular
45
46 // --- Step 3: Symmetric matrix M = L_c' * Po * L_c ---
47 const Eigen::MatrixXd M = L_c.transpose() * Po * L_c;
48
49 // --- Step 4: Symmetric eigendecomposition of M ---
50 // Eigenvalues lambda_i = sigma_i^2 where sigma_i are the Hankel singular values.
51 // SelfAdjointEigenSolver gives ascending order - we want descending.
52 Eigen::SelfAdjointEigenSolver<Eigen::MatrixXd> eig(M);
53
54 // Collect HSVs in descending order
55 const int n_eig = eig.eigenvalues().size();
56 Eigen::VectorXd hsv(n_eig);
57 Eigen::MatrixXd U_desc(n_eig, n_eig);
58 for (int i = 0; i < n_eig; ++i)
59 {
60 // Clamp negative eigenvalues (rounding errors) to 0
61 const double ev = std::max(eig.eigenvalues()(n_eig - 1 - i), 0.0);
62 hsv(i) = std::sqrt(ev);
63 U_desc.col(i) = eig.eigenvectors().col(n_eig - 1 - i);
64 }
65
66 // --- Step 5: Balancing transformation ---
67 // T_r = L_c * U * diag(hsv^{-1/2}) [right transformation]
68 // T_l = T_r^{-1} [left transformation]
69 const Eigen::VectorXd sig_inv_sqrt =
70 hsv.cwiseInverse().cwiseSqrt(); // sigma_i^{-1/2}, all sigma_i > 0 if gramians are PD
71
72 const Eigen::MatrixXd T_r = L_c * U_desc * sig_inv_sqrt.asDiagonal();
73 const Eigen::MatrixXd T_l = T_r.inverse(); // affordable for small n
74
75 // --- Step 6: Balanced realisation ---
76 const Eigen::MatrixXd A_bal = T_l * A * T_r;
77 const Eigen::MatrixXd B_bal = T_l * B;
78 const Eigen::MatrixXd C_bal = C * T_r;
79 // D is unchanged
80
81 // --- Step 7: Truncate to order r ---
82 StateSpace reduced(
83 A_bal.topLeftCorner(r, r),
84 B_bal.topRows(r),
85 C_bal.leftCols(r),
86 D,
87 sys.Ts);
88
89 // --- Step 8: Error bound ---
90 // ||G - Gᵣ||inf <= 2 . Sigma_i=ᵣ₊1^n sigma_i
91 const double error_bound = 2.0 * hsv.tail(n - r).sum();
92
93 const bool is_stable = SystemAnalysis::isDiscreteStable(reduced);
94
95 return {reduced, hsv, error_bound, is_stable};
96}

98int suggestOrder(const TruncationResult &res, double tol)
99{
100 const Eigen::VectorXd &hsv = res.hankelSingularValues;
101 const int n = static_cast<int>(hsv.size());
102 const double total_norm = 2.0 * hsv.sum(); // approx = 2.||G||inf (loose upper bound)
103 const double threshold = tol * total_norm;
104
105 // Return smallest r such that 2.Sigma_i=ᵣ₊1^n sigma_i < threshold
106 double tail_sum = 0.0;
107 for (int i = n - 1; i >= 1; --i)
108 {
109 tail_sum += hsv(i);
110 if (2.0 * tail_sum >= threshold)
111 return std::min(i + 1, n - 1); // order r = i: tail from i+1 is within tol
112 }
113 return 1; // single state is sufficient
114}
