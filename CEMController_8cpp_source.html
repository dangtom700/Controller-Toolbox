31double CEMController::rolloutCost(const Eigen::VectorXd& x0,
32 const Eigen::VectorXd& u_seq,
33 const Eigen::VectorXd& y_ref) const
34{
35 Eigen::VectorXd x = x0;
36 double cost = 0.0;
37 for (int k = 0; k < p_.Np; ++k) {
38 Eigen::VectorXd u_k(1); u_k(0) = u_seq(k);
39 x = f_(x, u_k);
40 Eigen::VectorXd e = C_ * x - y_ref;
41 cost += p_.Q * e.squaredNorm() + p_.R * u_k.squaredNorm();
42 }
43 return cost;
44}

50Eigen::VectorXd CEMController::computeRef(const Eigen::VectorXd& x_cur,
51 const Eigen::VectorXd& y_ref)
52{
53 const int N = p_.N_samples;
54 const int K = std::max(1, static_cast<int>(p_.elite_frac * N));
55
56 Eigen::VectorXd mu = mu_; // warm-start from previous solution
57 double sigma = p_.sigma_init;
58
59 std::normal_distribution<double> nd(0.0, 1.0);
60
61 std::vector<Eigen::VectorXd> samples(N, Eigen::VectorXd(p_.Np));
62 std::vector<double> costs(N);
63
64 for (int iter = 0; iter < p_.n_iter; ++iter) {
65 // Sample action sequences
66 for (int i = 0; i < N; ++i) {
67 for (int k = 0; k < p_.Np; ++k)
68 samples[i](k) = std::clamp(mu(k) + sigma * nd(rng_),
69 p_.uMin, p_.uMax);
70 costs[i] = rolloutCost(x_cur, samples[i], y_ref);
71 }
72
73 // Sort by cost and keep elite set
74 std::vector<int> idx(N);
75 std::iota(idx.begin(), idx.end(), 0);
76 std::partial_sort(idx.begin(), idx.begin() + K, idx.end(),
77 [&](int a, int b) { return costs[a] < costs[b]; });
78
79 // Update distribution from elite set
80 Eigen::VectorXd new_mu = Eigen::VectorXd::Zero(p_.Np);
81 double new_var = 0.0;
82 for (int j = 0; j < K; ++j) {
83 new_mu += samples[idx[j]];
84 new_var += (samples[idx[j]] - mu).squaredNorm() / p_.Np;
85 }
86 mu = new_mu / K;
87 sigma = std::max(p_.sigma_min, std::sqrt(new_var / K));
88 }
89
90 last_cost_ = rolloutCost(x_cur, mu, y_ref);
91 mu_ = mu; // warm-start next call
92
93 Eigen::VectorXd u_out(1);
94 u_out(0) = std::clamp(mu(0), p_.uMin, p_.uMax);
95 notifyObserver(u_out(0), x_cur.norm());
96 return u_out;
97}
