15// ============================================================================
16// MF::operator() - evaluate the tagged-union membership function at x.
17// No heap allocation; all branches are plain arithmetic.
18// ============================================================================
19
20double MF::operator()(double x) const
21{
22 switch (type) {
23 case Type::Triangular: {
24 // p: [a, c, b] peaks at c, zero at a and b
25 const double a = p[0], c = p[1], b = p[2];
26 if (x <= a || x >= b) return 0.0;
27 if (x <= c) return (x - a) / (c - a + 1e-12);
28 return (b - x) / (b - c + 1e-12);
29 }
30 case Type::Trapezoidal: {
31 // p: [a, b, c, d] flat-top [b,c], zero outside [a,d]
32 const double a = p[0], b = p[1], c = p[2], d = p[3];
33 if (x <= a || x >= d) return 0.0;
34 if (x >= b && x <= c) return 1.0;
35 if (x < b) return (x - a) / (b - a + 1e-12);
36 return (d - x) / (d - c + 1e-12);
37 }
38 case Type::Gaussian: {
39 // p: [mean, sigma]
40 const double z = (x - p[0]) / (p[1] + 1e-12);
41 return std::exp(-0.5 * z * z);
42 }
43 case Type::Singleton: {
44 // p: [value]
45 return (std::abs(x - p[0]) < 1e-9) ? 1.0 : 0.0;
46 }
47 case Type::ShoulderLeft: {
48 // p: [a, b] 1 for x<=a, linear 1->0 from a to b
49 if (x <= p[0]) return 1.0;
50 if (x >= p[1]) return 0.0;
51 return (p[1] - x) / (p[1] - p[0] + 1e-12);
52 }
53 case Type::ShoulderRight: {
54 // p: [a, b] 0 for x<=a, linear 0->1 from a to b, 1 for x>=b
55 if (x <= p[0]) return 0.0;
56 if (x >= p[1]) return 1.0;
57 return (x - p[0]) / (p[1] - p[0] + 1e-12);
58 }
59 }
60 return 0.0; // unreachable - suppresses compiler warning
61}

163void FuzzySystem::addRule(const Rule& rule)
164{
165 // Validate every antecedent against registered input variables.
166 // A typo in a term name silently produces zero rule strength without this check.
167 for (std::size_t a = 0; a < rule.antecedents.size(); ++a)
168 {
169 const auto& ant = rule.antecedents[a];
170 if (ant.input_idx < 0 || ant.input_idx >= static_cast<int>(inputs_.size()))
171 throw std::out_of_range(
172 "FuzzySystem::addRule: antecedent[" + std::to_string(a) +
173 "] input_idx=" + std::to_string(ant.input_idx) +
174 " is out of range (numInputs=" + std::to_string(inputs_.size()) + ").");
175 if (ant.term_idx < 0 ||
176 ant.term_idx >= static_cast<int>(inputs_[ant.input_idx].terms.size()))
177 throw std::out_of_range(
178 "FuzzySystem::addRule: antecedent[" + std::to_string(a) +
179 "] term_idx=" + std::to_string(ant.term_idx) +
180 " is out of range for input '" + inputs_[ant.input_idx].name +
181 "' (numTerms=" + std::to_string(inputs_[ant.input_idx].terms.size()) + ").");
182 }
183 // Validate consequent against registered output variable.
184 if (outputs_.empty())
185 throw std::logic_error("FuzzySystem::addRule: no output variable registered. "
186 "Call addOutput() before addRule().");
187 if (rule.consequent_term_idx < 0 ||
188 rule.consequent_term_idx >= static_cast<int>(outputs_[0].terms.size()))
189 throw std::out_of_range(
190 "FuzzySystem::addRule: consequent_term_idx=" +
191 std::to_string(rule.consequent_term_idx) +
192 " is out of range for output '" + outputs_[0].name +
193 "' (numTerms=" + std::to_string(outputs_[0].terms.size()) + ").");
194
195 rules_.push_back(rule);
196}

198double FuzzySystem::ruleStrength(const Rule& r,
199 const std::vector<std::vector<double>>& mu) const
200{
201 // Product t-norm over all antecedents (product-AND)
202 double strength = r.weight;
203 for (auto& ant : r.antecedents)
204 strength *= mu[ant.input_idx][ant.term_idx];
205 return strength;
206}

208double FuzzySystem::evaluate(const std::vector<double>& inputs)
209{
210 if (inputs.size() != inputs_.size())
211 throw std::invalid_argument("FuzzySystem::evaluate: wrong number of inputs");
212 if (outputs_.empty())
213 throw std::logic_error("FuzzySystem::evaluate: no output variable defined");
214
215 // Fuzzify all inputs - fill pre-allocated workspace mu_ in-place (no heap alloc)
216 for (std::size_t i = 0; i < inputs_.size(); ++i) {
217 const auto& terms = inputs_[i].terms;
218 for (std::size_t t = 0; t < terms.size(); ++t)
219 mu_[i][t] = std::clamp(terms[t].mf(inputs[i]), 0.0, 1.0);
220 }
221
222 // Fire all rules - accumulate activation strengths per output term (max aggregation)
223 const int nTerms = static_cast<int>(outputs_[0].terms.size());
224 std::fill(strengths_.begin(), strengths_.end(), 0.0);
225
226 for (auto& rule : rules_) {
227 double w = ruleStrength(rule, mu_);
228 int tidx = rule.consequent_term_idx;
229 if (tidx >= 0 && tidx < nTerms)
230 strengths_[tidx] = std::max(strengths_[tidx], w);
231 }
232
233 // Defuzzify
234 double result = 0.0;
235 if (params.inference == InferenceMethod::TakagiSugeno ||
236 params.defuzz == DefuzzMethod::WeightedAverage)
237 result = defuzzWeightedAvg(strengths_);
238 else
239 result = defuzzCoG(strengths_);
240
241 return std::clamp(result, params.uMin, params.uMax);
242}

244// Centre of Gravity over a discrete universe grid
245double FuzzySystem::defuzzCoG(const std::vector<double>& strengths) const
246{
247 const auto& out = outputs_[0];
248 const int N = params.cog_resolution;
249 const double lo = out.lo, hi = out.hi;
250 const double step = (hi - lo) / (N - 1);
251
252 double num = 0.0, den = 0.0;
253 for (int k = 0; k < N; ++k) {
254 double x = lo + k * step;
255 // Aggregate: max over all active rule-clipped output MFs
256 double agg = 0.0;
257 for (int t = 0; t < static_cast<int>(out.terms.size()); ++t) {
258 double clipped = std::min(strengths[t], out.terms[t].mf(x));
259 agg = std::max(agg, clipped);
260 }
261 num += x * agg;
262 den += agg;
263 }
264 if (den < 1e-12) return (lo + hi) * 0.5; // no rules fired - return centre
265 return num / den;
266}

268// Weighted average (Takagi-Sugeno / singleton consequents)
269double FuzzySystem::defuzzWeightedAvg(const std::vector<double>& strengths) const
270{
271 const auto& out = outputs_[0];
272 double num = 0.0, den = 0.0;
273 for (int t = 0; t < static_cast<int>(out.terms.size()); ++t) {
274 double centre;
275 if (out.terms[t].peak.has_value()) {
276 // Exact peak provided (singletons, or explicitly set TS terms) - no grid search.
277 centre = out.terms[t].peak.value();
278 } else {
279 // Fall back to grid search for continuous Mamdani-style output terms.
280 centre = (out.lo + out.hi) * 0.5;
281 constexpr int M = kTSPeakSearchResolution;
282 double best_mu = 0.0;
283 for (int k = 0; k < M; ++k) {
284 double x = out.lo + k * (out.hi - out.lo) / (M - 1);
285 double mu = out.terms[t].mf(x);
286 if (mu > best_mu) { best_mu = mu; centre = x; }
287 }
288 }
289 num += strengths[t] * centre;
290 den += strengths[t];
291 }
292 if (den < 1e-12) return (out.lo + out.hi) * 0.5;
293 return num / den;
294}

301// Canonical 5-term partition on [-1, 1]: NL, NS, ZE, PS, PL
302// 25-rule diagonal table (same as Mamdani textbook PD table)
303static void build5TermPartition(LinguisticVariable& v)
304{
305 v.terms.push_back({"NL", mfShoulderLeft(-1.0, -0.5)});
306 v.terms.push_back({"NS", mfTriangular (-1.0, -0.5, 0.0)});
307 v.terms.push_back({"ZE", mfTriangular (-0.5, 0.0, 0.5)});
308 v.terms.push_back({"PS", mfTriangular ( 0.0, 0.5, 1.0)});
309 v.terms.push_back({"PL", mfShoulderRight(0.5, 1.0)});
310}
311
312// Diagonal 25-rule table: e_term x de_term -> u_term
313// Index: NL=0, NS=1, ZE=2, PS=3, PL=4
314static const int kPDRuleTable[5][5] = {
315 // de: NL NS ZE PS PL e->
316 { 0, 0, 0, 1, 2 }, // NL
317 { 0, 0, 1, 2, 3 }, // NS
318 { 0, 1, 2, 3, 4 }, // ZE
319 { 1, 2, 3, 4, 4 }, // PS
320 { 2, 3, 4, 4, 4 }, // PL
321};

323void FuzzyPD::buildSystem()
324{
325 sys_ = FuzzySystem{};
326 sys_.params.inference = InferenceMethod::Mamdani;
327 sys_.params.defuzz = DefuzzMethod::CoG;
328 sys_.params.cog_resolution = kCoGResolutionDefault;
329 sys_.params.uMin = -1.0;
330 sys_.params.uMax = 1.0;
331
332 LinguisticVariable ve; ve.name = "e"; ve.lo = -1.0; ve.hi = 1.0;
333 LinguisticVariable vde; vde.name = "de"; vde.lo = -1.0; vde.hi = 1.0;
334 LinguisticVariable vu; vu.name = "u"; vu.lo = -1.0; vu.hi = 1.0;
335
336 build5TermPartition(ve);
337 build5TermPartition(vde);
338 build5TermPartition(vu);
339
340 sys_.addInput(ve);
341 sys_.addInput(vde);
342 sys_.addOutput(vu);
343
344 // Build 25 rules from the diagonal table
345 for (int ie = 0; ie < 5; ++ie) {
346 for (int ide = 0; ide < 5; ++ide) {
347 Rule r;
348 r.antecedents.push_back({0, ie});
349 r.antecedents.push_back({1, ide});
350 r.consequent_term_idx = kPDRuleTable[ie][ide];
351 r.weight = 1.0;
352 sys_.addRule(r);
353 }
354 }

369double FuzzyPD::compute(double error)
370{
371 if (!std::isfinite(error)) return u_prev_;
372
373 double de = (error - e_prev_) / Ts_;
374
375 // Normalise to [-1, 1] universe
376 double e_n = std::clamp(error / (p_.e_scale + 1e-12), -1.0, 1.0);
377 double de_n = std::clamp(de / (p_.de_scale + 1e-12), -1.0, 1.0);
378
379 double u_n = sys_.evaluate({e_n, de_n});
380
381 double u = std::clamp(u_n * p_.u_scale, p_.uMin, p_.uMax);
382
383 e_prev_ = error;
384 u_prev_ = u;
385 return u;
386}

409
410double FuzzyPID::compute(double error)
411{
412 if (!std::isfinite(error)) return u_prev_;
413
414 // FuzzyPD block provides the P+D part
415 double u_pd = pd_block_.compute(error);
416
417 // Integral accumulation (backward Euler) + anti-windup
418 double ki_update = p_.Ki * Ts_ * error;
419 double u_unsat = u_pd + integral_ + ki_update;
420 double u_sat = std::clamp(u_unsat, p_.uMin, p_.uMax);
421
422 integral_ += ki_update + p_.Kb * (u_sat - u_unsat);
423 u_prev_ = u_sat;
424 return u_sat;
425}

433
434void FuzzyPID::bumplessInit(double u_target, double error)
435{
436 pd_block_.reset();
437 // Estimate PD contribution at this error with de=0 (first call after reset).
438 double u_pd_est = pd_block_.compute(error);
439 pd_block_.reset(); // restore clean derivative state (e_prev_=0)
440 // Set integral so that u_pd_est + integral_ approx = u_target on the next compute().
441 integral_ = u_target - u_pd_est;
442 u_prev_ = u_target;
443}

450void FuzzySupervisor::buildSystem()
451{
452 sys_ = FuzzySystem{};
453 sys_.params.inference = InferenceMethod::Mamdani;
454 sys_.params.defuzz = DefuzzMethod::CoG;
455 sys_.params.cog_resolution = kTSPeakSearchResolution;
456 sys_.params.uMin = 0.0;
457 sys_.params.uMax = 1.0;
458
459 // Input 1: normalised error magnitude [0, 1+]
460 LinguisticVariable verr;
461 verr.name = "error_norm"; verr.lo = 0.0; verr.hi = 1.5;
462 verr.terms.push_back({"Small", mfTrapezoidal(0.0, 0.0, 0.25, 0.45)});
463 verr.terms.push_back({"Medium", mfTriangular (0.30, 0.50, 0.70)});
464 verr.terms.push_back({"Large", mfShoulderRight(0.60, 0.90)});
465
466 // Input 2: normalised error trend (d|e|/dt) [-1, 1]
467 LinguisticVariable vtrend;
468 vtrend.name = "trend"; vtrend.lo = -1.0; vtrend.hi = 1.0;
469 vtrend.terms.push_back({"Decreasing", mfShoulderLeft(-0.6, -0.1)});
470 vtrend.terms.push_back({"Steady", mfTriangular (-0.3, 0.0, 0.3)});
471 vtrend.terms.push_back({"Increasing", mfShoulderRight(0.1, 0.6)});
472
473 // Output: relinearize_signal [0, 1]
474 LinguisticVariable vsig;
475 vsig.name = "signal"; vsig.lo = 0.0; vsig.hi = 1.0;
476 vsig.terms.push_back({"None", mfTriangular(0.0, 0.0, 0.3)}); // 0
477 vsig.terms.push_back({"Maybe", mfTriangular(0.3, 0.5, 0.7)}); // 1
478 vsig.terms.push_back({"High", mfTriangular(0.6, 0.8, 1.0)}); // 2
479 vsig.terms.push_back({"Full", mfShoulderRight(0.8, 1.0)}); // 3
480
481 sys_.addInput(verr);
482 sys_.addInput(vtrend);
483 sys_.addOutput(vsig);
484
485 // 9-rule table: error_term(3) x trend_term(3) -> signal_term(4)
486 // Rows = error: Small(0), Medium(1), Large(2)
487 // Cols = trend: Decreasing(0), Steady(1), Increasing(2)
488 // Output terms: None=0, Maybe=1, High=2, Full=3
489 const int table[3][3] = {
490 { 0, 0, 1 }, // Small error
491 { 0, 1, 2 }, // Medium error
492 { 1, 2, 3 }, // Large error
493 };
494
495 for (int ie = 0; ie < 3; ++ie) {
496 for (int it = 0; it < 3; ++it) {
497 Rule r;
498 r.antecedents.push_back({0, ie});
499 r.antecedents.push_back({1, it});
500 r.consequent_term_idx = table[ie][it];
501 r.weight = 1.0;
502 sys_.addRule(r);
503 }
504 }
505}

519SupervisorDecision FuzzySupervisor::update(double abs_error)
520{
521 // Normalise error
522 double err_n = abs_error / (p_.e_threshold + 1e-12);
523
524 // Trend: rate of change of |error|, normalised
525 double d_err = (abs_error - abs_error_prev_) / (Ts_ + 1e-12);
526 double trend_n = std::clamp(d_err / (p_.trend_threshold + 1e-12), -1.0, 1.0);
527
528 abs_error_prev_ = abs_error;
529
530 // Fuzzy inference
531 double signal = sys_.evaluate({std::clamp(err_n, 0.0, 1.5), trend_n});
532
533 // Hysteresis / cooldown
534 bool trigger = false;
535 if (cooldown_remaining_ > 0) {
536 --cooldown_remaining_;
537 } else if (signal > p_.signal_threshold) {
538 trigger = true;
539 cooldown_remaining_ = p_.cooldown_steps;
540 }
541
542 return {signal, trigger, err_n, trend_n};
543}
